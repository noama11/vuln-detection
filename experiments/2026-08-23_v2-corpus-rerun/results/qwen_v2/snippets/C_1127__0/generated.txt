AutocompleteMatch ConvertScoredHistoryMatchToAutocompleteMatch(const ScoredHistoryMatch* history_match, int score) {
    AutocompleteMatch match;
    memset(&match, 0, sizeof(AutocompleteMatch));

    match.url = history_match->url;
    match.title = history_match->title;
    match.description = history_match->description;
    match.score = score;

    // Format URL and description for display
    char url_buffer[URL_MAX_LENGTH];
    snprintf(url_buffer, sizeof(url_buffer), "%s", history_match->url);
    match.display_url = url_buffer;

    // Handle inline autocomplete offsets
    if (history_match->has_inline_match) {
        match.inline_autocomplete_offset = history_match->inline_match_offset;
    }

    // Format term matches for URL and title
    char formatted_url[URL_MAX_LENGTH];
    char formatted_title[TITLE_MAX_LENGTH];

    FormatMatchForDisplay(history_match->url, history_match->url_match_positions, formatted_url, sizeof(formatted_url));
    FormatMatchForDisplay(history_match->title, history_match->title_match_positions, formatted_title, sizeof(formatted_title));

    match.formatted_url = formatted_url;
    match.formatted_title = formatted_title;

    return match;
}
