void pagevec_release(struct pagevec *pvec, struct zone *zone) {
    struct page *page;
    int i, unlocked = 0;
    struct page *backup[PAGEVEC_SIZE];
    int backup_count = 0;

    /* Phase 1: Isolate pages and prepare backup vector */
    for (i = 0; i < pagevec_count(pvec); i++) {
        page = pvec->pages[i];
        if (PageLocked(page)) {
            unlock_page(page);
            unlocked++;
        } else {
            backup[backup_count++] = page;
        }
    }

    /* Update the pagevec with the backup pages */
    for (i = 0; i < backup_count; i++) {
        pvec->pages[i] = backup[i];
    }
    pvec->nr = backup_count;

    /* Phase 2: Unlock pages and manage state */
    for (i = 0; i < unlocked; i++) {
        page = pvec->pages[i];
        if (page->mapping && page->index != 0) {
            page->mapping = NULL;
        }
        page->flags &= ~PG_locked;
    }

    /* Phase 3: Process fast path putback */
    for (i = 0; i < backup_count; i++) {
        page = pvec->pages[i];
        if (PageLRU(page)) {
            __pagevec_add(pvec, page);
        } else {
            __free_page(page);
        }
    }

    pagevec_reinit(pvec);
}
