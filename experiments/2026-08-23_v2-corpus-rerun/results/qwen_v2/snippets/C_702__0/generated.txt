int get_data_block(struct inode *inode, sector_t iblock, struct buffer_head *bh, int create, int flag, pgoff_t *next_pgofs) {
    struct address_space *mapping = inode->i_mapping;
    int ret;

    ret = block_map(mapping, iblock, create, flag, bh, next_pgofs);
    if (ret < 0)
        return ret;

    if (!bh)
        return -EIO;

    return 0;
}
