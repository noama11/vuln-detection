void prepare_page_callback(GtkNotebook *assistant, GtkWidget *page, gpointer user_data) {
    gboolean processing_finished = FALSE; // Assume this is determined by some logic
    GtkWidget *back_button = gtk_notebook_get_tab_label(GTK_NOTEBOOK(assistant), page);
    GtkWidget *next_button = gtk_notebook_get_nth_page(GTK_NOTEBOOK(assistant), gtk_notebook_get_current_page(assistant) + 1);

    // Save text fields before updating GUI
    // (Assume save_text_fields() is a helper function defined elsewhere)
    save_text_fields();

    if (processing_finished) {
        gtk_widget_hide(back_button);
        gtk_widget_hide(next_button);
        gtk_label_set_text(GTK_LABEL(warning_label), "");
    } else {
        gtk_widget_show(back_button);
        gtk_widget_show(next_button);
    }

    gint current_page_num = gtk_notebook_get_current_page(assistant);
    GtkWidget *current_page = gtk_notebook_get_nth_page(assistant, current_page_num);

    if (current_page == event_progress_page) {
        update_progress_display();
    } else if (current_page == selector_page) {
        update_selector_ui();
    }

    // Update checkboxes and highlight forbidden elements
    update_checkboxes();
    highlight_forbidden_elements();
}
