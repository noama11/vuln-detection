int online_pages(unsigned long pfn, unsigned long nr_pages) {
    int ret = 0;
    struct mem_notify {
        int status;
        int nid;
    } notify;

    lock_hotplug();

    if (!node_online(pfn_to_nid(pfn))) {
        notify.nid = pfn_to_nid(pfn);
        send_notify_memory_online(&notify);
        if (notify.status < 0) {
            ret = notify.status;
            goto unlock;
        }
    }

    if (!zone_populated()) {
        rebuild_zonelists();
    }

    ret = online_page_range(pfn, nr_pages);
    if (ret < 0) {
        goto unlock;
    }

    update_zone_statistics(pfn, nr_pages);
    if (should_rebuild_zonelists()) {
        rebuild_zonelists();
    }

    send_notify_memory_online_complete(&notify);

unlock:
    unlock_hotplug();
    return ret;
}
