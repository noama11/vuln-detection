int rds_atomic_op_process(struct rds_sock *rs, struct rds_message *rm, struct cmsghdr *cmsg)
{
    struct rds_atomic_op_args *args;
    struct rds_atomic_op *op;
    struct rds_notifier *notifier;
    int ret = -EINVAL;

    if (cmsg->cmsg_len < CMSG_LEN(sizeof(*args)))
        return ret;

    args = CMSG_DATA(cmsg);
    if (rs->atomic_op_active)
        return -EBUSY;

    op = kzalloc(sizeof(*op), GFP_KERNEL);
    if (!op)
        return -ENOMEM;

    op->type = args->type;
    op->flags = args->flags;
    op->length = args->length;

    op->sg = kmalloc_array(op->num_sg, sizeof(*op->sg), GFP_KERNEL);
    if (!op->sg) {
        kfree(op);
        return -ENOMEM;
    }

    if (!is_aligned((unsigned long)op->sg, L1_CACHE_BYTES)) {
        kfree(op->sg);
        kfree(op);
        return -EFAULT;
    }

    if (args->notify) {
        notifier = rds_notifier_alloc(rs, op);
        if (!notifier) {
            kfree(op->sg);
            kfree(op);
            return -ENOMEM;
        }
        op->notifier = notifier;
    }

    op->remote_addr = args->remote_addr;
    op->key = args->key;

    rs->atomic_op_active = true;
    rm->atomic_op = op;

    return 0;
}
