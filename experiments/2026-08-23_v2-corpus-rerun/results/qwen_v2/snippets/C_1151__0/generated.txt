bool ResolveRelativeURL(const char* base_url, const url_parse::Parsed& base_parsed,
                        const CHAR* relative_url, const url_parse::Component& relative_component,
                        CharsetConverter* query_converter, CanonOutput* output,
                        url_parse::Parsed* out_parsed) {
  // Parse the relative URL to extract its components
  url_parse::Parsed relative_parsed;
  if (!url_parse::ParseURL(relative_url, relative_component, &relative_parsed)) {
    return false;
  }

  // Create a set of replacements for the base URL using the parsed components of the relative URL
  url_parse::Replacements replacements;
  if (!url_parse::CreateReplacements(base_parsed, relative_parsed, &replacements)) {
    return false;
  }

  // Call another function to perform the actual replacement and output the resolved URL
  return url_parse::ReplaceURL(base_url, base_parsed, replacements, query_converter, output, out_parsed);
}
