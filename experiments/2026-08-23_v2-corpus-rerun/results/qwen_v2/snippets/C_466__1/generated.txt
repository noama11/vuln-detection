RList* get_global_entries(RBinWasmObj *bin, RBinWasmSection *sec) {
    RList *globals = r_list_new();
    if (!globals) {
        return NULL;
    }

    ut8 *buf = sec->data;
    int len = sec->size;
    int i = 0;

    while (i < len) {
        RBinWasmGlobalEntry *entry = R_NEW0(RBinWasmGlobalEntry);
        if (!entry) {
            r_list_free(globals);
            return globals;
        }

        if (i + 1 > len) {
            r_list_free(globals);
            free(entry);
            return NULL;
        }

        entry->content = buf[i];
        i++;

        if (i + 1 > len) {
            r_list_free(globals);
            free(entry);
            return NULL;
        }

        entry->mutability = buf[i];
        i++;

        if (!r_list_append(globals, entry)) {
            r_list_free(globals);
            free(entry);
            return globals;
        }
    }

    return globals;
}
