bool resize_shaper_arrays(ASS_Shaper *shaper, size_t new_size) {
    if (new_size <= shaper->n_glyphs)
        return true;

    void *temp;

    temp = realloc(shaper->event_text, new_size * sizeof(*shaper->event_text));
    if (!temp)
        return false;
    shaper->event_text = temp;

    temp = realloc(shaper->ctypes, new_size * sizeof(*shaper->ctypes));
    if (!temp)
        return false;
    shaper->ctypes = temp;

    temp = realloc(shaper->emblevels, new_size * sizeof(*shaper->emblevels));
    if (!temp)
        return false;
    shaper->emblevels = temp;

    temp = realloc(shaper->cmap, new_size * sizeof(*shaper->cmap));
    if (!temp)
        return false;
    shaper->cmap = temp;

    shaper->n_glyphs = new_size;
    return true;
}
