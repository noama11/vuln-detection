void cleanup_keys(struct list_head *keys) {
    struct list_head *pos, *tmp;
    struct key *key;

    list_for_each_safe(pos, tmp, keys) {
        key = list_entry(pos, struct key, list);
        list_del(&key->list);

        if (key->in_quota) {
            key->user->key_count--;
            key->user->quota--;
        }

        if (key->type->destroy) {
            key->type->destroy(key);
        }

        free(key);
    }
}
