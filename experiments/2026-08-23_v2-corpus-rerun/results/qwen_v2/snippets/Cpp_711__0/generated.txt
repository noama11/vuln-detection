std::string computeOValue(std::string const& user_password, std::string const& owner_password, QPDF::EncryptionData const& data) {
    // Derive RC4 key from user and owner passwords
    std::string key = deriveRC4Key(user_password, owner_password, data);

    // Pad or truncate the user password
    std::string padded_user_password = padOrTruncatePassword(user_password, data);

    // Initialize RC4 state
    RC4State rc4_state;
    rc4_init(&rc4_state, key);

    // Iterate RC4 algorithm
    std::string o_value;
    o_value.resize(data.key_length);
    rc4_crypt(&rc4_state, &padded_user_password[0], &o_value[0], padded_user_password.size());

    return o_value;
}
