int walk_huge_page_range(unsigned long addr, unsigned long end, struct mm_walk *walk) {
    int ret = 0;
    struct hstate *h;
    unsigned long mask;

    for (; addr < end; addr += mask) {
        h = hstate_addr(addr);
        if (!h)
            return -EFAULT;

        mask = huge_page_mask(h);
        if ((addr & (mask - 1)) != 0)
            return -EFAULT;

        ret = walk->vma ? walk->vma(addr, addr + mask, walk->vma) : 0;
        if (ret < 0)
            break;
    }

    return ret;
}
