int process_component(mif_hdr_t *hdr, char *buf) {
    component_t comp;
    tv_parser_t parser;
    int ret = -1;

    memset(&comp, 0, sizeof(comp));
    tv_parser_init(&parser, buf);

    if (!tv_parser_get_int(&parser, "TLX", &comp.tlx)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "TLY", &comp.tly)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "WIDTH", &comp.width)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "HEIGHT", &comp.height)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "XSR", &comp.xsr)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "YSR", &comp.ysr)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "PRECISION", &comp.precision)) {
        goto error;
    }

    if (!tv_parser_get_int(&parser, "SIGN", &comp.sign)) {
        goto error;
    }

    comp.data = tv_parser_get_string(&parser, "DATA");
    if (!comp.data) {
        goto error;
    }

    if (component_create(&comp) != 0) {
        goto error;
    }

    if (mif_hdr_add_component(hdr, &comp) != 0) {
        goto error;
    }

    ret = 0;

error:
    tv_parser_cleanup(&parser);
    return ret;
}
