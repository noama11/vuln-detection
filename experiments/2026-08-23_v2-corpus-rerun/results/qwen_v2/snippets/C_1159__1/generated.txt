bool is_media_request_allowed_by_default() {
    // Check if requests from internal objects are always allowed
    if (are_internal_requests_allowed()) {
        return true;
    }

    // Define structure to hold device capabilities and policies
    struct DevicePolicy {
        bool has_capability;
        enum Policy { ALWAYS_ALLOW, ALWAYS_DENY, NOT_SET } policy;
    };

    struct DevicePolicy mic_policy = get_microphone_policy();
    struct DevicePolicy cam_policy = get_webcam_policy();

    // Check microphone policy
    if (mic_policy.has_capability) {
        if (mic_policy.policy == ALWAYS_DENY) {
            return false;
        }
    }

    // Check webcam policy
    if (cam_policy.has_capability) {
        if (cam_policy.policy == ALWAYS_DENY) {
            return false;
        }
    }

    // If policy is not set, check if request is from a secure origin
    if (mic_policy.policy == NOT_SET || cam_policy.policy == NOT_SET) {
        if (!is_request_from_secure_origin()) {
            return false;
        }
    }

    // Check content settings for the request's security origin
    if (!are_content_settings_allowed()) {
        return false;
    }

    // If none of the checks deny the request, it is allowed by default
    return true;
}
