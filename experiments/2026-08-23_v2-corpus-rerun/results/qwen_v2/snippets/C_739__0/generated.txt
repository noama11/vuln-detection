int genl_register_family(struct genl_family *family)
{
    int err;
    struct genl_family *existing;

    if (!family || !family->name || !family->version)
        return -EINVAL;

    if (family->maxattr > 0 && !family->ops) {
        family->ops = genl_alloc_ops(family->maxattr);
        if (!family->ops)
            return -ENOMEM;
    }

    if (family->ops) {
        for (int i = 0; i < family->n_ops; i++) {
            if (!family->ops[i].cmd || !family->ops[i].doit)
                return -EINVAL;
        }
    }

    mutex_lock(&genl_mutex);

    existing = genl_find_family(family->name);
    if (existing) {
        mutex_unlock(&genl_mutex);
        return -EEXIST;
    }

    if (family->id == GENL_ID_GENERATE) {
        family->id = genl_get_free_id();
        if (family->id < 0) {
            mutex_unlock(&genl_mutex);
            return family->id;
        }
    }

    list_add(&family->list, &genl_families);

    for (int i = 0; i < family->n_mcgrps; i++) {
        if (family->mcgrps[i].name && family->mcgrps[i].id == 0) {
            family->mcgrps[i].id = genl_get_free_mcgrp_id(family);
            if (family->mcgrps[i].id < 0) {
                err = family->mcgrps[i].id;
                goto err_out;
            }
        }
    }

    genl_notify_family_registered(family);

    mutex_unlock(&genl_mutex);
    return 0;

err_out:
    genl_unregister_family(family);
    mutex_unlock(&genl_mutex);
    return err;
}
