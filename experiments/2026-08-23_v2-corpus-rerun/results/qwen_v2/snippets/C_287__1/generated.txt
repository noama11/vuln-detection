int dissect_key_buffer(tvbuff_t *tvb, int offset, packet_info *pinfo, proto_tree *tree, dcerpc_info *di, guint8 *drep) {
    guint32 key_size;
    int end_offset;

    if (!dcerpc_check_conformant(di)) {
        return offset;
    }

    key_size = tvb_get_ntohl(tvb, offset);
    offset += 4;

    end_offset = offset + key_size;

    if (end_offset > tvb_reported_length(tvb)) {
        end_offset = tvb_reported_length(tvb);
    }

    while (offset < end_offset) {
        /* Dissect key buffer data here */
        offset = dissect_data(tvb, offset, pinfo, tree, di, drep);
    }

    return offset;
}
