int hiddev_ioctl_usage(struct hiddev *hiddev, unsigned int cmd, void __user *user_arg)
{
    int ret = 0;
    struct hiddev_usage_ref *uref = NULL;
    int usage_length;

    switch (cmd) {
    case HIDIOCUSAGE:
        uref = kmalloc(sizeof(*uref), GFP_KERNEL);
        if (!uref)
            return -ENOMEM;

        if (copy_from_user(uref, user_arg, sizeof(*uref))) {
            ret = -EFAULT;
            goto out;
        }

        if (uref->usage_code == 0) {
            ret = -EINVAL;
            goto out;
        }

        usage_length = hiddev->hid->report_length;
        if (uref->usage_index >= usage_length) {
            ret = -EINVAL;
            goto out;
        }

        if (copy_to_user(user_arg, uref, sizeof(*uref))) {
            ret = -EFAULT;
            goto out;
        }

        break;
    default:
        ret = -EINVAL;
        break;
    }

out:
    kfree(uref);
    return ret;
}
