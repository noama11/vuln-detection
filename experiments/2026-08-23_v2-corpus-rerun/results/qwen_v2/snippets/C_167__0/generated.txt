int process_magic_input_event(struct hid_device *hdev, struct hid_report *report, u8 *data, int size)
{
    struct magic_drvdata *drvdata = hid_get_drvdata(hdev);
    int ret = 0;

    if (!drvdata)
        return 0;

    switch (report->id) {
    case TRACKPAD_REPORT_ID:
        if (size < TRACKPAD_MIN_SIZE)
            break;
        ret = process_trackpad_data(drvdata, data, size);
        break;
    case MOUSE_REPORT_ID:
        if (size < MOUSE_MIN_SIZE)
            break;
        ret = process_mouse_data(drvdata, data, size);
        break;
    case DOUBLE_REPORT_ID:
        if (size < DOUBLE_MIN_SIZE)
            break;
        ret = process_double_data(drvdata, data, size);
        break;
    default:
        break;
    }

    if (ret)
        input_sync(drvdata->input_dev);

    return ret ? 1 : 0;
}
