int setsockopt_pppol2tp(struct socket *sock, int level, int optname, char __user *optval, unsigned int optlen) {
    int err = 0;
    struct sock *sk = sock->sk;
    struct pppol2tp_session *session;

    if (level != SOL_PPPOL2TP)
        return -EINVAL;

    if (optlen < sizeof(int))
        return -EINVAL;

    if (!sock->sk->sk_bound_dev_if)
        return -EINVAL;

    if (!sock->sk->sk_state || sock->sk->sk_state != PPPOX_CONNECTED)
        return -ENOTCONN;

    session = pppol2tp_get_session(sk);
    if (!session)
        return -ENOENT;

    if (session->session_id == 0) {
        /* Tunnel options */
        err = pppol2tp_set_tunnel_options(session, optname, optval, optlen);
    } else {
        /* Session options */
        err = pppol2tp_set_session_options(session, optname, optval, optlen);
    }

    return err;
}
