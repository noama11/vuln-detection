ssize_t read_sound_timer(struct file *file, char __user *buffer, size_t count, loff_t *offset) {
    struct sound_timer_data *data = file->private_data;
    ssize_t bytes_read = 0;
    int ret;

    if (!data)
        return -EINVAL;

    spin_lock(&data->lock);

    while (bytes_read < (ssize_t)count) {
        if (kthread_should_stop()) {
            spin_unlock(&data->lock);
            return -ENODEV;
        }

        if (data->disconnected) {
            spin_unlock(&data->lock);
            return -EIO;
        }

        if (data->queue_head == data->queue_tail) {
            if (file->f_flags & O_NONBLOCK) {
                spin_unlock(&data->lock);
                return -EAGAIN;
            }

            ret = wait_event_interruptible(data->read_wait, data->queue_head != data->queue_tail);
            if (ret) {
                spin_unlock(&data->lock);
                return ret;
            }
        }

        if (data->queue_head >= SOUND_TIMER_QUEUE_SIZE) {
            spin_unlock(&data->lock);
            return -EFAULT;
        }

        size_t bytes_to_read = min(count - bytes_read, (size_t)(SOUND_TIMER_QUEUE_SIZE - data->queue_head));
        if (copy_to_user(buffer + bytes_read, data->queue + data->queue_head, bytes_to_read)) {
            spin_unlock(&data->lock);
            return -EFAULT;
        }

        bytes_read += bytes_to_read;
        data->queue_head += bytes_to_read;
        data->queue_head %= SOUND_TIMER_QUEUE_SIZE;
    }

    spin_unlock(&data->lock);
    return bytes_read;
}
