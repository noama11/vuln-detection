void dvb_usb_disconnect(struct usb_interface *intf) {
    struct dvb_usb_device *d = usb_get_intfdata(intf);
    int i;

    if (!d)
        return;

    for (i = 0; i < intf->cur_altsetting->desc.bNumEndpoints; i++)
        usb_free_urb(intf->cur_altsetting->endpoint[i].hcpriv);

    if (d->props.exit)
        d->props.exit(d);

    dvb_usbv2_exit(d);

    dev_info(&intf->dev, "DVB USB device %s successfully de-registered\n", d->name);
    kfree(d->name);
    kfree(d);
}
