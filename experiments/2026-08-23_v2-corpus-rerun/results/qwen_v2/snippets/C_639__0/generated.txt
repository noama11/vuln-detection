RBinJavaAttrInfo *r_bin_java_attr_info_line_number_table_new(ut8 *buffer, ut64 sz, ut64 buf_offset) {
	RBinJavaAttrInfo *attr = R_NEW0(RBinJavaAttrInfo);
	if (!attr) {
		return NULL;
	}

	ut64 line_number_table_length = read_ut16(buffer, buf_offset);
	buf_offset += 2;

	RList *line_number_table = r_list_new();
	if (!line_number_table) {
		free(attr);
		return NULL;
	}

	for (ut64 i = 0; i < line_number_table_length; i++) {
		RBinJavaAttrLine *line = R_NEW0(RBinJavaAttrLine);
		if (!line) {
			r_list_free(line_number_table);
			free(attr);

		}

		line->start_pc = read_ut16(buffer, buf_offset);
		buf_offset += 2;

}
