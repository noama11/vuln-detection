int validate_stream_offset(const cdf_stream_t *sst, const cdf_header_t *h, const void *p, size_t tail, int line) {
    size_t min_size = h->min_size;
    size_t dir_len = sst->dir_len;
    const char *stream_start = (const char *)sst->data;
    const char *stream_end = stream_start + dir_len;
    const char *check_end = (const char *)p + tail;

    if (check_end < stream_start || check_end > stream_end) {
        errno = EFTYPE;
        return -1;
    }

    return 0;
}
