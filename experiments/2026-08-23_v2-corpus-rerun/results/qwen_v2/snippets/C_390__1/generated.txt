ACPI_STATUS AcpiPsCompleteFinalOp(ACPI_WALK_STATE *WalkState, ACPI_PARSE_OBJECT *Op, ACPI_STATUS Status)
{
    ACPI_PARSE_OBJECT *ParentOp;
    ACPI_STATUS Status2;

    if (!WalkState || !Op) {
        return (AE_BAD_PARAMETER);
    }

    /* Check if there are any ascending callbacks to execute */
    if (WalkState->DescendingCallback || WalkState->AscendingCallback) {
        Status2 = WalkState->AscendingCallback(WalkState, Op, &Status);
        if (ACPI_FAILURE(Status2)) {
            return (Status2);
        }
    }

    /* Complete the operation and manage the scope stack */
    ParentOp = AcpiPsPopScope(WalkState);
    if (!ParentOp) {
        return (AE_NULL_OBJECT);
    }

    /* Ensure the parent operation is valid and finalize it */
    if (ParentOp != Op) {
        Status2 = AcpiPsCompleteFinalOp(WalkState, ParentOp, Status);
        if (ACPI_FAILURE(Status2)) {
            return (Status2);
        }
    }

    return (Status);
}
