struct rmap_item *get_next_rmap_item(struct page **page) {
    struct mm_slot *slot;
    struct vm_area_struct *vma;
    struct page *p;

    if (list_empty(&mm_slots_list))
        return NULL;

    slot = list_first_entry(&mm_slots_list, struct mm_slot, mm_list);
    list_del(&slot->mm_list);

    for (vma = slot->mm->mmap; vma; vma = vma->vm_next) {
        if (vma->vm_flags & VM_MERGEABLE) {
            p = find_mergeable_page(vma);
            if (p) {
                *page = p;
                return get_rmap_item(p);
            }
        }
    }

    free_mm_slot(slot);
    return get_next_rmap_item(page);
}
