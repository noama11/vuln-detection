int journal_unmap_buffer(journal_t *journal, struct buffer_head *bh) {
    struct journal_head *jh;
    int ret = 0;

    if (!buffer_journalled(bh))
        return 0;

    jh = journal_grab_journal_head(bh);
    if (!jh)
        return -EINVAL;

    journal_lock_journal(journal);

    if (jh->b_jlist == BJ_None) {
        jbd_zap_buffer(jh);
    } else if (jh->b_transaction == journal->j_running_transaction) {
        jbd_attach_to_commit(jh);
    } else {
        jbd_dispose_buffer(jh);
    }

    clear_bit(BH_JDirty, &bh->b_state);
    journal_unlock_journal(journal);
    journal_release_journal_head(jh);

    return ret;
}
