size_t RegisterJPEGImage(void) {
  static const char *JPEGVersion = "6b";
  static const char *JPEGDescription = "JPEG (Joint Photographic Experts Group)";
  size_t size = 0;

  MagickInfo *magick_info;

  magick_info = SetMagickInfo("JPE");
  magick_info->decoder = JPEGDecode;
  magick_info->encoder = JPEGEcoder;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) JPEGDescription;
  magick_info->version = (char *) JPEGVersion;
  size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("JPEG");
  magick_info->decoder = JPEGDecode;
  magick_info->encoder = JPEGEncoder;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) JPEGDescription;
  magick_info->version = (char *) JPEGVersion;
  size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("JPG");
  magick_info->decoder = JPEGDecode;
  magick_info->encoder = JPEGEncoder;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) JPEGDescription;
  magick_info->version = (char *) JPEGVersion;
  size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("JPS");
  magick_info->decoder = JPEGDecode;
  magick_info->encoder = JPEGEncoder;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) JPEGDescription;
  magick_info->version = (char *) JPEGVersion;
  size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("PJPEG");
  magick_info->decoder = JPEGDecode;
  magick_info->encoder = JPEGEncoder;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) JPEGDescription;
  magick_info->version = (char *) JPEGVersion;
  size = RegisterMagickInfo(magick_info);

  return size;
}
