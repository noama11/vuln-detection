int can_send_credentials(struct ucred *creds) {
    uid_t current_uid = getuid();
    gid_t current_gid = getgid();
    int has_admin_cap = (cap_get_proc() & (1 << CAP_SYS_ADMIN)) != 0;

    if (creds == NULL) {
        return -EINVAL;
    }

    if (creds->uid != current_uid || creds->gid != current_gid) {
        if (!has_admin_cap) {
            return -EPERM;
        }
    }

    return 0;
}
