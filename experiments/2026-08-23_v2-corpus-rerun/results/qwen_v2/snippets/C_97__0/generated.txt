unsigned long kvm_ioapic_read_indirect_register(struct kvm_ioapic *ioapic,
                                               unsigned long addr,
                                               unsigned long length)
{
    unsigned long val = 0;

    switch (ioapic->ioregsel) {
    case KVM_IOAPIC_REG_SELECT_ID:
        val = (ioapic->id << 2);
        break;
    case KVM_IOAPIC_REG_SELECT_VERSION:
        val = ((KVM_IOAPIC_NUM_REGS - 1) & 0xff) |
              (KVM_IOAPIC_REVISION << 8);
        break;
    case KVM_IOAPIC_REG_SELECT_RED_TBL_BASE ... KVM_IOAPIC_REG_SELECT_RED_TBL_END:
        if (ioapic->ioregsel >= KVM_IOAPIC_NUM_REGS)
            break;
        val = ioapic->regs[ioregsel];
        break;
    default:
        break;
    }

    return val;
}
