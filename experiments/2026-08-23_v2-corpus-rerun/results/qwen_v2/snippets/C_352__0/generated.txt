int check_pointer_alignment(struct bpf_verifier_env *env, const struct bpf_reg_state *reg, int off, int size) {
    int ptr_type = reg->type;
    int align = 1;

    switch (ptr_type) {
        case PTR_TO_PACKET:
        case PTR_TO_PACKET_META:
        case PTR_TO_PACKET_END:
            align = 4;
            break;
        case PTR_TO_CTX:
            align = 8;
            break;
        default:
            align = 1;
            break;
    }

    if ((off % align) != 0) {
        return -EINVAL;
    }

    if ((size % align) != 0) {
        return -EINVAL;
    }

    return 0;
}
