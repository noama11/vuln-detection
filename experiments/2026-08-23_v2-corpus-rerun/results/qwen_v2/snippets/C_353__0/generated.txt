int construct_dest_keyring(struct key **_dest_keyring) {
    struct key *dest_keyring = NULL;
    int ret = 0;

    if (*_dest_keyring != NULL) {
        dest_keyring = *(_dest_keyring);
    } else {
        // Attempt to find a suitable default keyring based on current credentials
        dest_keyring = find_default_keyring();
        if (IS_ERR(dest_keyring)) {
            return PTR_ERR(dest_keyring);
        }
    }

    // Perform permission checks
    ret = key_permission(dest_keyring, KEY_NEED_SEARCH);
    if (ret < 0) {
        key_put(dest_keyring);
        return ret;
    }

    // Update the provided pointer
    *_dest_keyring = dest_keyring;

    return 0;
}
