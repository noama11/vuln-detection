size_t RegisterJPEGImage(void) {
    static const char *jpeg_version = "JPEG Library Version 9.0";
    static const MagickInfo *magick_info;
    size_t size = 0;

    magick_info = SetMagickInfo("JPE");
    magick_info->decoder = JPEGDecoder;
    magick_info->encoder = JPEGLossyEncoder;
    magick_info->thread_support = MagickTrue;
    magick_info->mime_type = "image/jpeg";
    magick_info->description = "JPEG image format";
    magick_info->version = jpeg_version;
    RegisterMagickInfo(magick_info);
    size++;

    magick_info = SetMagickInfo("JPEG");
    magick_info->decoder = JPEGDecoder;
    magick_info->encoder = JPEGLossyEncoder;
    magick_info->thread_support = MagickTrue;
    magick_info->mime_type = "image/jpeg";
    magick_info->description = "JPEG image format";
    magick_info->version = jpeg_version;
    RegisterMagickInfo(magick_info);
    size++;

    magick_info = SetMagickInfo("JPG");
    magick_info->decoder = JPEGDecoder;
    magick_info->encoder = JPEGLossyEncoder;
    magick_info->thread_support = MagickTrue;
    magick_info->mime_type = "image/jpeg";
    magick_info->description = "JPEG image format";
    magick_info->version = jpeg_version;
    RegisterMagickInfo(magick_info);
    size++;

    magick_info = SetMagickInfo("JPS");
    magick_info->decoder = JPEGDecoder;
    magick_info->encoder = JPEGLossyEncoder;
    magick_info->thread_support = MagickTrue;
    magick_info->mime_type = "image/jpeg";
    magick_info->description = "JPEG image format";
    magick_info->version = jpeg_version;
    RegisterMagickInfo(magick_info);
    size++;

    magick_info = SetMagickInfo("PJPEG");
    magick_info->decoder = JPEGDecoder;
    magick_info->encoder = JPEGLossyEncoder;
    magick_info->thread_support = MagickTrue;
    magick_info->mime_type = "image/jpeg";
    magick_info->description = "JPEG image format";
    magick_info->version = jpeg_version;
    RegisterMagickInfo(magick_info);
    size++;

    return size;
}
