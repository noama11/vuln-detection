int vc4_prepare_binning(struct drm_device *dev, struct vc4_exec_info *exec)
{
    int ret;
    struct vc4_dev *vc4 = to_vc4_dev(dev);
    struct vc4_bo *bo;
    void *vaddr;
    size_t binning_size, shader_records_size, uniforms_size;
    size_t total_size;
    uint32_t *binning_cmd;
    struct vc4_shader_record *shader_records;
    uint32_t *uniforms;

    binning_size = exec->binning_cmd_size;
    shader_records_size = exec->shader_records_size;
    uniforms_size = exec->uniforms_size;
    total_size = binning_size + shader_records_size + uniforms_size;

    if (total_size > VC4_MAX_BINNING_SIZE)
        return -EINVAL;

    bo = vc4_bo_create(dev, total_size, 0, 0);
    if (IS_ERR(bo))
        return PTR_ERR(bo);

    vaddr = vc4_bo_map(bo);
    if (!vaddr) {
        ret = -ENOMEM;
        goto err_free_bo;
    }

    binning_cmd = vaddr;
    shader_records = vaddr + binning_size;
    uniforms = (uint32_t *)((void *)shader_records + shader_records_size);

    if (copy_from_user(binning_cmd, exec->binning_cmd, binning_size) ||
        copy_from_user(shader_records, exec->shader_records, shader_records_size) ||
        copy_from_user(uniforms, exec->uniforms, uniforms_size)) {
        ret = -EFAULT;
        goto err_unmap_bo;
    }

    ret = vc4_validate_binning(dev, binning_cmd, binning_size,
                               shader_records, shader_records_size,
                               uniforms, uniforms_size);
    if (ret)
        goto err_unmap_bo;

    ret = vc4_wait_for_rendering(dev, exec);
    if (ret)
        goto err_unmap_bo;

    exec->binning_bo = bo;
    exec->binning_cmd = binning_cmd;
    exec->shader_records = shader_records;
    exec->uniforms = uniforms;

    return 0;

err_unmap_bo:
    vc4_bo_unmap(bo);
err_free_bo:
    vc4_bo_unref(bo);
    return ret;
}
