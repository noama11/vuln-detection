int restore_fpu_checking(struct task_struct *tsk) {
    int result;

    if (cpu_has_fpu_feature) {
        __asm__ __volatile__(
            "fninit\n"
            "fldz\n"
            "fld1\n"
            "fldpi\n"
            "fldl2e\n"
            "fldl2t\n"
        );
    }

    result = perform_fpu_restore_check(tsk);

    return result;
}
