int yam_ioctl(struct net_device *dev, struct ifreq *ifr, int cmd)
{
    int err = 0;
    void __user *user_data = ifr->ifr_data;

    if (!dev || !ifr || !netif_running(dev)) {
        return -EINVAL;
    }

    switch (cmd) {
    case YAM_IOCTL_SET_BITRATE:
    {
        unsigned int bitrate;
        if (copy_from_user(&bitrate, user_data, sizeof(bitrate)))
            return -EFAULT;

        if (netif_running(dev)) {
            return -EBUSY;
        }

        dev->bitrate = bitrate;
        break;
    }

    case YAM_IOCTL_GET_BITRATE:
    {
        unsigned int bitrate = dev->bitrate;
        if (copy_to_user(user_data, &bitrate, sizeof(bitrate)))
            return -EFAULT;
        break;
    }

    case YAM_IOCTL_SET_BAUDRATE:
    {
        unsigned int baudrate;
        if (copy_from_user(&baudrate, user_data, sizeof(baudrate)))
            return -EFAULT;

        if (netif_running(dev)) {
            return -EBUSY;
        }

        dev->baudrate = baudrate;
        break;
    }

    case YAM_IOCTL_GET_BAUDRATE:
    {
        unsigned int baudrate = dev->baudrate;
        if (copy_to_user(user_data, &baudrate, sizeof(baudrate)))
            return -EFAULT;
        break;
    }

    case YAM_IOCTL_SET_IRQ:
    {
        unsigned int irq;
        if (copy_from_user(&irq, user_data, sizeof(irq)))
            return -EFAULT;

        if (netif_running(dev)) {
            return -EBUSY;
        }

        dev->irq = irq;
        break;
    }

    case YAM_IOCTL_GET_IRQ:
    {
        unsigned int irq = dev->irq;
        if (copy_to_user(user_data, &irq, sizeof(irq)))
            return -EFAULT;
        break;
    }

    default:
        err = -EINVAL;
        break;
    }

    return err;
}
