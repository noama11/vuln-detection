struct mobj *allocate_user_ta_memory(size_t size) {
    size_t granularity = sysconf(_SC_PAGESIZE); // Assuming standard allocation granularity is page size
    size_t aligned_size = ((size + granularity - 1) / granularity) * granularity;

    struct mobj *mobj = NULL;

    if (config_paged_user_ta_enabled()) {
        mobj = mobj_paged_alloc(aligned_size);
    } else {
        mobj = mobj_mm_alloc(aligned_size, 0);
    }

    if (mobj != NULL) {
        memset(mobj, 0, aligned_size);
    }

    return mobj;
}
