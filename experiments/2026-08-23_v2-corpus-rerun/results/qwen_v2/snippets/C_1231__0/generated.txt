#include <v8.h>

using namespace v8;

void createDOMException(Isolate* isolate, int errorCode, const char* message, Local<Context> context) {
  Local<Value> exception;
  switch (errorCode) {
    case 1:
      exception = Exception::Error(String::NewFromOneByte(isolate, (const uint8_t*)message));
      break;
    case 2:
      exception = Exception::TypeError(String::NewFromOneByte(isolate, (const uint8_t*)message));
      break;
    case 3:
      exception = Exception::RangeError(String::NewFromOneByte(isolate, (const uint8_t*)message));
      break;
    case 4:
      exception = Exception::SyntaxError(String::NewFromOneByte(isolate, (const uint8_t*)message));
      break;
    case 5:
      exception = Exception::ReferenceError(String::NewFromOneByte(isolate, (const uint8_t*)message));
      break;
    default:
      exception = Exception::Error(String::NewFromOneByte(isolate, (const uint8_t*)"Unknown DOM Exception"));
  }

  Local<Object> error = exception.As<Object>();
  Local<String> stack = String::NewFromOneByte(isolate, (const uint8_t*)"Stack trace goes here");
  error->Set(context, String::NewFromOneByte(isolate, (const uint8_t*)"stack"), stack).Check();
}

void throwException(Isolate* isolate, Local<Value> exception) {
  if (!isolate->InContext() || isolate->IsExecutionTerminating()) {
    return;
  }

  Local<Context> context = isolate->GetCurrentContext();
  Local<Value> result;
  Local<Function> uncaughtExceptionHandler;
  Local<Value> args[1] = { exception };

  if (uncaughtExceptionHandler.IsEmpty()) {
    context->ThrowException(exception);
  } else {
    uncaughtExceptionHandler->Call(context, context->Global(), 1, args).ToLocal(&result);
  }
}
