CURLcode escape_eob_smtp(struct connectdata *conn, ssize_t nread) {
  char *scratch = NULL;
  char *src, *dst;
  size_t i, j;
  size_t eob_escape_count = 0;
  size_t len = (size_t)nread;
  CURLcode result = CURLE_OK;

  scratch = malloc(len + 2); /* +2 for possible trailing CRLF */
  if (!scratch) {
    return CURLE_OUT_OF_MEMORY;
  }

  src = conn->upload.upload_buffer;
  dst = scratch;

  for (i = 0, j = 0; i < len; i++) {
    if (i + 1 < len && src[i] == '\r' && src[i + 1] == '\n') {
      *dst++ = src[i];
      *dst++ = src[i + 1];
      i++;
      eob_escape_count++;
    } else {
      *dst++ = src[i];
    }
  }

  if (eob_escape_count > 0) {
    /* Free previous buffer and update upload buffer */
    free(conn->upload.upload_buffer);
    conn->upload.upload_buffer = scratch;
    conn->upload.ulbuf = scratch;
    conn->upload.size = (size_t)(dst - scratch);
    conn->upload.ndone = 0;
  } else {
    free(scratch);
  }

  return result;
}
