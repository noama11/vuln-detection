int ext4_orphan_del(handle_t *handle, struct inode *inode) {
    struct ext4_inode_info *ei = EXT4_I(inode);
    struct ext4_sb_info *sbi = EXT4_SB(inode->i_sb);
    struct ext4_orphan *ohead = &sbi->s_orphan;
    struct ext4_orphan *op;
    int err = 0;

    if (!ext4_journal_available(inode->i_sb))
        return 0;

    mutex_lock(&ohead->o_mutex);

    list_for_each_entry(op, &ohead->o_list, o_list) {
        if (op->o_ino == inode->i_ino) {
            list_del(&op->o_list);
            kfree(op);
            break;
        }
    }

    if (handle) {
        err = ext4_journal_stop(handle);
        if (err < 0)
            return err;
    }

    mutex_unlock(&ohead->o_mutex);
    return 0;
}
