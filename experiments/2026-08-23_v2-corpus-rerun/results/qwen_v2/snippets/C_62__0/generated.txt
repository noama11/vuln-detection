int validate_perf_event(struct pmu_hw_events *hw_events, struct perf_event *event) {
    if (event->pmu == NULL)
        return 1;

    if (event->pmu != event->group_leader->pmu)
        return 0;

    if (event->state < PERF_EVENT_STATE_OFF)
        return 0;

    if (event->state == PERF_EVENT_STATE_OFF && !(event->hw.config_base & ARM_PMU_CONFIG_EXEC))
        return 1;

    int idx = arm_pmu_get_event_idx(hw_events, event);
    return (idx >= 0) ? 1 : 0;
}
