void adjust_bpf_jumps(struct bpf_prog *prog, int pos, int delta) {
    for (int i = 0; i < prog->len; i++) {
        struct bpf_insn *insn = &prog->insnsi[i];
        if ((insn->code & BPF_CLASS_MASK) == BPF_JMP) {
            if ((insn->code & BPF_OP_MASK) != BPF_JA && (insn->code & BPF_OP_MASK) != BPF_JMP32) {
                int target = insn->off;
                if (target >= pos) {
                    insn->off += delta;
                }
            }
        }
    }
}
