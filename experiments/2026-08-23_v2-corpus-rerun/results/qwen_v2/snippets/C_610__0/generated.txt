int xfs_attr_shortform_add(xfs_da_args_t *args) {
    int error;

    trace_xfs_attr_add(args->dp, args->name, args->val, args->namelen, args->valuelen, _RET_IP_);

    if ((args->flags & ATTR_REPLACE) && !(args->flags & ATTR_EXISTS)) {
        return -ENOATTR;
    }

    if ((args->flags & ATTR_CREATE) && (args->flags & ATTR_EXISTS)) {
        return -EEXIST;
    }

    if (args->namelen > XFS_ATTR_MAX_NAMELEN || args->valuelen > XFS_ATTR_MAX_VALUELEN) {
        return -EINVAL;
    }

    int new_size = xfs_attr_leaf_new_size(args->namelen, args->valuelen);
    if (new_size > XFS_ATTR_MAX_SHORTFORM_SIZE) {
        return -ENOSPC;
    }

    if (args->flags & ATTR_EXISTS) {
        error = xfs_attr_remove(args);
        if (error) {
            return error;
        }
    }

    return xfs_attr_leaf_add(args);
}
