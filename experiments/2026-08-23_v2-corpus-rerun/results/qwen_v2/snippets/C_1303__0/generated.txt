void SendWebUIMessage(gin::Arguments* args) {
  // Implementation logic follows the specification
  if (!ShouldRespondToRequest()) {
    return;
  }

  std::string message = GetMessageFromArguments(args);
  if (messageRequiresUserGesture(message) && !UserGestureDetected()) {
    LogError("User gesture required but not detected for message: " + message);
    return;
  }

  base::ListValue* content = nullptr;
  if (args->HasNext()) {
    v8::Local<v8::Value> contentValue;
    args->GetNext(&contentValue);
    content = ConvertV8ObjectToListValue(contentValue);
  }

  RenderFrame::Send(message.c_str(), content);
}
