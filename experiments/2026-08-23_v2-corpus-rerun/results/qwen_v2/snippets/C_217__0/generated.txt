void vhost_scsi_send_event(struct vhost_scsi *vs, struct vhost_scsi_tpg *tpg,
                           struct se_lun *lun, u32 event, u32 reason)
{
    struct vhost_scsi_event *e;

    e = kzalloc(sizeof(*e), GFP_KERNEL);
    if (!e)
        return;

    e->vs = vs;
    e->tpg = tpg;
    e->lun = lun;
    e->event = event;
    e->reason = reason;

    if (tpg) {
        e->tpg_id = tpg->tpg_id;
        if (lun)
            e->lun_id = lun->lun;
    }

    list_add_tail(&e->list, &vs->event_list);
    queue_work(vs->event_wq, &vs->event_work);
}
