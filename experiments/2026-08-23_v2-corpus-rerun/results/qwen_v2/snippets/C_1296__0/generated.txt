void ShutdownTaskQueue(const scoped_refptr<MainThreadTaskQueue>& task_queue) {
  if (!main_thread_shutdown) {
    if (throttler_available) {
      throttler_shutdown(task_queue);
    }

    if (task_runners_erase(task_queue)) {
      if (task_queue_class == TIMER) {
        remove_timer_task_observer(task_queue->cost_estimator);
      } else if (task_queue_class == LOADING) {
        remove_loading_task_observer(task_queue->cost_estimator);
      }
    }
  }
}
