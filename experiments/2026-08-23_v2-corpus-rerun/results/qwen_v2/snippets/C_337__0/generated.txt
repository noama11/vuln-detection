int kvm_update_pit_state(struct kvm *kvm, struct kvm_pit_state2 *ps)
{
    int i;

    mutex_lock(&kvm->pit_state->mutex);
    kvm->pit_state->hpet = ps->hpet;
    for (i = 0; i < 3; i++) {
        kvm->pit_state->channels[i].state = ps->channels[i].state;
        kvm->pit_state->channels[i].count = ps->channels[i].count;
        kvm->pit_state->channels[i].init_count = ps->channels[i].init_count;
        kvm->pit_state->channels[i].mode = ps->channels[i].mode;
        kvm->pit_state->channels[i].latched_count = ps->channels[i].latched_count;
        kvm->pit_state->channels[i].latched_flag = ps->channels[i].latched_flag;
        kvm->pit_state->channels[i].count_load_time = ps->channels[i].count_load_time;
        kvm->pit_state->channels[i].count_load_val = ps->channels[i].count_load_val;
    }
    mutex_unlock(&kvm->pit_state->mutex);

    return 0;
}
