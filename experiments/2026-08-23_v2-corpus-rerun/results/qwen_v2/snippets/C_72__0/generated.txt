int l2tp_ip6_recvmsg(struct kiocb *iocb, struct sock *sk, struct msghdr *msg,
                     size_t len, int noblock, int flags, int *addr_len)
{
    struct sk_buff *skb;
    int copied, err, peeked = 0;
    struct ipv6_pinfo *np = inet6_sk(sk);
    struct in6_addr *saddr, *daddr;
    struct sockaddr_in6 *sin6;
    int ret = 0;

    if (flags & MSG_OOB)
        return -EOPNOTSUPP;

    skb = skb_recv_datagram(sk, flags, noblock, &err);
    if (!skb)
        return err;

    copied = skb->len;
    if (len < copied) {
        if (!(flags & MSG_TRUNC))
            copied = len;
    }

    if (flags & MSG_TRUNC)
        copied = skb->len;

    if (iov_iter_count(&msg->msg_iter))
        ret = skb_copy_datagram_iovec(skb, 0, msg->msg_iter.iov, 0, copied);

    if (!ret) {
        struct sock_ext_ack *sack = inet6_sk(sk)->sk_extack;
        if (sack && sack->eol_ts)
            sock_skb_set_timestamp(msg->msg_iter.sock, skb);
    }

    if (addr_len)
        *addr_len = sizeof(struct sockaddr_in6);

    sin6 = (struct sockaddr_in6 *)msg->msg_name;
    if (sin6) {
        saddr = &skb->sk->sk_v6_rcv_saddr;
        daddr = &skb->sk->sk_v6_daddr;
        sin6->sin6_family = AF_INET6;
        sin6->sin6_port = 0;
        sin6->sin6_flowinfo = 0;
        sin6->sin6_scope_id = 0;
        ipv6_addr_copy(&sin6->sin6_addr, saddr);
    }

    skb_free_datagram(sk, skb);
    return copied;
}
