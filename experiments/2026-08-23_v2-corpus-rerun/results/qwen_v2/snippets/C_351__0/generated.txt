int check_stack_boundary_access(struct bpf_verifier_env *env, int regno, int access_size,
                                bool zero_size_allowed, struct bpf_call_arg_meta *meta)
{
    struct bpf_reg_state *reg = &env->cur_state->regs[regno];

    if (reg->type != PTR_TO_STACK)
        return -EACCES;

    if (access_size == 0) {
        if (zero_size_allowed)
            return 0;
        return -EACCES;
    }

    int off = reg->off + reg->var_off.value;
    int max_access = reg->var_off.max_value + access_size;

    if (off < 0 || off + access_size > MAX_BPF_STACK)
        return -EACCES;

    if (meta)
        meta->stack_depth = max(meta->stack_depth, off + access_size);

    for (int i = 0; i < access_size; i++) {
        int slot = -(off + i) / BPF_WORD_SIZE;
        if (slot < 0 || slot >= MAX_BPF_STACK / BPF_WORD_SIZE)
            return -EACCES;
    }

    return 0;
}
