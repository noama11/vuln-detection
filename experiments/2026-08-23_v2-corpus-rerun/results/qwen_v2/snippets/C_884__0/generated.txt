#include <linux/platform_device.h>
#include <linux/gpio/driver.h>
#include <linux/io.h>
#include <linux/irq.h>
#include <linux/err.h>
#include <linux/slab.h>
#include <linux/device.h>
#include <linux/module.h>

int amd_gpio_init(struct platform_device *pdev)
{
    struct gpio_chip *chip;
    struct resource *mem;
    void __iomem *regs;
    int irq;
    int ret;

    chip = devm_kzalloc(&pdev->dev, sizeof(*chip), GFP_KERNEL);
    if (!chip)
        return -ENOMEM;

    mutex_init(&chip->bgpio_lock);

    mem = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    if (!mem)
        return -ENODEV;

    regs = devm_ioremap_resource(&pdev->dev, mem);
    if (IS_ERR(regs))
        return PTR_ERR(regs);

    irq = platform_get_irq(pdev, 0);
    if (irq < 0)
        return irq;

    chip->label = "AMD_GPIO";
    chip->owner = THIS_MODULE;
    chip->get = amd_gpio_get;
    chip->set = amd_gpio_set;
    chip->direction_input = amd_gpio_dir_in;
    chip->direction_output = amd_gpio_dir_out;
    chip->read = amd_gpio_read;
    chip->write = amd_gpio_write;
    chip->debounce_set = amd_gpio_set_debounce;
    chip->base = -1;
    chip->ngpio = 32; /* Example value, adjust as needed */
    chip->can_sleep = false;
    chip->regs = regs;

    ret = devm_gpiochip_add_data(&pdev->dev, chip, NULL);
    if (ret) {
        dev_err(&pdev->dev, "Failed to add GPIO chip\n");
        return ret;
    }

    ret = devm_request_irq(&pdev->dev, irq, amd_gpio_irq_handler, 0,
                             dev_name(&pdev->dev), chip);
    if (ret) {
        dev_err(&pdev->dev, "Failed to request IRQ\n");
        return ret;
    }

    dev_info(&pdev->dev, "AMD GPIO driver loaded\n");
    return 0;
}
