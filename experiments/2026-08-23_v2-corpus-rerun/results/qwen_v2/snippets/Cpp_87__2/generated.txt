void parse_pdf_object(struct pdf_struct *pdf, struct pdf_obj *obj) {
    if (!pdf || !obj) {
        return;
    }

    size_t obj_size = obj->size;
    if (obj_size == 0) {
        return;
    }

    char *dict_start = obj->data;
    char *dict_end = obj->data + obj_size;

    while (dict_start < dict_end) {
        if (*dict_start == '/') {
            char *name_end = std::find(dict_start, dict_end, ' ');
            if (name_end != dict_end) {
                *name_end = '\0';
                std::string name(dict_start + 1);
                std::cout << "Found PDF name: " << name << std::endl;

                if (name == "Linearized") {
                    std::cout << "Handling linearized PDF..." << std::endl;
                    // Handle linearized PDF logic
                } else if (name == "Launch") {
                    std::cout << "Handling launch action..." << std::endl;
                    // Handle launch action logic
                } else if (name == "JS") {
                    std::cout << "Handling JavaScript reference..." << std::endl;
                    // Handle JavaScript reference logic
                } else if (name == "F") {
                    std::cout << "Filter detected. Updating flags..." << std::endl;
                    obj->flags |= PDF_OBJ_FILTERED;
                }
            }
        }

        dict_start++;
    }

    std::cout << "PDF object parsing complete." << std::endl;
}
