int handle_ioctl(unsigned long cmd, unsigned long arg, struct fb_info *info, int type, int fb_depth, unsigned long fb_size) {
    int ret = 0;
    void __user *argp = (void __user *)arg;

    switch (cmd) {
        case FBIOGET_FSCREENINFO: {
            struct fb_fix_screeninfo finfo;
            memset(&finfo, 0, sizeof(finfo));
            finfo.type = type;
            finfo.type_aux = 0;
            finfo.visual = fb_depth;
            finfo.xpanstep = 0;
            finfo.ypanstep = 0;
            finfo.ywrapstep = 0;
            finfo.accel = 0;
            finfo.size = fb_size;
            if (copy_to_user(argp, &finfo, sizeof(finfo)))
                ret = -EFAULT;
            break;
        }
        case FBIOGET_CMAP: {
            struct fb_cmap cmap;
            if (copy_from_user(&cmap, argp, sizeof(cmap)))
                ret = -EFAULT;
            if (cmap.len > 256)
                ret = -EINVAL;
            break;
        }
        case FBIOSET_CMAP: {
            struct fb_cmap cmap;
            if (copy_from_user(&cmap, argp, sizeof(cmap)))
                ret = -EFAULT;
            if (cmap.len > 256)
                ret = -EINVAL;
            break;
        }
        default:
            ret = -EINVAL;
            break;
    }

    return ret;
}
