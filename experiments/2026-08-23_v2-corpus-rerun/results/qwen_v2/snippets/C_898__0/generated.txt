int decode_vop_header(Mpeg4DecContext *ctx, GetBitContext *gb) {
    if (get_bits_left(gb) < 0)
        return 0;

    ctx->frame_type = get_bits1(gb);
    ctx->dct_type = get_bits1(gb);

    skip_bits(gb, 3); // temporal reference

    int picture_type = get_bits(gb, 2);
    if (picture_type == 0) {
        ctx->predictor_reset = 1;
        init_scanning_tables(ctx);
    }

    load_default_matrices(ctx);

    skip_bits(gb, 8); // user data

    return 0;
}
