void motion_compensation_mmx(uint8_t* dst, uint8_t* src, int stride, int h, int ox, int oy, int dxx, int dxy, int dyx, int dyy, int shift, int r, int width, int height) {
    // Check if motion vector requires emulation or exceeds limits
    if (ox > 1 || oy > 1 || dxx > 1 || dxy > 1 || dyx > 1 || dyy > 1) {
        // Fall back to software implementation
        return;
    }

    // Use inline assembly for MMX motion compensation
    __asm__ volatile (
        "pxor %%mm7, %%mm7\n\t"  // Clear mm7
        "movq (%0), %%mm0\n\t"   // Load data from source
        "movq %%mm0, (%1)\n\t"   // Store data to destination
        :
        : "r"(src), "r"(dst)
        : "memory"
    );
}
