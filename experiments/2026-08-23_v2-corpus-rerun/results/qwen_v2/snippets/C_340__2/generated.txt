int restore_xstate(struct xsave_struct *fx, u64 mask) {
    int ret = 0;
    u64 xcr0;

    __asm__ __volatile__(
        "1: "
        "xrstors %1, %0\n"
        "2:\n"
        ".section .fixup,\"ax\"\n"
        "3: "
        "xor %%rax, %%rax\n"
        "jmp 2b\n"
        ".previous\n"
        ".pushsection __ex_table,\"a\"\n"
        "  .align 8\n"
        "  .quad 1b,3b\n"
        ".popsection"
        : "+r"(mask), "+m"(*fx)
        : : "memory");

    return ret;
}
