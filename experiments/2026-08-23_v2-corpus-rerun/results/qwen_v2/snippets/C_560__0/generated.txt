static void RemoveICCProfileFromResourceBlock(StringInfo *icc_profile)
{
  size_t length;
  unsigned char *p;
  unsigned char *q;
  unsigned int id;

  if (icc_profile == (StringInfo *) NULL)
    return;
  length=GetStringInfoLength(icc_profile);
  p=(unsigned char *) GetStringInfoBody(icc_profile);
  q=p;
  while ((size_t) (p - q) < length)
  {
    id=ReadBlobMSBLong((const unsigned char *) p);
    p+=4;
    if (id == 0x0000040f)
      break;
    p+=ReadBlobMSBLong((const unsigned char *) p);
  }
  if ((size_t) (p - q) < length)
  {
    size_t offset;

    offset=(size_t) (p - q);
    (void) CopyMemory(q,p,length - (p - q));
    SetStringInfoLength(icc_profile,length - (p - q));
  }
}
