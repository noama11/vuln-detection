int wan_ioctl(struct net_device *dev, struct ifreq *ifr, int cmd)
{
    struct sync_serial_settings __user *user_settings;
    struct sync_serial_settings settings;
    int ret;

    if (cmd != SIOCWANDEV)
        return hdlc_ioctl(dev, ifr, cmd);

    switch (cmd) {
    case IF_GET_IFACE:
        if (!capable(CAP_NET_ADMIN))
            return -EPERM;

        user_settings = (struct sync_serial_settings __user *)ifr->ifr_ifsu.ifsu_data;
        if (copy_from_user(&settings, user_settings, sizeof(struct sync_serial_settings)))
            return -EFAULT;

        if (settings.size != sizeof(struct sync_serial_settings)) {
            if (copy_to_user(&settings.size, &ifr->ifr_ifsu.ifsu_data, sizeof(int)))
                return -EFAULT;
            return -ENOBUFS;
        }

        settings = get_current_settings(dev); // Assume this helper exists

        if (copy_to_user(user_settings, &settings, sizeof(struct sync_serial_settings)))
            return -EFAULT;
        return 0;

    case IF_IFACE_SYNC_SERIAL:
        if (!capable(CAP_NET_ADMIN))
            return -EPERM;

        if (dev->flags & IFF_UP)
            return -EBUSY;

        if (copy_from_user(&settings, ifr->ifr_ifsu.ifsu_data, sizeof(settings)))
            return -EFAULT;

        if (settings.size != sizeof(settings))
            return -EINVAL;

        if (settings.clocking != CLOCK_INT && settings.clocking != CLOCK_EXT)
            return -EINVAL;

        update_clocking_status(dev, settings.clocking); // Assume this helper exists
        return 0;

    default:
        return hdlc_ioctl(dev, ifr, cmd);
    }
}
