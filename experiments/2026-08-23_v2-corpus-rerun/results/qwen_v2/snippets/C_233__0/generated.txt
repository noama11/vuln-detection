int
readdir_device_volume(struct vnode *dvp, struct uio *uiop, struct cred *cred,
    int *eofp, caller_context_t *ct_unused, int flags_unused)
{
    int error = 0;
    struct dirent *dent;
    off_t *offset;
    char *name;
    size_t namelen;

    /* Check if the directory is a ZVOL_DIR */
    if (strcmp(vnode_getpath(dvp), ZVOL_DIR) == 0) {
        /* Handle special directory creation if needed */
        if (create_special_directories() != 0) {
            return (EIO);
        }

        /* Acquire read lock for thread safety */
        read_lock_acquire();

        /* Iterate over datasets and populate directory entries */
        while (next_dataset(&name, &namelen) == 0) {
            dent = (struct dirent *)malloc(sizeof(struct dirent) + namelen);
            if (dent == NULL) {
                read_lock_release();
                return (ENOMEM);
            }

            dent->d_ino = get_dataset_inode(name);
            dent->d_type = DT_DIR;
            dent->d_namlen = namelen;
            strcpy(dent->d_name, name);

            if ((error = uiomove(dent, sizeof(struct dirent) + namelen, uiop)) != 0) {
                free(dent);
                read_lock_release();
                return (error);
            }

            free(dent);
        }

        read_lock_release();
    }

    *eofp = 1;
    return (error);
}
