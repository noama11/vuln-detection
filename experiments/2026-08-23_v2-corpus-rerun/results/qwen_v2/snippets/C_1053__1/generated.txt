NativeBrowserFrame* create_native_browser_frame(BrowserFrame* browser_frame, BrowserView* browser_view) {
    if (is_using_pure_views_framework() && default_parent_view_available()) {
        return create_browser_frame_views(browser_frame, browser_view);
    } else {
        return create_browser_frame_gtk(browser_frame, browser_view);
    }
}
