MagickBooleanType WriteImages(const ImageInfo *image_info, Image *images,
                              const char *filename, ExceptionInfo *exception) {
  assert(image_info != (ImageInfo *) NULL);
  assert(image_info->signature == MagickSignature);
  assert(images != (Image *) NULL);
  assert(images->signature == MagickSignature);
  assert(exception != (ExceptionInfo *) NULL);
  assert(exception->signature == MagickSignature);

  Image
    *image;

  ImageInfo
    *clone_info;

  MagickBooleanType
    status;

  register ssize_t
    i;

  size_t
    scene;

  clone_info = CloneImageInfo(image_info);
  if (clone_info == (ImageInfo *) NULL)
    return (MagickBooleanType) MagickFalse;

  status = MagickTrue;
  if (images->adjoin != MagickFalse) {
    for (i = 0; i < (ssize_t) images->number_images; i++) {
      char
        *temp_filename;

      temp_filename = GetImageFileName(images, i, filename);
      if (temp_filename == (char *) NULL)
        status = MagickFalse;
      else {
        clone_info->filename = temp_filename;
        image = CloneImage(images, images->columns, images->rows, MagickFalse,
                           exception);
        if (image == (Image *) NULL)
          status = MagickFalse;
        else {
          image->scene = i + 1;
          status = WriteImage(clone_info, image, exception);
          if (status == MagickFalse)
            break;
        }
        if (temp_filename != (char *) NULL)
          MagickFreeMemory(temp_filename);
      }
    }
  }
  else {
    clone_info->filename = (char *) filename;
    status = WriteImage(clone_info, images, exception);
  }

  clone_info->filename = (char *) NULL;
  clone_info->scene = 0;
  DestroyImageInfo(clone_info);
  return status;
}
