FLAC__StreamDecoderWriteStatus write_callback(const FLAC__StreamDecoder *decoder, const FLAC__Frame *frame, const int32_t * const buffer[], void *client_data) {
    struct private_data *data = (struct private_data *)client_data;
    data->frame = frame;
    data->buffer = buffer;
    data->position = frame->header.number.sample_number;

    copy_buffer(data);

    return FLAC__STREAM_DECODER_WRITE_STATUS_CONTINUE;
}
