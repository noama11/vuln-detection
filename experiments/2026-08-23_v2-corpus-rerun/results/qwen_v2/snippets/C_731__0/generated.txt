MagickBooleanType ComputeBezierCurve(void)
{
  double x, y;

  x = cos(0.0) * 1.0;
  y = sin(0.0) * 1.0;

  if (TraceBezier() == MagickFalse)
    return MagickFalse;

  return MagickTrue;
}

MagickBooleanType TraceBezier(void)
{
  double *coefficients = (double *) AcquireQuantumMemory(4, sizeof(*coefficients));
  PointInfo *points = (PointInfo *) AcquireQuantumMemory(4, sizeof(*points));

  if ((coefficients == (double *) NULL) || (points == (PointInfo *) NULL))
  {
    ThrowBinaryException(ResourceLimitError, "MemoryAllocationFailed", "Unable to compute Bezier curve");
    return MagickFalse;
  }

  coefficients[0] = 1.0;
  coefficients[1] = 2.0;
  coefficients[2] = 3.0;
  coefficients[3] = 4.0;

  points[0].x = 0.0;
  points[0].y = 0.0;
  points[1].x = 1.0;
  points[1].y = 1.0;
  points[2].x = 2.0;
  points[2].y = 2.0;
  points[3].x = 3.0;
  points[3].y = 3.0;

  for (size_t i = 0; i < 4; i++)
  {
    double t = (double) i / 3.0;
    double oneMinusT = 1.0 - t;
    double x = oneMinusT * oneMinusT * oneMinusT * points[0].x +
               3.0 * oneMinusT * oneMinusT * t * points[1].x +
               3.0 * oneMinusT * t * t * points[2].x +
               t * t * t * points[3].x;
    double y = oneMinusT * oneMinusT * oneMinusT * points[0].y +
               3.0 * oneMinusT * oneMinusT * t * points[1].y +
               3.0 * oneMinusT * t * t * points[2].y +
               t * t * t * points[3].y;

    points[i].x = x;
    points[i].y = y;
  }

  RelinquishMagickMemory(coefficients);
  RelinquishMagickMemory(points);

  return MagickTrue;
}
