int read_index_entry(git_index_entry **out, size_t *out_size, git_index *index, const void *buffer, size_t buffer_size, const char *last) {
    if (buffer_size < sizeof(git_index_entry)) {
        return -1;
    }

    git_index_entry *entry = (git_index_entry *)malloc(sizeof(git_index_entry));
    if (!entry) {
        return -1;
    }

    memcpy(entry, buffer, sizeof(git_index_entry));
    *out = entry;
    *out_size = sizeof(git_index_entry);

    size_t path_len = 0;
    const char *path = git_index_entry_path(entry, &path_len);

    if (!path) {
        path = last;
        size_t full_path_len = path_len + strlen(last) + 1;
        char *full_path = (char *)malloc(full_path_len);
        if (!full_path) {
            free(entry);
            return -1;
        }

        snprintf(full_path, full_path_len, "%s", last);
        path = full_path;
        entry->path = path;
    }

    return 0;
}
