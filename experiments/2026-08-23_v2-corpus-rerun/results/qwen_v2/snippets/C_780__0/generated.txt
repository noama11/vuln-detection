int hidp_ioctl(struct socket *sock, unsigned int cmd, void __user *argp)
{
    int ret = 0;
    struct hidp_connadd conn_add;
    struct hidp_conndel conn_del;
    struct hidp_connlist conn_list;
    struct hidp_conninfo conn_info;

    if (!capable(CAP_NET_ADMIN))
        return -EPERM;

    switch (cmd) {
    case HIDPADDCONN:
        if (copy_from_user(&conn_add, argp, sizeof(conn_add)))
            return -EFAULT;
        ret = hidp_add_connection(sock, &conn_add);
        break;
    case HIDPDELCONN:
        if (copy_from_user(&conn_del, argp, sizeof(conn_del)))
            return -EFAULT;
        ret = hidp_del_connection(sock, &conn_del);
        break;
    case HIDPGETCONNLIST:
        if (copy_from_user(&conn_list, argp, sizeof(conn_list)))
            return -EFAULT;
        ret = hidp_get_connlist(sock, &conn_list);
        break;
    case HIDPGETCONNINFO:
        if (copy_from_user(&conn_info, argp, sizeof(conn_info)))
            return -EFAULT;
        ret = hidp_get_conninfo(sock, &conn_info);
        if (copy_to_user(argp, &conn_info, sizeof(conn_info)))
            return -EFAULT;
        break;
    default:
        ret = -EINVAL;
        break;
    }

    return ret;
}
