MagickBooleanType SkipMipmapImages(Image *image, DDSInfo *dds_info, int pixel_size, ExceptionInfo *exception) {
    MagickBooleanType status = MagickTrue;
    size_t width = image->columns;
    size_t height = image->rows;
    size_t skip_size;

    if ((dds_info->mipmap_count > 1) && (dds_info->is_texture || dds_info->is_cube_map)) {
        while (width > 1 || height > 1) {
            width = (width > 1) ? width / 2 : 1;
            height = (height > 1) ? height / 2 : 1;

            skip_size = width * height * pixel_size;

            if (SeekBlob(image, skip_size, SEEK_CUR) == (MagickOffsetType)(-1)) {
                (void) ThrowMagickException(exception, GetMagickModule(), CoderError, "EndOfFile", "Unable to skip mipmap data");
                status = MagickFalse;
                break;
            }
        }
    }

    return status;
}
