void add_suggestion_to_scene(UiScene* scene, UiBrowserInterface* browser, Model* model, SuggestionBinding* element_binding) {
    // Create UI components
    UiElement* icon = ui_element_create_icon(model, "icon_property");
    UiElement* content_text = ui_element_create_text(model, "content_property");
    UiElement* description_text = ui_element_create_text(model, "description_property");

    // Create layout and add components
    UiElement* layout = ui_element_create_layout(UI_LAYOUT_VERTICAL);
    ui_element_add_child(layout, icon);
    ui_element_add_child(layout, content_text);
    ui_element_add_child(layout, description_text);

    // Create background button and bind navigation
    UiElement* background_button = ui_element_create_button();
    ui_element_set_on_click(background_button, (void*)UiBrowserInterfaceNavigate, browser);
    ui_element_bind_color(background_button, model, "button_color_property");

    // Add to scene and bind
    ui_scene_add_element(scene, background_button);
    suggestion_binding_bind(element_binding, layout, model);
}
