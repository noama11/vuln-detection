int ceph_auth_client_process_ticket(struct ceph_auth_client *ac,
                                    struct ceph_crypto_key *secret,
                                    void **p, void *end)
{
    int ret = -EINVAL;
    void *ticket_start = *p;
    uint32_t type, version;
    struct ceph_auth_ticket_handler *handler = NULL;
    struct ceph_auth_ticket *ticket = NULL;
    struct ceph_crypto_key *session_key = NULL;

    if (*p + sizeof(type) > end) {
        goto out;
    }
    type = ceph_decode_32(p);

    if (*p + sizeof(version) > end) {
        goto out;
    }
    version = ceph_decode_32(p);

    handler = ceph_get_auth_ticket_handler(type);
    if (!handler) {
        goto out;
    }

    ticket = handler->decode_ticket(p, end);
    if (IS_ERR(ticket)) {
        ret = PTR_ERR(ticket);
        ticket = NULL;
        goto out;
    }

    ret = handler->decrypt_ticket(ticket, secret);
    if (ret < 0) {
        goto out;
    }

    session_key = ceph_crypto_key_alloc();
    if (!session_key) {
        ret = -ENOMEM;
        goto out;
    }

    ret = ceph_crypto_key_dup(session_key, &ticket->session_key);
    if (ret < 0) {
        goto out;
    }

    ac->session_key = session_key;
    ac->ticket_blob = ticket->blob;
    ac->ticket_blob_len = ticket->blob_len;
    ac->ticket_expires = ticket->expires;

    ret = 0;

out:
    if (ret < 0) {
        ceph_crypto_key_free(session_key);
        ceph_auth_ticket_free(ticket);
    }
    *p = end;
    return ret;
}
