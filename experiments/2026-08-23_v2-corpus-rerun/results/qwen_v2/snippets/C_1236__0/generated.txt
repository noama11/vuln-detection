std::vector<std::unique_ptr<ChooserContextBase::Object>> GetGrantedObjects(
    const GURL& requesting_origin,
    const GURL& embedding_origin) {
  std::vector<std::unique_ptr<ChooserContextBase::Object>> results;

  if (!requesting_origin.is_valid() || !embedding_origin.is_valid()) {
    return results;
  }

  if (!CanGrantPermission(requesting_origin, embedding_origin)) {
    return results;
  }

  WebsiteSettings settings = GetWebsiteSettings(requesting_origin, embedding_origin);
  const auto& object_list = settings.GetObjectList();

  for (const auto& obj : object_list) {
    if (IsObjectValid(obj)) {
      results.push_back(std::make_unique<ChooserContextBase::Object>(obj));
    }
  }

  return results;
}
