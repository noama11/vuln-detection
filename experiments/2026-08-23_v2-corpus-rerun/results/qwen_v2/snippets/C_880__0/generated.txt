#include <stdio.h>
#include <stdlib.h>
#include <sys/stat.h>
#include <limits.h>

int umask_set_from_cmdline = 0;

mode_t set_umask_from_string(const char *optarg) {
    char *endptr;
    long umask_value;

    if (!optarg) {
        fprintf(stderr, "Invalid input: NULL pointer provided.\n");
        return 0;
    }

    errno = 0;
    umask_value = strtol(optarg, &endptr, 10);

    if (errno != 0 || endptr == optarg || *endptr != '\0' || umask_value < 0 || umask_value > USHRT_MAX) {
        fprintf(stderr, "Invalid umask value: %s\n", optarg);
        return 0;
    }

    mode_t new_umask = (mode_t)umask_value;
    umask(new_umask);
    umask_set_from_cmdline = 1;

    return new_umask;
}
