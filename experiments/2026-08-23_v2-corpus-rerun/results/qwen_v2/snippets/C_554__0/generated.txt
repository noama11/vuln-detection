void calculate_coefficients(double mu[4], const int index[4], int prefilter, double r2, double mul) {
    double kernel[4];
    double freq_matrix[4][4];
    double transform_matrix[4][4];
    double vector[4];
    int i, j, k;

    // Compute kernel based on multiplier and r2
    for (i = 0; i < 4; i++) {
        kernel[i] = mul * exp(-index[i] / r2);
    }

    // Apply filtering and blurring operations on frequency matrices
    for (i = 0; i < 4; i++) {
        for (j = 0; j < 4; j++) {
            freq_matrix[i][j] = kernel[i] * kernel[j];
        }
    }

    // Calculate transformation matrix
    for (i = 0; i < 4; i++) {
        for (j = 0; j < 4; j++) {
            transform_matrix[i][j] = freq_matrix[i][j] / (i + j + 1);
        }
    }

    // Apply prefilter iterations
    for (k = 0; k < prefilter; k++) {
        for (i = 0; i < 4; i++) {
            for (j = 0; j < 4; j++) {
                transform_matrix[i][j] = (transform_matrix[i][j] + transform_matrix[i][(j + 1) % 4]) / 2;
            }
        }
    }

    // Compute vector based on frequency matrices
    for (i = 0; i < 4; i++) {
        vector[i] = 0;
        for (j = 0; j < 4; j++) {
            vector[i] += freq_matrix[i][j];
        }
    }

    // Update mu array with matrix-vector multiplication
    for (i = 0; i < 4; i++) {
        mu[i] = 0;
        for (j = 0; j < 4; j++) {
            mu[i] += transform_matrix[i][j] * vector[j];
        }
        mu[i] = (mu[i] < 0) ? 0 : mu[i]; // Ensure non-negative values
    }
}
