#include <linux/uaccess.h>
#include <linux/bitmap.h>

int compat_sys_mbind(int mode, compat_ulong_t __user *nmask, compat_ulong_t maxnode) {
    unsigned long *bitmap = NULL;
    int nbits = 0;
    int ret = -EFAULT;

    if (maxnode) {
        nbits = maxnode;
        bitmap = kmalloc(BITS_TO_LONGS(nbits) * sizeof(unsigned long), GFP_KERNEL);
        if (!bitmap)
            return -ENOMEM;

        if (nmask) {
            if (copy_from_user(bitmap, nmask, BITS_TO_LONGS(nbits) * sizeof(compat_ulong_t)))
                goto out;
        }
    }

    ret = sys_mbind(mode, nmask, maxnode, bitmap, nbits, MPOL_F_STATIC);

    if (nmask && copy_to_user(nmask, bitmap, BITS_TO_LONGS(nbits) * sizeof(compat_ulong_t)))
        ret = -EFAULT;

out:
    kfree(bitmap);
    return ret;
}
