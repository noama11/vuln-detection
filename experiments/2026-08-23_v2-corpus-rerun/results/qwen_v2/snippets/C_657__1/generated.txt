struct import_t* get_import_symbols(struct MACH0_(obj_t)* bin) {
    if (!bin || !bin->symtab) {
        return NULL;
    }

    size_t num_undef = bin->symtab->num_undef;
    struct import_t* imports = (struct import_t*)malloc(num_undef * sizeof(struct import_t));
    if (!imports) {
        return NULL;
    }

    for (size_t i = 0; i < num_undef; i++) {
        const char* name = MACH0_(sym_get_name)(bin, i);
        if (!name) {
            continue;
        }

        char* filtered_name = strdup(name);
        if (!filtered_name) {
            continue;
        }

        imports[i].name = filtered_name;
    }

    bin->import_size = num_undef;
    return imports;
}
