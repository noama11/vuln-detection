int check_token_in_authfile(const char *authfile, const char *username, const char *otp_id, int verbose, FILE *debug_file) {
    FILE *fp = fopen(authfile, "r");
    if (!fp) {
        if (verbose && debug_file) {
            fprintf(debug_file, "Failed to open authfile: %s\n", authfile);
        }
        return AUTH_ERROR;
    }

    char line[1024];
    int user_found = 0;
    int token_found = 0;

    while (fgets(line, sizeof(line), fp)) {
        char *trimmed_line = line;
        while (*trimmed_line && isspace(*trimmed_line)) {
            trimmed_line++;
        }

        if (*trimmed_line == '#' || *trimmed_line == '\0') {
            continue;
        }

        char *user = strtok(trimmed_line, " \t");
        if (!user) {
            continue;
        }

        if (strcmp(user, username) == 0) {
            user_found = 1;
            char *token = strtok(NULL, " \t");
            if (token && strcmp(token, otp_id) == 0) {
                token_found = 1;
                break;
            }
        }
    }

    fclose(fp);

    if (ferror(fp)) {
        if (verbose && debug_file) {
            fprintf(debug_file, "Error reading authfile: %s\n", authfile);
        }
        return AUTH_ERROR;
    }

    if (!user_found) {
        return AUTH_NOT_FOUND;
    }

    if (token_found) {
        return AUTH_FOUND;
    } else {
        return AUTH_NO_TOKENS;
    }
}
