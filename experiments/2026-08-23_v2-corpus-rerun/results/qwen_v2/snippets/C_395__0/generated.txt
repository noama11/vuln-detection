Image *
ReadPNGImage(const ImageInfo *image_info, ExceptionInfo *exception)
{
  Image *image = NULL;
  FILE *file = NULL;
  size_t length = 0;
  unsigned char signature[PNG_SIG_LENGTH];

  (void) LogMagickEvent(CoderEvent, GetMagickModule(), "Enter ReadPNGImage");

  if ((image_info == (ImageInfo *) NULL) || (exception == (ExceptionInfo *) NULL))
    ThrowBinaryException(OptionError, "InvalidArgument", "ReadPNGImage");

  file = fopen(image_info->filename, "rb");
  if (file == (FILE *) NULL)
    {
      (void) LogMagickEvent(CoderEvent, GetMagickModule(), "Unable to open file");
      return (Image *) NULL;
    }

  length = fread(signature, 1, PNG_SIG_LENGTH, file);
  if (length != PNG_SIG_LENGTH)
    {
      (void) fclose(file);
      ThrowBinaryException(FileError, "UnableToReadFile", image_info->filename);
    }

  if (png_sig_cmp(signature, 0, PNG_SIG_LENGTH))
    {
      (void) fclose(file);
      ThrowBinaryException(CoderError, "NotPNGFile", image_info->filename);
    }

  image = AcquireImage(image_info, exception);
  if (image == (Image *) NULL)
    {
      (void) fclose(file);
      return (Image *) NULL;
    }

  {
    MngInfo *mng_info = (MngInfo *) AcquireMagickMemory(sizeof(*mng_info));
    if (mng_info == (MngInfo *) NULL)
      {
        (void) fclose(file);
        image = DestroyImageList(image);
        return (Image *) NULL;
      }

      (void) memset(mng_info, 0, sizeof(*mng_info));
      mng_info->file = file;
      mng_info->image = image;
      mng_info->exception = exception;

      if (!ReadPNGImageData(mng_info))
        {
          (void) fclose(file);
          image = DestroyImageList(image);
          mng_info = (MngInfo *) RelinquishMagickMemory(mng_info);
          return (Image *) NULL;
        }

      if (image->columns == 0 || image->rows == 0)
        {
          (void) fclose(file);
          image = DestroyImageList(image);
          mng_info = (MngInfo *) RelinquishMagickMemory(mng_info);
          return (Image *) NULL;
        }

      if (image->colorspace == UndefinedColorspace)
        image->colorspace = sRGBColorspace;
  }

  (void) fclose(file);

  (void) LogMagickEvent(CoderEvent, GetMagickModule(), "Exit ReadPNGImage");

  return image;
}
