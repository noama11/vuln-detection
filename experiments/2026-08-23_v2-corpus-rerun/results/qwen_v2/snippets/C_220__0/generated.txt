int kvm_pit_read(struct kvm_io_device *this, gpa_t addr, int len, void *data)
{
    if (addr >= 0x40 && addr < 0x44) {
        struct kvm_pit *pit = this->private;
        int channel = addr - 0x40;
        u8 val = 0;

        mutex_lock(&pit->pit_state->mutex);

        switch (channel) {
        case 0:
        case 1:
            val = pit->pit_state->channels[channel].count;
            break;
        case 2:
            val = pit->pit_state->speaker_data;
            break;
        }

        mutex_unlock(&pit->pit_state->mutex);

        *(u8 *)data = val;
        return 0;
    }

    return -EOPNOTSUPP;
}
