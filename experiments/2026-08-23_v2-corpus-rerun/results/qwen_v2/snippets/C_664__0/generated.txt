mrb_value
mrb_io_init_copy(mrb_state *mrb, mrb_value copy)
{
  struct RClass *c;
  mrb_value orig;
  mrb_io *io, *orig_io;
  int oflags;

  orig = mrb_get_args(mrb, "o", &c);
  if (!mrb_io_p(orig)) {
    mrb_raise(mrb, E_TYPE_ERROR, "wrong argument type");
  }

  io = mrb_io_ptr(copy);
  orig_io = mrb_io_ptr(orig);

  if (io == orig_io) {
    return copy;
  }

  if (io->fd != -1) {
    mrb_io_close(mrb, copy);
  }

  io->fd = dup(orig_io->fd);
  if (io->fd == -1) {
    mrb_sys_fail(mrb, "dup");
  }

  io->mode = orig_io->mode;
  io->path = orig_io->path;
  if (orig_io->path) {
    io->path = mrb_strdup(mrb, orig_io->path);
  }

  oflags = fcntl(orig_io->fd, F_GETFL);
  if (oflags == -1) {
    mrb_sys_fail(mrb, "fcntl");
  }

  if (oflags & O_APPEND) {
    io->mode |= MRB_IO_APPEND;
  }

  if (oflags & O_SYNC) {
    io->mode |= MRB_IO_SYNC;
  }

  if (oflags & O_NONBLOCK) {
    io->mode |= MRB_IO_NONBLOCK;
  }

  return copy;
}
