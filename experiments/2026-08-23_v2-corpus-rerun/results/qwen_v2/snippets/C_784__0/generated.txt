void process_lru_command(conn *c, token_t *tokens, size_t ntokens) {
    if (ntokens < 2) {
        // Not enough tokens to process the command
        return;
    }

    if (strcmp(tokens[0].value, "tune") == 0) {
        if (ntokens != 5) {
            // Invalid number of tokens for 'tune' command
            return;
        }

        int hot_pct = atoi(tokens[1].value);
        int warm_pct = atoi(tokens[2].value);
        int hot_age_factor = atoi(tokens[3].value);
        int warm_age_factor = atoi(tokens[4].value);

        if (hot_pct < 0 || hot_pct > 100 || warm_pct < 0 || warm_pct > 100 ||
            hot_age_factor < 0 || warm_age_factor < 0) {
            // Invalid parameter values
            return;
        }

        // Update LRU settings with the new values
        c->hot_percentage = hot_pct;
        c->warm_percentage = warm_pct;
        c->hot_age_factor = hot_age_factor;
        c->warm_age_factor = warm_age_factor;
    } else if (strcmp(tokens[0].value, "mode") == 0) {
        if (ntokens != 2) {
            // Invalid number of tokens for 'mode' command
            return;
        }

        if (strcmp(tokens[1].value, "hot") == 0) {
            c->lru_mode = LRU_MODE_HOT;
        } else if (strcmp(tokens[1].value, "warm") == 0) {
            c->lru_mode = LRU_MODE_WARM;
        } else {
            // Invalid mode argument
            return;
        }
    } else if (strcmp(tokens[0].value, "temp_ttl") == 0) {
        if (ntokens != 2) {
            // Invalid number of tokens for 'temp_ttl' command
            return;
        }

        int temp_ttl = atoi(tokens[1].value);
        if (temp_ttl < 0) {
            // Invalid TTL value
            return;
        }

        c->temp_ttl = temp_ttl;
    } else {
        // Unknown command
        return;
    }
}
