int hidp_setup_hid_session(struct hidp_session *session, struct hidp_connadd_req *req)
{
	struct hid_device *hid_dev;
	int ret = -ENOMEM;

	hid_dev = kzalloc(sizeof(*hid_dev), GFP_KERNEL);
	if (!hid_dev)
		goto err_alloc_hid_dev;

	session->hid_dev = hid_dev;

	hid_dev->vendor = req->vendor;
	hid_dev->product = req->product;
	hid_dev->country = req->country;

	if (hid_is_blacklisted(hid_dev)) {
		ret = -ENODEV;
		goto err_blacklisted;
	}

	session->read_data = kmalloc(req->read_buffer_size, GFP_KERNEL);
	if (!session->read_data)
		goto err_alloc_read_data;

	session->read_size = req->read_buffer_size;

	/* Copy user space data if needed */
	if (copy_from_user(session->read_data, req->read_buffer, req->read_buffer_size)) {
		ret = -EFAULT;
		goto err_copy_data;
	}

	return 0;

err_copy_data:
	kfree(session->read_data);
err_alloc_read_data:
err_blacklisted:
	kfree(hid_dev);
err_alloc_hid_dev:
	return ret;
}
