CmdResult handleUnregisteredClientAuthSASL(const std::vector<std::string>& parameters, User* user) {
    if (!user->isUnregistered()) {
        return CMD_FAILURE;
    }

    if (!user->hasCap("sasl")) {
        return CMD_FAILURE;
    }

    if (parameters.empty() || parameters[0].empty()) {
        return CMD_FAILURE;
    }

    if (!user->saslAuthenticator) {
        user->saslAuthenticator = std::make_unique<SaslAuthenticator>(parameters[0]);
    }

    if (!user->saslAuthenticator->sendClientMessage()) {
        user->saslAuthenticator = nullptr;
        user->announce("SASL authentication failed.");
        return CMD_FAILURE;
    }

    return CMD_SUCCESS;
}
