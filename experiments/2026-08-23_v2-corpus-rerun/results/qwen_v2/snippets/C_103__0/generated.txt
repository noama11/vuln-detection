int process_diag_message(struct sk_buff *skb, struct nlmsghdr *nlh) {
    int err = -EINVAL;

    if (nlh->nlmsg_len < sizeof(*nlh))
        return err;

    struct sock_diag_handler *handler = find_diag_handler(nlh->nlmsg_type);
    if (!handler)
        return -ENOENT;

    if (handler->family != AF_UNSPEC && handler->family != skb->sk->sk_family)
        return -EAFNOSUPPORT;

    if (!try_module_get(handler->owner))
        return -ENOENT;

    err = handler->dump(skb, nlh);

    module_put(handler->owner);
    return err;
}
