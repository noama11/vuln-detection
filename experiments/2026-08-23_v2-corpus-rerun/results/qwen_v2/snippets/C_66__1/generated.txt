void adjust_mtu_and_maxfraglen(unsigned int *mtu, int *maxfraglen,
                               unsigned int fragheaderlen,
                               struct sk_buff *skb,
                               struct rt6_info *rt,
                               bool pmtuprobe) {
    if (!skb) {
        *mtu -= fragheaderlen;
    } else {
        *mtu = min(*mtu, rt->dst.dev->mtu);
    }

    if (pmtuprobe) {
        *mtu = min(*mtu, rt->dst.pmtu);
    }

    *maxfraglen = *mtu - fragheaderlen;
}
