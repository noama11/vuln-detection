struct berval** encode_krb5_key_data(krb5_key_data* key_data_in, int n_key_data, krb5_kvno mkvno) {
    struct berval** encoded_keys = NULL;
    krb5_key_data* key_data_copy = NULL;
    int i, j, unique_kvno_count = 0;
    int* unique_kvno_indices = NULL;

    if (n_key_data <= 0 || key_data_in == NULL) {
        return NULL;
    }

    key_data_copy = malloc(n_key_data * sizeof(krb5_key_data));
    if (key_data_copy == NULL) {
        return NULL;
    }

    for (i = 0; i < n_key_data; i++) {
        key_data_copy[i] = key_data_in[i];
        if (mkvno != 0) {
            key_data_copy[i].key_data_ver = mkvno;
        }
    }

    unique_kvno_indices = malloc(n_key_data * sizeof(int));
    if (unique_kvno_indices == NULL) {
        free(key_data_copy);
        return NULL;
    }

    for (i = 0; i < n_key_data; i++) {
        int is_unique = 1;
        for (j = 0; j < unique_kvno_count; j++) {
            if (key_data_copy[i].key_data_ver == key_data_copy[unique_kvno_indices[j]].key_data_ver) {
                is_unique = 0;
                break;
            }
        }
        if (is_unique) {
            unique_kvno_indices[unique_kvno_count++] = i;
        }
    }

    encoded_keys = calloc(unique_kvno_count + 1, sizeof(struct berval*));
    if (encoded_keys == NULL) {
        free(key_data_copy);
        free(unique_kvno_indices);
        return NULL;
    }

    for (i = 0; i < unique_kvno_count; i++) {
        krb5_key_data* kd = &key_data_copy[unique_kvno_indices[i]];
        struct berval* bv = calloc(1, sizeof(struct berval));
        if (bv == NULL) {
            for (j = 0; j < i; j++) {
                free(encoded_keys[j]);
            }
            free(encoded_keys);
            free(key_data_copy);
            free(unique_kvno_indices);
            return NULL;
        }

        bv->bv_len = kd->key_data_length;
        bv->bv_val = malloc(bv->bv_len);
        if (bv->bv_val == NULL) {
            free(bv);
            for (j = 0; j < i; j++) {
                free(encoded_keys[j]);
            }
            free(encoded_keys);
            free(key_data_copy);
            free(unique_kvno_indices);
            return NULL;
        }

        memcpy(bv->bv_val, kd->key_data_contents, bv->bv_len);
        encoded_keys[i] = bv;
    }

    encoded_keys[unique_kvno_count] = NULL;

    free(key_data_copy);
    free(unique_kvno_indices);

    return encoded_keys;
}
