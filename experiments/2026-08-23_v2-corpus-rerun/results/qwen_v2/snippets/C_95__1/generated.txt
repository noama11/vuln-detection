int sctp_get_assoc_stats(struct sock *sk, int len, char __user *optval, int __user *optlen)
{
    struct sctp_association *asoc;
    struct sctp_stats stats;
    int ret = 0;

    if (len < sizeof(stats))
        return -EINVAL;

    asoc = sctp_id2assoc(sk, 0);
    if (!asoc)
        return -EINVAL;

    memset(&stats, 0, sizeof(stats));
    stats.assoc_id = asoc->assoc_id;
    stats.cumu_flight = asoc->c.sack.outstanding;
    stats.a_rwnd = asoc->a_rwnd;
    stats.max_rwnd = asoc->max_rwnd;
    stats.cur_rwnd = asoc->rwnd;
    stats.pd_mode = asoc->param->pdapi_enable;
    stats.max_burst = asoc->param->max_burst;
    stats.max_init_timeo = asoc->param->init_max_timeo;
    stats.max_path_rto = asoc->param->rto_max;
    stats.min_path_rto = asoc->param->rto_min;
    stats.max_path_rtx = asoc->param->rto_max;
    stats.min_path_rtx = asoc->param->rto_min;
    stats.max_outstreams = asoc->param->max_out_streams;
    stats.min_outstreams = asoc->param->min_out_streams;
    stats.max_instreams = asoc->param->max_in_streams;
    stats.min_instreams = asoc->param->min_in_streams;

    if (copy_to_user(optval, &stats, sizeof(stats)))
        return -EFAULT;

    if (put_user(sizeof(stats), optlen))
        return -EFAULT;

    return 0;
}
