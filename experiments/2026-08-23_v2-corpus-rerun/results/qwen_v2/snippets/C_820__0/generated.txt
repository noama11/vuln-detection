int dvb_diseqc_send_master_cmd(struct dvb_frontend *fe, struct dvb_diseqc_master_cmd *d)
{
	int i, rc = 0;

	if (d->msg[0] > 6) {
		dprintk(1, "DiSEqC message length out of range (%d)\n", d->msg[0]);
		return -EINVAL;
	}

	if (dprintk_enabled(2)) {
		dprintk(2, "DiSEqC message to send:\n");
		for (i = 0; i <= d->msg[0]; i++)
			dprintk(2, "%02x ", d->msg[i]);
		dprintk(2, "\n");
	}

	if (fe->ops.diseqc_send_master_cmd) {
		rc = fe->ops.diseqc_send_master_cmd(fe, d);
		if (rc < 0)
			return rc;
	}

	return 0;
}
