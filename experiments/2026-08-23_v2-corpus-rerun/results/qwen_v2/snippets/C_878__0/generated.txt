bool verify_client_password(PgSocket *client, const char *passwd) {
    if (client->user == NULL) {
        elog(ERROR, "Client has no associated user");
        return false;
    }

    const char *stored_password = client->user->password;

    if (passwd[0] == '\0' || stored_password[0] == '\0') {
        return false;
    }

    if (strncmp(stored_password, "md5", 3) == 0) {
        if (strlen(passwd) < 1) {
            return false;
        }

        char *md5_hash = generate_md5_hash(passwd, client->user->username);
        if (md5_hash == NULL) {
            return false;
        }

        bool result = strcmp(stored_password, md5_hash) == 0;
        pfree(md5_hash);
        return result;
    } else if (strncmp(stored_password, "crypt", 5) == 0) {
        char *encrypted = crypt(passwd, stored_password);
        return encrypted != NULL && strcmp(stored_password, encrypted) == 0;
    } else {
        return strcmp(stored_password, passwd) == 0;
    }

    return false;
}
