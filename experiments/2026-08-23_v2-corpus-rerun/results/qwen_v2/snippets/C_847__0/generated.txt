void readSingleImageFrame(MagickCore::Image *image, MagickCore::ExceptionInfo *exceptionInfo) {
  if (image != (MagickCore::Image *) NULL) {
    if (image->next != (MagickCore::Image *) NULL) {
      MagickCore::Image *nextImage = image->next;
      MagickCore::Image *lastImage = image;

      while (nextImage != (MagickCore::Image *) NULL) {
        lastImage = nextImage;
        nextImage = nextImage->next;
      }

      lastImage->next = (MagickCore::Image *) NULL;
    }
  }

  if (exceptionInfo->severity == UndefinedException && image == (MagickCore::Image *) NULL) {
    exceptionInfo = (MagickCore::ExceptionInfo *) MagickRelinquishMemory(exceptionInfo);
    if (!IsEventLogging())
      (void) ThrowMagickException(exceptionInfo, GetMagickModule(), WarningError, "NoImage", "%s", "");
  }

  ThrowImageException(image);
}
