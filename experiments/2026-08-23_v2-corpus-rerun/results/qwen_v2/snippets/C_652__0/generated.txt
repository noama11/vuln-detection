mrb_value
mrb_clone(mrb_state *mrb, mrb_value self)
{
  mrb_value clone;

  if (mrb_immediate_p(self)) {
    mrb_raise(mrb, E_TYPE_ERROR, "can't clone immediate value");
  }
  if (mrb_singleton_class(mrb_class(self))) {
    mrb_raise(mrb, E_TYPE_ERROR, "can't clone singleton class");
  }

  clone = mrb_obj_new(mrb, mrb_class(self), 0, NULL);
  mrb_data_object_dup(mrb, mrb_obj_value(clone), mrb_obj_value(self));
  mrb_obj_iv_copy(mrb, mrb_obj_value(clone), mrb_obj_value(self));
  mrb_obj_set_flag(mrb_obj_value(clone), mrb_obj_flag(self));

  return clone;
}
