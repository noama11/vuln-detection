int create_or_update_file(const char *path, bool parents, usec_t stamp, uid_t uid, gid_t gid, mode_t mode) {
    int fd = -1;
    mode_t default_mode = S_IRUSR | S_IWUSR | S_IRGRP | S_IROTH; // 0644
    mode_t final_mode = mode ? mode : default_mode;
    struct timespec times[2];

    if (!path) {
        return -EINVAL;
    }

    if (parents) {
        if (mkdir_parents(path, 0755) < 0) {
            return -errno;
        }
    }

    fd = open(path, O_CREAT | O_WRONLY | O_TRUNC, final_mode);
    if (fd < 0) {
        return -errno;
    }

    if (uid != (uid_t)-1 || gid != (gid_t)-1) {
        if (chown(path, uid, gid) < 0) {
            int err = -errno;
            close(fd);
            return err;
        }
    }

    times[0].tv_nsec = USEC_TO_NSEC(stamp);
    times[0].tv_sec = stamp / USEC_PER_SEC;
    times[1].tv_nsec = USEC_TO_NSEC(stamp);
    times[1].tv_sec = stamp / USEC_PER_SEC;

    if (utimensat(AT_FDCWD, path, times, 0) < 0) {
        int err = -errno;
        close(fd);
        return err;
    }

    close(fd);
    return 0;
}
