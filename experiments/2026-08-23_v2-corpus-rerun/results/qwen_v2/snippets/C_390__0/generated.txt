ACPI_STATUS AcpiPsCompleteScope(ACPI_WALK_STATE *WalkState, ACPI_PARSE_OBJECT *Op, ACPI_STATUS Status)
{
    ACPI_PARSE_OBJECT *PrevOp;
    ACPI_PARSE_OBJECT *NextOp;

    if (WalkState->ScopeDepth > 0) {
        WalkState->ScopeDepth--;
    }

    if (WalkState->Control.ImmediateNotify) {
        WalkState->Control.ImmediateNotify = FALSE;
        return (AE_CTRL_IMEDIATE_NOTIFY);
    }

    if (Status == AE_CTRL_PENDING) {
        return (Status);
    }

    if (Status == AE_CTRL_TERMINATE) {
        return (Status);
    }

    if (WalkState->ParserState.Scope.Obj == Op) {
        PrevOp = Op->Common.Next;
        while (PrevOp) {
            NextOp = PrevOp->Common.Next;
            AcpiPsCompleteScope(WalkState, PrevOp, Status);
            PrevOp = NextOp;
        }
    }

    return (Status);
}
