int find_dnxhd_frame_end(DNXHDParserContext *dctx, const uint8_t *buf, int buf_size) {
    int i;
    int pos;

    if (!dctx->frame_start_found) {
        for (i = 0; i < buf_size - 3; i++) {
            if (buf[i] == 0x1f && buf[i+1] == 0x40 && (buf[i+2] & 0xf0) == 0x20) {
                dctx->frame_start_found = 1;
                dctx->frame_offset = i;
                break;
            }
        }
        if (!dctx->frame_start_found)
            return END_NOT_FOUND;
    }

    pos = dctx->frame_offset;

    if (pos + 4 > buf_size)
        return END_NOT_FOUND;

    dctx->width = ((buf[pos + 2] & 0x0f) << 4) | ((buf[pos + 3] >> 4) & 0x0f);
    dctx->height = ((buf[pos + 3] & 0x0f) << 8) | (buf[pos + 4] & 0xff);

    if (dctx->width == 0 || dctx->height == 0)
        return END_NOT_FOUND;

    dctx->frame_size = dctx->width * dctx->height * 3 / 2;

    if (buf_size - pos >= dctx->frame_size) {
        dctx->frame_start_found = 0;
        return buf_size - pos;
    }

    dctx->frame_offset += buf_size - pos;
    return END_NOT_FOUND;
}
