bool readDatabaseFromStarZone(StarZone &zone) {
    if (!zone.canReadDatabaseName()) {
        zone.logError("Failed to read database name from zone.");
        return false;
    }

    std::string dbName, tableName;
    if (!zone.readDatabaseName(dbName) || !zone.readTableName(tableName)) {
        zone.logError("Failed to read database or table name.");
        return false;
    }

    // Store in member variables
    this->databaseName = dbName;
    this->tableName = tableName;

    // Compatibility check
    if (zone.isCompatibleWithSQL()) {
        std::string sqlStatement;
        if (!zone.readSQLStatement(sqlStatement)) {
            zone.logError("Failed to read SQL statement.");
            return false;
        }
        this->sqlData.push_back(sqlStatement);
    }

    // Read additional data entries
    while (zone.hasMoreData()) {
        std::string dataEntry;
        if (!zone.readDataEntry(dataEntry)) {
            zone.logError("Failed to read data entry.");
            return false;
        }
        this->dataEntries.push_back(dataEntry);
    }

    zone.logState("Database read completed.");
    zone.closeRecord();
    return true;
}
