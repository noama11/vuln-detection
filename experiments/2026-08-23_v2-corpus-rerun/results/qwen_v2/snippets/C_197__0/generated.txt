int futex_wait_requeue(u32 __user *uaddr, unsigned int flags, u32 val,
                       ktime_t *abs_time, u32 bitset, u32 __user *uaddr2)
{
    int ret = 0;
    struct hrtimer s_timer;
    struct futex_waiter *waiter;
    struct task_struct *task = current;

    if (abs_time) {
        hrtimer_init(&s_timer, CLOCK_MONOTONIC, HRTIMER_MODE_ABS);
        s_timer.function = futex_hrtimer_func;
        hrtimer_set_expires(&s_timer, ktime_get());
        hrtimer_start(&s_timer, ktime_get(), HRTIMER_MODE_ABS);
    }

    waiter = kmalloc(sizeof(*waiter), GFP_KERNEL);
    if (!waiter)
        return -ENOMEM;

    waiter->bitset = bitset;
    waiter->task = task;
    waiter->uaddr = uaddr;
    waiter->uaddr2 = uaddr2;
    waiter->flags = flags;

    ret = futex_wait(uaddr, flags, val, abs_time, bitset, waiter);
    if (ret < 0)
        goto out;

    if (abs_time)
        hrtimer_cancel(&s_timer);

    if (waiter->requeue) {
        ret = futex_requeue(uaddr, uaddr2, flags, val, bitset, waiter);
        if (ret < 0)
            goto out;
    }

out:
    kfree(waiter);
    return ret;
}
