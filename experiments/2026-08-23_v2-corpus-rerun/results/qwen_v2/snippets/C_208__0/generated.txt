int get_bitmap_file(struct mddev *mddev, void __user *arg)
{
    struct bitmap_file *bf;
    int ret = 0;

    bf = kmalloc(sizeof(*bf), GFP_NOIO);
    if (!bf)
        return -ENOMEM;

    if (!mddev->bitmap)
        bf->pathname[0] = '\0';
    else
        strlcpy(bf->pathname, mddev->bitmap->file_path, sizeof(bf->pathname));

    if (copy_to_user(arg, bf, sizeof(*bf))) {
        ret = -EFAULT;
        goto out;
    }

out:
    kfree(bf);
    return ret;
}
