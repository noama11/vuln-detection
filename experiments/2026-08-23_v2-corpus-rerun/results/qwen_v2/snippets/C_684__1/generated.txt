#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <sys/stat.h>
#include <unistd.h>

#include "leptola.h"  // Assuming this is the header for l_int32 and related functions

l_int32 generate_html_viewer(const char* dirin, const char* dirout, const char* rootname, l_int32 thumbwidth, l_int32 viewwidth) {
    if (!dirin || !dirout || !rootname || thumbwidth <= 0 || viewwidth <= 0) {
        return 1;  // Invalid parameters
    }

    // Create output directory if it doesn't exist
    struct stat st = {0};
    if (stat(dirout, &st) == -1) {
        if (mkdir(dirout, 0777) != 0) {
            return 2;  // Failed to create output directory
        }
    }

    // Open input directory
    DIR* dir = opendir(dirin);
    if (!dir) {
        return 3;  // Failed to open input directory
    }

    // Read filenames from input directory
    struct dirent* entry;
    char** filenames = NULL;
    l_int32 count = 0;
    while ((entry = readdir(dir)) != NULL) {
        if (strcmp(entry->d_name, ".") == 0 || strcmp(entry->d_name, "..") == 0) {
            continue;
        }

        // Allocate memory for filenames array
        filenames = (char**)realloc(filenames, (count + 1) * sizeof(char*));
        if (!filenames) {
            closedir(dir);
            return 4;  // Memory allocation failed
        }

        // Copy filename to array
        l_int32 len = strlen(entry->d_name);
        filenames[count] = (char*)malloc(len + 1);
        if (!filenames[count]) {
            closedir(dir);
            return 4;  // Memory allocation failed
        }
        strcpy(filenames[count], entry->d_name);
        count++;
    }
    closedir(dir);

    // Generate thumbnails and view images
    for (l_int32 i = 0; i < count; i++) {
        char input_path[1024];
        char output_thumb_path[1024];
        char output_view_path[1024];

        // Construct input path
        snprintf(input_path, sizeof(input_path), "%s/%s", dirin, filenames[i]);

        // Construct output paths
        snprintf(output_thumb_path, sizeof(output_thumb_path), "%s/%s_thumb.jpg", dirout, filenames[i]);
        snprintf(output_view_path, sizeof(output_view_path), "%s/%s_view.jpg", dirout, filenames[i]);

        // Load image
        PIX* pix = pixRead(input_path);
        if (!pix) {
            for (l_int32 j = 0; j < count; j++) {
                free(filenames[j]);
            }
            free(filenames);
            return 5;  // Failed to read image
        }

        // Generate thumbnail
        PIX* pix_thumb = pixScaleToWidth(pix, thumbwidth);
        if (!pix_thumb) {
            pixDestroy(&pix);
            for (l_int32 j = 0; j < count; j++) {
                free(filenames[j]);
            }
            free(filenames);
            return 6;  // Failed to generate thumbnail
        }
        l_int32 result = pixWrite(output_thumb_path, pix_thumb, IFF_JPEG);
        if (result != 0) {
            pixDestroy(&pix);
            pixDestroy(&pix_thumb);
            for (l_int32 j = 0; j < count; j++) {
                free(filenames[j]);
            }
            free(filenames);
            return 7;  // Failed to write thumbnail
        }
        pixDestroy(&pix_thumb);

        // Generate view image
        PIX* pix_view = pixScaleToWidth(pix, viewwidth);
        if (!pix_view) {
            pixDestroy(&pix);
            for (l_int32 j = 0; j < count; j++) {
                free(filenames[j]);
            }
            free(filenames);
            return 8;  // Failed to generate view image
        }
        result = pixWrite(output_view_path, pix_view, IFF_JPEG);
        if (result != 0) {
            pixDestroy(&pix);
            pixDestroy(&pix_view);
            for (l_int32 j = 0; j < count; j++) {
                free(filenames[j]);
            }
            free(filenames);
            return 9;  // Failed to write view image
        }
        pixDestroy(&pix_view);

        pixDestroy(&pix);
    }

    // Generate HTML files
    char index_html_path[1024];
    char gallery_html_path[1024];

    // Construct HTML file paths
    snprintf(index_html_path, sizeof(index_html_path), "%s/%s.html", dirout, rootname);
    snprintf(gallery_html_path, sizeof(gallery_html_path), "%s/%s_gallery.html", dirout, rootname);

    // Write index.html
    FILE* index_file = fopen(index_html_path, "w");
    if (!index_file) {
        for (l_int32 j = 0; j < count; j++) {
            free(filenames[j]);
        }
        free(filenames);
        return 10;  // Failed to open index.html
    }

    fprintf(index_file, "<html>\n<head>\n<title>Image Viewer</title>\n</head>\n<body>\n");
    fprintf(index_file, "<h1>Image Viewer</h1>\n<ul>\n");

    for (l_int32 i = 0; i < count; i++) {
        char* filename = filenames[i];
        char thumb_path[1024];
        char view_path[1024];

        // Construct thumbnail and view image paths
        snprintf(thumb_path, sizeof(thumb_path), "%s/%s_thumb.jpg", dirout, filename);
        snprintf(view_path, sizeof(view_path), "%s/%s_view.jpg", dirout, filename);

        fprintf(index_file, "<li><a href=\"%s\"><img src=\"%s\" alt=\"%s\"></a></li>\n", view_path, thumb_path, filename);
    }

    fprintf(index_file, "</ul>\n</body>\n</html>");

    fclose(index_file);

    // Write gallery.html
    FILE* gallery_file = fopen(gallery_html_path, "w");
    if (!gallery_file) {
        for (l_int32 j = 0; j < count; j++) {
            free(filenames[j]);
        }
        free(filenames);
        return 11;  // Failed to open gallery.html
    }

    fprintf(gallery_file, "<html>\n<head>\n<title>Image Gallery</title>\n</head>\n<body>\n");
    fprintf(gallery_file, "<h1>Image Gallery</h1>\n<ul>\n");

    for (l_int32 i = 0; i < count; i++) {
        char* filename = filenames[i];
        char view_path[1024];

        // Construct view image path
        snprintf(view_path, sizeof(view_path), "%s/%s_view.jpg", dirout, filename);

        fprintf(gallery_file, "<li><a href=\"%s\">%s</a></li>\n", view_path, filename);
    }

    fprintf(gallery_file, "</ul>\n</body>\n</html>");

    fclose(gallery_file);

    // Clean up
    for (l_int32 i = 0; i < count; i++) {
        free(filenames[i]);
    }
    free(filenames);

    return 0;  // Success
}
