int write_task_data(struct task_struct *child, addr_t addr, addr_t data) {
    if (addr >= PSW_ADDR_MIN && addr <= PSW_ADDR_MAX) {
        /* Valid psw address range */
        if (data > MAX_PSW_VALUE)
            return -EINVAL;
        /* Write data to psw */
        *(addr_t *)((char *)&child->psw + (addr - PSW_ADDR_MIN)) = data;
        return 0;
    } else if (addr >= GPRS_ADDR_MIN && addr <= GPRS_ADDR_MAX) {
        /* Valid general purpose register address range */
        if (addr % sizeof(addr_t) != 0) {
            /* Prevent writes to padding holes */
            return -EINVAL;
        }
        /* Write data to gprs */
        *(addr_t *)((char *)&child->gprs + (addr - GPRS_ADDR_MIN)) = data;
        return 0;
    } else if (addr >= FPRS_ADDR_MIN && addr <= FPRS_ADDR_MAX) {
        /* Valid floating-point register address range */
        if (addr % sizeof(addr_t) != 0) {
            return -EINVAL;
        }
        /* Write data to fprs */
        *(addr_t *)((char *)&child->fprs + (addr - FPRS_ADDR_MIN)) = data;
        return 0;
    } else if (addr >= PER_INFO_ADDR_MIN && addr <= PER_INFO_ADDR_MAX) {
        /* Valid per_info address range */
        if (addr % sizeof(addr_t) != 0) {
            return -EINVAL;
        }
        /* Write data to per_info */
        *(addr_t *)((char *)&child->per_info + (addr - PER_INFO_ADDR_MIN)) = data;
        return 0;
    } else {
        /* Invalid address */
        return -EINVAL;
    }
}
