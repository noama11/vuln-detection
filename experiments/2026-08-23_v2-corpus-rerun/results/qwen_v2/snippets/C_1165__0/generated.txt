void handle_received_savable_resources(std::vector<GURL>& resources_list, std::vector<Referrer>& referrers_list, std::vector<GURL>& frames_list) {
    if (resources_list.size() != referrers_list.size()) {
        return;
    }

    size_t total_items = resources_list.size() + frames_list.size();
    if (download_in_progress) {
        download.total_bytes += total_items * AVERAGE_RESOURCE_SIZE;
    }

    for (size_t i = 0; i < resources_list.size(); ++i) {
        const GURL& resource_url = resources_list[i];
        const Referrer& referrer = referrers_list[i];

        if (resource_url.is_valid()) {
            SaveItem item;
            item.url = resource_url;
            item.referrer = referrer;
            waiting_queue.push(item);
        }
    }

    for (const GURL& frame_url : frames_list) {
        if (frame_url.is_valid()) {
            SaveItem item;
            item.url = frame_url;
            waiting_queue.push(item);
        }
    }

    if (waiting_queue.empty()) {
        handle_user_cancellation();
    }

    update_wait_state(WAIT_STATE_SAVING);
    start_saving_process();
}
