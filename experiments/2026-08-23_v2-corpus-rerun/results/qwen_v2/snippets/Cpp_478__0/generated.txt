int decodeWaveAdpcmBlock(const uint8_t *encoded, int16_t *decoded) {
    // Initialize previous values and indices for each channel
    int16_t prevValues[2] = {0};
    int8_t indices[2] = {0};

    // Read initial values from the encoded data
    prevValues[0] = (int16_t)((encoded[0] << 8) | encoded[1]);
    indices[0] = encoded[2];
    prevValues[1] = (int16_t)((encoded[3] << 8) | encoded[4]);
    indices[1] = encoded[5];

    // Pointer to the start of the encoded audio data
    const uint8_t *p = encoded + 6;
    int sampleIndex = 0;

    // Process the encoded samples in packets
    while (*p != 0xFF) {
        // Read the number of samples in this packet
        int numSamples = *p++;
        for (int i = 0; i < numSamples; ++i) {
            // Decode a sample for each channel
            for (int ch = 0; ch < 2; ++ch) {
                int8_t sample = *p++;
                int16_t step = adpcmStepSizeTable[indices[ch]];
                int16_t delta = (sample & 0x0F) * step;
                int16_t prediction = (sample & 0x10) ? -delta : delta;
                int16_t newValue = prevValues[ch] + prediction;

                // Clamp to 16-bit range
                if (newValue > 32767) newValue = 32767;
                if (newValue < -32768) newValue = -32768;

                decoded[sampleIndex++] = newValue;
                prevValues[ch] = newValue;

                // Update the index for the next sample
                indices[ch] += (sample & 0x0F);
                if (indices[ch] < 0) indices[ch] = 0;
                if (indices[ch] > 88) indices[ch] = 88;
            }
        }
    }

    // Return the total number of bytes written to the decoded buffer
    return sampleIndex * sizeof(int16_t);
}
