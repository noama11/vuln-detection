int pptp_connect(struct socket *sock, struct sockaddr *uservaddr, int sockaddr_len, int flags)
{
    int result = -EINVAL;
    struct sock *sk = sock->sk;
    struct net_device *dev;

    if (!sock || !uservaddr || sockaddr_len < sizeof(struct sockaddr_in))
        return -EINVAL;

    if (uservaddr->sa_family != AF_INET)
        return -EAFNOSUPPORT;

    lock_sock(sk);

    if (sk->sk_state != TCP_ESTABLISHED || sock->state != SS_UNCONNECTED) {
        release_sock(sk);
        return -EALREADY;
    }

    dev = ppp_register_new_channel();
    if (!dev) {
        release_sock(sk);
        return -ENOMEM;
    }

    result = setup_pptp_route(uservaddr, dev);
    if (result < 0) {
        ppp_unregister_channel(dev);
        release_sock(sk);
        return result;
    }

    sk->sk_state = TCP_ESTABLISHED;
    sock->state = SS_CONNECTED;

    release_sock(sk);

    return 0;
}
