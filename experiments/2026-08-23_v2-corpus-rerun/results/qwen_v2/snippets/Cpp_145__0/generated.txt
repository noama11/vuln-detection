#include <string>
#include <cstring>
#include <cstdlib>
#include <fstream>
#include <unistd.h>
#include <fcntl.h>
#include <iostream>

std::string generate_iv(int size, int source = 0) {
    if (size <= 0) {
        std::cerr << "Invalid size for IV." << std::endl;
        return "false";
    }

    char* iv = new (std::nothrow) char[size];
    if (!iv) {
        std::cerr << "Memory allocation failed." << std::endl;
        return "false";
    }

    int fd;
    if (source == 0) {
        fd = open("/dev/random", O_RDONLY);
    } else if (source == 1) {
        fd = open("/dev/urandom", O_RDONLY);
    } else {
        // Use user-defined f_rand function
        for (int i = 0; i < size; ++i) {
            iv[i] = static_cast<char>(f_rand());
        }
        std::string result(iv, size);
        delete[] iv;
        return result;
    }

    if (fd == -1) {
        std::cerr << "Failed to open randomness source." << std::endl;
        delete[] iv;
        return "false";
    }

    ssize_t bytes_read = read(fd, iv, size);
    close(fd);

    if (bytes_read != static_cast<ssize_t>(size)) {
        std::cerr << "Failed to read sufficient random bytes." << std::endl;
        delete[] iv;
        return "false";
    }

    std::string result(iv, size);
    delete[] iv;
    return result;
}
