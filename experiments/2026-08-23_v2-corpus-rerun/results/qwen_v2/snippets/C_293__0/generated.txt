int x25_negotiate_facilities(struct sk_buff *skb, struct sock *sk,
                             struct x25_facilities *new,
                             struct x25_dte_facilities *dte)
{
    struct x25_facilities *caller;
    struct x25_facilities *callee;
    int len = 0;
    u8 *ptr;

    caller = (struct x25_facilities *)skb->data;
    callee = &sk->x25->facilities;

    ptr = skb->data;
    len = 0;

    if (*ptr != X25_FACILITIES) {
        X25_SUBLOG("No facilities to negotiate.\n");
        return -EINVAL;
    }

    ptr++;
    len++;

    new->throughput = 0;
    new->window = 0;
    new->packet_size = 0;
    new->reverse = 0;

    if (*ptr & X25_FACILITY_REVERSE_CHARGE) {
        new->reverse = 1;
        X25_SUBLOG("Reverse charging requested.\n");
    }

    ptr++;
    len++;

    new->throughput = *ptr;
    if (new->throughput > X25_MAX_THROUGHPUT)
        new->throughput = X25_MAX_THROUGHPUT;

    ptr++;
    len++;

    new->window = *ptr;
    if (new->window > X25_MAX_WINDOW)
        new->window = X25_MAX_WINDOW;

    ptr++;
    len++;

    new->packet_size = *ptr;
    if (new->packet_size > X25_MAX_PACKET_SIZE)
        new->packet_size = X25_MAX_PACKET_SIZE;

    ptr++;
    len++;

    dte->throughput = *ptr;
    if (dte->throughput > X25_MAX_DTE_THROUGHPUT)
        dte->throughput = X25_MAX_DTE_THROUGHPUT;

    ptr++;
    len++;

    dte->window = *ptr;
    if (dte->window > X25_MAX_DTE_WINDOW)
        dte->window = X25_MAX_DTE_WINDOW;

    ptr++;
    len++;

    dte->packet_size = *ptr;
    if (dte->packet_size > X25_MAX_DTE_PACKET_SIZE)
        dte->packet_size = X25_MAX_DTE_PACKET_SIZE;

    ptr++;
    len++;

    if (new->throughput < callee->throughput)
        new->throughput = callee->throughput;

    if (new->window < callee->window)
        new->window = callee->window;

    if (new->packet_size < callee->packet_size)
        new->packet_size = callee->packet_size;

    if (dte->throughput < callee->dte_throughput)
        dte->throughput = callee->dte_throughput;

    if (dte->window < callee->dte_window)
        dte->window = callee->dte_window;

    if (dte->packet_size < callee->dte_packet_size)
        dte->packet_size = callee->dte_packet_size;

    X25_SUBLOG("Negotiated throughput: %d, window: %d, packet size: %d\n",
               new->throughput, new->window, new->packet_size);

    return len;
}
