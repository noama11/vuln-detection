ssize_t read_sound_timer(struct file *file, char __user *buffer, size_t count, loff_t *offset) {
    struct sound_timer_data *data = file->private_data;
    ssize_t bytes_read = 0;
    int ret;

    if (!data) {
        return -EINVAL;
    }

    spin_lock(&data->lock);

    while (bytes_read < count) {
        if (kthread_should_stop() || data->disconnected) {
            ret = -ENODEV;
            goto unlock_exit;
        }

        if (data->queue_head == data->queue_tail) {
            if (file->f_flags & O_NONBLOCK) {
                ret = -EAGAIN;
                goto unlock_exit;
            }

            ret = wait_event_interruptible(data->wait_queue,
                                           data->queue_head != data->queue_tail);
            if (ret < 0) {
                goto unlock_exit;
            }
            continue;
        }

        size_t bytes_available = data->queue[data->queue_tail].size;
        size_t bytes_to_copy = min(count - bytes_read, bytes_available);

        if (copy_to_user(buffer + bytes_read, data->queue[data->queue_tail].data, bytes_to_copy)) {
            ret = -EFAULT;
            goto unlock_exit;
        }

        bytes_read += bytes_to_copy;
        data->queue_tail = (data->queue_tail + 1) % SOUND_TIMER_QUEUE_SIZE;
    }

unlock_exit:
    spin_unlock(&data->lock);
    return (ret < 0) ? ret : bytes_read;
}
