int bind_pptp_socket(struct socket *sock, struct sockaddr *uservaddr, int sockaddr_len) {
    int err = 0;

    if (sockaddr_len < sizeof(struct sockaddr_in)) {
        return -EINVAL;
    }

    lock_sock(sock);

    memcpy(&sock->sk->sk_srcaddr, uservaddr, sockaddr_len);
    err = add_pptp_channel(sock);

    if (err < 0) {
        return err;
    }

    release_sock(sock);

    return 0;
}
