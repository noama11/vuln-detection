char *read_next_line(FILE *VFile, char *ptr) {
    if (fgets(ptr, MAX_LINE_LENGTH, VFile) == NULL) {
        return NULL;
    }

    // Remove the newline character if present
    size_t len = strlen(ptr);
    if (len > 0 && ptr[len - 1] == '\n') {
        ptr[len - 1] = '\0';
    }

    return ptr;
}
