MagickBooleanType WritePSDImage(const PSDInfo *psd_info, const ImageInfo *image_info, Image *image, Image *next_image, MagickBooleanType separate, ExceptionInfo *exception) {
  MagickBooleanType status = MagickTrue;
  size_t number_pixels;
  Quantum *pixels = NULL;
  ssize_t y;

  number_pixels = image->columns * image->rows;

  if (image->storage_class == PseudoClass)
    (void) SetImageColorspace(image, sRGBColorspace, exception);

  if (image->alpha_trait != UndefinedAlpha)
    image->alpha_trait = BlendAlpha;

  pixels = (Quantum *) AcquireQuantumMemory(number_pixels, sizeof(*pixels));
  if (pixels == (Quantum *) NULL)
    return MagickFalse;

  for (y = 0; y < (ssize_t) image->rows; y++) {
    if (image->compression == RLECompression)
      status = WriteRLEPixelData(psd_info, image_info, image, pixels, y, exception);
    else
      status = WriteRawPixelData(psd_info, image_info, image, pixels, y, exception);

    if (status == MagickFalse)
      break;
  }

  pixels = (Quantum *) RelinquishMagickMemory(pixels);
  return status;
}
