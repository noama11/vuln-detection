int process_ticket_reply(struct ceph_auth_client *ac, struct ceph_crypto_key *secret, void *buf, void *end) {
    int ret = 0;
    int version;
    int num_tickets;
    void **p = &buf;

    if ((ret = ceph_decode_version(p, end, &version)) < 0)
        return ret;

    if (version != 1) {
        return -EINVAL;
    }

    if ((ret = ceph_decode_num(&num_tickets, p, end)) < 0)
        return ret;

    for (int i = 0; i < num_tickets; i++) {
        ret = process_one_ticket(ac, secret, p, end);
        if (ret < 0) {
            return ret;
        }
    }

    return 0;
}
