CURLcode generate_plain_auth_message(struct Curl_easy *data, const char *userp, const char *passwdp, char **outptr, size_t *outlen) {
  size_t user_len = strlen(userp);
  size_t pass_len = strlen(passwdp);
  size_t plain_len = user_len + 1 + pass_len + 1 + pass_len;
  size_t encoded_len = Curl_base64_encode_len(plain_len);
  char *plain = NULL;
  char *encoded = NULL;
  CURLcode result = CURLE_OK;

  if (encoded_len == 0) {
    return CURLE_OUT_OF_MEMORY;
  }

  plain = malloc(plain_len);
  if (!plain) {
    return CURLE_OUT_OF_MEMORY;
  }

  snprintf(plain, plain_len, "%s\0%s\0%s", userp, userp, passwdp);

  encoded = Curl_base64_encode(data, plain, plain_len);
  if (!encoded) {
    result = CURLE_OUT_OF_MEMORY;
    goto cleanup;
  }

  *outptr = encoded;
  *outlen = encoded_len;

cleanup:
  free(plain);
  return result;
}
