int sit_init_net(struct net *net)
{
    struct sit_net *sit = net_generic(net, sit_net_id);
    struct net_device *dev;
    int err;

    if (!sit)
        return -ENOMEM;

    sit->tunnels = NULL;
    sit->fb_tunnel_dev = NULL;

    dev = alloc_netdev(sizeof(struct ip_tunnel), "sit%d", NET_NAME_UNKNOWN, ip_tunnel_setup);
    if (!dev)
        return -ENOMEM;

    dev->rtnl_link_ops = &sit_link_ops;
    dev->netdev_ops = &sit_netdev_ops;
    dev->needs_free_netdev = true;

    err = register_netdev(dev);
    if (err) {
        free_netdev(dev);
        return err;
    }

    sit->fb_tunnel_dev = dev;
    return 0;
}
