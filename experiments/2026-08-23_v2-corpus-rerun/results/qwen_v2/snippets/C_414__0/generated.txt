MagickBooleanType WriteMAPImage(const ImageInfo *image_info, Image *image)
{
  ExceptionInfo
    *exception;

  FILE
    *file;

  MagickBooleanType
    status;

  MapInfo
    *map_info;

  register IndexPacket
    *indexes;

  register PixelPacket
    *p;

  size_t
    count,
    number_pixels;

  StringInfo
    *colormap;

  unsigned char
    *pixels;

  if ((image->columns == 0) || (image->rows == 0))
    return(MagickFalse);

  exception = image->exception;
  status = MagickTrue;
  number_pixels = image->columns * image->rows;

  if (image->storage_class != PseudoClass)
    (void) TransformImageColorspace(image, sRGBColorspace);

  if (image->colors > 256)
    {
      (void) ThrowMagickException(exception, GetMagickModule(), ImageError,
        "InsufficientColormapEntries","`%s'", image->filename);
      return(MagickFalse);
    }

  if ((file = fopen(image->filename, "wb")) == (FILE *) NULL)
    {
      (void) ThrowMagickException(exception, GetMagickModule(), FileError,
        "UnableToCreate","%s", image->filename);
      return(MagickFalse);
    }

  map_info = (MapInfo *) AcquireMagickMemory(sizeof(*map_info));
  if (map_info == (MapInfo *) NULL)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
        "UnableToAllocateMemory","`%s'", image->filename);
      return(MagickFalse);
    }

  colormap = (StringInfo *) AcquireMagickMemory(sizeof(*colormap));
  if (colormap == (StringInfo *) NULL)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
        "UnableToAllocateMemory","`%s'", image->filename);
      return(MagickFalse);
    }

  pixels = (unsigned char *) AcquireQuantumMemory(image->colors,sizeof(*pixels));
  if (pixels == (unsigned char *) NULL)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
        "UnableToAllocateMemory","`%s'", image->filename);
      return(MagickFalse);
    }

  for (count = 0; count < image->colors; count++)
    pixels[count] = (unsigned char) count;

  if (fwrite(pixels, sizeof(*pixels), image->colors, file) != image->colors)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), FileError,
        "UnableToWriteFile","%s", image->filename);
      return(MagickFalse);
    }

  p = AcquireImagePixels(image,0,0,image->columns,image->rows,exception);
  if (p == (PixelPacket *) NULL)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
        "UnableToAllocateMemory","`%s'", image->filename);
      return(MagickFalse);
    }

  indexes = (IndexPacket *) AcquireQuantumMemory(image->columns*image->rows,
    sizeof(*indexes));
  if (indexes == (IndexPacket *) NULL)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
        "UnableToAllocateMemory","`%s'", image->filename);
      return(MagickFalse);
    }

  for (count = 0; count < number_pixels; count++)
    indexes[count] = p->index;

  if (fwrite(indexes, sizeof(*indexes), number_pixels, file) != number_pixels)
    {
      (void) fclose(file);
      (void) ThrowMagickException(exception, GetMagickModule(), FileError,
        "UnableToWriteFile","%s", image->filename);
      return(MagickFalse);
    }

  (void) fclose(file);
  return(status);
}
