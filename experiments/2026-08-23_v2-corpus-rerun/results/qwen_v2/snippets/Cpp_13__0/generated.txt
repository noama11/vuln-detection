ByteVector extract_subrange(uint index, uint length) {
    if (index >= data.size()) {
        return ByteVector();
    }

    uint end_index = std::min(index + length, static_cast<uint>(data.size()));
    ByteVector result;
    result.data.insert(result.data.begin(), data.begin() + index, data.begin() + end_index);
    return result;
}
