int read_quant_matrices(MpegEncContext *s, GetBitContext *gb) {
    int i, j, matrix_index;
    int matrix_data[64];

    // Check for intra matrix
    if (get_bits1(gb)) {
        for (i = 0; i < 64; i++) {
            if (get_remaining_bits(gb) < 8) {
                return AVERROR_INVALIDDATA;
            }
            matrix_data[i] = get_bits8(gb);
        }
        for (i = 0; i < 64; i++) {
            s->intra_matrix[i] = matrix_data[i];
        }
    }

    // Check for non-intra matrix
    if (get_bits1(gb)) {
        for (i = 0; i < 64; i++) {
            if (get_remaining_bits(gb) < 8) {
                return AVERROR_INVALIDDATA;
            }
            matrix_data[i] = get_bits8(gb);
        }
        for (i = 0; i < 64; i++) {
            s->non_intra_matrix[i] = matrix_data[i];
        }
    }

    // Check for chroma intra matrix
    if (get_bits1(gb)) {
        for (i = 0; i < 64; i++) {
            if (get_remaining_bits(gb) < 8) {
                return AVERROR_INVALIDDATA;
            }
            matrix_data[i] = get_bits8(gb);
        }
        for (i = 0; i < 64; i++) {
            s->chroma_intra_matrix[i] = matrix_data[i];
        }
    }

    // Check for chroma non-intra matrix
    if (get_bits1(gb)) {
        for (i = 0; i < 64; i++) {
            if (get_remaining_bits(gb) < 8) {
                return AVERROR_INVALIDDATA;
            }
            matrix_data[i] = get_bits8(gb);
        }
        for (i = 0; i < 64; i++) {
            s->chroma_non_intra_matrix[i] = matrix_data[i];
        }
    }

    // Advance to the next start code
    skip_bits(gb, 8 - (get_bits_count(gb) & 7));
    while (get_bits1(gb)) {
        if (get_bits_count(gb) >= get_bits_left(gb) * 8) {
            break;
        }
    }

    return 0;
}
