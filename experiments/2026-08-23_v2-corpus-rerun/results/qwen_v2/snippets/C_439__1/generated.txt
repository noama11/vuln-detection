int create_graphics_buffer_surface(struct drm_device *dev, void *data, struct drm_file *file_priv) {
    int ret = -EINVAL;
    struct drm_mode_create_dumb args;
    struct drm_gem_object *obj = NULL;
    struct drm_plane *plane = NULL;

    if (!dev || !data || !file_priv)
        return ret;

    memcpy(&args, data, sizeof(args));

    obj = drm_gem_object_create(dev, args.size);
    if (IS_ERR(obj)) {
        ret = PTR_ERR(obj);
        goto out;
    }

    plane = kzalloc(sizeof(*plane), GFP_KERNEL);
    if (!plane) {
        ret = -ENOMEM;
        goto out_unref;
    }

    plane->possible_crtcs = dev->mode_config.crtc_mask;
    drm_plane_init(dev, plane, DRM_PLANE_TYPE_PRIMARY);

    ret = drm_mode_create_dumb(dev, &args, file_priv);
    if (ret)
        goto out_free_plane;

    return 0;

out_free_plane:
    drm_plane_cleanup(plane);
    kfree(plane);
out_unref:
    drm_gem_object_put(obj);
out:
    return ret;
}
