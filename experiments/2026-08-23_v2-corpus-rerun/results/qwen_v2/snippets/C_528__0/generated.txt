int git_pkt_parse(git_pkt **head, const char *line, const char **out, size_t bufflen) {
    if (bufflen < 4)
        return GIT_EBUFS;

    if (!strncmp(line, "data", 4)) {
        return git_pkt_data(head, line + 4, out, bufflen - 4);
    } else if (!strncmp(line, "sideband", 8)) {
        return git_pkt_sideband(head, line + 8, out, bufflen - 8);
    } else if (!strncmp(line, "ack", 3)) {
        return git_pkt_ack(head, line + 3, out, bufflen - 3);
    } else if (!strncmp(line, "flush", 5)) {
        *head = git_pkt_flush();
        *out = line + 5;
        return 0;
    } else {
        return GIT_ERROR;
    }
}
