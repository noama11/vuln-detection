void
process_bgp_update(netdissect_options *ndo, const u_char *dat, int length)
{
    const u_char *cp;
    int total_length;
    int rem;
    u_int16_t withdrawn_routes_length;
    u_int16_t path_attributes_length;

    ND_PRINT((ndo, "BGP Update Message: "));

    if (length < BGP_HDR_LEN) {
        ND_PRINT((ndo, "truncated BGP update message"));
        return;
    }

    cp = dat + BGP_HDR_LEN;
    rem = length - BGP_HDR_LEN;

    if (rem < 4) {
        ND_PRINT((ndo, "truncated BGP update message"));
        return;
    }

    withdrawn_routes_length = GET_U_2_BYTES(cp);
    cp += 2;
    rem -= 2;

    if (withdrawn_routes_length > rem) {
        ND_PRINT((ndo, "truncated BGP update message"));
        return;
    }

    ND_PRINT((ndo, "Withdrawn Routes Length: %u bytes", withdrawn_routes_length));
    cp += withdrawn_routes_length;
    rem -= withdrawn_routes_length;

    if (rem < 2) {
        ND_PRINT((ndo, "truncated BGP update message"));
        return;
    }

    path_attributes_length = GET_U_2_BYTES(cp);
    cp += 2;
    rem -= 2;

    if (path_attributes_length > rem) {
        ND_PRINT((ndo, "truncated BGP update message"));
        return;
    }

    ND_PRINT((ndo, "Path Attributes Length: %u bytes", path_attributes_length));
    cp += path_attributes_length;
    rem -= path_attributes_length;

    if (rem != 0) {
        ND_PRINT((ndo, "NLRI Length: %u bytes", rem));
    } else {
        ND_PRINT((ndo, "Empty NLRI"));
    }
}
