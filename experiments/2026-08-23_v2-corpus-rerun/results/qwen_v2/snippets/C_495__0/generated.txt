int update_inode_acl(struct inode *inode, struct posix_acl *acl, int type) {
    int ret;

    if (type == ACL_TYPE_ACCESS) {
        ret = posix_acl_update_mode(inode, &inode->i_mode, &acl);
        if (ret)
            return ret;
    }

    inode->i_ctime = current_time(inode);
    inode->i_acl = acl;
    return 0;
}
