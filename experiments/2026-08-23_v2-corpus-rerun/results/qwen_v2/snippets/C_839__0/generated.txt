void cleanup_keys(struct list_head *keys) {
    struct key *pos, *next;
    list_for_each_entry_safe(pos, next, keys, list) {
        list_del(&pos->list);
        if (pos->type->destroy && !pos->negative) {
            pos->type->destroy(pos);
        }
        if (pos->uid) {
            atomic_sub(1, &pos->uid->keys);
            atomic_sub(pos->bytes, &pos->uid->key_bytes);
        }
        kfree(pos->description);
        debug_log("Key %d cleaned up", pos->id);
        kfree(pos);
    }
}
