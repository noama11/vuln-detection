struct sk_buff *reassemble_ipv6_fragments(struct nf_ct_frag6_queue *fq, struct net_device *dev) {
    struct sk_buff *skb;
    int payload_len;
    int i;

    if (!fq || !dev) {
        return NULL;
    }

    payload_len = 0;
    for (i = 0; i < fq->q.pf.holes; i++) {
        payload_len += fq->q.fragments[i]->len;
    }

    if (payload_len == 0) {
        return NULL;
    }

    skb = alloc_skb(payload_len, GFP_ATOMIC);
    if (!skb) {
        pr_debug("Memory allocation failed for reassembly.\n");
        return NULL;
    }

    for (i = 0; i < fq->q.pf.holes; i++) {
        struct sk_buff *frag = fq->q.fragments[i];
        int frag_len = frag->len;

        if (skb_tailroom(skb) < frag_len) {
            kfree_skb(skb);
            pr_debug("Insufficient space in skb for fragment.\n");
            return NULL;
        }

        skb_put_data(skb, frag->data, frag_len);
    }

    skb->dev = dev;
    skb->protocol = htons(ETH_P_IPV6);

    ip6_recalc(skb);

    return skb;
}
