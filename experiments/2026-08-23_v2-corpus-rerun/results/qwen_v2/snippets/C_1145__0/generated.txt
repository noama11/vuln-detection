void format_text_selection(const VisiblePosition& startOfSelection, const VisiblePosition& endOfSelection) {
    if (is_unsplittable_element(startOfSelection)) {
        create_block_element(startOfSelection);
        insert_placeholder(startOfSelection);
    }

    VisiblePosition currentPos = startOfSelection;
    while (!currentPos.is_equal(endOfSelection)) {
        if (is_paragraph(currentPos)) {
            format_paragraph(currentPos);
            if (is_table_cell(currentPos)) {
                ensure_selection_validity(currentPos);
            }
        }
        currentPos = move_to_next_position(currentPos);
    }

    perform_sanity_checks();
}
