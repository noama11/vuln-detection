int git_pkt_parse(git_pkt **out, const char *line, size_t len) {
    git_pkt_ng *pkt;
    char *ref, *msg;
    size_t ref_len, msg_len;
    const char *end;

    pkt = (git_pkt_ng *)malloc(sizeof(git_pkt_ng));
    if (!pkt)
        return -1;

    // Parse reference
    ref = (char *)line;
    end = strchr(ref, '\0');
    if (!end)
        goto error;

    ref_len = end - ref;
    pkt->ref = (char *)malloc(ref_len + 1);
    if (!pkt->ref)
        goto error;

    memcpy(pkt->ref, ref, ref_len);
    pkt->ref[ref_len] = '\0';

    // Parse message
    msg = (char *)malloc(len - ref_len + 1);
    if (!msg)
        goto error;

    memcpy(msg, line + ref_len + 1, len - ref_len - 1);
    msg[len - ref_len - 1] = '\0';
    pkt->msg = msg;

    *out = (git_pkt *)pkt;
    return 0;

error:
    if (pkt->ref)
        free(pkt->ref);
    if (pkt->msg)
        free(pkt->msg);
    free(pkt);
    return -1;
}
