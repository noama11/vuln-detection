CustomStreamInfo *destroy_custom_stream_info(CustomStreamInfo *custom_stream) {
    if (custom_stream == NULL) {
        return NULL;
    }

    // Log event
    log_event("Destroying CustomStreamInfo object.");

    // Check for valid signature
    if (custom_stream->signature != VALID_SIGNATURE) {
        return NULL;
    }

    // Modify signature to indicate invalid
    custom_stream->signature = INVALID_SIGNATURE;

    // Release memory
    free(custom_stream);

    return custom_stream;
}

void *detach_blob(BlobInfo *blob_info) {
    // Assert that blob_info is not NULL
    assert(blob_info != NULL);

    // Log event if debugging is enabled
    if (DEBUG_ENABLED) {
        log_event("Detaching blob from BlobInfo.");
    }

    void *blob_data = blob_info->blob_data;

    if (blob_info->is_mapped) {
        unmap_blob_data(blob_data);
        blob_info->is_mapped = false;
    }

    blob_info->blob_data = NULL;

    return blob_data;
}
