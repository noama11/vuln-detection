int skb_copy_datagram_from_iovec(struct sk_buff *skb, const struct iovec *from, int offset, size_t count)
{
    size_t len = 0;
    int i;

    for (i = 0; i < count; i++) {
        len += from[i].iov_len;
    }

    if (len == 0)
        return 0;

    if (offset >= len)
        return 0;

    if (offset + skb->data_len > len)
        return -EFAULT;

    for (i = 0; i < count && offset; i++) {
        if (offset < from[i].iov_len)
            break;
        offset -= from[i].iov_len;
    }

    while (i < count && len > 0) {
        size_t copy = from[i].iov_len - offset;
        if (copy > len)
            copy = len;

        if (copy > skb_headroom(skb))
            return -EFAULT;

        if (copy > 0) {
            char *data = skb_push(skb, copy);
            if (!data)
                return -EFAULT;

            if (copy_from_user(data, (void __user *)((unsigned long)from[i].iov_base + offset), copy))
                return -EFAULT;

            skb->len += copy;
            skb->data_len += copy;
            len -= copy;
            offset = 0;
        }

        i++;
    }

    return 0;
}
