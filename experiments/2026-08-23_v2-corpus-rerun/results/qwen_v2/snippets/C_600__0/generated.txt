int build_indirect_sgl(Sg_scatter_hold *schp, Sg_fd *sfp, int buff_size) {
    int ret = 0;
    unsigned int sector_size = 512; // Assuming standard sector size
    unsigned int aligned_size = ((buff_size + sector_size - 1) / sector_size) * sector_size;

    if (aligned_size < buff_size)
        return -ENOMEM;

    schp->k_use = 0;
    schp->k_nents = 0;

    while (schp->k_nents < SG_DEF_MAX) {
        void *page = kmalloc(aligned_size, GFP_KERNEL);
        if (!page) {
            if (schp->k_nents == 0)
                return -ENOMEM;
            break;
        }

        sg_set_page(&schp->k_sgt[schp->k_nents], page, aligned_size, 0);
        schp->k_nents++;
        schp->k_use += aligned_size;
    }

    if (schp->k_nents == 0)
        return -ENOMEM;

    return 0;
}
