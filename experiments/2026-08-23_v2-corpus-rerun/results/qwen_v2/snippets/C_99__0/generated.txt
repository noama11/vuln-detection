void edgeport_drain_data(struct edgeport_port *port, unsigned long timeout, int flush)
{
    struct tty_struct *tty;
    wait_queue_t wait;
    unsigned long flags;
    int discon = 0;

    if (!port)
        return;

    tty = port->port.tty;
    if (!tty)
        return;

    init_waitqueue_entry(&wait, current);

    add_wait_queue(&port->drain_wait, &wait);

    while (1) {
        set_current_state(TASK_INTERRUPTIBLE);

        if (port->disconnected)
            discon = 1;

        if (!tty->flip.count && !port->write_qlen && (port->port.flags & ASYNC_CLOSING) &&
            !port->write_qlen && !port->port.xmit_fifo_size)
            break;

        if (timeout) {
            if (signal_pending(current))
                break;
            schedule_timeout(timeout);
            if (port->disconnected)
                discon = 1;
        } else {
            if (signal_pending(current))
                break;
            schedule();
        }
    }

    set_current_state(TASK_RUNNING);
    remove_wait_queue(&port->drain_wait, &wait);

    if (discon)
        port->disconnected = 1;

    if (flush) {
        spin_lock_irqsave(&port->port.lock, flags);
        port->port.xmit_fifo_size = 0;
        spin_unlock_irqrestore(&port->port.lock, flags);
    }

    if (port->port.flags & ASYNC_CLOSING)
        wait_for_xmit_chars(tty);
}
