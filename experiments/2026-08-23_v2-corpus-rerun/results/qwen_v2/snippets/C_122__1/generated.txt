#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

// Assuming config_exclusive_filename is a global variable defined elsewhere
extern char *config_exclusive_filename;

// Placeholder for the function signature
int load_performance_configurations(config_fn_t fn, void *data) {
    int count = 0;
    char *home_dir = getenv("HOME");
    char system_config_path[] = "/etc/performance.conf";
    char user_config_path[1024];

    // Check if an exclusive configuration file is specified
    if (config_exclusive_filename && strlen(config_exclusive_filename) > 0) {
        if (access(config_exclusive_filename, R_OK) == 0) {
            FILE *file = fopen(config_exclusive_filename, "r");
            if (file) {
                count += fn(file, data);
                fclose(file);
            }
        }
    }

    // Check for system-wide configuration file
    if (access(system_config_path, R_OK) == 0) {
        FILE *file = fopen(system_config_path, "r");
        if (file) {
            count += fn(file, data);
            fclose(file);
        }
    }

    // Check for user-specific configuration file
    if (home_dir && strlen(home_dir) > 0) {
        snprintf(user_config_path, sizeof(user_config_path), "%s/.performance.conf", home_dir);
        if (access(user_config_path, R_OK) == 0) {
            FILE *file = fopen(user_config_path, "r");
            if (file) {
                count += fn(file, data);
                fclose(file);
            }
        }
    }

    return (count > 0) ? count : -1;
}
