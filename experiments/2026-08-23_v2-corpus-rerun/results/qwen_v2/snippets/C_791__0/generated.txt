GF_Err parse_hex_to_bin128(const char *string, bin128 value) {
    if (!string) return GF_BAD_PARAM;

    const char *hex_str = string;
    if (hex_str[0] == '0' && hex_str[1] == 'x') {
        hex_str += 2;
    }

    size_t len = strlen(hex_str);
    if (len % 2 != 0) {
        return GF_BAD_PARAM;
    }

    for (size_t i = 0; i < len; i += 2) {
        char byte_str[3] = { hex_str[i], hex_str[i + 1], '\0' };
        char *end;
        unsigned int nibble = (unsigned int)strtol(byte_str, &end, 16);
        if (*end != '\0') {
            return GF_BAD_PARAM;
        }
        value[i / 2] = (uint8_t)nibble;
    }

    if (len / 2 != 16) {
        return GF_BAD_PARAM;
    }

    return GF_OK;
}
