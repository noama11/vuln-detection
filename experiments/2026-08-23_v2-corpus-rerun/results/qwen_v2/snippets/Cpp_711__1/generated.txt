std::string computeUValue(std::string const& user_password, QPDF::EncryptionData const& data) {
    // Retrieve the encryption key using the password and encryption data
    std::string key = getEncryptionKey(user_password, data);

    // Prepare buffer for the U value
    std::vector<unsigned char> buffer(data.getInitializationVector().begin(),
                                      data.getInitializationVector().end());

    // Process the key with RC4 encryption
    rc4_encrypt(buffer, key);

    // Convert the buffer to a hexadecimal string
    std::ostringstream oss;
    for (unsigned char byte : buffer) {
        oss << std::hex << std::setw(2) << std::setfill('0') << static_cast<int>(byte);
    }

    return oss.str();
}
