MagickBooleanType WriteMAPImage(const ImageInfo *image_info, Image *image)
{
  ColormapType
    *colormap;

  ExceptionInfo
    *exception;

  FILE
    *file;

  MagickBooleanType
    status;

  register ssize_t
    i;

  size_t
    length;

  StringInfo
    *profile;

  assert(image_info != (ImageInfo *) NULL);
  assert(image != (Image *) NULL);
  assert(image->signature == MagickSignature);
  if (image->debug != MagickFalse)
    (void) DebugTraceImage("enter", "WriteMAPImage", image);
  status = MagickTrue;
  exception = &image->exception;
  if ((file = fopen(image_info->filename,"wb")) == (FILE *) NULL)
  {
    (void) ThrowMagickException(exception,GetMagickModule(),FileOpenError,
      "Unable to open file","`%s'",image_info->filename);
    return(MagickFalse);
  }
  if (image->colors > 256)
    (void) SetImageColorspace(image,sRGBColorspace);
  colormap = AcquireColormap(image,256);
  if (colormap == (ColormapType *) NULL)
  {
    (void) fclose(file);
    (void) ThrowMagickException(exception,GetMagickModule(),ResourceLimitError,
      "Unable to allocate colormap","`%s'",image_info->filename);
    return(MagickFalse);
  }
  length = 256 * sizeof(PixelPacket);
  (void) fwrite(colormap,sizeof(*colormap),length,file);
  for (i=0; i < (ssize_t) image->rows; i++)
  {
    register PixelPacket
      *p;

    p = GetImagePixels(image,0,i,image->columns,1);
    if (p == (PixelPacket *) NULL)
      break;
    for ( ; image->x != image->columns; image->x++)
    {
      register ssize_t
        x;

      for (x=0; x < (ssize_t) image->columns; x++)
      {
        register ssize_t
          y;

        for (y=0; y < (ssize_t) image->rows; y++)
        {
          register ssize_t
            z;

          for (z=0; z < (ssize_t) image->depth; z++)
          {
            register ssize_t
              index;

            index = (ssize_t) (GetPixelRed(image,p) * 255.0 + 0.5);
            if (index < 0)
              index = 0;
            if (index > 255)
              index = 255;
            (void) fwrite(&index,sizeof(index),1,file);
          }
        }
      }
    }
    if (SyncImagePixels(image) == MagickFalse)
      break;
  }
  (void) fclose(file);
  if (image->debug != MagickFalse)
    (void) DebugTraceImage("exit", "WriteMAPImage", image);
  return(status);
}
