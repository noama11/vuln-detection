int set_auth_parameters(struct net_device *dev, struct iw_request_info *info,
                        union iwreq_data *wrqu, char *extra)
{
    int ret = 0;
    struct orinoco_private *priv = (struct orinoco_private *)dev->priv;

    /* Lock the device for thread-safe access */
    spin_lock_irq(&priv->lock);

    switch (wrqu->data.flags) {
    case IW_AUTH_KEY_MGMT:
        priv->key_mgmt = wrqu->data.value;
        break;
    case IW_AUTH_TKIP_COUNTERMEASURES:
        priv->tkip_countermeasures = wrqu->data.value;
        break;
    case IW_AUTH_ALG:
        if (wrqu->data.value & IW_AUTH_ALG_OPEN_SYSTEM)
            priv->auth_alg = IW_AUTH_ALG_OPEN_SYSTEM;
        else if (wrqu->data.value & IW_AUTH_ALG_SHARED_KEY)
            priv->auth_alg = IW_AUTH_ALG_SHARED_KEY;
        else if (wrqu->data.value & IW_AUTH_ALG_LEAP)
            priv->auth_alg = IW_AUTH_ALG_LEAP;
        else
            ret = -EINVAL;
        break;
    case IW_AUTH_WPA_VERSION:
        if (wrqu->data.value & IW_AUTH_WPA_VERSION_1)
            priv->wpa_version = IW_AUTH_WPA_VERSION_1;
        else if (wrqu->data.value & IW_AUTH_WPA_VERSION_2)
            priv->wpa_version = IW_AUTH_WPA_VERSION_2;
        else
            ret = -EINVAL;
        break;
    default:
        ret = -EOPNOTSUPP;
        break;
    }

    /* Unlock the device */
    spin_unlock_irq(&priv->lock);

    return ret;
}
