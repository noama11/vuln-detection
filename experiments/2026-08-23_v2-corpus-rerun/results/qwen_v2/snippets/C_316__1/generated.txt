enum integrity_status verify_hmac_xattr(struct dentry *dentry, const char *xattr_name,
                                        char *xattr_value, size_t xattr_value_len,
                                        struct integrity_iint_cache *iint)
{
    enum integrity_status status;
    void *digest;
    size_t digest_len;
    int rc;

    if (iint->flags & INTEGRITY_SIGNATURE_PRESENT)
        return INTEGRITY_STATUS_NO_CHANGE;

    digest = get_xattr_digest(dentry, xattr_name, &digest_len);
    if (IS_ERR_OR_NULL(digest))
        return INTEGRITY_STATUS_NO_ATTRIBUTE;

    rc = verify_hmac_or_hash(xattr_value, xattr_value_len, digest, digest_len);
    if (rc != 0) {
        status = INTEGRITY_STATUS_FAILURE;
    } else {
        status = INTEGRITY_STATUS_SUCCESS;
    }

    kfree(digest);
    return status;
}
