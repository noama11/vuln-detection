bool PP_ImageDataInit(PP_ImageDataFormat format, int width, int height, bool init_to_zero) {
    if (format < 0 || format >= PP_IMAGEDATAFORMAT_COUNT) {
        return false;
    }

    if (width <= 0 || height <= 0) {
        return false;
    }

    // Check for potential overflow in size calculation
    if ((unsigned int)width > ((unsigned int)~0 / (unsigned int)height)) {
        return false;
    }

    // Set member variables (example placeholder logic)
    imageData.format = format;
    imageData.width = width;
    imageData.height = height;
    imageData.init_to_zero = init_to_zero;

    return backend_Init(&imageData);
}
