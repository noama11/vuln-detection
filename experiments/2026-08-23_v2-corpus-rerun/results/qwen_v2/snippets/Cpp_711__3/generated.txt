bool checkUserPassword(std::string& user_password, const std::string& owner_password, const QPDF::EncryptionData& data) {
    // Compute RC4 key based on user and owner passwords
    std::string rc4_key = computeRC4Key(user_password, owner_password);

    // Retrieve O data from encryption data
    const std::string& o_data = data.o;

    // Iterate the RC4 algorithm to generate a new user password
    std::string new_user_password = applyRC4Iteration(rc4_key, o_data);

    // Check if the new password matches the provided data
    if (new_user_password == data.u) {
        user_password = new_user_password;
        return true;
    }

    return false;
}
