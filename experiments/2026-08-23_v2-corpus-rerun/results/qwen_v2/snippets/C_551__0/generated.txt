#include <stdio.h>
#include <stdlib.h>
#include <stdarg.h>
#include <wchar.h>

void safe_fprintf(FILE *f, const char *fmt, ...) {
    char buffer[1024];
    va_list args;
    va_start(args, fmt);
    int result = vsnprintf(buffer, sizeof(buffer), fmt, args);
    va_end(args);

    if (result < 0 || result >= (int)sizeof(buffer)) {
        // Buffer was too small, allocate a larger one
        size_t size = (size_t)(result < 0 ? 2048 : result + 1);
        char *heap_buffer = malloc(size);
        if (!heap_buffer) {
            // Handle allocation failure gracefully
            fputs("Memory allocation failed.\n", stderr);
            return;
        }

        va_start(args, fmt);
        result = vsnprintf(heap_buffer, size, fmt, args);
        va_end(args);

        if (result < 0) {
            free(heap_buffer);
            fputs("Failed to format the string.\n", stderr);
            return;
        }

        // Check for printable wide characters and expand unprintable ones
        size_t i = 0;
        while (i < size) {
            wchar_t wc;
            char *p = &heap_buffer[i];
            mbtowc(&wc, p, MB_CUR_MAX);
            if (iswprint(wc)) {
                fputc((unsigned char)p[0], f);
                i += 1;
            } else {
                // Handle unprintable characters
                for (int j = 0; j < 4 && p[j]; ++j) {
                    fprintf(f, "\\x%02x", (unsigned char)p[j]);
                }
                i += 1;
            }
        }

        free(heap_buffer);
    } else {
        // Check for printable wide characters and expand unprintable ones
        size_t i = 0;
        while (i < sizeof(buffer)) {
            wchar_t wc;
            char *p = &buffer[i];
            mbtowc(&wc, p, MB_CUR_MAX);
            if (iswprint(wc)) {
                fputc((unsigned char)p[0], f);
                i += 1;
            } else {
                // Handle unprintable characters
                for (int j = 0; j < 4 && p[j]; ++j) {
                    fprintf(f, "\\x%02x", (unsigned char)p[j]);
                }
                i += 1;
            }
        }
    }
}
