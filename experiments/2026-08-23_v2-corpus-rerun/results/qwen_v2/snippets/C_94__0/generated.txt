struct block_device *get_block_device(dev_t dev, struct super_block *sb) {
    struct block_device *bdev = NULL;
    int ret;

    bdev = blkdev_get(dev, FMODE_READ | FMODE_WRITE | FMODE_EXCL, sb);
    if (IS_ERR_OR_NULL(bdev)) {
        pr_err("Failed to get block device for dev %d\n", dev);
        return NULL;
    }

    return bdev;
}
