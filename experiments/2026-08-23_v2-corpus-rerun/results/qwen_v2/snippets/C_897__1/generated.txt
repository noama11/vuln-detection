OPJ_BOOL read_rle4_bitmap(FILE *IN, OPJ_UINT8 *pData, OPJ_UINT32 stride, OPJ_UINT32 width, OPJ_UINT32 height) {
    OPJ_UINT32 expected_pixels = width * height;
    OPJ_UINT32 written_pixels = 0;
    OPJ_UINT8 *row_start = pData;
    OPJ_UINT8 byte;

    while (written_pixels < expected_pixels && fread(&byte, sizeof(OPJ_UINT8), 1, IN) == 1) {
        if (byte & 0x80) {
            /* Absolute mode: next byte is literal data for the next 2 pixels */
            OPJ_UINT8 literal;
            if (fread(&literal, sizeof(OPJ_UINT8), 1, IN) == 1) {
                pData[0] = (literal >> 4) & 0x0F;
                pData[1] = literal & 0x0F;
                pData += 2;
                written_pixels += 2;
            } else {
                return OPJ_FALSE;
            }
        } else {
            /* Encoded mode: byte is run length (low 4 bits) of the value (high 4 bits) */
            OPJ_UINT8 run_length = byte & 0x0F;
            OPJ_UINT8 value = (byte >> 4) & 0x0F;
            for (OPJ_UINT32 i = 0; i <= run_length; ++i) {
                *pData++ = value;
                written_pixels++;
                if (written_pixels % width == 0 && i < run_length) {
                    /* If we reach the end of a row, pad the row to the next stride */
                    pData = row_start + ((written_pixels / width) * stride);
                }
            }
        }
    }

    if (written_pixels != expected_pixels) {
        return OPJ_FALSE;
    }

    return OPJ_TRUE;
}
