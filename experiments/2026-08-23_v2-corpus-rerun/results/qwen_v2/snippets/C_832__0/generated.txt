int copy_templates_to_skb(struct xfrm_policy *xp, struct sk_buff *skb) {
    int i, len = 0;
    struct xfrm_tmpl *tmpls = xp->tp;
    int num_tmpls = xp->num_tmpl;

    if (!num_tmpls)
        return 0;

    for (i = 0; i < num_tmpls; i++) {
        struct xfrm_user_tmpl utmpl;

        utmpl = (struct xfrm_user_tmpl){
            .id = tmpls[i].id,
            .saddr = tmpls[i].saddr,
            .family = tmpls[i].family,
            .mode = tmpls[i].mode,
            .reqid = tmpls[i].reqid,
            .lifetime = tmpls[i].lifetime,
            .aalgos = tmpls[i].aalgos,
            .ealgos = tmpls[i].ealgos,
            .flags = tmpls[i].flags,
        };

        len += nla_put(skb, XFRMA_TMPL, sizeof(utmpl), &utmpl);
    }

    return len;
}
