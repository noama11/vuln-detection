int handle_sysenter(struct x86_emulate_ctxt *ctxt) {
    uint64_t efer;

    if (ctxt->mode == X86EMUL_MODE_REAL) {
        return inject_gp_fault(ctxt, 0);
    }

    if (!ctxt->ops->get_msr(ctxt, MSR_EFER, &efer)) {
        return inject_ud_fault(ctxt);
    }

    if ((ctxt->mode != X86EMUL_MODE_PROT) && (efer & EFER_SCE) == 0) {
        return inject_ud_fault(ctxt);
    }

    uint64_t cs, eip, esp;
    if (!ctxt->ops->get_msr(ctxt, MSR_IA32_SYSENTER_CS, &cs) ||
        !ctxt->ops->get_msr(ctxt, MSR_IA32_SYSENTER_EIP, &eip) ||
        !ctxt->ops->get_msr(ctxt, MSR_IA32_SYSENTER_ESP, &esp)) {
        return inject_gp_fault(ctxt, 0);
    }

    ctxt->regs->cs = cs;
    ctxt->regs->ip = eip;
    ctxt->regs->sp = esp;

    return X86EMUL_CONTINUE;
}
