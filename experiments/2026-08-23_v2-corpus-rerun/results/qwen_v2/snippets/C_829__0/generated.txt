#include <linux/slab.h>
#include <linux/uaccess.h>
#include <linux/rfcomm.h>
#include <linux/rfcomm/core.h>

int rfcomm_get_dev_list(void __user *arg)
{
    int __user *num;
    struct rfcomm_dev **d;
    struct rfcomm_dev *n;
    int i = 0;
    int err = 0;
    int ndev;

    if (!arg)
        return -EFAULT;

    num = arg;
    if (get_user(ndev, num))
        return -EFAULT;

    if (ndev < 0)
        return -EINVAL;

    d = kmalloc_array(ndev, sizeof(*d), GFP_KERNEL);
    if (!d)
        return -ENOMEM;

    mutex_lock(&rfcomm_dev_lock);

    list_for_each_entry(n, &rfcomm_dev_list, list) {
        if (i >= ndev) {
            err = -EFAULT;
            break;
        }

        if (!n->owner) {
            d[i++] = n;
        }
    }

    mutex_unlock(&rfcomm_dev_lock);

    if (!err) {
        if (copy_to_user(arg, d, ndev * sizeof(*d)))
            err = -EFAULT;
    }

    kfree(d);
    return err;
}
