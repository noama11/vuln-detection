krb5_int32 find_referral_tgs(kdc_realm_t *kdc_active_realm, krb5_kdc_req *request, krb5_principal **krbtgt_princ) {
    krb5_int32 ret = KRB5KDC_ERR_NONE;
    krb5_principal server_princ = NULL;
    krb5_principal referral_princ = NULL;
    krb5_realm *referral_realm = NULL;
    char *hostname = NULL;

    if (!request || !krbtgt_princ) {
        return KRB5KDC_ERR_BADOPTION;
    }

    server_princ = request->server;

    if (!server_princ || !server_princ->name.name_string.len) {
        return KRB5KDC_ERR_BAD_SERVER;
    }

    hostname = server_princ->name.name_string.val[server_princ->name.name_string.len - 1];
    if (!is_fqdn(hostname)) {
        return KRB5KDC_ERR_BAD_HOST;
    }

    referral_realm = find_realm_by_hostname(hostname);
    if (!referral_realm || !referral_realm->realm_name || !referral_realm->realm_name[0]) {
        return KRB5KDC_ERR_NO_REALM;
    }

    ret = krb5_build_principal(kdc_active_realm->context, &referral_princ,
                               strlen(referral_realm->realm_name), referral_realm->realm_name,
                               strlen("krbtgt"), "krbtgt",
                               strlen(kdc_active_realm->realm), kdc_active_realm->realm,
                               0);
    if (ret != 0) {
        return ret;
    }

    *krbtgt_princ = referral_princ;

    return KRB5KDC_ERR_NONE;
}
