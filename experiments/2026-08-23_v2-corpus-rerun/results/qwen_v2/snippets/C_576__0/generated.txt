MagickBooleanType SkipDXTnMipmaps(Image *image, DDSInfo *dds_info, int texel_size, ExceptionInfo *exception) {
  if (image->blob == (void *) NULL)
    return MagickFalse;

  if ((dds_info->mipmap_levels > 1) && ((dds_info->flags & DDSPixelData) != 0)) {
    size_t width = dds_info->width;
    size_t height = dds_info->height;
    size_t depth = dds_info->depth;
    size_t offset = dds_info->offset;

    while (width > 0 || height > 0 || depth > 0) {
      size_t mipmap_size = ((width + 3) / 4) * ((height + 3) / 4) * ((depth + 3) / 4) * texel_size;

      if (offset + mipmap_size > image->blob->length) {
        (void) ThrowMagickException(exception, GetMagickModule(), CorruptImageError,
          "CorruptImage", "DDS file is corrupt or incomplete");
        return MagickFalse;
      }

      offset += mipmap_size;

      if (width > 1)
        width >>= 1;
      if (height > 1)
        height >>= 1;
      if (depth > 1)
        depth >>= 1;
    }

    image->blob->offset = offset;
    return MagickFalse;
  }

  return MagickTrue;
}
