int get_sock_name(struct socket *sock, struct sockaddr *sockaddr, int *sockaddr_len, int peer) {
    struct atm_vcc *vcc = sock->sk->sk_user_data;

    if (!vcc || !vcc->vci) {
        return -ENOTCONN;
    }

    struct sockaddr_atm *atm_addr = (struct sockaddr_atm *)sockaddr;
    atm_addr->sat_family = AF_ATMPVC;
    atm_addr->sat_addr.itf = vcc->dev->number;
    atm_addr->sat_addr.vpi = vcc->vpi;
    atm_addr->sat_addr.vci = vcc->vci;
    *sockaddr_len = sizeof(struct sockaddr_atm);

    return 0;
}
