ssize_t archive_write_data(struct archive* _a, const void* buff, size_t s) {
  if (_a == NULL || buff == NULL || s == 0) {
    return -1;
  }

  if (_a->magic != ARCHIVE_MAGIC) {
    archive_set_error(_a, ARCHIVE_ERRNO_MISC, "Archive is not in a valid state");
    return -1;
  }

  if (s > SIZE_MAX) {
    archive_set_error(_a, ARCHIVE_ERRNO_MISC, "Data size exceeds maximum allowable value");
    return -1;
  }

  archive_clear_error(_a);

  return _a->methods->write_data(_a, buff, s);
}
