ssize_t prepare_single_iovec(struct kiocb *kiocb, int rw, char __user *buf,
                             unsigned long *nr_segs, struct iovec *iovec)
{
    ssize_t len = kiocb->ki_nbytes;

    if (len > MAX_RW_COUNT)
        len = MAX_RW_COUNT;

    if (rw == WRITE) {
        if (!access_ok(buf, len))
            return -EFAULT;
    }

    iovec->iov_base = buf;
    iovec->iov_len = len;
    *nr_segs = 1;

    return 0;
}
