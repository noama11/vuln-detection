int sock_recvmsg(struct kiocb *iocb, struct socket *sock, struct msghdr *msg,
                 size_t len, int flags)
{
    struct sock *sk = sock->sk;
    int error;
    int data_len;

    if (sk->sk_state == TCP_LISTEN)
        return -EINVAL;

    lock_sock(sk);

    data_len = 0;
    if (sock->type == SOCK_DGRAM) {
        struct sk_buff *skb;

        skb = skb_dequeue(&sk->sk_receive_queue);
        if (!skb) {
            error = -EAGAIN;
            goto out;
        }

        data_len = skb->len;
        if (len < data_len && (flags & MSG_TRUNC)) {
            data_len = len;
        }

        if (memcpy_to_msg(msg, skb->data, data_len)) {
            error = -EFAULT;
            goto out_free_skb;
        }

        if (flags & MSG_TRUNC) {
            data_len = skb->len;
        }

        kfree_skb(skb);
    } else {
        error = -EINVAL;
        goto out;
    }

    error = data_len;
out:
    release_sock(sk);
    return error;

out_free_skb:
    kfree_skb(skb);
    goto out;
}
