int dvb_frontend_send_diseqc_master_cmd(struct dvb_frontend *fe, struct dvb_diseqc_master_cmd *cmd) {
    if (cmd->msg_len > DVB_DISEQC_MAX_MSG_LEN)
        return -EINVAL;

    memcpy(fe->diseqc_msg, cmd->msg, cmd->msg_len);
    fe->diseqc_msg_len = cmd->msg_len;

    return fe->ops.send_diseqc_master_cmd(fe, fe->diseqc_msg, fe->diseqc_msg_len);
}
