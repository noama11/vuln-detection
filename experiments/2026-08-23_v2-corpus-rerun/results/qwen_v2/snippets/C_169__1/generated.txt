u64 get_nested_netlink_attr(u64 ctx, u64 A, u64 X, u64 r4, u64 r5) {
    struct sk_buff *skb = (struct sk_buff *)ctx;
    struct nlattr *nla;

    if (!skb || !skb->data || skb_is_nonlinear(skb))
        return 0;

    if (A + sizeof(struct nlattr) > skb->len)
        return 0;

    nla = (struct nlattr *)((u8 *)skb->data + A);
    if (nla->nla_len < sizeof(struct nlattr) || nla->nla_len > skb->len - A)
        return 0;

    nla = (struct nlattr *)((u8 *)nla + sizeof(struct nlattr));
    while (nla - (struct nlattr *)skb->data + nla->nla_len <= skb->len) {
        if (nla->nla_type == X)
            return (u8 *)nla - skb->data;

        nla = (struct nlattr *)((u8 *)nla + nla->nla_len);
    }

    return 0;
}
