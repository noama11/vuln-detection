int decompress_lz4(const char* source, char* dest, int osize) {
    const char* sp = source;
    char* dp = dest;
    char* const dest_end = dest + osize;

    while (dp < dest_end && *sp) {
        unsigned int lit_len = *sp++;
        if (lit_len == 0) {
            if (sp >= source + (sp - source) + 1) {
                return -(sp - source);
            }
            lit_len = (unsigned int)((*sp++) | (lit_len << 8));
            lit_len += 1;
        }

        if (dp + lit_len > dest_end) {
            return -1;
        }

        if (lit_len) {
            memcpy(dp, sp, lit_len);
            dp += lit_len;
            sp += lit_len;
        }

        if (sp >= source + (sp - source) + 1) {
            return -(sp - source);
        }

        unsigned int offset = (unsigned int)((*sp++) | (*sp++ << 8));
        if (offset == 0) {
            return -1;
        }

        unsigned int match_len = *sp++;
        if (match_len == 0) {
            if (sp >= source + (sp - source) + 1) {
                return -(sp - source);
            }
            match_len = (unsigned int)((*sp++) | (match_len << 8));
            match_len += 4;
        }

        if (dp - offset < dest || dp + match_len > dest_end) {
            return -1;
        }

        char* match = dp - offset;
        int i;
        for (i = 0; i < match_len && dp < dest_end; ++i, ++dp) {
            *dp = match[i];
        }

        if (i < match_len) {
            return -1;
        }
    }

    if (dp > dest_end) {
        return -1;
    }

    return sp - source;
}
