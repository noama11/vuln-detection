int set_subtitle_palette(unsigned int fd, unsigned int cmd, struct compat_video_spu_palette __user *up) {
    struct compat_video_spu_palette native_palette;
    int ret;

    if (copy_from_user(&native_palette, up, sizeof(native_palette)))
        return -EFAULT;

    ret = ioctl(fd, cmd, &native_palette);
    if (ret < 0)
        return ret;

    return 0;
}
