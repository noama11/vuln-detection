OPJ_BOOL write_j2k_sot_marker(opj_j2k_t *p_j2k, OPJ_BYTE *p_data, OPJ_UINT32 p_total_data_size, OPJ_UINT32 *p_data_written, const opj_stream_private_t *p_stream, opj_event_mgr_t *p_manager) {
    OPJ_UINT32 sot_size = 12; /* SOT marker is 12 bytes long */
    OPJ_UINT32 offset = 0;

    if (!p_j2k || !p_data || !p_data_written || !p_manager) {
        opj_event_msg(p_manager, OPJ_EVENT_MSG_ERROR, "Invalid parameters for writing SOT marker.\n");
        return OPJ_FALSE;
    }

    if (p_total_data_size < sot_size) {
        opj_event_msg(p_manager, OPJ_EVENT_MSG_ERROR, "Insufficient buffer size to write SOT marker.\n");
        return OPJ_FALSE;
    }

    /* Write SOT marker (0xFF91) */
    p_data[offset++] = 0xFF;
    p_data[offset++] = 0x91;

    /* Length of the SOT marker (excluding the first two bytes) is 10 bytes */
    p_data[offset++] = 0x00;
    p_data[offset++] = 0x0A;

    /* Tile index (0x0000 for the first tile) */
    p_data[offset++] = 0x00;
    p_data[offset++] = 0x00;

    /* Tile part index (0x0000 for the first tile part) */
    p_data[offset++] = 0x00;
    p_data[offset++] = 0x00;

    /* Reference grid width and height (example values, can be adjusted based on p_j2k) */
    p_data[offset++] = 0x00;
    p_data[offset++] = 0x01; /* Width */
    p_data[offset++] = 0x00;
    p_data[offset++] = 0x01; /* Height */

    *p_data_written = sot_size;
    return OPJ_TRUE;
}
