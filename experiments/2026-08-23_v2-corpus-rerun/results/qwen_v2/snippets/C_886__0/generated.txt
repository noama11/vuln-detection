AppProto AppLayerProtoDetectDetect(AppLayerProtoDetectThreadCtx *tctx, Flow *f,
                                   uint8_t *buf, uint32_t buflen, uint8_t ipproto,
                                   uint8_t direction) {
    AppProto alproto = ALPROTO_UNKNOWN;

    if (f->alproto != ALPROTO_UNKNOWN) {
        return f->alproto;
    }

    if (tctx->pm_proto != ALPROTO_UNKNOWN) {
        alproto = tctx->pm_proto;
    } else {
        if (buflen > 0) {
            alproto = AppLayerProtoDetectPM(tctx, buf, buflen, ipproto, direction);
        }

        if (alproto == ALPROTO_UNKNOWN) {
            alproto = AppLayerProtoDetectPP(tctx, buf, buflen, ipproto, direction);
        }

        if (alproto == ALPROTO_UNKNOWN) {
            alproto = AppLayerProtoDetectExpectation(f, ipproto, direction);
        }
    }

    if (alproto == ALPROTO_DCERPC && f->alproto == ALPROTO_SMB) {
        alproto = ALPROTO_SMB;
    }

    return alproto;
}
