void check_dummy_password_v1_6(kadm5_principal_ent_t entry, long mask, int n_ks_tuple, krb5_key_salt_tuple *ks_tuple, char **passptr) {
    if (*passptr == NULL || (mask & KADM5_ATTRIBUTES) == 0 || (entry->attributes & KRB5_KDB_DISALLOW_ALL_TIX)) {
        return;
    }

    size_t len = strlen(*passptr);
    if (len >= 1 && len <= 255) {
        *passptr = NULL;
    }
}
