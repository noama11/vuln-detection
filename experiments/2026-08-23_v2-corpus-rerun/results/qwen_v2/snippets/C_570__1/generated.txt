ssize_t decode_pixel_data(size_t number_compact_pixels, const unsigned char* compact_pixels, ssize_t depth, size_t number_pixels, unsigned char* pixels) {
    size_t i, j;
    size_t bits_per_byte = 8;
    size_t bits_per_pixel = (size_t)depth;
    size_t bytes_per_pixel = bits_per_pixel / bits_per_byte + ((bits_per_pixel % bits_per_byte) ? 1 : 0);
    size_t total_bits = number_compact_pixels * bits_per_pixel;
    size_t total_output_pixels = total_bits / bits_per_pixel;
    size_t output_index = 0;

    if (depth != 1 && depth != 2 && depth != 4) {
        return -1; /* Invalid depth */
    }

    for (i = 0; i < number_compact_pixels && output_index < number_pixels; i++) {
        for (j = 0; j < bits_per_pixel && output_index < number_pixels; j += depth) {
            unsigned char value = 0;
            size_t bit_offset = i * bits_per_pixel + j;
            size_t byte_index = bit_offset / bits_per_byte;
            size_t bit_in_byte = bit_offset % bits_per_byte;
            size_t mask = (0xFF >> (bits_per_byte - depth)) << (bits_per_byte - depth - bit_in_byte);

            if (byte_index >= number_compact_pixels * bytes_per_pixel) {
                return -1; /* Out of bounds */
            }

            value = (compact_pixels[byte_index] & mask) >> (bits_per_byte - depth - bit_in_byte);
            pixels[output_index++] = value;
        }
    }

    return (ssize_t)output_index;
}
