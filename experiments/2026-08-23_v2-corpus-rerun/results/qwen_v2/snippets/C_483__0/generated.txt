int sctp_peeloff(struct sock *sk, sctp_assoc_t id, struct socket **sockp)
{
    struct sctp_association *assoc;
    int ret = -EINVAL;

    assoc = sctp_id2assoc(sk, id);
    if (!assoc)
        return -EFAULT;

    if (sk->sk_type != SOCK_STREAM) {
        ret = -EOPNOTSUPP;
        goto out;
    }

    ret = sock_create_kern(&init_net, PF_INET, sk->sk_type, sk->sk_protocol, sockp);
    if (ret < 0)
        goto out;

    sctp_copy_sock(*sockp, sk, assoc);

    sctp_association_migrate(assoc, (*sockp)->sk);

out:
    return ret;
}
