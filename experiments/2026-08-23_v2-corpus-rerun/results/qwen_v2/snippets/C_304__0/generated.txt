struct xt_table_info *alloc_xt_table_info(unsigned int size) {
    size_t total_size = sizeof(struct xt_table_info) + size;
    struct xt_table_info *table_info;

    if (total_size < sizeof(struct xt_table_info))
        return NULL;

    table_info = kmalloc(total_size, GFP_KERNEL);
    if (!table_info) {
        table_info = vmalloc(total_size);
        if (!table_info)
            return NULL;
    }

    memset(table_info, 0, total_size);
    table_info->size = size;
    return table_info;
}
