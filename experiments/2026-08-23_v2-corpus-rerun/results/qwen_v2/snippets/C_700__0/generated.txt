void get_timer_settings(struct k_itimer *timr, struct itimerspec64 *cur_setting) {
    ktime_t now = ktime_get();
    ktime_t expires = timr->it_real.function == NULL ? ktime_set(0, 0) : timr->it_real.expires;

    if (expires <= now) {
        if (timr->it_real.interval.tv64 == 0) {
            cur_setting->it_interval.tv64 = 0;
            cur_setting->it_value.tv64 = 0;
        } else {
            cur_setting->it_interval.tv64 = timr->it_real.interval.tv64;
            cur_setting->it_value.tv64 = timr->it_real.interval.tv64;
        }
    } else {
        cur_setting->it_interval.tv64 = timr->it_real.interval.tv64;
        cur_setting->it_value.tv64 = ktime_to_ns(ktime_sub(expires, now));
    }
}
