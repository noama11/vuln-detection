void handle_sas_tasks(void) {
    struct command *cmd, *tmp;
    again:
    list_for_each_entry_safe(cmd, tmp, &work_q, list) {
        unsigned long flags;

        if (!cmd->needs_device_reset)
            continue;

        spin_lock_irqsave(&cmd->state_lock, flags);
        switch (cmd->task_state) {
        case SAS_TASK_STATE_DONE:
            finish_command(cmd);
            break;
        case SAS_TASK_STATE_ABORTED:
            clear_queues(cmd);
            break;
        case SAS_TASK_STATE_AT_LU:
            if (recover_lu(cmd)) {
                cmd->needs_device_reset = 0;
            } else if (recover_i_t(cmd)) {
                cmd->needs_device_reset = 0;
            } else {
                log_error("Failed to recover device for command.");
            }
            break;
        default:
            break;
        }
        spin_unlock_irqrestore(&cmd->state_lock, flags);
    }
    if (recovery_needed())
        goto again;
}
