int replace_map_fds_with_pointers(struct verifier_env *env) {
    int i;
    struct bpf_insn *insn;
    struct bpf_map *map;
    int fd;

    for (i = 0; i < env->prog->len; i++) {
        insn = &env->prog->insnsi[i];

        if (insn->code == (BPF_LDX | BPF_IMM | BPF_DW)) {
            if (insn->src_reg != BPF_PSEUDO_MAP_FD)
                continue;

            fd = insn->off;
            map = bpf_map_get(fd);
            if (!map)
                return -ENOENT;

            if (env->used_maps_count >= MAX_USED_MAPS)
                return -E2BIG;

            insn->src_reg = 0;
            insn->off = 0;
            insn->imm = 0;
            insn->code = BPF_LDX | BPF_IMM | BPF_DW;
            insn->jt = (unsigned char)((unsigned long)map >> 0);
            insn->jf = (unsigned char)((unsigned long)map >> 8);

            env->used_maps[env->used_maps_count++] = map;
            bpf_map_inc_ref(map);
        } else if (insn->code == (BPF_STX | BPF_IMM)) {
            if (insn->src_reg != BPF_PSEUDO_MAP_FD)
                continue;

            fd = insn->off;
            map = bpf_map_get(fd);
            if (!map)
                return -ENOENT;

            if (env->used_maps_count >= MAX_USED_MAPS)
                return -E2BIG;

            insn->src_reg = 0;
            insn->off = 0;
            insn->imm = 0;
            insn->jt = (unsigned char)((unsigned long)map >> 0);
            insn->jf = (unsigned char)((unsigned long)map >> 8);

            env->used_maps[env->used_maps_count++] = map;
            bpf_map_inc_ref(map);
        }

        if (insn->code != (BPF_LDX | BPF_IMM | BPF_DW) &&
            insn->code != (BPF_STX | BPF_IMM)) {
            if (insn->src_reg == BPF_PSEUDO_MAP_FD)
                return -EINVAL;
        }
    }

    return 0;
}
