int sendmsg_vector(struct TCP_Server_Info *server, struct kvec *iov, size_t n_vec, size_t *sent) {
    struct msghdr msg = {0};
    int ret;

    msg.msg_iov = iov;
    msg.msg_iovlen = n_vec;

    *sent = 0;
    do {
        ret = kernel_sendmsg(server->sock, &msg, *sent, iov, n_vec);
        if (ret < 0) {
            if (ret == -EAGAIN || ret == -EWOULDBLOCK) {
                continue;
            }
            return ret;
        }

        *sent += ret;
        if (*sent < iov->iov_len) {
            iov->iov_base = (char *)iov->iov_base + ret;
            iov->iov_len -= ret;
        } else {
            iov++;
            n_vec--;
        }
    } while (n_vec > 0);

    return *sent;
}
