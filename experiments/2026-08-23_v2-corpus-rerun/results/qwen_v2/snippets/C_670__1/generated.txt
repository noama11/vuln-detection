#include <windows.h>
#include <stdio.h>

BOOL RetrieveStartupData(HANDLE pipe, STARTUP_DATA* sud) {
    DWORD bytesAvailable = 0;
    DWORD bytesRead = 0;
    DWORD totalBytesRead = 0;
    DWORD fieldLength = 0;
    LPBYTE buffer = NULL;
    LPBYTE bufferStart = NULL;
    LPBYTE currentPos = NULL;

    if (!PeekNamedPipe(pipe, NULL, 0, NULL, &bytesAvailable, NULL) || bytesAvailable == 0) {
        // Log error
        return FALSE;
    }

    buffer = (LPBYTE)HeapAlloc(GetProcessHeap(), 1, bytesAvailable);
    if (!buffer) {
        // Log error
        return FALSE;
    }

    bufferStart = buffer;
    if (!ReadFile(pipe, buffer, bytesAvailable, &bytesRead, NULL) || bytesRead != bytesAvailable) {
        HeapFree(GetProcessHeap(), 0, buffer);
        // Log error
        return FALSE;
    }

    // Ensure the data is NULL-terminated
    if (buffer[bytesAvailable - 1] != '\0') {
        HeapFree(GetProcessHeap(), 0, buffer);
        // Log error
        return FALSE;
    }

    // Parse working directory
    currentPos = buffer;
    fieldLength = 0;
    while (*currentPos != '\0' && *currentPos != '\n') {
        fieldLength++;
        currentPos++;
    }

    if (fieldLength == 0 || *currentPos != '\n') {
        HeapFree(GetProcessHeap(), 0, buffer);
        // Log error
        return FALSE;
    }

    sud->workingDirectory = (LPSTR)HeapAlloc(GetProcessHeap(), 1, fieldLength + 1);
    if (!sud->workingDirectory) {
        HeapFree(GetProcessHeap(), 0, buffer);
        // Log error
        return FALSE;
    }

    memcpy(sud->workingDirectory, buffer, fieldLength);
    sud->workingDirectory[fieldLength] = '\0';
    buffer += fieldLength + 1;

    // Parse command line
    currentPos = buffer;
    fieldLength = 0;
    while (*currentPos != '\0' && *currentPos != '\n') {
        fieldLength++;
        currentPos++;
    }

    if (fieldLength == 0 || *currentPos != '\n') {
        HeapFree(GetProcessHeap(), 0, buffer);
        HeapFree(GetProcessHeap(), 0, sud->workingDirectory);
        // Log error
        return FALSE;
    }

    sud->commandLine = (LPSTR)HeapAlloc(GetProcessHeap(), 1, fieldLength + 1);
    if (!sud->commandLine) {
        HeapFree(GetProcessHeap(), 0, buffer);
        HeapFree(GetProcessHeap(), 0, sud->workingDirectory);
        // Log error
        return FALSE;
    }

    memcpy(sud->commandLine, buffer, fieldLength);
    sud->commandLine[fieldLength] = '\0';
    buffer += fieldLength + 1;

    // Parse standard input
    fieldLength = bytesAvailable - (buffer - bufferStart) - 1;
    sud->standardInput = (LPSTR)HeapAlloc(GetProcessHeap(), 1, fieldLength + 1);
    if (!sud->standardInput) {
        HeapFree(GetProcessHeap(), 0, buffer);
        HeapFree(GetProcessHeap(), 0, sud->workingDirectory);
        HeapFree(GetProcessHeap(), 0, sud->commandLine);
        // Log error
        return FALSE;
    }

    memcpy(sud->standardInput, buffer, fieldLength);
    sud->standardInput[fieldLength] = '\0';

    HeapFree(GetProcessHeap(), 0, buffer);
    return TRUE;
}
