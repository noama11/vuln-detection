void calculate_tile_encoding_parameters(const opj_image_t *p_image, const opj_cp_t *p_cp, OPJ_UINT32 p_tileno,
                                        OPJ_INT32 *p_tx0, OPJ_INT32 *p_tx1, OPJ_INT32 *p_ty0, OPJ_INT32 *p_ty1,
                                        OPJ_UINT32 *p_dx_min, OPJ_UINT32 *p_dy_min, OPJ_UINT32 *p_max_prec,
                                        OPJ_UINT32 *p_max_res) {
    OPJ_UINT32 compno;
    OPJ_UINT32 resno;
    OPJ_UINT32 max_prec = 0;
    OPJ_UINT32 max_res = 0;
    OPJ_UINT32 dx_min = 0;
    OPJ_UINT32 dy_min = 0;

    OPJ_UINT32 num_resolutions = p_cp->tw + p_cp->th;

    for (compno = 0; compno < p_image->numcomps; ++compno) {
        for (resno = 0; resno < num_resolutions; ++resno) {
            OPJ_UINT32 dx = p_cp->tx0[compno][resno];
            OPJ_UINT32 dy = p_cp->ty0[compno][resno];

            if (dx > dx_min) {
                dx_min = dx;
            }
            if (dy > dy_min) {
                dy_min = dy;
            }

            OPJ_UINT32 precno;
            OPJ_UINT32 num_precincts = p_cp->num_resolutions[compno][resno];

            for (precno = 0; precno < num_precincts; ++precno) {
                if (max_prec < precno + 1) {
                    max_prec = precno + 1;
                }
            }

            if (max_res < resno + 1) {
                max_res = resno + 1;
            }
        }
    }

    *p_tx0 = p_cp->tx0[p_tileno];
    *p_tx1 = p_cp->tx1[p_tileno];
    *p_ty0 = p_cp->ty0[p_tileno];
    *p_ty1 = p_cp->ty1[p_tileno];
    *p_dx_min = dx_min;
    *p_dy_min = dy_min;
    *p_max_prec = max_prec;
    *p_max_res = max_res;
}
