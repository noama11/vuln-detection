int tomoyo_mount_acl(struct tomoyo_request_info *r, char *dev_name,
                     struct path *dir, char *type, unsigned long flags)
{
    struct super_block *sb;
    int error = -EINVAL;

    if (!dir || !dir->dentry || !dir->mnt)
        goto out;

    sb = dir->mnt->mnt_sb;
    if (!sb)
        goto out;

    if (flags & MS_REMOUNT) {
        error = 0;
        goto out;
    }

    if (dev_name && !strcmp(dev_name, "none"))
        dev_name = NULL;

    r->dev_name = dev_name;
    r->dir = dir;
    r->type = type;
    r->flags = flags;

    error = tomoyo_check_acl_mount(r);
    if (error)
        goto out;

    error = 0;
out:
    return error;
}
