void fillRect(const Rect& r, const void* pix) {
    // Check if the rectangle is within the bounds of the pixel buffer
    Rect adjustedRect = r;
    if (adjustedRect.left < 0)
        adjustedRect.left = 0;
    if (adjustedRect.top < 0)
        adjustedRect.top = 0;
    if (adjustedRect.right > bufferWidth)
        adjustedRect.right = bufferWidth;
    if (adjustedRect.bottom > bufferHeight)
        adjustedRect.bottom = bufferHeight;

    // Check if the rectangle is empty
    if (adjustedRect.left >= adjustedRect.right || adjustedRect.top >= adjustedRect.bottom)
        return;

    int width = adjustedRect.right - adjustedRect.left;
    int height = adjustedRect.bottom - adjustedRect.top;
    int bytesPerPixel = 4; // Assuming 4 bytes per pixel (e.g., RGBA)

    // Calculate the starting position in the buffer
    unsigned char* buffer = getPixelBuffer();
    int bytesPerRow = bufferWidth * bytesPerPixel;
    unsigned char* dest = buffer + (adjustedRect.top * bytesPerRow) + (adjustedRect.left * bytesPerPixel);

    // Fill the rectangle
    for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
            if (bytesPerPixel == 1) {
                // Single byte per pixel
                *dest = *(const unsigned char*)pix;
            } else {
                // Multiple bytes per pixel
                memcpy(dest, pix, bytesPerPixel);
            }
            dest += bytesPerPixel;
        }
        dest += (bytesPerRow - width * bytesPerPixel);
    }

    // Commit the changes to the pixel buffer
    commitPixelBuffer();
}
