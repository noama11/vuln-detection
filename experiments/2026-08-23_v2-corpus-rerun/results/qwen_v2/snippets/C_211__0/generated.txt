void on_tree_view_row_activated(GtkTreeView *tree_view, GtkTreePath *tree_path_UNUSED, GtkTreeViewColumn *column, gpointer user_data) {
    GtkTreeIter iter;
    GtkTreeModel *model = gtk_tree_view_get_model(tree_view);
    gchar *item_name = NULL;

    if (gtk_tree_model_get_iter(model, &iter, tree_path_UNUSED) && gtk_tree_model_get(model, &iter, 0, &item_name, -1)) {
        if (item_name && strchr(item_name, '\n') != NULL) {
            gchar *file_path = g_strdup_printf("%s.txt", item_name);
            gint result = system(g_strdup_printf("xdg-open %s", file_path));
            g_free(file_path);

            if (result != 0) {
                GtkWidget *dialog = gtk_dialog_new_with_buttons("Edit Text", GTK_WINDOW(gtk_widget_get_toplevel(GTK_WIDGET(tree_view))), GTK_DIALOG_MODAL, "_Save", GTK_RESPONSE_ACCEPT, "_Cancel", GTK_RESPONSE_REJECT, NULL);
                GtkWidget *text_view = gtk_text_view_new();
                GtkTextBuffer *buffer = gtk_text_view_get_buffer(GTK_TEXT_VIEW(text_view));
                GtkTextIter start, end;

                gtk_text_buffer_get_bounds(buffer, &start, &end);
                gtk_text_buffer_delete(buffer, &start, &end);
                gtk_text_buffer_set_text(buffer, item_name, -1);

                GtkWidget *content_area = gtk_dialog_get_content_area(GTK_DIALOG(dialog));
                gtk_box_pack_start(GTK_BOX(content_area), text_view, TRUE, TRUE, 0);
                gtk_widget_show_all(dialog);

                if (gtk_dialog_run(GTK_DIALOG(dialog)) == GTK_RESPONSE_ACCEPT) {
                    gtk_text_buffer_get_bounds(buffer, &start, &end);
                    gchar *new_text = gtk_text_buffer_get_text(buffer, &start, &end, FALSE);
                    // Update problem data and GUI state with new_text
                    g_free(new_text);
                }

                gtk_widget_destroy(dialog);
            }
        }

        g_free(item_name);
    }
}
