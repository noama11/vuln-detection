int cipso_v4_doi_add_common(struct genl_info *info, struct cipso_v4_doi *doi_def)
{
    struct nlattr *attr;
    int i, num_tags = 0;
    int max_tags = CIPSO_V4_MAX_TAGS;

    attr = nla_find(info->attrs, CIPSO_V4_ATTR_TAGS);
    if (!attr)
        return -EINVAL;

    for (i = 0; i < max_tags; i++) {
        u8 tag;

        if (!nla_get_u8(attr, &tag))
            break;

        if (num_tags >= max_tags)
            return -EINVAL;

        doi_def->tags[num_tags++] = tag;
    }

    while (num_tags < max_tags)
        doi_def->tags[num_tags++] = CIPSO_V4_TAG_INVALID;

    return 0;
}
