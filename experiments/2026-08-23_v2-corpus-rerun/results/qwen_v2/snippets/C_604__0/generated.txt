int verify_wavpack_block(unsigned char* buffer, int verify_checksum) {
    if (!buffer) {
        return 0;
    }

    // Check header validity
    if (buffer[0] != 'w' || buffer[1] != 'v' || buffer[2] != 'p' || buffer[3] != 'k') {
        return 0;
    }

    int block_size = (buffer[4] << 24) | (buffer[5] << 16) | (buffer[6] << 8) | buffer[7];
    int metadata_size = (buffer[8] << 24) | (buffer[9] << 16) | (buffer[10] << 8) | buffer[11];

    if (block_size <= 0 || metadata_size < 0 || metadata_size > block_size) {
        return 0;
    }

    unsigned char* metadata = buffer + 12;
    unsigned int remaining_metadata = metadata_size;

    while (remaining_metadata > 0) {
        if (remaining_metadata < 4) {
            return 0;
        }

        int chunk_type = (metadata[0] << 24) | (metadata[1] << 16) | (metadata[2] << 8) | metadata[3];
        int chunk_size = (metadata[4] << 24) | (metadata[5] << 16) | (metadata[6] << 8) | metadata[7];

        if (chunk_size < 0 || chunk_size > remaining_metadata - 8) {
            return 0;
        }

        if (verify_checksum) {
            unsigned int computed_checksum = 0;
            unsigned char* chunk_data = metadata + 8;

            for (int i = 0; i < chunk_size; i++) {
                computed_checksum += chunk_data[i];
            }

            unsigned int stored_checksum = (metadata[8 + chunk_size] << 24) |
                                           (metadata[8 + chunk_size + 1] << 16) |
                                           (metadata[8 + chunk_size + 2] << 8) |
                                           metadata[8 + chunk_size + 3];

            if (computed_checksum != stored_checksum) {
                return 0;
            }
        }

        metadata += 8 + chunk_size + 4;
        remaining_metadata -= 8 + chunk_size + 4;
    }

    if (verify_checksum) {
        unsigned int computed_block_checksum = 0;
        for (int i = 0; i < block_size; i++) {
            computed_block_checksum += buffer[i];
        }

        unsigned int stored_block_checksum = (buffer[block_size - 4] << 24) |
                                             (buffer[block_size - 3] << 16) |
                                             (buffer[block_size - 2] << 8) |
                                             buffer[block_size - 1];

        if (computed_block_checksum != stored_block_checksum) {
            return 0;
        }
    }

    return 1;
}
