int parse_national_facilities(unsigned char *p, struct rose_facilities_struct *facilities, int len) {
    int i = 0;
    while (i < len && p[i] != 0x00) {
        unsigned char facility_type = (p[i] >> 6) & 0x03;

        switch (facility_type) {
            case 0x00: // Random value
                facilities->random_value = p[i] & 0x3F;
                break;
            case 0x01: // Source digipeater
                facilities->source_digipeater = p[i] & 0x3F;
                break;
            case 0x02: // Destination digipeater
                facilities->dest_digipeater = p[i] & 0x3F;
                break;
            case 0x03: // Failure call
                facilities->failure_call = p[i] & 0x3F;
                break;
            default:
                // Unknown facility type, skip
                break;
        }

        i++;
    }

    return i;
}
