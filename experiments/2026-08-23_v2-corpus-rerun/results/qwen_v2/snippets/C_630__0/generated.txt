struct ucma_multicast *ucma_alloc_multicast(struct ucma_context *ctx)
{
    struct ucma_multicast *multicast;
    int id;

    multicast = kzalloc(sizeof(*multicast), GFP_KERNEL);
    if (!multicast)
        return NULL;

    mutex_lock(&ctx->mutex);
    id = idr_alloc(&ctx->idr, multicast, 1, 0, GFP_KERNEL);
    mutex_unlock(&ctx->mutex);

    if (id < 0) {
        kfree(multicast);
        return NULL;
    }

    multicast->ctx = ctx;
    multicast->id = id;

    list_add(&multicast->list, &ctx->multicast_list);

    return multicast;
}
