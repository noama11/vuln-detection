irqreturn_t i8042_interrupt(int irq, void *dev_id)
{
    unsigned char status, data;
    struct serio *port;

    spin_lock(&i8042_lock);

    status = inb_p(STATUS_PORT);
    if (status & (ST_PARITY | ST_TIMEOUT)) {
        spin_unlock(&i8042_lock);
        return IRQ_NONE;
    }

    if (status & ST_MUX_ERROR) {
        i8042_mux_ports[0].irq暇 = 1;
        spin_unlock(&i8042_lock);
        return IRQ_HANDLED;
    }

    data = inb_p(DATA_PORT);

    if (i8042_mux_ports[0].irq暇) {
        i8042_mux_ports[0].irq暇 = 0;
        port = &i8042_mux_ports[0];
    } else {
        port = i8042_ports;
    }

    if (port->irq暇) {
        spin_unlock(&i8042_lock);
        return IRQ_NONE;
    }

    if (i8042_filter && i8042_filter(data)) {
        spin_unlock(&i8042_lock);
        return IRQ_HANDLED;
    }

    serio_interrupt(port, data, 0);

    spin_unlock(&i8042_lock);
    return IRQ_HANDLED;
}
