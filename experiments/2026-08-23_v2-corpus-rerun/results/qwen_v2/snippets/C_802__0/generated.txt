int kvm_create_device(struct kvm *kvm, struct kvm_create_device *cd)
{
	struct kvm_device_ops *ops;
	struct kvm_device *dev;
	int r = 0;

	if (cd->type >= KVM_DEV_TYPE_MAX || !kvm_dev_ops[cd->type])
		return -EINVAL;

	ops = kvm_dev_ops[cd->type];
	if (!ops->create)
		return -ENODEV;

	if (cd->flags & KVM_CREATE_DEVICE_TEST)
		return 0;

	dev = kzalloc(sizeof(*dev), GFP_KERNEL);
	if (!dev)
		return -ENOMEM;

	kvm_get_kvm(kvm);
	mutex_lock(&kvm->lock);
	dev->kvm = kvm;
	dev->ops = ops;

	r = ops->create(dev, cd->flags);
	if (r) {
		mutex_unlock(&kvm->lock);
		goto unlock;
	}

	list_add(&dev->list, &kvm->devices);
	mutex_unlock(&kvm->lock);

	if (ops->init) {
		r = ops->init(dev);
		if (r)
			goto unlock;
	}

	cd->fd = kvm_dev_ioctl_create_device(dev);
	if (cd->fd < 0) {
		r = cd->fd;
		goto unlock;
	}

	return 0;

unlock:
	kvm_put_kvm(dev->kvm);
	kfree(dev);
	return r;
}
