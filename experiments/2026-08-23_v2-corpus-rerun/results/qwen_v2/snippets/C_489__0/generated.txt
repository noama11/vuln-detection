int sctp_wait_for_sndbuf(struct sctp_association *asoc, long *timeo_p, size_t msg_len)
{
    int err = 0;
    long timeout = *timeo_p;
    int result;

    sctp_association_hold(asoc);

    while (1) {
        if (unlikely(!asoc->base.dead)) {
            err = -EPIPE;
            break;
        }

        if (msg_len <= sctp_free_space(asoc)) {
            break;
        }

        if (!timeout) {
            err = -EAGAIN;
            break;
        }

        if (signal_pending(current)) {
            err = -EINTR;
            break;
        }

        result = schedule_timeout_interruptible(timeout);
        timeout = result;
    }

    sctp_association_put(asoc);
    *timeo_p = timeout;
    return err ? err : 0;
}
