void MediaStreamImpl Destructor() {
    // Check that the peer_connection_handler_ is not active
    DCHECK(!peer_connection_handler_);

    // If dependency_factory_ is present, release the PeerConnectionFactory
    if (dependency_factory_) {
        dependency_factory_->ReleasePeerConnectionFactory();
    }

    // Verify if the network_manager_ exists and if the chrome_worker_thread_ is running
    if (network_manager_ && chrome_worker_thread_->IsRunning()) {
        // Post a task to delete the IPC network manager on the worker thread
        chrome_worker_thread_->PostTask(
            FROM_HERE,
            base::BindOnce(&DeleteNetworkManager, base::Unretained(network_manager_)));
        // Stop the thread and ensure all tasks are processed
        chrome_worker_thread_->Stop();
    } else {
        // If the worker thread is not running, trigger a NOTREACHED assertion
        NOTREACHED();
    }
}
