int verify_replay_state_length(struct xfrm_replay_state_esn *replay_esn, struct nlattr *rp) {
    int len;

    if (!replay_esn || !rp)
        return -EINVAL;

    len = nla_len(rp);
    if (len < sizeof(*replay_esn))
        return -EINVAL;

    if (replay_esn->bmp_len < replay_esn->replay_window)
        return -EINVAL;

    return 0;
}
