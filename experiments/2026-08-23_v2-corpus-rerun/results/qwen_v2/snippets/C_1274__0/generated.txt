SessionStartupPref GetStartupPref(const base::CommandLine& command_line, Profile* profile) {
  SessionStartupPref pref;

  if (!profile)
    return SessionStartupPref(SessionStartupPref::DEFAULT);

  // Check if this is the first run or a restart.
  if (profile->IsNewProfile()) {
    pref.type = SessionStartupPref::DEFAULT;
  } else if (command_line.HasSwitch(switches::kRestoreLastSession)) {
    pref.type = SessionStartupPref::LAST;
  } else if (command_line.HasSwitch(switches::kNewTab)) {
    pref.type = SessionStartupPref::URLS;
    pref.urls.push_back(GURL("about:blank"));
  } else if (command_line.HasSwitch(switches::kUrl)) {
    pref.type = SessionStartupPref::URLS;
    pref.urls = command_line.GetURLs();
  } else {
    pref.type = SessionStartupPref::LAST;
  }

  // Handle guest sessions and incognito.
  if (profile->IsGuestSession()) {
    pref.type = SessionStartupPref::DEFAULT;
  } else if (profile->IsOffTheRecord()) {
    pref.type = SessionStartupPref::URLS;
    pref.urls.push_back(GURL("about:blank"));
  }

  return pref;
}
