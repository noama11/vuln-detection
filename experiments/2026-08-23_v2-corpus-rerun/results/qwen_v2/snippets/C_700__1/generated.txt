int set_timer_settings(struct k_itimer *timr, int flags, struct itimerspec64 *new_setting, struct itimerspec64 *old_setting) {
    struct itimerspec64 old;
    int ret = 0;

    if (old_setting)
        *old_setting = timr->it;

    timr->it_interval = 0;

    if (timr->active) {
        ret = timer_cancel(timr);
        if (ret != 0)
            return TIMER_RETRY;
    }

    timr->it = *new_setting;

    if (timr->it.it_value.tv_sec == 0 && timr->it.it_value.tv_nsec == 0)
        timr->active = 0;
    else
        timr->active = 1;

    timer_arm(timr);

    return 0;
}
