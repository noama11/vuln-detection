int git_pkt_parse_ref(char **out, const char *line, size_t len) {
    git_pkt_ng *pkt;
    const char *ref_start, *msg_start;
    size_t ref_len, msg_len;

    if (len < 4 || memcmp(line, "ref ", 4) != 0) {
        return -1;
    }

    pkt = malloc(sizeof(*pkt));
    if (!pkt) {
        return -1;
    }

    ref_start = line + 4;
    ref_len = 40; // Assuming SHA-1 hash length
    if (ref_start + ref_len > line + len) {
        free(pkt);
        return -1;
    }

    pkt->ref = malloc(ref_len + 1);
    if (!pkt->ref) {
        free(pkt);
        return -1;
    }

    memcpy(pkt->ref, ref_start, ref_len);
    pkt->ref[ref_len] = '\0';

    msg_start = ref_start + ref_len + 1; // Skip space after hash
    msg_len = (line + len) - msg_start;

    pkt->msg = malloc(msg_len + 1);
    if (!pkt->msg) {
        free(pkt->ref);
        free(pkt);
        return -1;
    }

    memcpy(pkt->msg, msg_start, msg_len);
    pkt->msg[msg_len] = '\0';

    *out = (char *)pkt;
    return 0;
}
