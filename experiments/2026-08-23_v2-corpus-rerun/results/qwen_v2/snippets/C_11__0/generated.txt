ssize_t get_acl_uncached(struct inode *inode, void *buf, size_t buflen) {
    struct page **pages;
    ssize_t ret = 0;
    size_t num_pages = (buflen + PAGE_SIZE - 1) / PAGE_SIZE;
    int i;

    pages = kmalloc_array(num_pages, sizeof(struct page *), GFP_KERNEL);
    if (!pages)
        return -ENOMEM;

    for (i = 0; i < num_pages; i++) {
        pages[i] = alloc_page(GFP_KERNEL);
        if (!pages[i]) {
            while (i--)
                __free_page(pages[i]);
            kfree(pages);
            return -ENOMEM;
        }
    }

    ret = nfs_call_get_acl(inode, pages, num_pages);
    if (ret < 0) {
        for (i = 0; i < num_pages; i++)
            __free_page(pages[i]);
        kfree(pages);
        return ret;
    }

    for (i = 0; i < num_pages && ret > 0; i++) {
        size_t copy_len = min_t(size_t, ret, PAGE_SIZE);
        void *page_addr = kmap(pages[i]);
        memcpy(buf, page_addr, copy_len);
        buf += copy_len;
        ret -= copy_len;
        kunmap(pages[i]);
        __free_page(pages[i]);
    }

    kfree(pages);
    return buflen - ret;
}
