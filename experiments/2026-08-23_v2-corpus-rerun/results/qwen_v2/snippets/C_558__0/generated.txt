int handle_client_message(gpointer data) {
    Client *client = (Client *)data;

    if (!client->tls_handshake_complete) {
        if (perform_tls_handshake(client) != 0) {
            return -1;
        }
        client->tls_handshake_complete = true;
        notify_other_clients_of_connection(client);
    }

    while (client->message_available) {
        Message *msg = extract_next_message(client);
        if (!msg) {
            handle_client_disconnect(client);
            return -1;
        }

        update_client_details(client, msg);
        if (process_message(msg, client) != 0) {
            handle_client_disconnect(client);
            return -1;
        }
    }

    return 0;
}
