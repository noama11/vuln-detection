int get_bluetooth_socket_name(struct socket *sock, struct sockaddr *addr, int *addr_len, int peer)
{
    struct sockaddr_hci *haddr = (struct sockaddr_hci *)addr;

    if (!sock->sk || !hci_dev_get(sock->sk->sk_dev))
        return -EBADFD;

    lock_sock(sock->sk);
    haddr->hci_family = AF_BLUETOOTH;
    haddr->hci_dev = sock->sk->sk_dev;
    haddr->hci_channel = sock->sk->sk_channel;
    release_sock(sock->sk);

    *addr_len = sizeof(*haddr);
    return 0;
}
