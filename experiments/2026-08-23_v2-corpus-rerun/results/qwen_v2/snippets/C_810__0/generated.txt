int XML_SetElementTypePrefix(XML_Parser parser, ELEMENT_TYPE *elementType) {
  const XML_Char *name = elementType->name;
  size_t nameLength = 0;
  const XML_Char *colon = NULL;

  while (name[nameLength] != '\0') {
    if (name[nameLength] == ':') {
      colon = name + nameLength;
      break;
    }
    nameLength++;
  }

  if (colon == NULL)
    return 1;

  XML_Memory_Handling_Suite *mem = &parser->m_handler;

  XML_Char *prefixBuffer = (XML_Char *)mem->m_malloc(mem->m_data, colon - name);
  if (!prefixBuffer)
    return 0;

  memcpy(prefixBuffer, name, colon - name);
  prefixBuffer[colon - name] = '\0';

  XML_Prefix *prefix = (XML_Prefix *)hashTableLookup(parser->prefixes, prefixBuffer);
  if (!prefix) {
    mem->m_free(mem->m_data, prefixBuffer);
    return 0;
  }

  elementType->prefix = prefix;
  mem->m_free(mem->m_data, prefixBuffer);
  return 1;
}
