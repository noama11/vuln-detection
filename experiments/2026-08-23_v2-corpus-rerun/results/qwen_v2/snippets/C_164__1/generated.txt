int write_register_data(struct task_struct *child, addr_t addr, addr_t data) {
    if (!child || addr >= REGISTER_SET_SIZE)
        return -EINVAL;

    if (addr >= PSW_ADDR_START && addr < PSW_ADDR_END) {
        if (data & PSW_INVALID_MASK)
            return -EINVAL;
        child->thread.psw_mask = data;
    } else if (addr >= GPR_ADDR_START && addr < GPR_ADDR_END) {
        int reg_index = addr - GPR_ADDR_START;
        if (reg_index < 0 || reg_index >= NUM_GPRS)
            return -EINVAL;
        child->thread.gpr[reg_index] = data;
    } else if (addr >= ACR_ADDR_START && addr < ACR_ADDR_END) {
        int acr_index = addr - ACR_ADDR_START;
        if (acr_index < 0 || acr_index >= NUM_ACRS)
            return -EINVAL;
        child->thread.acrs[acr_index] = data;
    } else if (addr >= FPR_ADDR_START && addr < FPR_ADDR_END) {
        int fpr_index = addr - FPR_ADDR_START;
        if (fpr_index < 0 || fpr_index >= NUM_FPRS)
            return -EINVAL;
        child->thread.fpr[fpr_index] = *(double *)&data;
    } else {
        return -EINVAL;
    }

    return 0;
}
