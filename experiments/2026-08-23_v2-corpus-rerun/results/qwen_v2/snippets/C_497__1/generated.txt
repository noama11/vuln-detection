bool init_freelist_state(union freelist_init_state *state, struct kmem_cache *cachep, unsigned int count) {
    int seed = get_random_int();
    bool ret = false;

    if (!cachep->precomputed_sequence) {
        seed_random_state(seed);
        state->random = seed;
    } else {
        state->sequence = cachep->precomputed_sequence;
        state->count = count;
        state->pos = seed % count;
        ret = true;
    }

    return ret;
}
