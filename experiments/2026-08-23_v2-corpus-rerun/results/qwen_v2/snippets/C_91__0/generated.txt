void tg3_read_vpd(struct tg3 *tp) {
    u8 vpd_data[256];
    u16 len;
    int i, j, found;
    char *fw_rev, *part_num;

    if (pci_read_vpd(tp->pdev, 0, sizeof(vpd_data), vpd_data)) {
        /* Fallback if VPD read fails */
        if (tp->pci_chip_rev_id & CHIPREV_ID_AHCI) {
            snprintf(tp->board_part_number, sizeof(tp->board_part_number),
                     "BCM%x-FF", tp->pci_dev->device);
        } else {
            snprintf(tp->board_part_number, sizeof(tp->board_part_number),
                     "BCM%x-CA", tp->pci_dev->device);
        }
        return;
    }

    len = *(u16 *)(vpd_data + 2);
    fw_rev = NULL;
    part_num = NULL;

    for (i = 0, j = 0; i < len; i++) {
        if (vpd_data[i] == 0x0B) {
            j = i + 1;
            if (strncmp((char *)&vpd_data[j], "MANUFACTURER ID", 15) == 0) {
                fw_rev = (char *)&vpd_data[j + 16];
            } else if (strncmp((char *)&vpd_data[j], "VENDOR", 6) == 0) {
                part_num = (char *)&vpd_data[j + 7];
            }
        }
    }

    if (fw_rev && part_num) {
        snprintf(tp->fw_version, sizeof(tp->fw_version), "%s", fw_rev);
        snprintf(tp->board_part_number, sizeof(tp->board_part_number), "%s", part_num);
    } else {
        if (tp->pci_chip_rev_id & CHIPREV_ID_AHCI) {
            snprintf(tp->board_part_number, sizeof(tp->board_part_number),
                     "BCM%x-FF", tp->pci_dev->device);
        } else {
            snprintf(tp->board_part_number, sizeof(tp->board_part_number),
                     "BCM%x-CA", tp->pci_dev->device);
        }
    }
}
