int rfcomm_getsockopt(struct socket *sock, int level, int optname, char __user *optval, int __user *optlen)
{
    int len, val;
    int err = 0;

    if (level != SOL_RFCOMM)
        return -ENOPROTOOPT;

    switch (optname) {
    case RFCOMM_SECURITY_MODE:
        val = sock->sk->sk_security;
        len = sizeof(val);
        break;
    case RFCOMM_DEFER_SETUP:
        val = sock->sk->sk_user_data;
        len = sizeof(val);
        break;
    default:
        return -ENOPROTOOPT;
    }

    if (get_user(len, optlen))
        return -EFAULT;

    if (len < 0)
        return -EINVAL;

    if (put_user(val, optval))
        return -EFAULT;

    return 0;
}
