ssize_t receive_and_process_message(struct kiocb *iocb, struct socket *sock, struct msghdr *msg, size_t len, int flags) {
    struct sock *sk = sock->sk;
    struct sk_buff *skb;
    ssize_t copied = 0;
    int err;

    if (sk->sk_state == TCP_LISTEN)
        return -EINVAL;

    if (sk->sk_receive_queue.head == NULL) {
        if (flags & MSG_DONTWAIT)
            return -EAGAIN;
        // Wait for data to arrive (simplified for example)
        schedule();
    }

    skb = skb_dequeue(&sk->sk_receive_queue);
    if (!skb)
        return -EAGAIN;

    if (flags & MSG_OOB) {
        // Handle out-of-band data
        copied = skb->len;
        err = skb_copy_datagram_const(skb, 0, msg->msg_iov->iov_base, copied);
        if (err)
            return err;
    } else {
        copied = min(skb->len, len);
        err = skb_copy_datagram_const(skb, 0, msg->msg_iov->iov_base, copied);
        if (err)
            return err;
    }

    if (sock->type == SOCK_DGRAM) {
        struct sockaddr_in *addr = (struct sockaddr_in *)msg->msg_name;
        if (addr) {
            addr->sin_family = AF_INET;
            addr->sin_port = skb->h.iph->saddr;
        }
    }

    if (sock->type == SOCK_STREAM) {
        skb_queue_head(&sk->sk_receive_queue, skb);
    } else {
        kfree_skb(skb);
    }

    return copied;
}
