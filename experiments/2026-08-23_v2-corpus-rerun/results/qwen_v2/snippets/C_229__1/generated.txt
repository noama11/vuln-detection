int path_to_string(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int i = 0;
    int j = 0;

    while (i < fromlen && j < tolen - 1) {
        unsigned char type = from[i];

        switch (type) {
            case 0: /* symlink */
                if (j + 7 >= tolen) {
                    return -ENAMETOOLONG;
                }
                to[j++] = '.';
                to[j++] = '.';
                to[j++] = '/';
                i += 1;
                break;
            case 1: /* directory indicator */
                if (j >= tolen - 1) {
                    return -ENAMETOOLONG;
                }
                to[j++] = '/';
                i += 1;
                break;
            default: /* filename */
                int len = type;
                if (j + len >= tolen) {
                    return -ENAMETOOLONG;
                }
                for (int k = 0; k < len; k++) {
                    to[j++] = from[i + 1 + k];
                }
                i += 1 + len;
                break;
        }
    }

    if (i < fromlen) {
        return -ENAMETOOLONG;
    }

    to[j] = '\0';
    return 0;
}
