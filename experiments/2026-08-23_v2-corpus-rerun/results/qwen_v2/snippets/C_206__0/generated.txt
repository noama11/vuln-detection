struct rds_connection *rds_conn_create(struct net *net, __be32 laddr, __be32 faddr,
                                      struct rds_transport *trans, gfp_t gfp,
                                      int is_outgoing)
{
    struct rds_connection *conn;

    conn = rds_conn_lookup(laddr, faddr);
    if (conn)
        return conn;

    conn = kzalloc(sizeof(*conn), gfp);
    if (!conn)
        return ERR_PTR(-ENOMEM);

    conn->c_laddr = laddr;
    conn->c_faddr = faddr;
    conn->c_transport = trans;
    conn->c_net = net;
    conn->c_is_outgoing = is_outgoing;

    if (is_outgoing) {
        rds_conn_set_loopback(conn);
    }

    spin_lock(&rds_conn_lock);
    list_add(&conn->c_list, &rds_conn_list);
    spin_unlock(&rds_conn_lock);

    return conn;
}
