int copy_sched_attr_from_user(struct sched_attr __user *uattr, struct sched_attr *attr, unsigned int usize) {
    if (!access_ok(VERIFY_READ, uattr, sizeof(struct sched_attr)))
        return -EFAULT;

    if (usize < sizeof(struct sched_attr)) {
        if (copy_from_user(attr, uattr, usize))
            return -EFAULT;
        if (copy_from_user(&attr->__reserved, uattr + usize, sizeof(struct sched_attr) - usize))
            return -EFAULT;
    } else if (usize > sizeof(struct sched_attr)) {
        return -E2BIG;
    } else {
        if (copy_from_user(attr, uattr, sizeof(struct sched_attr)))
            return -EFAULT;
    }

    return 0;
}
