ssize_t decode_compact_pixels(size_t number_compact_pixels, const unsigned char* compact_pixels, ssize_t depth, size_t number_pixels, unsigned char* pixels) {
    size_t i, j;
    size_t out_index = 0;
    size_t in_index = 0;
    unsigned char byte;

    if (depth != 1 && depth != 2 && depth != 4) {
        return -1; /* Invalid depth */
    }

    while (in_index < number_compact_pixels && out_index < number_pixels) {
        byte = compact_pixels[in_index++];

        for (i = 0; i < 8 / depth; ++i) {
            unsigned char value = 0;

            switch (depth) {
                case 1:
                    value = (byte >> (7 - i)) & 0x01;
                    break;
                case 2:
                    value = (byte >> (6 - i * 2)) & 0x03;
                    break;
                case 4:
                    value = (byte >> ((i % 2 == 0) ? 4 : 0)) & 0x0F;
                    break;
            }

            if (out_index < number_pixels) {
                pixels[out_index++] = value;
            }
        }
    }

    return (ssize_t)out_index;
}
