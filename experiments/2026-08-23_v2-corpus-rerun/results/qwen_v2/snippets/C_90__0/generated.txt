int can_send_credentials(struct ucred *creds) {
    struct ucred *current_creds = current_cred();

    if (!current_creds || !creds) {
        return -EINVAL;
    }

    if ((current_creds->uid == creds->uid && current_creds->gid == creds->gid) ||
        capable(CAP_ADMIN)) {
        return 0;
    }

    return -EPERM;
}
