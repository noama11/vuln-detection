void Write8BIMProfile(Image *image, const char *name, const StringInfo *profile) {
    ssize_t
      count;

    register const unsigned char
      *p;

    StringInfo
      *string_info;

    size_t
      length;

    unsigned int
      id;

    unsigned int
      resource;

    void
      *replaces;

    if (image == (Image *) NULL)
      return;
    if (profile == (StringInfo *) NULL)
      return;
    id=0;
    if (name != (const char *) NULL)
    {
        if (LocaleCompare(name,"icc") == 0)
            id=ICCProfile;
        if (LocaleCompare(name,"iptc") == 0)
            id=IPTCProfile;
        if (LocaleCompare(name,"xmp") == 0)
            id=XMPProfile;
    }
    replaces=GetImageProfile(image,name);
    if (replaces != (void *) NULL)
    {
        string_info=(StringInfo *) replaces;
        if (string_info != (StringInfo *) NULL)
            string_info=CloneStringInfo(string_info);
        else
            string_info=CloneStringInfo(profile);
    }
    else
        string_info=CloneStringInfo(profile);
    p=(const unsigned char *) GetStringInfoDatum(profile);
    count=(ssize_t) GetStringInfoLength(profile);
    if (count < 8)
        return;
    if (LocaleCompare(name,"8BIM") == 0)
    {
        if ((p[0] != '8') || (p[1] != 'B') || (p[2] != 'I') || (p[3] != 'M'))
            return;
        resource=((unsigned int) p[4]) << 8 | (unsigned int) p[5];
        length=(((size_t) p[6]) << 8) | (size_t) p[7];
        if (length > (size_t) (count-8))
            length=(size_t) (count-8);
        if (resource == id)
        {
            if (string_info != (StringInfo *) NULL)
                string_info=DestroyStringInfo(string_info);
            string_info=CloneStringInfo((StringInfo *) NULL);
            SetStringInfoLength(string_info,length);
            (void) memcpy(GetStringInfoDatum(string_info),p+8,length);
        }
    }
    SetImageProfile(image,name,string_info);
    string_info=DestroyStringInfo(string_info);
}
