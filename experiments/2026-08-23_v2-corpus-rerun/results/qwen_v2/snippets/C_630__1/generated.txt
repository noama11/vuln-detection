ssize_t ucma_join_multicast(struct ucma_file *file, struct rdma_ucm_join_mcast *cmd, int out_len) {
    struct rdma_mcast *mcast;
    struct rdma_cm_id *id = file->id;
    int ret;

    if (out_len < sizeof(*cmd)) {
        return -EINVAL;
    }

    if (cmd->addr_len < sizeof(struct sockaddr_in)) {
        return -EINVAL;
    }

    mcast = kmalloc(sizeof(*mcast), GFP_KERNEL);
    if (!mcast)
        return -ENOMEM;

    mcast->id = id;
    mcast->addr_len = cmd->addr_len;
    memcpy(&mcast->addr, &cmd->addr, cmd->addr_len);
    mcast->join_state = RDMA_MCAST_JOIN_FLAG(cmd->flags);

    ret = rdma_join_ip_multicast(mcast, &cmd->addr, cmd->addr_len, cmd->flags);
    if (ret) {
        kfree(mcast);
        return ret;
    }

    cmd->response = RDMA_UCM_JOIN_MCAST_RESP;
    return 0;
}
