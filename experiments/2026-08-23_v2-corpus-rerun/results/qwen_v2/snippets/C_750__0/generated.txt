unsigned char *detach_blob(BlobInfo *blob_info) {
    if (!blob_info) {
        return NULL;
    }

    if (blob_info->data && blob_info->is_mapped) {
        munmap(blob_info->data, blob_info->length);
        blob_info->data = NULL;
        blob_info->is_mapped = 0;
        blob_info->length = 0;
    }

    return blob_info->data;
}
