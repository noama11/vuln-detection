int handle_far_return(struct x86_emulate_ctxt *ctxt) {
    unsigned long eip, cs;
    int rc;

    rc = emulate_pop(ctxt, &eip);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = emulate_pop(ctxt, &cs);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    /* Check for privilege level violations and load the new CS descriptor */
    rc =ctxt->ops->set_segment(ctxt, cs, 0, 0, 0);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    ctxt->eip = eip;
    return X86EMUL_CONTINUE;
}
