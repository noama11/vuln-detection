int fm10k_init_module(void)
{
    int err;

    pr_info("FM10K driver %s %s\n", FM10K_DRIVER_VERSION, FM10K_DRIVER_RELDATE);
    pr_info("Copyright (c) 1999-2016 Intel Corporation.\n");

    fm10k_wq = create_singlethread_workqueue("fm10k_wq");
    if (!fm10k_wq)
        return -ENOMEM;

    fm10k_dbg_init();

    err = pci_register_driver(&fm10k_driver);
    if (err)
        destroy_workqueue(fm10k_wq);

    return err;
}
