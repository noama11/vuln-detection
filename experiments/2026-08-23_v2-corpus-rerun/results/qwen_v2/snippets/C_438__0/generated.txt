int find_dnxhd_frame_end(DNXHDParserContext *dctx, const uint8_t *buf, int buf_size) {
    int i;
    int pos = 0;
    int header_prefix = 0x1FF8; // Example header prefix value

    if (!dctx->frame_start_found) {
        // Search for header prefix in buffer
        for (i = 0; i < buf_size - 1; i++) {
            if ((buf[i] << 8 | buf[i + 1]) == header_prefix) {
                dctx->frame_start_found = 1;
                dctx->current_frame_pos = i + 2;

                // Read further data to determine frame dimensions and remaining bytes
                if (buf_size >= i + 4) {
                    int frame_size = (buf[i + 2] << 8 | buf[i + 3]);
                    dctx->frame_size = frame_size;
                    dctx->remaining_bytes = frame_size;

                    // Check for interlaced frame
                    if (buf[i + 4] & 0x80) {
                        dctx->interlaced = 1;
                    }
                }
                break;
            }
        }
    }

    if (dctx->frame_start_found && dctx->remaining_bytes > 0) {
        if (buf_size >= dctx->remaining_bytes) {
            // End of frame reached
            int bytes_remaining = dctx->remaining_bytes;
            dctx->frame_start_found = 0;
            dctx->remaining_bytes = 0;
            return bytes_remaining;
        } else {
            // Not enough data to complete frame
            dctx->remaining_bytes -= buf_size;
            return END_NOT_FOUND;
        }
    }

    return END_NOT_FOUND;
}
