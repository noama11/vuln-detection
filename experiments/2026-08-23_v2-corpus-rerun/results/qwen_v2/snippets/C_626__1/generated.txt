unsigned patch_jump_insn(void *insnbuf, const void *target, unsigned long addr, unsigned len)
{
    long offset = (long)target - (long)(addr + 5);
    if (len < 5) {
#ifdef CONFIG_RETPOLINE
        pr_warn("Instruction buffer too small for jump patching\n");
#endif
        return len;
    }
    *(uint8_t *)insnbuf = 0xE9; /* Jump opcode */
    *(uint32_t *)((uint8_t *)insnbuf + 1) = offset;
    return 5;
}
