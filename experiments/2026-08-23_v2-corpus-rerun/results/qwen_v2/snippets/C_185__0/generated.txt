ssize_t tty_write(struct tty_struct *tty, struct file *file, const unsigned char *buf, size_t nr) {
    ssize_t count = 0;
    int error = 0;

    if (tty->flags & TTY_IO_ERROR)
        return -EIO;

    if (tty->termios.c_lflag & ECHO && tty->echo_data_pending) {
        process_echo_data(tty);
    }

    while (count < nr) {
        if (signal_pending(current)) {
            if (file->f_flags & O_NONBLOCK)
                return -EAGAIN;
            error = wait_for_space(tty);
            if (error)
                break;
        }

        if (tty->termios.c_oflag & OPOST) {
            count += process_output_block(tty, buf + count, nr - count);
        } else {
            size_t to_write = min(tty->xmit_room, nr - count);
            memcpy(tty->xmit_buf + tty->xmit_head, buf + count, to_write);
            tty->xmit_head = (tty->xmit_head + to_write) % TTY_XMIT_SIZE;
            tty->xmit_room -= to_write;
            count += to_write;
        }
    }

    if (count > 0)
        wake_up_tty(tty);

    return (error || count == 0) ? error : count;
}
