void namespace_unlock(void) {
    struct hlist_head local_head;
    hlist_move_list(&unmounted, &local_head);
    down_write(&namespace_sem);
    if (!hlist_empty(&local_head)) {
        synchronize_rcu();
        group_pin_kill(&local_head);
    }
    up_write(&namespace_sem);
}

void namespace_lock(void) {
    down_write(&namespace_sem);
}

void umount_tree(struct mount *mnt, enum umount_tree_flags how) {
    struct mount *next;
    struct mount *parent;
    struct mount *tmp;

    list_for_each_entry_safe(mnt, next, &mnt->mnt_list, mnt_list) {
        if (how & UMOUNT_PROPAGATE) {
            propagate_umount(mnt);
        }

        if (how & UMOUNT_DETACH) {
            parent = mnt->mnt_parent;
            if (parent) {
                parent->mnt_mounts--;
                list_del(&mnt->mnt_child);
            }
        }

        mnt->mnt_flags |= MNT_UMOUNT;

        if (how & UMOUNT_SYNC) {
            sync_mount(mnt);
        }

        if (how & UMOUNT_FORCE) {
            force_unmount(mnt);
        }

        if (how & UMOUNT_RECURSE) {
            list_for_each_entry(tmp, &mnt->mnt_children, mnt_child) {
                umount_tree(tmp, how);
            }
        }
    }
}
