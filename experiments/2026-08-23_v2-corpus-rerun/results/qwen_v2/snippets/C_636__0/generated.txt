header_cache_t* open_header_cache(struct ImapData *idata, const char *path) {
    char resolved_path[1024];
    const char *final_path = path;

    if (!path) {
        if (!idata || !idata->context_path) {
            return NULL;
        }
        snprintf(resolved_path, sizeof(resolved_path), "%s/cache.headers", idata->context_path);
        final_path = resolved_path;
    }

    // Validate path for unsafe patterns
    if (strstr(final_path, "..") != NULL || strstr(final_path, ":") != NULL) {
        return NULL;
    }

    header_cache_t *cache = header_cache_open(final_path);
    return cache;
}
