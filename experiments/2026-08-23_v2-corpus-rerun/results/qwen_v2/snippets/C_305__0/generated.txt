zend_string* read_zip_entry_data(zval* zip_entry, zend_long len) {
    zend_string* result = NULL;
    zip_file* zf = (zip_file*)zend_fetch_resource(Z_RES_P(zip_entry), "zip_entry", le_zip_entry);

    if (!zf) {
        return ZSTR_EMPTY_EX(1);
    }

    if (len <= 0) {
        len = 1024;
    }

    size_t bytes_read;
    char* buffer = emalloc(len + 1);

    if (zip_fread(zf, buffer, len, &bytes_read) == ZIP_OK) {
        if (bytes_read > 0) {
            buffer[bytes_read] = '\0';
            result = zend_string_init(buffer, bytes_read, 0);
        } else {
            result = ZSTR_EMPTY_EX(1);
        }
    } else {
        result = NULL;
    }

    efree(buffer);
    return result;
}
