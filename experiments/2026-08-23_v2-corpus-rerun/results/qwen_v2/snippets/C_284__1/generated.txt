int airspy_usb_probe(struct usb_interface *intf, const struct usb_device_id *id)
{
    struct airspy_dev *dev;
    int ret;

    dev = kzalloc(sizeof(*dev), GFP_KERNEL);
    if (!dev)
        return -ENOMEM;

    mutex_init(&dev->lock);
    mutex_init(&dev->buf_lock);

    ret = usb_fill_control_urb(dev->ctrl_urb, dev->udev,
                               USB_DIR_IN | USB_TYPE_VENDOR | USB_RECIP_DEVICE,
                               AIRSPY_USB_CTRL_GET_BOARD_ID, dev->buf,
                               sizeof(dev->buf), airspy_usb_ctrl_callback, dev);
    if (ret < 0)
        goto error_free_dev;

    ret = usb_submit_urb(dev->ctrl_urb, GFP_KERNEL);
    if (ret < 0)
        goto error_free_dev;

    ret = videobuf2_queue_init(&dev->vb2_queue, &airspy_queue_ops,
                               &dev->udev->dev, &airspy_lock_ops,
                               V4L2_BUF_TYPE_VIDEO_CAPTURE, V4L2_MEMORY_MMAP,
                               NULL, dev, NULL);
    if (ret < 0)
        goto error_free_dev;

    ret = video_register_device(&dev->vdev, VFL_TYPE_VIDEO, 0, 0);
    if (ret < 0)
        goto error_free_dev;

    v4l2_ctrl_handler_init(&dev->ctrl_handler, 4);
    dev->gain_ctrl = v4l2_ctrl_new_std(&dev->ctrl_handler, &airspy_ctrl_ops,
                                       V4L2_CID_GAIN, 0, 100, 1, 0);
    if (dev->ctrl_handler.error) {
        ret = dev->ctrl_handler.error;
        goto error_unregister_video;
    }

    dev_info(&dev->udev->dev, "Airspy board ID: %d, firmware version: %d.%d\n",
             dev->board_id, dev->fw_version_major, dev->fw_version_minor);

    usb_set_intfdata(intf, dev);
    return 0;

error_unregister_video:
    video_unregister_device(&dev->vdev);
error_free_dev:
    kfree(dev);
    return ret;
}
