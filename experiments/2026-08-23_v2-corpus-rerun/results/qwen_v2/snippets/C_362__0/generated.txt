struct usb_cdc_union_desc *get_cdc_union_desc(struct usb_interface *intf)
{
    struct usb_host_interface *host_intf;
    struct usb_cdc_union_desc *desc = NULL;
    unsigned char *extra;
    int extralen;

    if (!intf)
        return NULL;

    host_intf = &intf->altsetting[0];
    extra = host_intf->extra;
    extralen = host_intf->extralen;

    if (!extra || extralen <= 0) {
        dev_dbg(&intf->dev, "No extra descriptors found\n");
        return NULL;
    }

    while (extralen > 0) {
        if (extralen < sizeof(*desc)) {
            dev_dbg(&intf->dev, "CDC union descriptor too short\n");
            return NULL;
        }

        desc = (struct usb_cdc_union_desc *)extra;

        if (desc->bDescriptorType != USB_DT_CS_INTERFACE ||
            desc->bDescriptorSubtype != USB_CDC_SUBTYPE_UNION) {
            desc = NULL;
            extra++;
            extralen--;
            continue;
        }

        if (desc->bLength > extralen) {
            dev_dbg(&intf->dev, "CDC union descriptor length exceeds available data\n");
            return NULL;
        }

        break;
    }

    if (!desc)
        dev_dbg(&intf->dev, "CDC union descriptor not found\n");

    return desc;
}
