void set_original_addr(struct msghdr *m, struct tipc_msg *msg)
{
    struct sockaddr_tipc *dst = (struct sockaddr_tipc *)m->msg_name;

    if (!dst)
        return;

    dst->family = AF_TIPC;
    dst->addrtype = TIPC_ADDR_ID;
    memset(&dst->addr, 0, sizeof(dst->addr));

    dst->addr.id.port = tipc_msg_orig_port(msg);
    dst->addr.id.node = tipc_msg_orig_node(msg);

    m->msg_namelen = sizeof(struct sockaddr_tipc);
}
