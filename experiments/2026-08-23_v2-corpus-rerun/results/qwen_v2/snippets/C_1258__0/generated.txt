void handle_gpu_process_crash(int exit_code) {
    // Record the crash for debugging purposes
    log_gpu_crash(exit_code);

    // Check if the crash occurred while compiling a shader
    if (is_shader_compilation_in_progress()) {
        // Clear the shader cache to remove potentially invalid binaries
        clear_shader_cache();
    }

    // Send outstanding replies indicating the GPU host is invalid
    send_invalid_gpu_host_replies();

    // Retrieve termination information about the crashed process
    ProcessTerminationInfo termination_info = get_process_termination_info();

    // Update the GPU data manager with the crash status
    update_gpu_data_manager_with_crash_status(termination_info);
}
