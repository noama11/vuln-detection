ext3_fsblk_t get_superblock_block(void **data, struct super_block *sb) {
    char *options = * (char **) data;
    ext3_fsblk_t block = 1;

    if (options && strncmp(options, "sb=", 3) == 0) {
        char *endptr;
        block = simple_strtoul(options + 3, &endptr, 0);

        if (endptr == options + 3) {
            /* No digits found after "sb=" */
            return 1;
        }

        if (*endptr != '\0' && *endptr != ',' && *endptr != ';') {
            /* Invalid characters after the block number */
            printk(KERN_ERR "Invalid characters in superblock option\n");
            return 1;
        }

        *data = (void *) endptr;
    }

    return block;
}
