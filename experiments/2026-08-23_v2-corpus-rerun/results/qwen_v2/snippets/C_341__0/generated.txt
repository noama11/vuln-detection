int cuse_channel_release(struct inode *inode, struct file *file) {
    struct fuse_conn *fc = file->private_data;
    int res;

    if (!fc)
        return -ENODEV;

    mutex_lock(&fc->lock);
    list_del(&fc->entry);
    mutex_unlock(&fc->lock);

    if (fc->dev) {
        device_unregister(fc->dev);
        fc->dev = NULL;
    }

    cdev_del(&fc->cdev);
    put_device(&fc->dev);

    res = fuse_dev_release(inode, file);
    return res;
}
