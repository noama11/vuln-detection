int update_char_class_node(CClassNode* cc, OnigCodePoint* vs, enum CCVALTYPE* type, enum CCSTATE* state, ScanEnv* env) {
    if (*state == CCSTATE_END) {
        return ONIGERR_INVALID_RANGE;
    }

    if (*type == CCVALTYPE_BITSET) {
        if (onig_code_to_8bit_code_table[*vs] != 0) {
            cc->bits[onig_code_to_8bit_code_table[*vs] >> 3] |= (1 << (onig_code_to_8bit_code_table[*vs] & 0x07));
        }
    } else if (*type == CCVALTYPE_CODE_RANGE) {
        if (cc->range_end == 0) {
            cc->range_start = *vs;
            cc->range_end = *vs;
        } else {
            if (*vs <= cc->range_start) {
                cc->range_end = *vs;
            } else {
                cc->range_end = *vs;
                *state = CCSTATE_END;
            }
        }
    }

    *state = CCSTATE_NORMAL;
    return 0;
}
