int handle_mgmt_tx(struct wiphy *wiphy, struct wireless_dev *wdev,
                   struct cfg80211_mgmt_tx_params *params, u64 *cookie)
{
    const u8 *frame = params->buf;
    size_t len = params->len;

    if (len < IEEE80211_MIN_HEADER_LEN)
        return -EINVAL;

    if (frame[0] & IEEE80211_FCTL_TODS) {
        struct ieee80211_mgmt *mgmt = (struct ieee80211_mgmt *)frame;

        if (len < ieee80211_mgmt_len(mgmt))
            return -EINVAL;

        if (mgmt->frame_control & cpu_to_le16(IEEE80211_FTYPE_MGMT |
                                              IEEE80211_STYPE_PROBE_RESP)) {
            struct ieee80211_probe_resp *probe_resp = &mgmt->u.probe_resp;
            struct ieee80211_mgmt_ie *mgmt_ie;

            mgmt_ie = kzalloc(sizeof(*mgmt_ie), GFP_KERNEL);
            if (!mgmt_ie)
                return -ENOMEM;

            mgmt_ie->len = len - IEEE80211_PROBE_RESP_HEADER_LEN;
            mgmt_ie->data = frame + IEEE80211_PROBE_RESP_HEADER_LEN;

            probe_resp->capability_info = le16_to_cpu(mgmt->u.probe_resp.capability_info);
            probe_resp->listen_interval = le16_to_cpu(mgmt->u.probe_resp.listen_interval);

            *cookie = (u64)(uintptr_t)mgmt_ie;
            return 0;
        } else if (mgmt->frame_control & cpu_to_le16(IEEE80211_FTYPE_MGMT |
                                                    IEEE80211_STYPE_ACTION)) {
            struct ieee80211_mgmt_action *action = &mgmt->u.action;

            if (len < IEEE80211_ACTION_HEADER_LEN)
                return -EINVAL;

            action->category = frame[IEEE80211_ACTION_HEADER_LEN];
            action->data = frame + IEEE80211_ACTION_HEADER_LEN + 1;
            action->len = len - IEEE80211_ACTION_HEADER_LEN - 1;

            return send_action_frame(wdev, action, cookie);
        }
    }

    return -EINVAL;
}
