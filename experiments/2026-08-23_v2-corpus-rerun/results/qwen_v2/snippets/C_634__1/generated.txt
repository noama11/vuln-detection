int parse_imap_fetch_response(struct ImapHeader *h, char *s) {
    char *ptr = s;
    while (*ptr && isspace(*ptr)) {
        ptr++;
    }

    while (*ptr) {
        if (strncmp(ptr, "FLAGS", 5) == 0) {
            ptr += 5;
            while (*ptr && isspace(*ptr)) {
                ptr++;
            }
            if (*ptr != '(') {
                return -1;
            }
            ptr++;
            h->flags = 0;
            while (*ptr && *ptr != ')') {
                if (*ptr == ' ') {
                    ptr++;
                    continue;
                }
                if (strncmp(ptr, "\\Seen", 6) == 0) {
                    h->flags |= 1 << 0;
                    ptr += 6;
                } else if (strncmp(ptr, "\\Answered", 11) == 0) {
                    h->flags |= 1 << 1;
                    ptr += 11;
                } else if (strncmp(ptr, "\\Flagged", 9) == 0) {
                    h->flags |= 1 << 2;
                    ptr += 9;
                } else if (strncmp(ptr, "\\Deleted", 9) == 0) {
                    h->flags |= 1 << 3;
                    ptr += 9;
                } else if (strncmp(ptr, "\\Draft", 7) == 0) {
                    h->flags |= 1 << 4;
                    ptr += 7;
                } else if (strncmp(ptr, "\\Recent", 8) == 0) {
                    h->flags |= 1 << 5;
                    ptr += 8;
                } else {
                    return -1;
                }
            }
            if (*ptr != ')') {
                return -1;
            }
            ptr++;
        } else if (strncmp(ptr, "UID", 3) == 0) {
            ptr += 3;
            while (*ptr && isspace(*ptr)) {
                ptr++;
            }
            if (!isdigit(*ptr)) {
                return -1;
            }
            h->uid = 0;
            while (*ptr && isdigit(*ptr)) {
                h->uid = h->uid * 10 + (*ptr - '0');
                ptr++;
            }
        } else if (strncmp(ptr, "INTERNALDATE", 12) == 0) {
            ptr += 12;
            while (*ptr && isspace(*ptr)) {
                ptr++;
            }
            if (*ptr != '"') {
                return -1;
            }
            ptr++;
            h->internal_date[0] = '\0';
            int i = 0;
            while (*ptr && *ptr != '"') {
                h->internal_date[i++] = *ptr++;
                if (i >= sizeof(h->internal_date) - 1) {
                    return -1;
                }
            }
            h->internal_date[i] = '\0';
            if (*ptr != '"') {
                return -1;
            }
            ptr++;
        } else if (strncmp(ptr, "RFC822.SIZE", 11) == 0) {
            ptr += 11;
            while (*ptr && isspace(*ptr)) {
                ptr++;
            }
            if (!isdigit(*ptr)) {
                return -1;
            }
            h->size = 0;
            while (*ptr && isdigit(*ptr)) {
                h->size = h->size * 10 + (*ptr - '0');
                ptr++;
            }
        } else if (strncmp(ptr, "BODY", 4) == 0 || strncmp(ptr, "RFC822.HEADER", 13) == 0) {
            return -2;
        } else {
            return -1;
        }

        while (*ptr && isspace(*ptr)) {
            ptr++;
        }
    }

    return 0;
}
