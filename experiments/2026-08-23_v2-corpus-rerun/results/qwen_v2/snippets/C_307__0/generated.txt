int attach_usb_serial_for_treo_kyocera(struct usb_serial *serial) {
    int i;
    struct usb_host_endpoint *ep;

    if ((serial->num_interrupt_endpoints < 1) || (serial->num_bulk_in_endpoints < 2)) {
        dev_err(&serial->dev, "Missing required endpoints for Treo/Kyocera device\n");
        return -ENODEV;
    }

    for (i = 0; i < serial->num_interrupt_endpoints; i++) {
        ep = serial->interrupt_in_endpoint[i];
        if (ep && (ep->desc.bmAttributes & USB_ENDPOINT_XFERTYPE_MASK) == USB_ENDPOINT_XFER_INT) {
            break;
        }
    }

    if (i == serial->num_interrupt_endpoints) {
        dev_err(&serial->dev, "No valid interrupt endpoint found for Treo/Kyocera device\n");
        return -ENODEV;
    }

    // Swap the first and second bulk in endpoints
    struct usb_host_endpoint *temp_bulk_in = serial->bulk_in_endpoint[0];
    serial->bulk_in_endpoint[0] = serial->bulk_in_endpoint[1];
    serial->bulk_in_endpoint[1] = temp_bulk_in;

    // Swap the first and second interrupt endpoints
    struct usb_host_endpoint *temp_interrupt = serial->interrupt_in_endpoint[0];
    serial->interrupt_in_endpoint[0] = serial->interrupt_in_endpoint[1];
    serial->interrupt_in_endpoint[1] = temp_interrupt;

    return 0;
}
