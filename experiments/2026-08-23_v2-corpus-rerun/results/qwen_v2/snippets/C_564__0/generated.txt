void remove_resolution_info(StringInfo *bim_profile) {
    if (bim_profile->len < 16)
        return;

    char *data = (char *)bim_profile->data;
    for (size_t i = 0; i <= bim_profile->len - 4; ++i) {
        if (data[i] == '8' && data[i + 1] == 'B' && data[i + 2] == 'I' && data[i + 3] == 'M') {
            uint16_t pixel_info = *(uint16_t *)(data + i + 4);
            uint16_t count = *(uint16_t *)(data + i + 6);

            if (pixel_info == 0x0001 && count > 0) {
                memmove(data + i, data + i + 12, bim_profile->len - (i + 12));
                bim_profile->len -= 12;
            }
            break;
        }
    }
}
