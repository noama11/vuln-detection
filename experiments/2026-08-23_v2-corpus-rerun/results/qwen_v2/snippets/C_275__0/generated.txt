long handle_dedupe_ioctl(struct file *file, void __user *arg)
{
    struct dedupe_args __user *uargs = arg;
    struct dedupe_args *kargs;
    long ret = 0;

    kargs = kmalloc(sizeof(*kargs), GFP_KERNEL);
    if (!kargs)
        return -ENOMEM;

    if (copy_from_user(kargs, uargs, sizeof(*kargs))) {
        ret = -EFAULT;
        goto out_free;
    }

    ret = do_dedupe(file, kargs);
    if (ret < 0)
        goto out_free;

    if (copy_to_user(uargs, kargs, sizeof(*kargs)))
        ret = -EFAULT;

out_free:
    kfree(kargs);
    return ret;
}
