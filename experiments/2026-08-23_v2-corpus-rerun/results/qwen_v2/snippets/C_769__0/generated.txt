int nfc_target_deactivate(struct sk_buff *skb, struct genl_info *info)
{
	int dev_idx, target_idx;
	struct nfc_dev *nfc_dev;
	int rc = 0;

	if (!info->attrs[NFC_ATTR_DEVICE_INDEX] ||
	    !info->attrs[NFC_ATTR_TARGET_INDEX])
		return -EINVAL;

	dev_idx = nla_get_u32(info->attrs[NFC_ATTR_DEVICE_INDEX]);
	target_idx = nla_get_u32(info->attrs[NFC_ATTR_TARGET_INDEX]);

	nfc_dev = nfc_get_device(dev_idx);
	if (!nfc_dev)
		return -ENODEV;

	rc = nfc_dev->ops->target_deactivate(nfc_dev, target_idx);
	nfc_put_device(nfc_dev);

	return rc;
}
