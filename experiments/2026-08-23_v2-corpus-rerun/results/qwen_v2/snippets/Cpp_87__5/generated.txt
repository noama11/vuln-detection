void check_user_password_validity(struct pdf_struct *pdf, int R, const char *O, const char *U, int32_t P, int EM, unsigned length, unsigned oulen) {
    // Compute candidate encryption key based on passwords and revision
    unsigned char key_candidate[32];
    if (R == 2) {
        // Use SHA256 for revision 2
        SHA256_CTX sha256;
        SHA256_Init(&sha256);
        SHA256_Update(&sha256, O, strlen(O));
        SHA256_Final(key_candidate, &sha256);
    } else {
        // Use MD5 for other revisions
        MD5_CTX md5;
        MD5_Init(&md5);
        MD5_Update(&md5, O, strlen(O));
        MD5_Final(key_candidate, &md5);
    }

    // Compare computed key with expected value from U
    if (memcmp(key_candidate, U, length) == 0) {
        // Mark password as empty and set decryptable
        pdf->password_empty = true;
        pdf->decryptable = true;
    } else {
        // Password is invalid
        pdf->password_empty = false;
        pdf->decryptable = false;
    }
}
