int mov_read_dref(MOVContext *c, AVIOContext *pb, MOVAtom atom) {
    int entries;
    int i;
    int size;

    if (!c->nb_streams) {
        avio_skip(pb, atom.size);
        return 0;
    }

    entries = avio_rb32(pb);
    if (entries < 0)
        return AVERROR_INVALIDDATA;

    for (i = 0; i < entries; i++) {
        char type[5];
        int flags;
        int len;

        avio_read(pb, type, 4);
        type[4] = 0;
        flags = avio_rb32(pb);
        len = avio_rb32(pb);

        if (len < 0)
            return AVERROR_INVALIDDATA;

        if (!strcmp(type, "alis")) {
            avio_skip(pb, 4); // skip alias data
        } else if (!strcmp(type, "fods")) {
            avio_skip(pb, len);
        } else if (!strcmp(type, "url ")) {
            int is_self_contained = flags & 1;
            if (!is_self_contained) {
                char *filename = av_malloc(len + 1);
                if (!filename)
                    return AVERROR(ENOMEM);
                avio_read(pb, filename, len);
                filename[len] = 0;
                // Process filename as needed
                av_free(filename);
            } else {
                avio_skip(pb, len);
            }
        } else {
            avio_skip(pb, len);
        }
    }

    return 0;
}
