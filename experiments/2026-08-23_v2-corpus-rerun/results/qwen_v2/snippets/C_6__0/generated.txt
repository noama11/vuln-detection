int epoll_ctl(int epfd, int op, int fd, struct epoll_event __user *event) {
    // Implementation of the logic described in the docstring
    // This is a simplified and illustrative version based on the specification

    if (epfd < 0 || fd < 0 || !event) {
        return -EINVAL;
    }

    // Assume necessary structures and functions exist for retrieving file objects
    // and checking for poll support (e.g., f_get_file, is_poll_capable, etc.)

    // Retrieve the epoll instance and target file
    struct file *epfile = f_get_file(epfd);
    struct file *target_file = f_get_file(fd);

    if (!epfile || !target_file) {
        return -EBADF;
    }

    if (epfile == target_file) {
        return -EINVAL;
    }

    if (!is_poll_capable(target_file)) {
        return -EPFNOSUPPORT;
    }

    // Acquire necessary locks
    acquire_mutex(&epfile->lock);

    int result = -EINVAL;

    switch (op) {
        case EPOLL_CTL_ADD:
            result = add_event(epfile, fd, event);
            break;
        case EPOLL_CTL_DEL:
            result = remove_event(epfile, fd);
            break;
        case EPOLL_CTL_MOD:
            result = modify_event(epfile, fd, event);
            break;
        default:
            break;
    }

    release_mutex(&epfile->lock);

    return result;
}
