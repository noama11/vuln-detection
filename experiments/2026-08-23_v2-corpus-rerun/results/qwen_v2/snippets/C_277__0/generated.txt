int set_apparmor_process_attr(struct task_struct *task, char *name, void *value, size_t size) {
    if (size == 0) {
        return -EINVAL;
    }

    if (current != task) {
        return -EPERM;
    }

    char *str_value = value;
    if (str_value[size - 1] != '\0') {
        return -EINVAL;
    }

    if (strcmp(name, "profile") == 0) {
        int ret = apparmor_set_profile(task, str_value);
        if (ret < 0) {
            audit_log_message(AUDIT_AA_SETATTR, "Failed to set profile: %s", str_value);
            return ret;
        }
    } else if (strcmp(name, "permissions") == 0) {
        int ret = apparmor_set_permissions(task, str_value);
        if (ret < 0) {
            audit_log_message(AUDIT_AA_SETATTR, "Failed to set permissions: %s", str_value);
            return ret;
        }
    } else {
        return -EINVAL;
    }

    return size;
}
