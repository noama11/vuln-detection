int ati_remote_wonder_init(struct usb_interface *interface, const struct usb_device_id *id)
{
    struct usb_host_interface *iface_desc;
    struct usb_endpoint_descriptor *endpoint;
    struct ati_remote_wonder_dev *dev;
    int i, retval = -ENOMEM;

    iface_desc = &interface->cur_altsetting[0];
    if (iface_desc->desc.bNumEndpoints < 1)
        return -ENODEV;

    dev = kzalloc(sizeof(*dev), GFP_KERNEL);
    if (!dev)
        return -ENOMEM;

    dev->udev = usb_get_dev(interface_to_usbdev(interface));
    dev->interface = interface;

    if (iface_desc->desc.bNumEndpoints < 2) {
        kfree(dev);
        return -ENODEV;
    }

    for (i = 0; i < iface_desc->desc.bNumEndpoints; ++i) {
        endpoint = &iface_desc->endpoint[i].desc;
        if (usb_endpoint_is_int_in(endpoint)) {
            dev->int_in_endpoint = endpoint;
            break;
        }
    }

    if (!dev->int_in_endpoint) {
        kfree(dev);
        return -ENODEV;
    }

    usb_set_intfdata(interface, dev);

    if (usb_driver_claim_interface(&ati_remote_wonder_driver, interface, dev)) {
        kfree(dev);
        return -EBUSY;
    }

    if (ati_remote_wonder_setup_dev(dev)) {
        usb_set_intfdata(interface, NULL);
        kfree(dev);
        return -EFAULT;
    }

    if (sysfs_create_group(&interface->dev.kobj, &ati_remote_wonder_attr_group)) {
        ati_remote_wonder_cleanup_dev(dev);
        usb_set_intfdata(interface, NULL);
        kfree(dev);
        return -EFAULT;
    }

    dev->enabled = 1;
    return 0;
}
