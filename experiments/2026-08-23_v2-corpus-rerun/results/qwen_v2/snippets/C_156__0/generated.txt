OM_uint32 process_spnego_continuation(OM_uint32 *minstat, gss_buffer_t buf, gss_ctx_id_t *ctx,
                                      gss_buffer_t *responseToken, gss_buffer_t *mechListMIC,
                                      OM_uint33 *negState, send_token_flag *return_token) {
    OM_uint32 status = 0;
    OM_uint32 major_status, minor_status;
    gss_OID mech_type = GSS_C_NULL_OID;
    gss_buffer_desc token = GSS_C_EMPTY_BUFFER;

    if (!buf || !buf->length || !buf->value) {
        *minstat = GSS_S_BAD_SIG;
        return GSS_S_FAILURE;
    }

    if (!ctx || !negState || !return_token) {
        *minstat = GSS_S_FAILURE;
        return GSS_S_FAILURE;
    }

    major_status = gss_get_mech_type(&minor_status, buf, &mech_type);
    if (major_status != GSS_S_COMPLETE) {
        *minstat = minor_status;
        return major_status;
    }

    major_status = gss_init_sec_context(&minor_status, GSS_C_NO_CREDENTIAL, ctx, mech_type,
                                        GSS_C_MUTUAL_FLAG | GSS_C_REPLAY_FLAG, GSS_C_INDEFINITE,
                                        GSS_C_NO_CHANNEL_BINDINGS, NULL, NULL, &token, NULL, NULL, NULL);
    if (major_status != GSS_S_CONTINUE_NEEDED && major_status != GSS_S_COMPLETE) {
        *minstat = minor_status;
        return major_status;
    }

    if (responseToken) {
        responseToken->length = token.length;
        responseToken->value = token.value;
    }

    if (mechListMIC) {
        mechListMIC->length = 0;
        mechListMIC->value = NULL;
    }

    *negState = (major_status == GSS_S_COMPLETE) ? 1 : 0;
    *return_token = (major_status == GSS_S_CONTINUE_NEEDED) ? SEND_TOKEN : NO_TOKEN;

    return GSS_S_COMPLETE;
}
