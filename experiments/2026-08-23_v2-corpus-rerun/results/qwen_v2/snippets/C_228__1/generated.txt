int path_component_to_string(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int i = 0;
    int j = 0;

    while (i < fromlen && j < tolen - 1) {
        unsigned char type = from[i];

        if (type == 0) {
            if (j > 0 && to[j - 1] != '/') {
                to[j++] = '/';
            }
        } else if (type == 1) {
            if (j > 0 && to[j - 1] != '.') {
                to[j++] = '.';
                if (j < tolen && to[j - 1] != '/') {
                    to[j++] = '.';
                }
            }
        } else {
            while (i < fromlen && from[i] != 0) {
                to[j++] = from[i++];
            }
            if (j < tolen && i < fromlen && from[i] == 0 && j > 0 && to[j - 1] != '/') {
                to[j++] = '/';
            }
        }

        i++;
    }

    if (j >= tolen) {
        return -1;
    }

    to[j] = '\0';
    return 0;
}
