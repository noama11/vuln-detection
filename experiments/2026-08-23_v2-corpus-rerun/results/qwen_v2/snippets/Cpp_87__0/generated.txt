void pdf_flag_object(struct pdf_struct *pdf, struct pdf_obj *obj, enum pdf_flag flag) {
    if (!pdf || !obj) {
        return;
    }

    // Set the bit corresponding to the specified flag
    obj->flags |= (1 << flag);

    // Early exit if debug flag is not enabled
    if (!(pdf->flags & PDF_DEBUG_FLAG)) {
        return;
    }

    // Determine the appropriate debug message based on the flag
    const char *message = nullptr;
    switch (flag) {
        case PDF_FLAG_UNCLOSED_DICT:
            message = "Unclosed dictionary";
            break;
        case PDF_FLAG_BAD_STREAM_FILTER:
            message = "Bad stream filter";
            break;
        case PDF_FLAG_ENCRYPTED:
            message = "Object is encrypted";
            break;
        default:
            message = "Unknown flag";
            break;
    }

    if (message) {
        std::cerr << message << " (Object ID: " << obj->id << ")\n";
    }
}
