#include <linux/slab.h>
#include <linux/cred.h>
#include <linux/key.h>
#include <linux/errno.h>

int copy_credentials(struct task_struct *p, unsigned long clone_flags)
{
    struct cred *new;
    int ret = 0;

    new = prepare_creds();
    if (!new)
        return -ENOMEM;

    if (clone_flags & CLONE_NEWUSER) {
        struct user_namespace *ns;

        ns = create_user_namespace(current_cred());
        if (IS_ERR(ns)) {
            ret = PTR_ERR(ns);
            goto error;
        }

        new->user_ns = ns;
    }

    if (clone_flags & CLONE_THREAD) {
        new->thread_keyring = NULL;
        new->process_keyring = p->cred->process_keyring;
        new->session_keyring = p->cred->session_keyring;
    } else {
        new->thread_keyring = NULL;
        new->process_keyring = NULL;
        new->session_keyring = NULL;
    }

    if (clone_flags & CLONE_NEWKEYS) {
        key_put(new->thread_keyring);
        key_put(new->process_keyring);
        key_put(new->session_keyring);
        new->thread_keyring = NULL;
        new->process_keyring = NULL;
        new->session_keyring = NULL;
    }

    if (clone_flags & CLONE_CREDENTIALS) {
        if (atomic_read(&p->cred->usage) > 1) {
            get_cred(new);
        } else {
            new->uid = p->cred->uid;
            new->euid = p->cred->euid;
            new->fsuid = p->cred->fsuid;
            new->gid = p->cred->gid;
            new->egid = p->cred->egid;
            new->fsgid = p->cred->fsgid;
        }
    }

    ret = commit_creds(new);
    if (ret < 0)
        goto error;

    return 0;

error:
    abort_creds(new);
    return ret;
}
