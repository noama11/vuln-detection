long enumerate_media_entities(struct media_device *mdev, struct media_entity_desc __user *uent)
{
    struct media_entity_desc kent;
    int ret;

    if (!mdev || !uent)
        return -EINVAL;

    memset(&kent, 0, sizeof(kent));

    ret = media_entity_get(mdev, &kent);
    if (ret < 0)
        return ret;

    if (copy_to_user(uent, &kent, sizeof(kent)))
        return -EFAULT;

    return 0;
}
