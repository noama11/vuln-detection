int comedi_ioctl_devinfo(struct comedi_device *dev, struct comedi_devinfo __user *arg, struct file *file)
{
	struct comedi_devinfo info;
	int rsubdev = -1;
	int wsubdev = -1;
	int i;

	memset(&info, 0, sizeof(info));
	info.version_code = dev->version_code;
	info.n_subdevs = dev->n_subdevices;

	for (i = 0; i < dev->n_subdevices; i++) {
		if (dev->subdevices[i].type != COMEDI_SUBD_UNUSED) {
			if (dev->subdevices[i].subdev_flags & SDF_READ)
				rsubdev = i;
			if (dev->subdevices[i].subdev_flags & SDF_WRITE)
				wsubdev = i;
		}
	}

	strcpy(info.driver_name, dev->driver->driver_name);
	strcpy(info.board_name, dev->board_name);
	info.read_subdev = rsubdev;
	info.write_subdev = wsubdev;

	if (copy_to_user(arg, &info, sizeof(info)))
		return -EFAULT;

	return 0;
}
