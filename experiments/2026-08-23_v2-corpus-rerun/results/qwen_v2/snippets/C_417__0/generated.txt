long read_key_value(key_serial_t keyid, char __user *buffer, size_t buflen) {
    struct key *key;
    long ret;

    key = key_lookup(keyid);
    if (IS_ERR_OR_NULL(key)) {
        return PTR_ERR(key);
    }

    if (key->flags & KEY_FLAG_NEGATIVE) {
        key_put(key);
        return -ENOKEY;
    }

    if (!key_permission(current_cred(), key, KEY_NEED_READ)) {
        key_put(key);
        return -EACCES;
    }

    ret = key_read(key, buffer, buflen);
    if (ret < 0) {
        key_put(key);
        return ret;
    }

    key_put(key);
    return ret;
}
