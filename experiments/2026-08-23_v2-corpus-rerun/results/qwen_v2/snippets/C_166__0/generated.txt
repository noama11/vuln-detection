int send_raw_hid_request(struct hid_device *hid, unsigned char reportnum, __u8 *buf, size_t count, unsigned char report_type, int reqtype) {
    int result;
    __u8 *output_buf;

    if (buf[0] != report_type) {
        return -EINVAL;
    }

    output_buf = (__u8 *)malloc(count);
    if (!output_buf) {
        return -ENOMEM;
    }

    memcpy(output_buf, buf, count);

    result = hid_hw_raw_request(hid, reportnum, output_buf, count, reqtype, HID_REQ_GET_REPORT);

    free(output_buf);
    return result;
}
