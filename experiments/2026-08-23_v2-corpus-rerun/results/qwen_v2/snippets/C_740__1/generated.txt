int smb2_write(unsigned int xid, struct cifs_io_parms *io_parms, unsigned int *nbytes,
               struct kvec *iov, int n_vec) {
    struct smb2_write_request *req;
    struct smb2_write_response *resp;
    int rc = 0;
    int bytes_written = 0;

    if (!io_parms || !nbytes || !iov || n_vec <= 0) {
        return -EINVAL;
    }

    req = cifs_smb2_allocate_request(io_parms->tcon, SMB2_WRITE, sizeof(*req));
    if (!req) {
        return -ENOMEM;
    }

    req->hdr.MessageId = cpu_to_le64(xid);
    req->PersistentFileId = io_parms->persistent_fid;
    req->VolatileFileId = io_parms->volatile_fid;
    req->Offset = cpu_to_le64(io_parms->offset);
    req->RemainingBytes = 0;

    resp = cifs_smb2_allocate_response(SMB2_WRITE, sizeof(*resp));
    if (!resp) {
        rc = -ENOMEM;
        goto out;
    }

    rc = cifs_send_recv(io_parms->tcon->ses->server, req, iov, n_vec, resp);
    if (rc < 0) {
        goto out;
    }

    bytes_written = le32_to_cpu(resp->DataLength);
    *nbytes = bytes_written;

out:
    kfree(req);
    kfree(resp);
    return rc;
}
