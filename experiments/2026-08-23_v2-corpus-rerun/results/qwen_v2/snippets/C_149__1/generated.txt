int get_pppol2tp_sock_opt(struct socket *sock, int level, int optname,
                          char __user *optval, int __user *optlen)
{
    int ret = 0;
    struct sock *sk = sock->sk;
    struct pppol2tp_session *session;
    int len;

    if (level != SOL_PPPOL2TP)
        return -EINVAL;

    if (get_user(len, optlen))
        return -EFAULT;

    if (len < 0)
        return -EINVAL;

    if (!sock->sk->sk_state || sock->sk->sk_state != TCP_ESTABLISHED)
        return -ENOTCONN;

    session = pppol2tp_get_session(sk);
    if (!session)
        return -ENOENT;

    if (session->session_id == 0) {
        ret = pppol2tp_get_tunnel_opt(sk, optname, optval, len);
    } else {
        ret = pppol2tp_get_session_opt(session, optname, optval, len);
    }

    if (ret < 0)
        return ret;

    if (put_user(ret, optlen))
        return -EFAULT;

    return 0;
}
