void append_checksum_to_control_message(struct msghdr *msg, struct sk_buff *skb, int tlen, int offset) {
    if (!skb->ip_summed) {
        return;
    }

    if (offset > 0) {
        __sum16 partial_csum = csum_partial(skb->data + tlen, offset, 0);
        skb->csum = csum_sub(skb->csum, partial_csum);
    }

    msg->msg_controllen = CMSG_SPACE(sizeof(__sum16));
    struct cmsghdr *cmsg = CMSG_FIRSTHDR(msg);
    cmsg->cmsg_level = SOL_IP;
    cmsg->cmsg_type = IP_RECV_CHECKSUM;
    cmsg->cmsg_len = CMSG_LEN(sizeof(__sum16));
    *(__sum16 *)CMSG_DATA(cmsg) = skb->csum;
}
