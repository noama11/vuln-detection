int
print_ipv6_routing_header(netdissect_options *ndo, const u_char *bp, const u_char *bp2)
{
    const struct ip6_rthdr *rthdr;
    int len, i;
    u_int hdrlen;

    rthdr = (const struct ip6_rthdr *)bp;

    ND_PRINT((ndo, "Routing Header: Type %u, Segments Left %u", rthdr->ip6r_type, rthdr->ip6r_segleft));

    hdrlen = (rthdr->ip6r_len + 1) * 8;
    if (ndo->ndo_snapend < bp + hdrlen) {
        ND_PRINT((ndo, "[|truncated]"));
        return -1;
    }

    len = 0;
    ND_PRINT((ndo, "\n    "));

    for (i = 0; i <= rthdr->ip6r_segleft; i++) {
        const struct in6_addr *addr = &rthdr->ip6r_seg[i];
        ND_PRINT((ndo, "%s", ip6addr_string(ndo, addr)));
        if (i < rthdr->ip6r_segleft)
            ND_PRINT((ndo, ", "));
    }

    return hdrlen;
}
