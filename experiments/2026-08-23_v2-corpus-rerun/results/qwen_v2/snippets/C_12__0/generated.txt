int set_binprm_credentials(struct linux_binprm *bprm) {
    const struct cred *old_cred = current_cred();
    struct cred *new_cred;
    int ret;

    new_cred = prepare_creds();
    if (!new_cred)
        return -ENOMEM;

    if (bprm->capable) {
        if (old_cred->uid != 0 || old_cred->euid != 0) {
            ret = -EPERM;
            goto error;
        }
    }

    if (bprm->unsafe & BINPRM_SAFE) {
        ret = -EACCES;
        goto error;
    }

    if (bprm->file_caps) {
        if (cap_isclear(bprm->cap_effective)) {
            cap_clear(new_cred->cap_effective);
        } else {
            cap_set_flag(new_cred->cap_effective, CAP_EFFECTIVE, bprm->cap_effective, CAP_SET);
        }
    }

    if (ptrace_capable(old_cred)) {
        new_cred->cap_inheritable = cap_intersect(new_cred->cap_inheritable, old_cred->cap_permitted);
    }

    if (cap_isclear(new_cred->cap_effective)) {
        new_cred->cap_permitted = cap_dup(new_cred->cap_effective);
    }

    audit_log_cap(new_cred->cap_effective);

    ret = commit_creds(new_cred);
    if (ret < 0)
        return ret;

    return 0;

error:
    abort_creds(new_cred);
    return ret;
}
