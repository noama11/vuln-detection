#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <unistd.h>

int set_interface_variable(const char *iface, const char *var, const char *name, uint32_t val) {
    char path[256];
    FILE *fp;

    // Construct the file path using the provided interface name and variable format
    snprintf(path, sizeof(path), var, iface);

    // Check for path traversal vulnerabilities
    if (strstr(path, "../") != NULL) {
        fprintf(stderr, "Error: Path traversal attempt detected for variable '%s'.\n", name);
        return -1;
    }

    // Check if the file exists and can be opened for writing
    fp = fopen(path, "w");
    if (!fp) {
        perror("Error opening file");
        return -1;
    }

    // Write the value to the file
    if (fprintf(fp, "%u\n", val) < 0) {
        perror("Error writing to file");
        fclose(fp);
        return -1;
    }

    // Close the file
    fclose(fp);
    return 0;
}
