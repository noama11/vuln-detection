void process_sas_commands(void) {
    struct sas_command *cmd, *tmp;

    list_for_each_entry_safe(cmd, tmp, &command_list, entry) {
        struct sas_task *task = cmd->task;
        unsigned long flags;

        if (!task)
            continue;

        spin_lock_irqsave(&task->state_lock, flags);

        if (task_needs_reset(task)) {
            int rc;

            if (task->state == SAS_TASK_STATE_LOGGED_IN) {
                rc = sas_recover_lu(task);
                if (rc) {
                    dev_err(&task->dev, "Failed to recover LU for task %p\n", task);
                } else {
                    dev_info(&task->dev, "Recovered LU for task %p\n", task);
                }
            } else if (task->state == SAS_TASK_STATE_I_T_NEXUS_LOST) {
                rc = sas_recover_i_t(task);
                if (rc) {
                    dev_err(&task->dev, "Failed to recover I_T nexus for task %p\n", task);
                } else {
                    dev_info(&task->dev, "Recovered I_T nexus for task %p\n", task);
                }
            } else {
                dev_warn(&task->dev, "Unexpected task state %d for task %p during recovery\n",
                         task->state, task);
            }

            task_clear_reset_needed(task);
        }

        spin_unlock_irqrestore(&task->state_lock, flags);
    }
}
