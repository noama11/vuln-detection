void CreateBufferResource(PP_Instance instance, uint32_t size, HostResource* result_resource, ppapi::proxy::SerializedHandle* result_shm_handle) {
  // Check for necessary permissions and resource creation validity
  if (!HasPermission(instance)) {
    return;
  }

  // Create the buffer resource
  HostResource buffer_resource = CreateHostResource(instance, size);
  if (!buffer_resource.IsValid()) {
    return;
  }

  // Retrieve the shared memory handle
  ppapi::proxy::SerializedHandle shm_handle;
  if (!GetSharedMemoryHandle(buffer_resource, &shm_handle)) {
    return;
  }

  // Assign the result values
  *result_resource = buffer_resource;
  *result_shm_handle = shm_handle;
}
