int virtio_gpu_object_create(struct virtio_gpu_device *vgdev, unsigned long size, bool kernel, bool pinned, struct virtio_gpu_object **bo_ptr) {
    struct virtio_gpu_object *bo;
    int ret;

    if (!vgdev || !bo_ptr) {
        return -EINVAL;
    }

    *bo_ptr = NULL;

    bo = kzalloc(sizeof(*bo), GFP_KERNEL);
    if (!bo) {
        return -ENOMEM;
    }

    bo->size = size;
    bo->kernel = kernel;
    bo->pinned = pinned;

    if (kernel) {
        bo->type = VIRTIO_GPU_OBJECT_KERNEL;
    } else {
        bo->type = VIRTIO_GPU_OBJECT_USER;
    }

    ret = virtio_gpu_mem_alloc(vgdev, size, &bo->mem);
    if (ret < 0) {
        kfree(bo);
        return ret;
    }

    if (pinned) {
        ret = virtio_gpu_pin_object(vgdev, bo);
        if (ret < 0) {
            virtio_gpu_mem_free(vgdev, bo->mem);
            kfree(bo);
            return ret;
        }
    }

    *bo_ptr = bo;
    return 0;
}
