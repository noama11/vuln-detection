long get_device_id_from_symlink(const char *link) {
    char *resolved_link = NULL;
    struct stat file_stat;
    long device_id = -1;

    resolved_link = strdup(link);
    if (!resolved_link) {
        return -1;
    }

    if (VCWD_STAT(resolved_link, &file_stat) == -1) {
        free(resolved_link);
        return -1;
    }

    device_id = (long) file_stat.st_dev;
    free(resolved_link);
    return device_id;
}
