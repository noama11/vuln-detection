EbmlElement* findNextEbmlElement(IOCallback &DataStream, const EbmlSemanticContext &Context, int &UpperLevel, uint64 MaxDataSize, bool AllowDummyElt, unsigned int MaxLowerLevel) {
    // Read potential ID from the data stream
    uint64 id = 0;
    size_t bytesRead = DataStream.read(reinterpret_cast<char*>(&id), sizeof(uint64));
    if (bytesRead != sizeof(uint64))
        return nullptr;

    // Check if the ID is valid
    if (!Context.isValidElementID(id))
        return nullptr;

    // Read the size of the element
    uint64 size = 0;
    bytesRead = DataStream.read(reinterpret_cast<char*>(&size), sizeof(uint64));
    if (bytesRead != sizeof(uint64))
        return nullptr;

    // Check if the size exceeds the maximum allowed data size
    if (size > MaxDataSize)
        return nullptr;

    // Check if the element level is within the allowed range
    unsigned int lowerLevel = getEbmlElementLevel(id);
    if (lowerLevel > MaxLowerLevel)
        return nullptr;

    // Check if dummy elements are allowed
    if (!AllowDummyElt && isDummyElement(id))
        return nullptr;

    // Create the EBML element using the context
    EbmlElement* element = Context.createElement(id, size);
    if (!element)
        return nullptr;

    // Update the upper level
    UpperLevel = lowerLevel;

    return element;
}
