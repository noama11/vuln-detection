int parse_rose_facilities(unsigned char *p, struct rose_facilities_struct *facilities, int len) {
    int bytes_processed = 0;
    unsigned char *end = p + len;

    while (p < end && *p != 0x00) {
        unsigned char byte = *p++;
        bytes_processed++;

        switch ((byte >> 6) & 0x03) {
            case 0x00:
                // Process facility type 0
                if (p + 1 > end)
                    return -1;

                facilities->source_address = *p++;
                bytes_processed++;

                if (p + 1 > end)
                    return -1;

                facilities->source_call_length = *p++;
                bytes_processed++;

                if (facilities->source_call_length > ROSE_MAX_CALLSIGN_LENGTH)
                    return -1;

                if (p + facilities->source_call_length > end)
                    return -1;

                memcpy(facilities->source_call, p, facilities->source_call_length);
                p += facilities->source_call_length;
                bytes_processed += facilities->source_call_length;
                break;

            case 0x01:
                // Process facility type 1
                if (p + 1 > end)
                    return -1;

                facilities->dest_address = *p++;
                bytes_processed++;

                if (p + 1 > end)
                    return -1;

                facilities->dest_call_length = *p++;
                bytes_processed++;

                if (facilities->dest_call_length > ROSE_MAX_CALLSIGN_LENGTH)
                    return -1;

                if (p + facilities->dest_call_length > end)
                    return -1;

                memcpy(facilities->dest_call, p, facilities->dest_call_length);
                p += facilities->dest_call_length;
                bytes_processed += facilities->dest_call_length;
                break;

            case 0x02:
                // Process facility type 2
                if (p + 1 > end)
                    return -1;

                facilities->facility_type = *p++;
                bytes_processed++;
                break;

            case 0x03:
                // Termination byte
                return bytes_processed;

            default:
                // Invalid facility type
                return -1;
        }
    }

    if (p >= end)
        return -1;

    return bytes_processed;
}
