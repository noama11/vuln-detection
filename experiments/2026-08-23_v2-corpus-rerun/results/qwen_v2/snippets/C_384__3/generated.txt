OPJ_BOOL write_jpeg2000_tile_parts(opj_j2k_t *p_j2k, OPJ_BYTE *p_data, OPJ_UINT32 *p_data_written,
                                   OPJ_UINT32 p_total_data_size, opj_stream_private_t *p_stream,
                                   struct opj_event_mgr *p_manager) {
    OPJ_UINT32 data_written = 0;
    OPJ_BOOL success = OPJ_TRUE;

    for (OPJ_UINT32 tile_index = 0; tile_index < p_j2k->num_tiles; ++tile_index) {
        opj_tile_t *tile = &p_j2k->tiles[tile_index];

        for (OPJ_UINT32 part_index = 0; part_index < tile->num_parts; ++part_index) {
            opj_tile_part_t *tile_part = &tile->tile_parts[part_index];

            if (data_written + tile_part->size > p_total_data_size) {
                opj_event_msg(p_manager, EVT_ERROR, "Not enough space to write tile part data.\n");
                return OPJ_FALSE;
            }

            OPJ_UINT32 header_size = tile_part->header_size;
            OPJ_UINT32 data_size = tile_part->data_size;

            if (tile_part->is_first_part) {
                header_size += 4; /* Add space for marker */
            }

            if (data_written + header_size > p_total_data_size) {
                opj_event_msg(p_manager, EVT_ERROR, "Not enough space to write tile part header.\n");
                return OPJ_FALSE;
            }

            memcpy(p_data + data_written, tile_part->header, header_size);
            data_written += header_size;

            memcpy(p_data + data_written, tile_part->data, data_size);
            data_written += data_size;

            if (!opj_stream_write(p_stream, p_data + data_written - header_size - data_size, header_size + data_size)) {
                opj_event_msg(p_manager, EVT_ERROR, "Failed to write tile part to stream.\n");
                success = OPJ_FALSE;
                break;
            }

            opj_event_msg(p_manager, EVT_INFO, "Wrote tile part %u of tile %u.\n", part_index, tile_index);
        }

        if (!success) {
            break;
        }
    }

    *p_data_written = data_written;
    return success;
}
