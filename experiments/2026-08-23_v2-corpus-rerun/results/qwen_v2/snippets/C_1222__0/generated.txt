void initiate_zero_suggest(const AutocompleteInput* input, bool minimal_changes) {
    // Clear previous matches
    clear_previous_matches();

    // Check if input is valid for zero-suggest
    if (!is_valid_for_zero_suggest(input)) {
        return;
    }

    // Prepare suggest URL based on current context and eligibility criteria
    char* suggest_url = prepare_suggest_url(input, minimal_changes);

    // Update the suggest URL
    update_suggest_url(suggest_url);

    // Handle caching of suggestions
    cache_suggestions(input, suggest_url);

    // Run the suggestion process
    run_suggestion_process();

    // Free allocated memory
    free(suggest_url);
}
