void* allocate_non_paged_pool(size_t size) {
    if (size == 0) {
        return NULL;
    }

    // Calculate total size including header
    size_t total_size = sizeof(size_t) + size;

    // Check for overflow
    if (total_size < sizeof(size_t) || total_size < size) {
        return NULL;
    }

    void* pool = ExAllocatePoolWithTag(NonPagedPool, total_size, 'MyTag');

    if (!pool) {
        return NULL;
    }

    // Store the size in the header
    size_t* header = (size_t*)pool;
    *header = size;

    // Return pointer to the data portion
    return (void*)(header + 1);
}
