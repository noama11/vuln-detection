int unpack_summary_info(const cdf_stream_t *sst, const cdf_header_t *h,
                        cdf_summary_info_header_t *ssi, cdf_property_info_t **info,
                        size_t *count) {
    if (!sst || !h || !ssi || !info || !count) {
        return -1;
    }

    if (h->offset_summary_info > sst->size) {
        return -1;
    }

    ssi->byte_order = h->byte_order;
    ssi->os_version = h->os_version;
    ssi->os_type = h->os_type;
    ssi->class = h->class;
    ssi->property_count = h->property_count;

    if (h->section_decl_offset > sst->size) {
        return -1;
    }

    *count = h->property_count;
    *info = (cdf_property_info_t *)malloc(*count * sizeof(cdf_property_info_t));
    if (!*info) {
        return -1;
    }

    for (size_t i = 0; i < *count; ++i) {
        size_t prop_offset = h->section_decl_offset + i * sizeof(cdf_property_info_t);
        if (prop_offset + sizeof(cdf_property_info_t) > sst->size) {
            free(*info);
            *info = NULL;
            *count = 0;
            return -1;
        }

        // Assuming a helper function to read from the stream
        if (read_property_info(sst, prop_offset, &(*info)[i]) != 0) {
            free(*info);
            *info = NULL;
            *count = 0;
            return -1;
        }
    }

    return 0;
}
