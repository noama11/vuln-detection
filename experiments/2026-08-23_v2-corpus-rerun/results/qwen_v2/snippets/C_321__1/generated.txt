void *dma_alloc_coherent(struct device *dev, size_t size, dma_addr_t *dma_handle, gfp_t flags, struct dma_attrs *attrs)
{
    if (!dev)
        return NULL;

    if (dev->dma_mask && *dev->dma_mask < DMA_BIT_MASK(32))
        flags &= ~(__GFP_DMA | __GFP_HIGHMEM);

    if (attrs && dma_get_attr(DMA_ATTR_FORCE_CONTIGUOUS, attrs)) {
        void *cpu_addr = __dma_alloc(dev, size, dma_handle, flags, attrs);
        if (cpu_addr)
            return cpu_addr;
    }

    return dma_alloc_from_contiguous(dev, size, dma_handle, flags, attrs);
}
