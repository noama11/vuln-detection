int ext4_ext_grow(struct inode *inode, handle_t *handle, unsigned int flags)
{
    struct buffer_head *bh = NULL;
    struct ext4_extent_header *eh;
    ext4_lblk_t goal;
    int depth = ext4_ext_depth(inode);
    int ret = 0;

    goal = ext4_ext_find_goal(inode, 0);

    bh = ext4_get_block(inode, goal, flags);
    if (!bh)
        return -ENOMEM;

    lock_buffer(bh);
    if (ext4_journal_get_write_access(handle, bh)) {
        ret = -EIO;
        goto out;
    }

    eh = ext4_ext_header(bh);
    ext4_ext_move_extents(inode, eh, goal, flags);

    ext4_ext_init_header(inode, eh);
    ext4_set_inode_flag(inode, EXT4_INODE_EXTENTS);

    mark_inode_dirty(handle, inode);

out:
    unlock_buffer(bh);
    brelse(bh);
    return ret;
}
