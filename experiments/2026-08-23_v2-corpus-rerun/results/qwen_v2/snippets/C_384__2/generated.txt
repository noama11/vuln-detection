OPJ_BOOL write_j2k_tile_part_first(opj_j2k_t *p_j2k, OPJ_BYTE *p_data, OPJ_UINT32 *p_data_written, OPJ_UINT32 p_total_data_size, opj_stream_private_t *p_stream, struct opj_event_mgr *p_manager) {
    OPJ_UINT32 written = 0;
    OPJ_BYTE *data = p_data;

    /* Write SOT marker */
    if (written + 6 > p_total_data_size) {
        return OPJ_FALSE;
    }
    data[0] = 0xFF;
    data[1] = 0x90;
    data[2] = (OPJ_UINT8)((p_j2k->current_tile_number >> 8) & 0xFF);
    data[3] = (OPJ_UINT8)(p_j2k->current_tile_number & 0xFF);
    data[4] = (OPJ_UINT8)((p_j2k->tile_part_number >> 8) & 0xFF);
    data[5] = (OPJ_UINT8)(p_j2k->tile_part_number & 0xFF);
    written += 6;
    data += 6;

    /* Write tile component information if needed */
    if (p_j2k->cinfo->tile_component_info_needed) {
        OPJ_UINT32 i;
        for (i = 0; i < p_j2k->cinfo->num_components; ++i) {
            if (written + 2 > p_total_data_size) {
                return OPJ_FALSE;
            }
            data[0] = 0x00;
            data[1] = 0x00;
            written += 2;
            data += 2;
        }
    }

    /* Write SOD marker */
    if (written + 1 > p_total_data_size) {
        return OPJ_FALSE;
    }
    data[0] = 0xFF;
    written += 1;
    data += 1;

    /* Handle cinema format specific cases if needed */
    if (p_j2k->cinfo->cinema_mode) {
        if (written + 2 > p_total_data_size) {
            return OPJ_FALSE;
        }
        data[0] = 0x00;
        data[1] = 0x00;
        written += 2;
        data += 2;
    }

    *p_data_written = written;
    return OPJ_TRUE;
}
