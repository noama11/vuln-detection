int skip_id3_tag(SF_PRIVATE *psf) {
    unsigned char header[10];
    int32_t tag_length;

    if (psf->filelength < 10)
        return 0;

    if (sf_readf_short(psf, header, 10) != 10)
        return 0;

    if (header[0] != 'I' || header[1] != 'D' || header[2] != '3')
        return 0;

    tag_length = (header[6] << 21) | (header[7] << 14) | (header[8] << 7) | header[9];
    tag_length += 10;

    if (tag_length > psf->filelength)
        return 0;

    psf->fileoffset = tag_length;
    return 1;
}
