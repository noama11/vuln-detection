ret_t validate_user_credentials(cherokee_validator_ldap_t *ldap, cherokee_connection_t *conn) {
    ret_t result = ret_error;
    LDAPMessage *search_result = NULL;
    char *user_dn = NULL;

    if (!ldap || !conn || !conn->user || !conn->password) {
        return ret_error;
    }

    if (ldap_search_s(ldap->ld, ldap->base_dn, LDAP_SCOPE_SUBTREE,
                      ldap_build_filter(ldap, conn->user), NULL, 0, &search_result) != LDAP_SUCCESS) {
        return ret_error;
    }

    if (ldap_count_entries(ldap->ld, search_result) != 1) {
        ldap_msgfree(search_result);
        return ret_not_found;
    }

    LDAPMessage *entry = ldap_first_entry(ldap->ld, search_result);
    user_dn = ldap_get_dn(ldap->ld, entry);

    if (!user_dn) {
        ldap_msgfree(search_result);
        return ret_error;
    }

    if (ldap_bind_s(ldap->ld, user_dn, conn->password, LDAP_AUTH_SIMPLE) != LDAP_SUCCESS) {
        ldap_memfree(user_dn);
        ldap_msgfree(search_result);
        return ret_not_found;
    }

    result = ret_ok;

    ldap_memfree(user_dn);
    ldap_msgfree(search_result);
    ldap_unbind_s(ldap->ld);

    return result;
}
