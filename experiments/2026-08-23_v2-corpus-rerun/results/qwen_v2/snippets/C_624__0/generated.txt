bool resolve_bool_expr(struct xkb_context *ctx, const ExprDef *expr, bool *set_rtrn) {
    if (!expr || !set_rtrn)
        return false;

    switch (expr->type) {
        case EXPR_BOOL: {
            if (expr->value.bool_val == NULL)
                return false;
            *set_rtrn = expr->value.bool_val;
            return true;
        }

        case EXPR_IDENT: {
            if (strcmp(expr->value.ident, "true") == 0) {
                *set_rtrn = true;
                return true;
            } else if (strcmp(expr->value.ident, "false") == 0) {
                *set_rtrn = false;
                return true;
            } else {
                xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Unknown identifier for boolean: %s", expr->value.ident);
                return false;
            }
        }

        case EXPR_FIELD_REF:
        case EXPR_UNARY_OP:
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Unsupported operation for boolean resolution");
            return false;

        default:
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Unknown expression type for boolean resolution");
            return false;
    }
}
