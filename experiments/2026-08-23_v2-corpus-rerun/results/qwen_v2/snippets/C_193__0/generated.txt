#include <rdma/rdma_cma.h>
#include <netinet/in.h>
#include <errno.h>

int check_rdma_ip_validity(__be32 addr) {
    struct rdma_cm_id *id;
    struct sockaddr_in sin;
    int ret;

    id = rdma_create_id(NULL, NULL, 0);
    if (!id)
        return -ENOMEM;

    memset(&sin, 0, sizeof(sin));
    sin.sin_family = AF_INET;
    sin.sin_addr = addr;
    sin.sin_port = 0;

    ret = rdma_bind_addr(id, (struct sockaddr *)&sin);
    if (ret) {
        rdma_destroy_id(id);
        return ret;
    }

    rdma_destroy_id(id);
    return 0;
}
