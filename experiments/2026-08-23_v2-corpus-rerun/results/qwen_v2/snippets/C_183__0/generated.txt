BOOL ReadLicenseScopes(wStream* s, SCOPE_LIST* scopeList)
{
    UINT32 scopeCount;

    if (Stream_GetRemainingLength(s) < 4)
        return FALSE;

    Stream_Read_UINT32(s, scopeCount);

    if (Stream_GetRemainingLength(s) < (scopeCount * sizeof(LICENSE_BLOB)))
        return FALSE;

    scopeList->count = scopeCount;
    scopeList->scopes = (LICENSE_BLOB*)malloc(scopeCount * sizeof(LICENSE_BLOB));

    if (!scopeList->scopes)
        return FALSE;

    for (UINT32 i = 0; i < scopeCount; i++)
    {
        if (!Stream_Read(s, &scopeList->scopes[i], sizeof(LICENSE_BLOB)))
            return FALSE;
    }

    return TRUE;
}
