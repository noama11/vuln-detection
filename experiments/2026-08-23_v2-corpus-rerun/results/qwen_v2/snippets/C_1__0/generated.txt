void pango_glyph_string_set_size(PangoGlyphString *string, gint new_len) {
    gint max_size = G_MAXINT / 2;
    gint size = string->num_glyphs;

    g_return_if_fail(new_len >= 0);

    if (new_len > size) {
        while (size < new_len) {
            size = (size == 0) ? 1 : size * 2;
            if (size > max_size)
                g_error("Exceeded maximum glyph string size limit.");
        }
        string->glyphs = g_renew(PangoGlyph, string->glyphs, size);
        string->log_clusters = g_renew(gint, string->log_clusters, size);
    }

    string->num_glyphs = new_len;
}
