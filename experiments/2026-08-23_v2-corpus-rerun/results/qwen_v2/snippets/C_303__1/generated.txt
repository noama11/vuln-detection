int open_cypress_usb_port(struct tty_struct *tty, struct usb_serial_port *port)
{
    int result;

    if (!port || !tty)
        return -EINVAL;

    if (!usb_serial_port_is_operational(port)) {
        dev_err(&port->dev, "Port is not operational\n");
        return -EIO;
    }

    usb_clear_halt(port->serial->dev, port->read_urb->pipe);
    usb_clear_halt(port->serial->dev, port->write_urb->pipe);

    tty->low_latency = 1;
    tty->no_job_control = 1;

    result = usb_submit_urb(port->read_urb, GFP_KERNEL);
    if (result) {
        dev_err(&port->dev, "Failed to submit read urb: %d\n", result);
        port->port.dead = 1;
        return result;
    }

    return 0;
}
