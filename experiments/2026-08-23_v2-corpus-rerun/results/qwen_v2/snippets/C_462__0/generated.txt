int probe_usb_interface(struct usb_interface *intf, const struct usb_device_id *id)
{
    struct host_config *host_cfg = NULL;
    struct device_config *dev_cfg = NULL;
    int ret = 0;

    host_cfg = kmalloc(sizeof(*host_cfg), GFP_KERNEL);
    if (!host_cfg) {
        ret = -ENOMEM;
        goto error;
    }

    dev_cfg = kmalloc(sizeof(*dev_cfg), GFP_KERNEL);
    if (!dev_cfg) {
        ret = -ENOMEM;
        goto free_host_cfg;
    }

    ret = usb_control_msg(intf->cur_altsetting->desc.bInterfaceNumber,
                          USB_DIR_OUT | USB_TYPE_CLASS | USB_RECIP_INTERFACE,
                          USB_REQ_SET_INTERFACE, 0, 0, NULL, 0, USB_CTRL_MSG_TIMEOUT);
    if (ret < 0)
        goto free_dev_cfg;

    ret = get_device_configuration(intf, dev_cfg);
    if (ret < 0)
        goto free_dev_cfg;

    if (dev_cfg->num_interfaces == 0) {
        ret = -ENODEV;
        goto free_dev_cfg;
    }

    initialize_device_structure(dev_cfg);

    for (int i = 0; i < dev_cfg->num_interfaces; i++) {
        ret = create_can_device(intf, i);
        if (ret < 0)
            goto cleanup_can_devices;
    }

    return 0;

cleanup_can_devices:
    for (int i = 0; i < dev_cfg->num_interfaces; i++) {
        cleanup_can_device(intf, i);
    }
free_dev_cfg:
    kfree(dev_cfg);
free_host_cfg:
    kfree(host_cfg);
error:
    return ret;
}
