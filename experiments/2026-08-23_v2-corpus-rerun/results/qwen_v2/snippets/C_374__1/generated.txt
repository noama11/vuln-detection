int compat_waitid(int which, compat_pid_t pid, struct compat_siginfo __user *infop, int options, struct compat_rusage __user *uru)
{
    struct compat_siginfo info;
    int ret;

    ret = kernel_waitid(which, pid, &info, options, uru);
    if (ret < 0)
        return ret;

    if (infop) {
        if (copy_to_user(infop, &info, sizeof(info)))
            return -EFAULT;
    }

    return 0;
}
