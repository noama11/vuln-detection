int read_socket_data(int s, void* arg, int* len) {
    int header;
    if (recv(s, &header, sizeof(int), 0) <= 0) {
        return -1;
    }

    if (header <= 0) {
        *len = 0;
        return header;
    }

    if (header > *len) {
        *len = 0;
        return -1;
    }

    if (recv(s, arg, header, 0) <= 0) {
        *len = 0;
        return -1;
    }

    *len = header;
    return header;
}
