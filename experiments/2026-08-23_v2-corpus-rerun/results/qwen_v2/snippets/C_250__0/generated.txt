int map_user_io(struct request_queue *q, struct request *rq, struct rq_map_data *map_data,
                const struct iov_iter *iter, gfp_t gfp_mask) {
    int ret = 0;
    size_t iov_offset = 0;
    struct bio *bio = NULL;

    if (!iter || !iov_iter_count(iter))
        return -EINVAL;

    if (unlikely(iter->nr_segs > BIO_MAX_PAGES)) {
        if (iter->type & ITER_IOVEC)
            return -EINVAL;
    }

    while (iov_iter_count(iter)) {
        struct bio_vec bvec;
        unsigned int len;
        void *buf;

        len = min(iov_iter_count(iter), (size_t)PAGE_SIZE);
        ret = iov_iter_get_pages(iter, &bvec, len, 1, &iov_offset);
        if (ret < 0)
            break;

        buf = page_address(bvec.bv_page) + bvec.bv_offset;
        bio = bio_alloc(GFP_NOIO, 1);
        if (!bio) {
            ret = -ENOMEM;
            break;
        }

        bio_add_page(rq->bio, bvec.bv_page, len, iov_offset);
        bio->bi_iter.bi_sector = map_data->offset >> 9;
        bio->bi_bdev = map_data->bdev;
        bio->bi_end_io = NULL;

        if (!rq->bio)
            rq->bio = bio;
        else
            rq->biotail->bi_next = bio;
        rq->biotail = bio;
    }

    if (ret < 0) {
        bio = rq->bio;
        while (bio) {
            struct bio *next = bio->bi_next;
            bio_put(bio);
            bio = next;
        }
        rq->bio = NULL;
        return ret;
    }

    rq->cmd_flags |= REQ_USER_MAPPED;
    return 0;
}
