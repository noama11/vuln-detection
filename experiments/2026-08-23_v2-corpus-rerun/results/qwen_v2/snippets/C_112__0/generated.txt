#include <linux/fs.h>
#include <linux/cpu.h>
#include <linux/errno.h>

int open_msr(struct inode *inode, struct file *file) {
    int cpu;

    if (!capable(CAP_SYS_RAWIO))
        return -EPERM;

    cpu = smp_processor_id();
    if (cpu >= nr_cpu_ids || !cpu_online(cpu))
        return -ENXIO;

    if (!boot_cpu_has(X86_FEATURE_MSR))
        return -EIO;

    return 0;
}
