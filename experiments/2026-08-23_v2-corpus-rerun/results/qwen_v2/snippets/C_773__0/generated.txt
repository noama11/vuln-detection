void handle_outbound_pinhole_timeout(struct upnphttp *h, const char *action, const char *ns) {
    if (!ipv6_forwarding_enabled()) {
        send_error_response(h, "IPv6 forwarding is disabled.");
        return;
    }

    char *internal_client_ip = parse_request_parameter(h, "InternalClient");
    char *port = parse_request_parameter(h, "Port");
    char *remote_host = parse_request_parameter(h, "RemoteHost");
    char *protocol = parse_request_parameter(h, "Protocol");

    if (!internal_client_ip || !port || !remote_host || !protocol) {
        send_error_response(h, "Missing required parameters.");
        return;
    }

    log_request_details(internal_client_ip, port, remote_host, protocol);

    int timeout = check_outbound_pinhole_timeout(internal_client_ip, port, remote_host, protocol);
    if (timeout < 0) {
        send_soap_error(h, "Unsupported protocol or action failed.");
        return;
    }

    char soap_response[1024];
    snprintf(soap_response, sizeof(soap_response),
             "<s:Envelope><s:Body><u:%sResponse xmlns:u=\"%s\">"
             "<NewTimeout>%d</NewTimeout></u:%sResponse></s:Body></s:Envelope>",
             action, ns, timeout, action);

    send_soap_response(h, soap_response);
}
