unsigned flac_process_audio_data(SF_PRIVATE *psf, unsigned len) {
    unsigned bytes_processed = 0;
    unsigned remaining = len;
    FLAC__StreamDecoder *decoder = psf->flac_decoder;
    FLAC__StreamDecoderState state;

    if (decoder == NULL)
        return 0;

    while (remaining > 0) {
        state = FLAC__stream_decoder_get_state(decoder);
        if (state != FLAC__STREAM_DECODER_SEARCHING_FOR_METADATA &&
            state != FLAC__STREAM_DECODER_READING_METADATA &&
            state != FLAC__STREAM_DECODER_DECODING) {
            psf_log_printf(psf, "FLAC decoder in unexpected state: %s\n",
                           FLAC__StreamDecoderStateString[state]);
            return bytes_processed;
        }

        if (psf->flac_decoder_buffer_pos > 0) {
            unsigned copy = SF_MIN(remaining, psf->flac_decoder_buffer_pos);
            memcpy(psf->data + bytes_processed, psf->flac_decoder_buffer, copy);
            psf->flac_decoder_buffer_pos -= copy;
            bytes_processed += copy;
            remaining -= copy;
            continue;
        }

        if (!FLAC__stream_decoder_process_single(decoder)) {
            psf_log_printf(psf, "FLAC decoding error: %s\n",
                           FLAC__StreamDecoderStateString[FLAC__stream_decoder_get_state(decoder)]);
            return bytes_processed;
        }

        if (FLAC__stream_decoder_get_state(decoder) == FLAC__STREAM_DECODER_END_OF_STREAM) {
            psf_log_printf(psf, "End of FLAC stream reached.\n");
            return bytes_processed;
        }
    }

    return bytes_processed;
}
