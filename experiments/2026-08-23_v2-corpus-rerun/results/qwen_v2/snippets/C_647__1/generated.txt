void verify_image_integrity(Gimp *gimp, gboolean with_unusual_stuff, gboolean compat_paths, gboolean use_gimp_2_8_features) {
  GimpImage *image;
  gchar *temp_file;
  GError *error = NULL;
  gboolean success;

  image = create_main_image(gimp, with_unusual_stuff, compat_paths, use_gimp_2_8_features);
  g_assert(image != NULL);

  temp_file = g_strdup_temp("image_XXXXXX", NULL);
  success = gimp_file_save(gimp, image, temp_file, &error);
  g_assert(success && error == NULL);

  GimpImage *loaded_image = gimp_file_load(temp_file, &error);
  g_assert(loaded_image != NULL && error == NULL);

  g_assert(compare_images(image, loaded_image));

  g_unlink(temp_file);
  g_free(temp_file);
  g_object_unref(image);
  g_object_unref(loaded_image);
}
