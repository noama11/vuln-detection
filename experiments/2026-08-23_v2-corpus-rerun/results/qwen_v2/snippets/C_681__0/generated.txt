int blkcg_init_queue(struct request_queue *q)
{
    struct blkg_policy_data *pd;
    int ret;

    pd = kzalloc(sizeof(*pd), GFP_KERNEL);
    if (!pd)
        return -ENOMEM;

    radix_tree_preload(GFP_KERNEL);
    ret = blkg_create(q, pd);
    if (ret) {
        kfree(pd);
        return ret;
    }

    ret = blkg_init(q, pd);
    if (ret) {
        blkg_destroy(q, pd);
        kfree(pd);
        return ret;
    }

    q->blkcg = pd;
    return 0;
}
