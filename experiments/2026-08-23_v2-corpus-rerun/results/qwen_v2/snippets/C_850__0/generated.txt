MagickBooleanType ReadPSDChannel(Image *image, const ImageInfo *image_info, const PSDInfo *psd_info, LayerInfo *layer_info, const size_t channel, PSDCompressionType compression, ExceptionInfo *exception) {
  MagickBooleanType status = MagickTrue;

  if (layer_info->has_mask[channel])
    /* Handle mask logic if needed, but not detailed in the spec. */

  switch (compression) {
    case PSDNoCompression:
      status = ReadPSDUncompressedChannel(image, image_info, psd_info, layer_info, channel, exception);
      break;
    case PSDRLECompression:
      status = ReadPSDRLECompressedChannel(image, image_info, psd_info, layer_info, channel, exception);
      break;
    default:
      (void) ThrowMagickException(exception, GetMagickModule(), CoderError, "UnsupportedPSDCompressionType", "compression=%d", compression);
      status = MagickFalse;
  }

  if (status == MagickFalse)
    return MagickFalse;

  /* Seek back to the correct position in the image blob if needed. */
  (void) SeekBlob(image, psd_info->channel_data_offset[channel], SEEK_SET);

  return status;
}
