int irda_sock_recvmsg(struct kiocb *iocb, struct socket *sock, struct msghdr *msg,
                      size_t size, int flags)
{
    struct sock *sk = sock->sk;
    struct sk_buff *skb;
    int copied, err;

    skb = skb_dequeue(&sk->sk_receive_queue);
    if (!skb) {
        return -EAGAIN;
    }

    copied = skb->len;
    if (size < copied) {
        copied = size;
        msg->msg_flags |= MSG_TRUNC;
    }

    err = skb_copy_datagram_msg(skb, 0, msg, copied);
    if (err) {
        kfree_skb(skb);
        return err;
    }

    kfree_skb(skb);

    if ((sk->sk_shutdown & RCV_SHUTDOWN) && !skb_peek(&sk->sk_receive_queue)) {
        return 0;
    }

    if (sk->sk_type == SOCK_DGRAM && skb_queue_len(&sk->sk_receive_queue) < IRDA_SKB_QUEUE_HI) {
        irtt_flow_start(sk);
    }

    return copied;
}
