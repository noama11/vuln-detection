struct RClass* get_real_class(struct RClass* cl) {
  if (cl == NULL) {
    return NULL;
  }

  while (1) {
    if (RCLASS_CLASSTYPE(cl) == 0) {
      return NULL;
    }

    if (!RCLASS_IS_SINGLETON(cl) && !RCLASS_IS_INCLUDED(cl)) {
      return cl;
    }

    cl = RCLASS_SUPER(cl);
    if (cl == NULL) {
      return NULL;
    }
  }
}
