int ieee80211_radiotap_iterator_init(struct ieee80211_radiotap_iterator *iterator,
                                     struct ieee80211_radiotap_header *radiotap_header,
                                     int max_length,
                                     const struct ieee80211_radiotap_vendor_namespaces *vns) {
    uint36_t align;
    uint36_t offset;
    uint32_t bitmap;
    uint8_t *cursor;
    uint8_t *end;

    iterator->hdr = radiotap_header;
    iterator->max_length = max_length;
    iterator->vns = vns;
    iterator->is_radiotap = 1;
    iterator->align = 1;
    iterator->cursor = (uint8_t *)(radiotap_header + 1);
    iterator->parse = 0;

    if (max_length < sizeof(*radiotap_header))
        return -EINVAL;

    if (radiotap_header->it_version != 0)
        return -EINVAL;

    if (radiotap_header->it_pad != 0)
        return -EINVAL;

    if (radiotap_header->it_len < sizeof(*radiotap_header))
        return -EINVAL;

    if (radiotap_header->it_len > max_length)
        return -EINVAL;

    bitmap = le16toh(radiotap_header->it_present);
    if (radiotap_header->it_present >> 15)
        bitmap |= le32toh(*((uint32_t *)radiotap_header + 1)) << 16;

    offset = 0;
    cursor = (uint8_t *)(radiotap_header + 1);
    end = (uint8_t *)radiotap_header + radiotap_header->it_len;

    while (offset < 32) {
        if (bitmap & 1) {
            align = 1;
            switch (offset) {
                case IEEE80211_RADIOTAP_TSFT:
                    align = 8;
                    break;
                case IEEE80211_RADIOTAP_FLAGS:
                case IEEE80211_RADIOTAP_RATE:
                case IEEE80211_RADIOTAP_CHANNEL:
                case IEEE80211_RADIOTAP_FHSS:
                case IEEE80211_RADIOTAP_ANTENNA:
                case IEEE80211_RADIOTAP_DBM_TX_POWER:
                case IEEE80211_RADIOTAP_DB_TX_POWER:
                case IEEE80211_RADIOTAP_DBM_RX_POWER:
                case IEEE80211_RADIOTAP_ANTENNA_SIGNAL:
                case IEEE80211_RADIOTAP_ANTENNA_NOISE:
                case IEEE80211_RADIOTAP_LOCK_QUALITY:
                case IEEE80211_RADIOTAP_TX_ATTENUATION:
                case IEEE80211_RADIOTAP_DB_TX_ATTENUATION:
                case IEEE80211_RADIOTAP_DBM_TX_ATTENUATION:
                case IEEE80211_RADIOTAP_VENDOR_NAMESPACE:
                case IEEE80211_RADIOTAP_EXT:
                    align = 1;
                    break;
                case IEEE80211_RADIOTAP_XCHANNEL:
                    align = 4;
                    break;
                default:
                    align = 1;
                    break;
            }

            if ((cursor - (uint8_t *)radiotap_header + align - 1) % align)
                cursor = (uint8_t *)(((uintptr_t)cursor + align - 1) / align * align);

            if (cursor + 1 > end)
                return -EINVAL;

            iterator->this_arg = cursor;
            iterator->arg_index = offset;
            iterator->parse |= (1 << offset);
            cursor += 1;
        }

        bitmap >>= 1;
        offset++;
    }

    iterator->cursor = cursor;
    return 0;
}
