int sctp_get_assoc_stats(struct sock *sk, int len, char __user *optval, int __user *optlen)
{
    struct sctp_association *asoc;
    struct sctp_assoc_stats stats;
    int ret = 0;

    if (len < sizeof(stats.assoc_id)) {
        return -EINVAL;
    }

    if (copy_from_user(&stats, optval, sizeof(stats))) {
        return -EFAULT;
    }

    asoc = sctp_id2assoc(sk, stats.assoc_id);
    if (!asoc) {
        return -EINVAL;
    }

    stats.cnt_packets = asoc->stats.cnt_packets;
    stats.cnt_bytes = asoc->stats.cnt_bytes;
    stats.cnt_retrans = asoc->stats.cnt_retrans;
    stats.cnt_fast_retrans = asoc->stats.cnt_fast_retrans;
    stats.cnt_rto = asoc->stats.cnt_rto;
    stats.cnt_timo = asoc->stats.cnt_timo;
    stats.cnt_gap = asoc->stats.cnt_gap;
    stats.cnt_sack = asoc->stats.cnt_sack;
    stats.cnt_dupack = asoc->stats.cnt_dupack;
    stats.cnt_reord = asoc->stats.cnt_reord;
    stats.max_rto = asoc->rto_max;

    if (copy_to_user(optval, &stats, sizeof(stats))) {
        return -EFAULT;
    }

    if (put_user(sizeof(stats), optlen)) {
        return -EFAULT;
    }

    return ret;
}
