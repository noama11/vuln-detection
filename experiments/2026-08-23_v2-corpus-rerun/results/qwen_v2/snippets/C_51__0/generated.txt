u32 kvm_lapic_get_timer_count(struct kvm_lapic *apic)
{
    u64 period;
    s64 remaining;

    if (!apic)
        return 0;

    if (apic->base.read(APIC_TMICT) == 0 || apic->base.read(APIC_TCR) == 0)
        return 0;

    period = apic->base.read(APIC_TMICT) * 1000000000ULL;
    period = period / apic->base.read(APIC_TCR);

    remaining = apic->base.read(APIC_TMICT) - apic->base.read(APIC_TMR);
    if (remaining < 0)
        remaining = 0;

    return (u32)((remaining * apic->divide_count) / apic->base.read(APIC_TCR));
}
