int videobuf_mmap(struct videobuf_queue *q, struct vm_area_struct *vma)
{
    struct videobuf_buffer *b = NULL;
    int ret = -EINVAL;

    if (vma->vm_flags & (VM_WRITE | VM_SHARED)) {
        list_for_each_entry(b, &q->bs_list, list) {
            if (vma->vm_start == b->boff) {
                struct videobuf_mapping *map;

                map = kzalloc(sizeof(*map), GFP_KERNEL);
                if (!map)
                    return -ENOMEM;

                map->vma = vma;
                map->buf = b;
                list_add(&map->list, &q->mappings);

                vma->vm_ops = &videobuf_vm_ops;
                vma->vm_private_data = map;

                ret = remap_pfn_range(vma, vma->vm_start,
                                      b->boff >> PAGE_SHIFT,
                                      vma->vm_end - vma->vm_start,
                                      vma->vm_page_prot);
                if (ret)
                    break;

                pr_debug("Mapped buffer %p to vma %p\n", b, vma);
                return 0;
            }
        }
    }

    vma->vm_private_data = q;
    return ret;
}
