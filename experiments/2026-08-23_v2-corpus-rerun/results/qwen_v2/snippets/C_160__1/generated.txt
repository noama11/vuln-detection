int load_tss_state(struct x86_emulate_ctxt *ctxt, struct tss_segment_16 *tss) {
    int rc;

    /* Initialize instruction pointer and flags from TSS */
    ctxt->regs->eip = tss->eip;
    ctxt->regs->eflags = tss->eflags;

    /* Load general-purpose registers from TSS */
    ctxt->regs->eax = tss->eax;
    ctxt->regs->ecx = tss->ecx;
    ctxt->regs->edx = tss->edx;
    ctxt->regs->ebx = tss->ebx;
    ctxt->regs->esp = tss->esp;
    ctxt->regs->ebp = tss->ebp;
    ctxt->regs->esi = tss->esi;
    ctxt->regs->edi = tss->edi;

    /* Set segment selectors */
    ctxt->regs->cs = tss->cs;
    ctxt->regs->ds = tss->ds;
    ctxt->regs->es = tss->es;
    ctxt->regs->fs = tss->fs;
    ctxt->regs->gs = tss->gs;
    ctxt->regs->ss = tss->ss;

    /* Attempt to load segment descriptors */
    if ((rc = load_segment_descriptor(ctxt, tss->cs, SEGMENT_CS)) != X86EMUL_CONTINUE ||
        (rc = load_segment_descriptor(ctxt, tss->ds, SEGMENT_DS)) != X86EMUL_CONTINUE ||
        (rc = load_segment_descriptor(ctxt, tss->es, SEGMENT_ES)) != X86EMUL_CONTINUE ||
        (rc = load_segment_descriptor(ctxt, tss->fs, SEGMENT_FS)) != X86EMUL_CONTINUE ||
        (rc = load_segment_descriptor(ctxt, tss->gs, SEGMENT_GS)) != X86EMUL_CONTINUE ||
        (rc = load_segment_descriptor(ctxt, tss->ss, SEGMENT_SS)) != X86EMUL_CONTINUE) {
        return rc;
    }

    return X86EMUL_CONTINUE;
}
