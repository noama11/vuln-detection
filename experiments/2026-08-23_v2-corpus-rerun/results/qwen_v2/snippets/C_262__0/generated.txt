void process_hid_field(struct hid_device *hid, struct hid_field *field, __u8 *data, int interrupt)
{
    int i;
    int *value = kmalloc(field->maxusage * sizeof(int), GFP_KERNEL);
    if (!value)
        return;

    for (i = 0; i < field->maxusage; i++) {
        int val = hid_input_data(hid, field, data, i);
        if (val < field->logical_minimum || val > field->logical_maximum)
            continue;

        if (field->flags & HID_MAIN_ITEM_CONSTANT)
            continue;

        hid_process_event(hid, field->usage->hid, val, interrupt);
    }

    memcpy(field->value, value, field->maxusage * sizeof(int));
    kfree(value);
}
