TEE_Result populate_crypto_object(unsigned long obj, struct utee_attribute *usr_attrs, unsigned long attr_count) {
    TEE_Result res = TEE_ERROR_GENERIC;
    struct tee_session *session = tee_get_current_session();
    struct tee_object *crypto_obj;

    if (!session) {
        return TEE_ERROR_BAD_STATE;
    }

    crypto_obj = tee_get_object_by_id(obj);
    if (!crypto_obj || !crypto_obj->is_transient || crypto_obj->is_initialized) {
        return TEE_ERROR_BAD_PARAMETERS;
    }

    struct tee_attribute *attrs = calloc(attr_count, sizeof(struct tee_attribute));
    if (!attrs) {
        return TEE_ERROR_OUT_OF_MEMORY;
    }

    for (unsigned long i = 0; i < attr_count; i++) {
        attrs[i] = usr_attrs[i];
    }

    res = validate_attributes_against_type(attrs, attr_count, crypto_obj->type);
    if (res != TEE_SUCCESS) {
        free(attrs);
        return res;
    }

    crypto_obj->attributes = attrs;
    crypto_obj->attr_count = attr_count;
    crypto_obj->is_initialized = 1;

    return TEE_SUCCESS;
}
