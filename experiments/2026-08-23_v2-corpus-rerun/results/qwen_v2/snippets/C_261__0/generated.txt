ssize_t read_env_vars(struct file *file, char __user *buf, size_t count, loff_t *ppos) {
    struct mm_struct *mm = file->f_mapping->host->mm;
    char *tmp_page;
    ssize_t total_read = 0;
    int ret;

    if (!mm || !mm->env_size)
        return -EINVAL;

    tmp_page = (char *)__get_free_page(GFP_KERNEL);
    if (!tmp_page)
        return -ENOMEM;

    down_read(&mm->mmap_lock);

    while (total_read < count && *ppos < mm->env_size) {
        size_t bytes_to_read = min(count - total_read, PAGE_SIZE);
        ret = read_env_chunk(mm, *ppos, tmp_page, bytes_to_read);
        if (ret <= 0)
            break;

        if (copy_to_user(buf + total_read, tmp_page, ret)) {
            ret = -EFAULT;
            break;
        }

        *ppos += ret;
        total_read += ret;
    }

    up_read(&mm->mmap_lock);
    free_page((unsigned long)tmp_page);

    return total_read ? total_read : ret;
}
