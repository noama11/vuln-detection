void GetExtensionViews(const v8::FunctionCallbackInfo<v8::Value>& args) {
  v8::Isolate* isolate = args.GetIsolate();
  v8::Local<v8::Context> context = isolate->GetCurrentContext();

  if (args.Length() != 2 || !args[0]->IsNumber() || !args[1]->IsString()) {
    isolate->ThrowException(v8::Exception::TypeError(
        v8::String::NewFromUtf8(isolate, "Invalid arguments: expected window ID and view type string").ToLocalChecked()));
    return;
  }

  int window_id = args[0].As<v8::Number>()->Value();
  v8::String::Utf8Value view_type_str(isolate, args[1]);
  std::string view_type(*view_type_str);

  std::string extension_id = GetExtensionIDFromWindow(window_id);
  if (extension_id.empty()) {
    args.GetReturnValue().Set(v8::Array::New(isolate, 0));
    return;
  }

  std::vector<int> frames = GetRelevantFrames(extension_id);
  std::vector<int> main_views;

  for (int frame_id : frames) {
    if (!IsIframe(frame_id)) {
      main_views.push_back(frame_id);
    }
  }

  v8::Local<v8::Array> result = v8::Array::New(isolate, main_views.size());
  for (size_t i = 0; i < main_views.size(); ++i) {
    result->Set(context, i, v8::Integer::New(isolate, main_views[i])).Check();
  }

  args.GetReturnValue().Set(result);
}
