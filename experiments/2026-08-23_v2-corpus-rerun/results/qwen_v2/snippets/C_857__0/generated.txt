size_t RegisterMPCImage(void) {
  MagickInfo
    *magick_info;

  magick_info = AcquireMagickInfo("MPC", "MPC", "Magick Persistent Cache");
  magick_info->decoder = (MagickDecoderHandler) ReadMPCImage;
  magick_info->encoder = (MagickEncoderHandler) WriteMPCImage;
  magick_info->flags |= MagickSupportsSingleFrame;
  magick_info->flags |= MagickSupportsMultiFrame;
  return (RegisterMagickInfo(magick_info));
}
