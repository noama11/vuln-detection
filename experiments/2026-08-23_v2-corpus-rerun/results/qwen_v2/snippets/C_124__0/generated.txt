void iriap_get_attr_by_class(struct iriap_cb *self, struct sk_buff *skb)
{
    char *className, *attrName;
    int classNameLen, attrNameLen;
    struct iriap_object *obj;
    struct iriap_attribute *attr;

    if (!self || !skb)
        return;

    className = skb->data;
    classNameLen = strlen(className);
    if (classNameLen > IRIAP_MAX_NAME_LEN)
        return;

    attrName = className + classNameLen + 1;
    attrNameLen = strlen(attrName);
    if (attrNameLen > IRIAP_MAX_NAME_LEN)
        return;

    obj = iriap_find_object(self, className);
    if (!obj) {
        iriap_send_error(skb, IRIAP_ERR_NO_OBJECT);
        return;
    }

    attr = iriap_find_attribute(obj, attrName);
    if (!attr) {
        iriap_send_error(skb, IRIAP_ERR_NO_ATTRIBUTE);
        return;
    }

    iriap_send_value(skb, attr->value);

    IRIAP_DBG("Retrieved attribute value: %s", attr->value);
}
