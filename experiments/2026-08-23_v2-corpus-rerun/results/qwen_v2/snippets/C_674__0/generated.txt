int git_index_entry_read(git_index_entry **out, size_t *out_size, git_index *index, const void *buffer, size_t buffer_size, const char *last) {
    if (!buffer || buffer_size < sizeof(git_index_entry)) {
        return -1;
    }

    git_index_entry *entry = (git_index_entry *)malloc(buffer_size);
    if (!entry) {
        return -1;
    }

    memcpy(entry, buffer, buffer_size);

    if (entry->flags & GIT_INDEX_ENTRY_COMPRESSED) {
        size_t path_len = strlen(last) + 1 + strlen(entry->path);
        char *full_path = (char *)malloc(path_len);
        if (!full_path) {
            free(entry);
            return -1;
        }

        snprintf(full_path, path_len, "%s/%s", last, entry->path);
        free(entry->path);
        entry->path = full_path;
    }

    *out = entry;
    *out_size = buffer_size;
    return 0;
}
