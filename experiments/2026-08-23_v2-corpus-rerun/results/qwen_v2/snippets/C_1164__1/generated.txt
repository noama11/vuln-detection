bool IsDownloadDangerous(const DownloadItem& download, const FilePath& suggested_path, bool visited_referrer_before) {
  // Check if the download is an extension without approval
  if (download.is_extension && !download.approved) {
    // If off-store extension installations are not enabled, it's dangerous
    if (!IsOffStoreExtensionInstallEnabled()) {
      return true;
    }
  }

  // Check if the file extension is dangerous
  if (IsDangerousExtension(suggested_path)) {
    // If the user did not gesture for the download and hasn't visited the referrer before,
    // it's considered dangerous
    if (!download.user_gesture && !visited_referrer_before) {
      return true;
    }
  }

  return false;
}
