xfs_buf_t*
xfs_buf_find(
	struct xfs_buftarg	*btp,
	struct xfs_buf_map	*map,
	int			nmaps,
	xfs_buf_flags_t		flags,
	xfs_buf_t		*new_bp)
{
	xfs_buf_t		*bp;
	struct xfs_buf_hash	*hashp;
	struct xfs_mount	*mp = btp->bt_mount;
	xfs_daddr_t		blkno;
	int			nblocks;
	int			i;

	blkno = 0;
	nblocks = 0;
	for (i = 0; i < nmaps; i++) {
		blkno += map[i].bm_bn;
		nblocks += XFS_BUF_COUNT(map + i);
	}

	if (blkno + nblocks > mp->m_ddev_brasize)
		return NULL;

	hashp = XFS_BUF_HASH(btp, blkno);
	bp = xfs_buf_find_hash(hashp, blkno, nblocks, flags, NULL);

	if (bp) {
		if (xfs_buf_isstale(bp)) {
			xfs_buf_stale(bp);
			xfs_buf_relse(bp);
			return NULL;
		}
		if (xfs_buf_lock(bp)) {
			if (xfs_buf_isstale(bp)) {
				xfs_buf_stale(bp);
				xfs_buf_relse(bp);
				return NULL;
			}
		}
		return bp;
	}

	if (new_bp) {
		new_bp->b_bn = blkno;
		new_bp->b_bno = blkno;
		new_bp->b_bcount = nblocks;
		new_bp->b_flags = flags;
		xfs_buf_hold(new_bp);
		xfs_buf_insert_hash(new_bp, hashp);
		return new_bp;
	}

	return NULL;
}
