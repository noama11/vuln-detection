int kvm_arch_complete_emulated_mmio(struct kvm_vcpu *vcpu) {
    if (!vcpu->mmio_needed)
        return 0;

    if (vcpu->mmio_is_write)
        kvm_mmio_write(vcpu, vcpu->mmio_data, vcpu->mmio_size);
    else
        kvm_mmio_read(vcpu, vcpu->mmio_data, vcpu->mmio_size);

    vcpu->mmio_needed = 0;
    vcpu->run->exit_reason = KVM_EXIT_MMIO;
    vcpu->run->mmio.is_write = vcpu->mmio_is_write;
    vcpu->run->mmio.phys_addr = vcpu->mmio_phys_addr;
    memcpy(vcpu->run->mmio.data, vcpu->mmio_data, vcpu->mmio_size);
    vcpu->run->mmio.len = vcpu->mmio_size;

    return 1;
}
