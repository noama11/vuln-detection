void scsi_abort_command_complete(struct scsi_cmnd *cmd)
{
    struct sas_hba *sas_hba = cmd->device->host->hostdata;
    struct sas_rphy *rphy = dev_to_rphy(cmd->device->sdev_gendev.parent);
    struct sas_device *sas_dev = rphy ? rphy_to_sas_device(rphy) : NULL;
    struct ata_port *ap = NULL;

    if (!sas_dev)
        return;

    if (sas_dev->sata_dev) {
        ap = sas_dev->sata_dev->ap;
        if (ap)
            ata_scsi_request(ap);
        return;
    }

    cmd->result = DID_ABORT << 16;
    scsi_finish_command(cmd);
}
