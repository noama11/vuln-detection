void *allocate_from_pool(size_t size, struct page **ret_page, gfp_t flags) {
    void *ptr = NULL;

    if (!atomic_pool_initialized) {
        pr_warn("Atomic pool is not initialized.\n");
        return NULL;
    }

    *ret_page = alloc_pages(flags, get_order(size));
    if (!*ret_page) {
        return NULL;
    }

    ptr = page_address(*ret_page);
    if (!ptr) {
        __free_pages(*ret_page, get_order(size));
        *ret_page = NULL;
        return NULL;
    }

    memset(ptr, 0, size);
    return ptr;
}
