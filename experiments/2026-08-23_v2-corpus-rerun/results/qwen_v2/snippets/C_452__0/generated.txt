struct sock *clone_lock_sock(struct sock *sk, const struct request_sock *req, gfp_t priority)
{
    struct sock *new_sk;

    new_sk = sk_clone_lock(sk, priority);
    if (!new_sk)
        return NULL;

    new_sk->sk_state = TCP_SYN_RECV;
    new_sk->sk_daddr = req->rsk_rcv_addr;
    new_sk->sk_v6_daddr = req->rsk_v6_rcv_addr;
    new_sk->sk_num = req->rsk_isn;
    new_sk->sk_ack_backlog = 0;
    new_sk->sk_err = 0;
    new_sk->sk_err_soft = 0;
    new_sk->sk_priority = 0;
    new_sk->sk_type = req->rsk_sk->sk_type;
    new_sk->sk_protocol = req->rsk_sk->sk_protocol;
    new_sk->sk_security = req->rsk_sk->sk_security;

    sk_acceptq_removed(new_sk);
    security_sk_clone(new_sk);

    return new_sk;
}
