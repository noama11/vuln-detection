void readSingleImageFrame(MagickCore::Image *image, MagickCore::ExceptionInfo *exceptionInfo) {
    if (image != (MagickCore::Image *) NULL) {
        MagickCore::Image *nextImage = image->next;
        while (nextImage != (MagickCore::Image *) NULL) {
            MagickCore::Image *temp = nextImage->next;
            nextImage->next = (MagickCore::Image *) NULL;
            DestroyImage(nextImage);
            nextImage = temp;
        }
        image->next = (MagickCore::Image *) NULL;
    }

    if (exceptionInfo != (MagickCore::ExceptionInfo *) NULL) {
        if (image == (MagickCore::Image *) NULL) {
            exceptionInfo->severity = UndefinedException;
            if (exceptionInfo->message[0] != '\0') {
                (void) ThrowMagickException(exceptionInfo, GetMagickModule(), WarningException, "ImageIsNull", "Image is null");
            }
            DestroyExceptionInfo(exceptionInfo);
        } else {
            (void) ThrowMagickException(exceptionInfo, GetMagickModule(), image->exception.severity, image->exception.reason, "%s", image->exception.description);
        }
    }
}
