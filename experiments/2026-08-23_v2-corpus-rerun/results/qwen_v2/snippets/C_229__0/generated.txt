int path_to_string(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int i = 0;
    int j = 0;

    if (!from || !to || fromlen <= 0 || tolen <= 0) {
        return -EINVAL;
    }

    while (i < fromlen && j < tolen - 1) {
        unsigned char c = from[i++];

        if (c == '.') {
            if (i < fromlen && from[i] == '.') {
                if (j > 0) {
                    j = 0;
                    while (j > 0 && j > 0 && to[j - 1] != '/') {
                        j--;
                    }
                    if (j > 0) {
                        j--;
                    }
                } else if (j == 0) {
                    j = 0;
                }
                i++;
            } else {
                if (j > 0 || fromlen > 1) {
                    to[j++] = '.';
                }
            }
        } else if (c == '/') {
            to[j++] = '/';
        } else {
            to[j++] = c;
        }
    }

    if (i < fromlen) {
        return -ENAMETOOLONG;
    }

    if (j >= tolen) {
        return -ENAMETOOLONG;
    }

    to[j] = '\0';
    return 0;
}
