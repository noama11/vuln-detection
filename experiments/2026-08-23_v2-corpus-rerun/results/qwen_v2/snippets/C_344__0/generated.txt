#include <stdlib.h>

// Assuming LineContribType is a defined type in the codebase
typedef struct {
    double weight;
} LineContribType;

LineContribType* calculate_line_contributions(unsigned int line_size, unsigned int src_size, double scale_d, double (*pFilter)(double)) {
    if (line_size == 0 || src_size == 0 || scale_d <= 0.0 || pFilter == NULL) {
        return NULL;
    }

    // Allocate memory for the contributions
    LineContribType* contributions = (LineContribType*)malloc(line_size * sizeof(LineContribType));
    if (contributions == NULL) {
        return NULL;
    }

    for (unsigned int i = 0; i < line_size; ++i) {
        double left_edge = (double)i / scale_d;
        double right_edge = left_edge + 1.0 / scale_d;

        // Ensure edges are within the source size
        double left = (left_edge < 0.0) ? 0.0 : left_edge;
        double right = (right_edge > (double)src_size) ? (double)src_size : right_edge;

        double total_weight = pFilter(right - left);

        if (total_weight < 0.0) {
            free(contributions);
            return NULL;
        }

        contributions[i].weight = total_weight;
    }

    return contributions;
}
