int bnep_add_connection(struct bnep_connadd_req *req, struct socket *sock) {
    struct net_device *dev;
    struct bnep_session *session;
    int err = -EINVAL;

    if (!sock) {
        return err;
    }

    dev = alloc_netdev(sizeof(struct bnep_session), "bnep%d", NET_NAME_UNKNOWN, ether_setup);
    if (!dev) {
        return -ENOMEM;
    }

    session = netdev_priv(dev);
    session->sock = sock;
    session->connected = 0;

    if (req->role & BNEP_ROLE_SRC) {
        memcpy(session->src_addr, req->src_addr, 6);
    }

    if (req->role & BNEP_ROLE_DST) {
        memcpy(session->dst_addr, req->dst_addr, 6);
    }

    if (check_existing_session(session->dst_addr)) {
        free_netdev(dev);
        return -EEXIST;
    }

    if (req->flags & BNEP_CONN_FLAGS_MULTICAST) {
        session->multicast_enabled = 1;
    }

    if (req->flags & BNEP_CONN_FLAGS_PROTO_FILTER) {
        session->proto_filter = req->proto_filter;
    }

    err = register_netdev(dev);
    if (err < 0) {
        free_netdev(dev);
        return err;
    }

    session->thread = kthread_run(bnep_session_handler, session, "bnep_session");
    if (IS_ERR(session->thread)) {
        unregister_netdev(dev);
        free_netdev(dev);
        return PTR_ERR(session->thread);
    }

    memcpy(req->dev_name, dev->name, IFNAMSIZ);

    return 0;
}
