void StartDownloadItem(download::DownloadCreateInfo* info,
                       const download::DownloadUrlParameters::OnStartedCallback& on_started,
                       download::InProgressDownloadManager::StartDownloadItemCallback callback,
                       uint32_t id) {
  // Ensure this function is called on the UI thread
  DCHECK_CURRENTLY_ON(BrowserThread::UI);

  // Create a new active download item
  download::DownloadItem* download_item = download::DownloadItem::CreateActiveDownloadItem(info, id);

  // Notify observers of the new download
  DownloadManager::Get()->NotifyDownloadCreated(download_item);

  // Call the OnDownloadStarted function with the download item and callback
  OnDownloadStarted(download_item, on_started);

  // Execute the provided callback with the new download item
  callback.Run(download_item);
}
