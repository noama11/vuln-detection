int parse_facilities(unsigned char *p, struct rose_facilities_struct *facilities) {
    int length = *p++;

    if (length == 0)
        return 1;

    if (length < 2)
        return 0;

    while (length >= 2) {
        int type = *p++;
        int len = *p++;

        length -= 2;

        if (len < 0 || len > length)
            return 0;

        switch (type) {
        case ROSE_FACILITY_NATIONAL:
            if (!parse_national_facility(p, len, facilities))
                return 0;
            break;
        case ROSE_FACILITY_CCITT:
            if (!parse_ccitt_facility(p, len, facilities))
                return 0;
            break;
        default:
            // Log debug message for unknown facility type
            fprintf(stderr, "Unknown facility type: %d\n", type);
            break;
        }

        p += len;
        length -= len;
    }

    return 1;
}
