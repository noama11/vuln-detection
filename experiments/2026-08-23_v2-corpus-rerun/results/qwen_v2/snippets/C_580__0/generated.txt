MagickBooleanType WriteImages(const ImageInfo *image_info, Image *images,
                              const char *filename, ExceptionInfo *exception)
{
  Image
    *image;

  ImageInfo
    *clone_info;

  MagickBooleanType
    status;

  register ssize_t
    i;

  clone_info = CloneImageInfo(image_info);
  if (clone_info == (ImageInfo *) NULL)
    return(MagickFalse);

  status = MagickTrue;
  for (i = 0; i < (ssize_t) GetImageListLength(images); i++)
  {
    image = GetImageList(images, i);
    if (image == (Image *) NULL)
      continue;

    if (clone_info->adjoin != MagickFalse)
    {
      clone_info->scene = i;
      clone_info->filename = (char *) filename;
    }
    else
    {
      char
        *output_filename;

      output_filename = (char *) AcquireQuantumMemory(MaxTextExtent,
        sizeof(*output_filename));
      if (output_filename == (char *) NULL)
        return(MagickFalse);

      FormatString(output_filename, "%s-%d", filename, i);
      clone_info->filename = output_filename;
    }

    status = WriteImage(clone_info, image, exception);
    if (status == MagickFalse)
      break;
  }

  clone_info = DestroyImageInfo(clone_info);
  return(status);
}
