void sock_to_str(const conn *c, const int af, char *addr, struct sockaddr *sock_addr) {
    const char *proto = (c->transport == tcp_transport) ? "tcp" : "udp";
    char buffer[INET6_ADDRSTRLEN];
    char port[NI_MAXSERV];
    int port_len = 0;

    if (af == AF_INET || af == AF_INET6) {
        void *ip;
        if (af == AF_INET) {
            ip = &((struct sockaddr_in *)sock_addr)->sin_addr;
            port_len = snprintf(port, sizeof(port), ":%d", ntohs(((struct sockaddr_in *)sock_addr)->sin_port));
        } else {
            ip = &((struct sockaddr_in6 *)sock_addr)->sin6_addr;
            port_len = snprintf(port, sizeof(port), ":%d", ntohs(((struct sockaddr_in6 *)sock_addr)->sin6_port));
        }

        if (inet_ntop(af, ip, buffer, sizeof(buffer)) == NULL) {
            snprintf(addr, MAX_SOCKET_STRLEN, "%s [%s]", proto, (af == AF_INET) ? "IPv4" : "IPv6");
        } else {
            snprintf(addr, MAX_SOCKET_STRLEN, "%s %s%s", proto, buffer, port);
        }
    } else if (af == AF_UNIX) {
        struct sockaddr_un *un = (struct sockaddr_un *)sock_addr;
        if (un->sun_path[0] == '\0') {
            snprintf(addr, MAX_SOCKET_STRLEN, "%s [UNIX abstract]", proto);
        } else {
            snprintf(addr, MAX_SOCKET_STRLEN, "%s %s", proto, un->sun_path);
        }
    } else {
        snprintf(addr, MAX_SOCKET_STRLEN, "%s [unknown address family]", proto);
    }
}
