int sysctl_dir_read(struct file *file, struct dir_context *ctx) {
    struct sysctl_dir_header *header;
    int ret;

    header = get_sysctl_dir_header(file);
    if (!header)
        return -ENOENT;

    ret = dir_emit_dot(ctx);
    if (ret)
        goto out;

    ret = dir_emit_dotdot(ctx);
    if (ret)
        goto out;

    while (header->entries) {
        ret = scan_sysctl_entry(header->entries, ctx);
        if (ret) {
            finish_sysctl_header(header);
            goto out;
        }
        header->entries = header->entries->next;
    }

    finish_sysctl_header(header);
out:
    return 0;
}
