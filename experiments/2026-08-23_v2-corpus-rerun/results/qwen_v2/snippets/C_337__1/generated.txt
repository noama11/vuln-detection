int kvm_set_pit_state(struct kvm *kvm, struct kvm_pit_state2 *ps)
{
    int i;

    mutex_lock(&kvm->pit_state->pit_mutex);
    kvm->pit_state->legacy_hpet = ps->legacy_hpet;
    for (i = 0; i < 3; i++) {
        kvm->pit_state->channels[i].count = ps->channels[i].count;
        kvm->pit_state->channels[i].init_count = ps->channels[i].init_count;
        kvm->pit_state->channels[i].mode = ps->channels[i].mode;
        kvm->pit_state->channels[i].latch = ps->channels[i].latch;
        kvm->pit_state->channels[i].configured = ps->channels[i].configured;
        kvm->pit_state->channels[i].retrieved = ps->channels[i].retrieved;
    }
    mutex_unlock(&kvm->pit_state->pit_mutex);

    return 0;
}
