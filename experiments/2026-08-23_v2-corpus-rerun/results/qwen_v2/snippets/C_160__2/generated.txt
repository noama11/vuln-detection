int load_tss_state(struct x86_emulate_ctxt *ctxt, struct tss_segment_32 *tss) {
    int rc;

    /* Load control register CR3 from TSS */
    ctxt->cr[3] = tss->cr3;

    /* Update instruction pointer and flags */
    ctxt->eip = tss->eip;
    ctxt->eflags = tss->eflags;

    /* Load general-purpose registers */
    ctxt->regs[VCPU_REGS_RAX] = tss->eax;
    ctxt->regs[VCPU_REGS_RCX] = tss->ecx;
    ctxt->regs[VCPU_REGS_RDX] = tss->edx;
    ctxt->regs[VCPU_REGS_RBX] = tss->ebx;
    ctxt->regs[VCPU_REGS_RSP] = tss->esp;
    ctxt->regs[VCPU_REGS_RBP] = tss->ebp;
    ctxt->regs[VCPU_REGS_RSI] = tss->esi;
    ctxt->regs[VCPU_REGS_RDI] = tss->edi;

    /* Load segment selectors before descriptors */
    ctxt->segment_selector[VCPU_SREG_LDT] = tss->ldt;
    ctxt->segment_selector[VCPU_SREG_ES] = tss->es;
    ctxt->segment_selector[VCPU_SREG_CS] = tss->cs;
    ctxt->segment_selector[VCPU_SREG_SS] = tss->ss;
    ctxt->segment_selector[VCPU_SREG_DS] = tss->ds;
    ctxt->segment_selector[VCPU_SREG_FS] = tss->fs;
    ctxt->segment_selector[VCPU_SREG_GS] = tss->gs;

    /* Load segment descriptors and check for faults */
    rc = load_segment_descriptor(ctxt, VCPU_SREG_LDT);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = load_segment_descriptor(ctxt, VCPU_SREG_ES);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = load_segment_descriptor(ctxt, VCPU_SREG_CS);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = load_segment_descriptor(ctxt, VCPU_SREG_SS);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = load_segment_descriptor(ctxt, VCPU_SREG_DS);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = load_segment_descriptor(ctxt, VCPU_SREG_FS);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    rc = load_segment_descriptor(ctxt, VCPU_SREG_GS);
    if (rc != X86EMUL_CONTINUE)
        return rc;

    return X86EMUL_CONTINUE;
}
