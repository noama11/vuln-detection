int generate_random_k(WC_RNG* rng, int size, mp_int* k, mp_int* order) {
    int ret = 0;
    byte* randBytes = NULL;
    mp_int temp;

    if (rng == NULL || k == NULL || order == NULL || size <= 0) {
        return BAD_FUNC_ARG;
    }

    randBytes = (byte*)malloc(size);
    if (randBytes == NULL) {
        return MEMORY_E;
    }

    if ((ret = wc_RNG_generate_block(rng, randBytes, size)) != 0) {
        goto cleanup;
    }

    mp_init(&temp);
    if ((ret = mp_read_unsigned_octets(&temp, randBytes, size)) != 0) {
        goto cleanup;
    }

    if (mp_cmp(&temp, order) >= 0 || mp_cmp_d(&temp, 0) <= 0) {
        ret = RNG_FAILURE_E;
        goto cleanup;
    }

    mp_copy(&temp, k);

cleanup:
    if (randBytes != NULL) {
        XMEMSET(randBytes, 0, size);
        free(randBytes);
    }
    mp_clear(&temp);
    return ret;
}
