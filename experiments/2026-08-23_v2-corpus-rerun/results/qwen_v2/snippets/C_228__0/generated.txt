int convert_path_component(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int ret = 0;
    int pos = 0;
    int i = 0;

    while (i < fromlen) {
        if (pos >= tolen - 1) {
            return -ENAMETOOLONG;
        }

        if (from[i] == 0) {
            to[pos++] = '.';
            i++;
        } else if (from[i] == 1) {
            to[pos++] = '.';
            to[pos++] = '.';
            i++;
        } else if (from[i] > 1) {
            int len = from[i];
            if (i + len > fromlen) {
                return -EIO;
            }

            if (pos + len >= tolen) {
                return -ENAMETOOLONG;
            }

            for (int j = 0; j < len; j++) {
                to[pos++] = from[i + 1 + j];
            }

            i += len + 1;
        } else {
            return -EIO;
        }
    }

    to[pos] = '\0';
    return ret;
}
