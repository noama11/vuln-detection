MagickBooleanType LoadTileImage(Image *image, Image *tile_image, XCFDocInfo *inDocInfo, XCFLayerInfo *inLayerInfo, size_t data_length, ExceptionInfo *exception) {
    register ssize_t x;
    ssize_t y;
    Quantum *q;
    ssize_t row;
    ssize_t column;
    size_t number_pixels;
    size::size_t length;
    unsigned char *pixel_data;

    if (inDocInfo->image_type == GIMP_GRAY || inDocInfo->image_type == GIMP_GRAY_ALPHA)
    {
        length = data_length * sizeof(*pixel_data);
        pixel_data = (unsigned char *) AcquireMemory(length);
        if (pixel_data == (unsigned char *) NULL)
            return MagickFalse;

        for (y = 0; y < (ssize_t) tile_image->rows; y++)
        {
            q = SetImagePixels(tile_image, 0, y, tile_image->columns, 1);
            if (q == (Quantum *) NULL)
                break;
            if (!ReadBlob(image, length, pixel_data))
            {
                SyncCacheViewPixelChannels(tile_image, q);
                continue;
            }
            for (x = 0; x < (ssize_t) tile_image->columns; x++)
            {
                *q = ScaleCharToQuantum(pixel_data[x]);
                q += 1;
            }
            if (!SyncCacheViewPixelChannels(tile_image, q))
                break;
        }
        RelinquishMemory(pixel_data);
    }
    else if (inDocInfo->image_type == GIMP_RGB || inDocInfo->image_type == GIMP_RGBA)
    {
        length = data_length * sizeof(*pixel_data);
        pixel_data = (unsigned char *) AcquireMemory(length);
        if (pixel_data == (unsigned char *) NULL)
            return MagickFalse;

        for (y = 0; y < (ssize_t) tile_image->rows; y++)
        {
            q = SetImagePixels(tile_image, 0, y, tile_image->columns, 1);
            if (q == (Quantum *) NULL)
                break;
            if (!ReadBlob(image, length, pixel_data))
            {
                SyncCacheViewPixelChannels(tile_image, q);
                continue;
            }
            for (x = 0; x < (ssize_t) tile_image->columns; x++)
            {
                *q = ScaleCharToQuantum(pixel_data[x * 4 + 0]);
                q += 1;
                *q = ScaleCharToQuantum(pixel_data[x * 4 + 1]);
                q += 1;
                *q = ScaleCharToQuantum(pixel_data[x * 4 + 2]);
                q += 1;
                *q = ScaleCharToQuantum(pixel_data[x * 4 + 3]);
                q += 1;
            }
            if (!SyncCacheViewPixelChannels(tile_image, q))
                break;
        }
        RelinquishMemory(pixel_data);
    }

    return MagickTrue;
}
