CURLcode close_curl_easy_handle(struct Curl_easy *data) {
    if (!data) {
        return CURLE_OK;
    }

    /* Clear timers */
    Curl_cleanup_timer(data);

    /* Detach from multi handle if part of one */
    if (data->multi) {
        Curl_multi_remove_handle(data->multi, data);
    }

    /* Clean up multi handles */
    if (data->multi_easy) {
        Curl_multi_easy_cleanup(data->multi_easy);
    }

    /* Destroy timeout lists */
    Curl_llist_destroy(&data->state.timeoutlist, NULL);

    /* Free allocated memory for states and configurations */
    Curl_safefree(data->set.str[STRING_COOKIEJAR]);
    Curl_safefree(data->set.str[STRING_COOKIE_FILE]);
    Curl_safefree(data->set.str[STRING_USERPWD]);
    Curl_safefree(data->set.str[STRING_OPTIONS]);
    Curl_safefree(data->set.str[STRING_CERT]);
    Curl_safefree(data->set.str[STRING_KEY]);
    Curl_safefree(data->set.str[STRING_CA_CACHE]);
    Curl_safefree(data->set.str[STRING_CAPATH]);
    Curl_safefree(data->set.str[STRING_RANDOM_FILE]);
    Curl_safefree(data->set.str[STRING_EGDSOCKET]);
    Curl_safefree(data->set.str[STRING_SSL_PINNEDPUBLICKEY]);
    Curl_safefree(data->set.str[STRING_ALTSVC]);
    Curl_safefree(data->set.str[STRING_ALTSVC_CTRL]);
    Curl_safefree(data->set.str[STRING_MAIL_FROM]);
    Curl_safefree(data->set.str[STRING_MAIL_RCPT]);

    /* Close SSL sessions */
    Curl_ssl_close(data);

    /* Clean up redirect info */
    Curl_cleanup_cookies(data);
    Curl_cleanup_http2(data);
    Curl_cleanup_connectdata(data);

    /* Clean up share and wildcard structures */
    if (data->share) {
        Curl_share_cleanup(data->share);
    }

    if (data->wildcard) {
        Curl_cleanup_wildcardmatch(data->wildcard);
    }

    /* Free the easy handle itself */
    Curl_safefree(data);

    return CURLE_OK;
}
