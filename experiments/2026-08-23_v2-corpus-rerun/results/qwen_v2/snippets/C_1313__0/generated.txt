download::DownloadItemImpl* CreateActiveDownloadItem(uint32_t id, const download::DownloadCreateInfo& info) {
    // Check if a download with the given id already exists
    if (download::DownloadManager::GetInstance()->HasDownload(id)) {
        return nullptr;
    }

    // Create a new download item using the provided information
    download::DownloadItemImpl* download_item = new download::DownloadItemImpl(id, info);

    // Store it in the downloads map
    download::DownloadManager::GetInstance()->AddDownload(download_item);

    // Attach additional information to the download item
    download_item->AttachAdditionalInfo();

    return download_item;
}
