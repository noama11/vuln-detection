Image *ExtractPostscriptImage(Image *image, const ImageInfo *image_info, MagickOffsetType PS_Offset, ssize_t PS_Size, ExceptionInfo *exception) {
    Image *new_image = NULL;
    ImageInfo *clone_info = CloneImageInfo(image_info);
    char filename[MagickPathExtent];

    GetPathComponent(image->filename, HeadPath, filename, MagickPathExtent);
    (void) CopyMagickString(filename, GetImageBlobFilename(image), MagickPathExtent);

    if (WriteBlobImage(clone_info, filename, image, exception) == MagickFalse)
        goto cleanup;

    if (IsPostscriptFormat(filename, exception) == MagickFalse)
        goto cleanup;

    new_image = ReadImage(clone_info, exception);
    if (new_image == NULL)
        goto cleanup;

    (void) CloneImageAttributes(image, new_image);
    (void) DestroyImageList(image);
    image = new_image;

cleanup:
    if (clone_info != NULL)
        clone_info = DestroyImageInfo(clone_info);
    return image;
}
