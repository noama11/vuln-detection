u64 get_netlink_attr_offset(u64 ctx, u64 A, u64 X, u64 r4, u64 r5) {
    struct sk_buff *skb = (struct sk_buff *)ctx;
    struct nlattr *nla;

    if (!skb || skb_is_nonlinear(skb) || skb->len < sizeof(*nla))
        return 0;

    if (A >= skb->len)
        return 0;

    nla = (struct nlattr *)((u8 *)skb->data + A);
    if (nla->nla_type == X)
        return (u64)((u8 *)nla - (u8 *)skb->data);

    return 0;
}
