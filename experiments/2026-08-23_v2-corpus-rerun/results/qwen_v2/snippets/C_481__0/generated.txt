plist_t parse_string_node(const char **bnode, uint64_t size) {
    plist_t node = NULL;
    char *str = (char *)malloc(size + 1);

    if (!str) {
        return NULL;
    }

    memcpy(str, *bnode, size);
    str[size] = '\0';

    node = plist_new_node(PLIST_STRING, str, size + 1);
    if (!node) {
        free(str);
    }

    return node;
}
