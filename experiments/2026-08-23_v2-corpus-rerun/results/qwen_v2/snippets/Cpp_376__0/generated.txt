AP4_HdlrAtom::AP4_HdlrAtom(AP4_UI32 size, AP4_UI08 version, AP4_UI32 flags, AP4_ByteStream& stream) :
    AP4_Atom(AP4_ATOM_TYPE_HDLR, size, version, flags)
{
    AP4_UI32  pre_defined;
    AP4_UI32  handler_type;
    AP4_UI32  reserved[2];

    // read the fields
    stream.ReadBEUI32(pre_defined);
    stream.ReadBEUI32(handler_type);
    stream.ReadBEUI32(reserved[0]);
    stream.ReadBEUI32(reserved[1]);

    // read the name
    AP4_Size name_size = m_size - (stream.Tell() - m_start_position);
    if (name_size > 0) {
        AP4_String name;
        if (name_size > 1 && (name_size - 1) < 256) {
            // Pascal string
            AP4_UI08 pascal_length;
            stream.Read(pascal_length);
            name_size--;
            if (pascal_length < name_size) {
                name.Resize(pascal_length);
                stream.Read(name, pascal_length);
                name_size -= pascal_length;
            } else {
                // invalid pascal string, treat as c string
                name.Resize(name_size);
                stream.Read(name, name_size);
            }
        } else {
            // C string
            name.Resize(name_size);
            stream.Read(name, name_size);
        }
        m_name = name;
    }
}
