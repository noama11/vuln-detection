#include <string.h>
#include <stdlib.h>

void normalize_path(char* file) {
    if (file == NULL) return;

    char* result = (char*)malloc(strlen(file) + 1);
    if (result == NULL) return;

    char* src = file;
    char* dst = result;
    int depth = 0;

    while (*src) {
        if (*src == '/') {
            // Skip consecutive slashes
            while (*src == '/') src++;
            // Write a single slash
            *dst++ = '/';
        } else if (*src == '.' && (src[1] == '/' || src[1] == '\0')) {
            // Skip './'
            src++;
            while (*src == '/') src++;
        } else if (*src == '.' && src[1] == '.' && (src[2] == '/' || src[2] == '\0')) {
            // Handle '../'
            src += 2;
            if (depth > 0) {
                dst--;
                while (dst > result && *dst != '/') dst--;
                if (*dst == '/') dst++;
                depth--;
            }
            // Skip the slash after '../'
            while (*src == '/') src++;
        } else {
            // Copy normal character
            *dst++ = *src++;
            if (*dst == '/') {
                depth++;
            }
        }
    }

    // Null-terminate the result
    *dst = '\0';

    // If the result is empty or just a slash, set it to a single slash
    if (result[0] == '\0') {
        strcpy(file, "/");
    } else {
        // Remove trailing slash if not at root
        if (strlen(result) > 1 && result[strlen(result) - 1] == '/') {
            result[strlen(result) - 1] = '\0';
        }
        strcpy(file, result);
    }

    free(result);
}
