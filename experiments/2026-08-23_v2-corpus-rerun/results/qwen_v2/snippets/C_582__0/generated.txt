int init_efi_state(struct boot_params *params, unsigned long params_load_addr,
                   unsigned int efi_map_offset, unsigned int efi_map_sz,
                   unsigned int efi_setup_data_offset) {
    if (!efi_map_sz)
        return 0;

    if (!is_1to1_mapping_enabled())
        return 0;

    memcpy(params->efi_info, boot_params.efi_info, sizeof(params->efi_info));
    params->efi_map = params_load_addr + efi_map_offset;
    params->efi_map_size = efi_map_sz;
    params->efi_setup_data = params_load_addr + efi_setup_data_offset;

    return 0;
}
