gboolean handle_new_file_transfer(gpointer data, gint fd, b_input_condition cond) {
    TransferData *transfer_data = (TransferData *)data;
    Transfer *transfer = transfer_data->transfer;
    Connection *connection = transfer_data->connection;
    Buddy *buddy = connection_find_buddy(connection, transfer->buddy_handle);

    if (!buddy || !transfer_start(transfer)) {
        return FALSE;
    }

    transfer->callbacks.on_data_received = handle_file_data_received;
    transfer->callbacks.on_complete = handle_file_transfer_complete;
    transfer->callbacks.on_error = handle_file_transfer_error;

    transfer_set_data(transfer, transfer_data);

    return TRUE;
}
