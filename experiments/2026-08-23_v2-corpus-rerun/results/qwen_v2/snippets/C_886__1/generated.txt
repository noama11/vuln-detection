AppProto AppLayerProtoDetectDetect(AppLayerProtoDetectThreadCtx *tctx, Flow *f,
                                   uint8_t *buf, uint32_t buflen, uint8_t ipproto,
                                   uint8_t direction) {
    AppProto alproto = ALPROTO_UNKNOWN;

    if (!FlowGetAppProto(f)) {
        if (tctx->pm_proto != ALPROTO_UNKNOWN) {
            alproto = tctx->pm_proto;
            if (alproto == ALPROTO_DCERPC && ipproto == IPPROTO_UDP) {
                AppProto pp_proto = AppLayerProtoDetectPPDetect(tctx, f, buf, buflen, ipproto, direction);
                if (pp_proto != ALPROTO_UNKNOWN) {
                    alproto = pp_proto;
                }
            }
        } else {
            alproto = AppLayerProtoDetectPMDetect(tctx, f, buf, buflen, ipproto, direction);
        }

        if (alproto == ALPROTO_UNKNOWN) {
            alproto = AppLayerProtoDetectExpectationListCheck(f, ipproto);
        }

        if (alproto != ALPROTO_UNKNOWN) {
            FlowSetAppProto(f, alproto);
        }
    } else {
        alproto = FlowGetAppProto(f);
    }

    return alproto;
}
