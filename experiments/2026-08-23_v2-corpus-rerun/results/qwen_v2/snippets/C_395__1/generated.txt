Image *ReadJNGImage(const ImageInfo *image_info, ExceptionInfo *exception)
{
    Image *image = NULL;
    FILE *file = fopen(image_info->filename, "rb");
    if (!file)
    {
        ThrowFileException(exception, FileOpenError, "UnableToOpenFile", image_info->filename);
        return NULL;
    }

    unsigned char signature[8];
    if (fread(signature, 1, 8, file) != 8 || memcmp(signature, "\x89JNG\r\n\x1a\n", 8) != 0)
    {
        fclose(file);
        ThrowException(exception, CorruptImageError, "NotAJNGFile", image_info->filename);
        return NULL;
    }

    fseek(file, 0, SEEK_END);
    size_t file_size = (size_t)ftell(file);
    rewind(file);

    if (file_size < 8)
    {
        fclose(file);
        ThrowException(exception, CorruptImageError, "InvalidFileSize", image_info->filename);
        return NULL;
    }

    MngInfo *mng_info = (MngInfo *)AcquireMagickMemory(sizeof(*mng_info));
    if (!mng_info)
    {
        fclose(file);
        ThrowException(exception, ResourceLimitError, "UnableToAllocateMemory", "MngInfo");
        return NULL;
    }

    // Read image data and populate the image structure
    image = CloneImageInfoToImage(image_info, exception);
    if (!image)
    {
        RelinquishMagickMemory(mng_info);
        fclose(file);
        return NULL;
    }

    // Read image dimensions and validate
    unsigned int width = 0;
    unsigned int height = 0;
    if (fread(&width, sizeof(width), 1, file) != 1 || fread(&height, sizeof(height), 1, file) != 1)
    {
        RelinquishMagickMemory(mng_info);
        fclose(file);
        DestroyImage(image);
        ThrowException(exception, CorruptImageError, "InvalidImageDimensions", image_info->filename);
        return NULL;
    }

    if (width == 0 || height == 0)
    {
        RelinquishMagickMemory(mng_info);
        fclose(file);
        DestroyImage(image);
        ThrowException(exception, CorruptImageError, "InvalidImageDimensions", image_info->filename);
        return NULL;
    }

    image->columns = width;
    image->rows = height;

    // Allocate pixel data
    image->pixels = (PixelPacket *)AcquireQuantumMemory(width, height * sizeof(*image->pixels));
    if (!image->pixels)
    {
        RelinquishMagickMemory(mng_info);
        fclose(file);
        DestroyImage(image);
        ThrowException(exception, ResourceLimitError, "UnableToAllocateMemory", "PixelPacket");
        return NULL;
    }

    // Read pixel data
    if (fread(image->pixels, sizeof(*image->pixels), width * height, file) != width * height)
    {
        RelinquishMagickMemory(mng_info);
        fclose(file);
        DestroyImage(image);
        ThrowException(exception, CorruptImageError, "UnableToReadImageData", image_info->filename);
        return NULL;
    }

    fclose(file);
    RelinquishMagickMemory(mng_info);
    return image;
}
