long compat_sock_recvmsg_many(int fd, struct compat_mmsghdr __user *mmsg,
                              unsigned int vlen, unsigned int flags,
                              struct compat_timespec __user *timeout) {
    struct timespec ktimeout;
    long ret;

    if (timeout) {
        if (get_user(ktimeout.tv_sec, &timeout->tv_sec) ||
            get_user(ktimeout.tv_nsec, &timeout->tv_nsec))
            return -EFAULT;

        if (ktimeout.tv_nsec > 999999999)
            return -EINVAL;
    } else {
        ktimeout.tv_sec = 0;
        ktimeout.tv_nsec = 0;
    }

    ret = kernel_recvmsg(fd, mmsg, vlen, flags, &ktimeout);

    if (ret >= 0 && timeout) {
        if (put_user(ktimeout.tv_sec, &timeout->tv_sec) ||
            put_user(ktimeout.tv_nsec, &timeout->tv_nsec))
            return -EFAULT;
    }

    return ret;
}
