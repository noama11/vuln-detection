int path_to_string(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int i = 0, j = 0;

    if (tolen < 1)
        return -ENAMETOOLONG;

    while (i < fromlen) {
        unsigned char type = from[i++];
        unsigned char len = from[i++];

        if (i + len > fromlen)
            return -EINVAL;

        if (j + len + 1 > tolen)
            return -ENAMETOOLONG;

        switch (type) {
            case DT_DIR:
                to[j++] = '/';
                break;
            case DT_LNK:
                to[j++] = '@';
                break;
            default:
                break;
        }

        for (int k = 0; k < len; k++) {
            to[j++] = from[i++];
        }
    }

    to[j] = '\0';
    return 0;
}
