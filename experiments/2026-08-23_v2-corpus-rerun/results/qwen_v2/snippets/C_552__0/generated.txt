int check_memory_range_validity(struct rxe_mem* mem, u64 iova, size_t length) {
    if (mem->type == MEM_TYPE_DMA) {
        return 0;
    } else if (mem->type == MEM_TYPE_MR || mem->type == MEM_TYPE_FMR) {
        if (iova < mem->start || length > mem->allocated_length) {
            return -EFAULT;
        }
        return 0;
    } else {
        return -EFAULT;
    }
}
