void process_lru_command(conn *c, token_t *tokens, size_t ntokens) {
    if (ntokens < 2) {
        conn_send(c, "ERROR: Insufficient arguments.\r\n", strlen("ERROR: Insufficient arguments.\r\n"));
        return;
    }

    if (strcmp(tokens[0].value, "tune") == 0) {
        if (ntokens != 5) {
            conn_send(c, "ERROR: Invalid number of arguments for 'tune'.\r\n", strlen("ERROR: Invalid number of arguments for 'tune'.\r\n"));
            return;
        }

        int hot_pct = atoi(tokens[1].value);
        int warm_pct = atoi(tokens[2].value);
        int hot_age = atoi(tokens[3].value);
        int warm_age = atoi(tokens[4].value);

        if (hot_pct < 0 || hot_pct > 100 || warm_pct < 0 || warm_pct > 100 ||
            hot_age < 0 || warm_age < 0) {
            conn_send(c, "ERROR: Invalid values for 'tune'.\r\n", strlen("ERROR: Invalid values for 'tune'.\r\n"));
            return;
        }

        // Update the LRU settings
        c->hot_percentage = hot_pct;
        c->warm_percentage = warm_pct;
        c->hot_age_factor = hot_age;
        c->warm_age_factor = warm_age;

        conn_send(c, "OK: LRU settings updated.\r\n", strlen("OK: LRU settings updated.\r\n"));
    } else if (strcmp(tokens[0].value, "mode") == 0) {
        if (ntokens != 2) {
            conn_send(c, "ERROR: Invalid number of arguments for 'mode'.\r\n", strlen("ERROR: Invalid number of arguments for 'mode'.\r\n"));
            return;
        }

        if (strcmp(tokens[1].value, "flat") == 0) {
            c->lru_mode = LRU_MODE_FLAT;
            conn_send(c, "OK: LRU mode set to flat.\r\n", strlen("OK: LRU mode set to flat.\r\n"));
        } else if (strcmp(tokens[1].value, "segmented") == 0) {
            c->lru_mode = LRU_MODE_SEGMENTED;
            conn_send(c, "OK: LRU mode set to segmented.\r\n", strlen("OK: LRU mode set to segmented.\r\n"));
        } else {
            conn_send(c, "ERROR: Invalid mode argument.\r\n", strlen("ERROR: Invalid mode argument.\r\n"));
        }
    } else if (strcmp(tokens[0].value, "temp_ttl") == 0) {
        if (ntokens != 2) {
            conn_send(c, "ERROR: Invalid number of arguments for 'temp_ttl'.\r\n", strlen("ERROR: Invalid number of arguments for 'temp_ttl'.\r\n"));
            return;
        }

        int temp_ttl = atoi(tokens[1].value);
        if (temp_ttl < 0) {
            conn_send(c, "ERROR: Invalid TTL value.\r\n", strlen("ERROR: Invalid TTL value.\r\n"));
            return;
        }

        c->temp_ttl = temp_ttl;
        conn_send(c, "OK: Temporary TTL set.\r\n", strlen("OK: Temporary TTL set.\r\n"));
    } else {
        conn_send(c, "ERROR: Unknown command.\r\n", strlen("ERROR: Unknown command.\r\n"));
    }
}
