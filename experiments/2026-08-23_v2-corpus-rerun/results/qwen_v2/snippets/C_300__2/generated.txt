int configure_icmpv6_socket(struct ndp *ndp) {
    int sockfd;
    int hoplimit = 1;
    int pktinfo = 1;
    int recvhoplimit = 1;
    int opt;

    sockfd = socket(AF_INET6, SOCK_RAW, IPPROTO_ICMPV6);
    if (sockfd < 0) {
        return -1;
    }

    opt = setsockopt(sockfd, SOL_IPV6, IPV6_RECVPKTINFO, &pktinfo, sizeof(pktinfo));
    if (opt < 0) {
        close(sockfd);
        return -2;
    }

    opt = setsockopt(sockfd, SOL_IPV6, IPV6_MULTICAST_HOPS, &hoplimit, sizeof(hoplimit));
    if (opt < 0) {
        close(sockfd);
        return -3;
    }

    opt = setsockopt(sockfd, SOL_IPV6, IPV6_RECVHOPLIMIT, &recvhoplimit, sizeof(recvhoplimit));
    if (opt < 0) {
        close(sockfd);
        return -4;
    }

    ndp->sockfd = sockfd;
    return 0;
}
