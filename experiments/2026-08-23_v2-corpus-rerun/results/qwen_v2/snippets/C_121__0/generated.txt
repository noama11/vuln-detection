int set_apparmor_task_attribute(struct task_struct *task, char *name, void *value, size_t size) {
    if (size == 0 || !value || *(char *)((char *)value + size - 1) != '\0') {
        return -EINVAL;
    }

    if (task != current) {
        return -EPERM;
    }

    if (strcmp(name, "profile") == 0) {
        return set_apparmor_profile(value);
    } else if (strcmp(name, "label") == 0) {
        return set_apparmor_label(value);
    } else {
        audit_log_untrusted_string(name);
        return -EINVAL;
    }

    return size;
}
