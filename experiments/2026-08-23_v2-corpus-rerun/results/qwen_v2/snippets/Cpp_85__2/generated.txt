void applyToneMapping(float shift, float smooth) {
    // Clamp shift and smooth to their valid ranges
    shift = std::clamp(shift, 0.25f, 8.0f);
    smooth = std::clamp(smooth, 0.0f, 1.0f);

    const int TBLN = 256;
    float* lut = new float[TBLN];

    // Compute the lookup table based on shift
    for (int i = 0; i < TBLN; ++i) {
        float x = static_cast<float>(i) / (TBLN - 1);
        if (shift <= 1.0f) {
            // Linear scaling
            lut[i] = x * shift;
        } else {
            // Cubic function
            float a = 2.0f * (1.0f - smooth) * (1.0f - smooth) * (1.0f - smooth);
            float b = 3.0f * (1.0f - smooth) * (1.0f - smooth) * smooth;
            float c = 3.0f * (1.0f - smooth) * smooth * smooth;
            float d = 2.0f * smooth * smooth * smooth;
            lut[i] = a * x * x * x + b * x * x + c * x + d;
        }
    }

    // Apply the LUT to each pixel in the image data
    for (int y = 0; y < C->height; ++y) {
        for (int x = 0; x < C->width; ++x) {
            float pixelValue = C->imageData[y][x];
            int index = static_cast<int>(pixelValue * (TBLN - 1));
            C->imageData[y][x] = lut[index];
        }
    }

    // Update the maximum data values in the context object
    C->maxDataValue = lut[TBLN - 1];

    // Free the allocated memory for the LUT
    delete[] lut;
}
