long sys_semtimedop(int semid, struct oabi_sembuf __user *tsops, unsigned nsops, const struct timespec __user *timeout) {
    struct oabi_sembuf *sops;
    long ret = -EINVAL;
    struct timespec to;

    if (nsops == 0)
        return -EINVAL;

    sops = kmalloc(sizeof(struct oabi_sembuf) * nsops, GFP_KERNEL);
    if (!sops)
        return -ENOMEM;

    if (copy_from_user(sops, tsops, sizeof(struct oabi_sembuf) * nsops))
        goto out_free;

    if (timeout) {
        if (copy_from_user(&to, timeout, sizeof(to)))
            goto out_free;
    }

    ret = do_semtimedop(semid, sops, nsops, timeout ? &to : NULL);

out_free:
    kfree(sops);
    return ret;
}
