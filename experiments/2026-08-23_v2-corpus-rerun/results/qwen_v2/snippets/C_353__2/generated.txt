struct key *construct_and_link_key(struct keyring_search_context *ctx,
                                   const char *callout_info,
                                   size_t callout_len,
                                   void *aux,
                                   struct key *dest_keyring,
                                   unsigned long flags) {
    struct key *key = NULL;
    struct key *index_key = NULL;
    struct user_struct *user = NULL;
    int ret;

    if (!dest_keyring || !callout_info || !callout_len) {
        return ERR_PTR(-EINVAL);
    }

    index_key = keyring_search(ctx, NULL, NULL);
    if (IS_ERR(index_key)) {
        return index_key;
    }

    user = get_current_user();
    if (!user) {
        key_put(index_key);
        return ERR_PTR(-ENOMEM);
    }

    key = key_alloc(index_key, callout_len, user, NULL, NULL, flags, NULL);
    if (IS_ERR(key)) {
        put_user(user);
        key_put(index_key);
        return key;
    }

    ret = key_instantiate(key, callout_info, callout_len, aux);
    if (ret < 0) {
        key_put(key);
        put_user(user);
        key_put(index_key);
        return ERR_PTR(ret);
    }

    ret = key_link(key, dest_keyring);
    if (ret < 0) {
        key_put(key);
        put_user(user);
        key_put(index_key);
        return ERR_PTR(ret);
    }

    key_put(index_key);
    put_user(user);

    return key;
}
