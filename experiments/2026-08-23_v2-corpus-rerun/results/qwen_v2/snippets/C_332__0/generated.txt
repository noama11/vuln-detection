bool_t xdr_string(XDR *xdrs, char **objp)
{
    u_int len;
    u_int i;
    char *sp;
    char *newsp;

    if (xdrs->x_op == XDR_ENCODE) {
        if (*objp == NULL) {
            return (XDR_PUTLONG(xdrs, (long *)0));
        }
        len = strlen(*objp) + 1;
        if (!XDR_PUTLONG(xdrs, (long *)&len)) {
            return (FALSE);
        }
        for (i = 0; i < len; i++) {
            if (!XDR_PUTBYTE(xdrs, (*objp)[i])) {
                return (FALSE);
            }
        }
        return (TRUE);
    } else if (xdrs->x_op == XDR_DECODE) {
        if (!XDR_GETLONG(xdrs, (long *)&len)) {
            return (FALSE);
        }
        if (len == 0) {
            *objp = NULL;
            return (TRUE);
        }
        if ((newsp = (char *)malloc(len)) == NULL) {
            return (FALSE);
        }
        *objp = newsp;
        for (i = 0; i < len; i++) {
            if (!XDR_GETBYTE(xdrs, &newsp[i])) {
                free(newsp);
                *objp = NULL;
                return (FALSE);
            }
        }
        return (TRUE);
    } else if (xdrs->x_op == XDR_FREE) {
        if (*objp != NULL) {
            free(*objp);
            *objp = NULL;
        }
        return (TRUE);
    }
    return (FALSE);
}
