int handle_far_jump(struct x86_emulate_ctxt *ctxt) {
    uint16_t selector;
    struct desc_struct descriptor;
    int status;

    status = get_segment_selector(ctxt, &selector);
    if (status != X86EMUL_CONTINUE)
        return status;

    status = load_segment_descriptor(ctxt, selector, &descriptor);
    if (status != X86EMUL_CONTINUE)
        return status;

    ctxt->eip = get_descriptor_ip(&descriptor);
    return X86EMUL_CONTINUE;
}
