void OpenMediaDevice(int32_t page_request_id, const char* device_id, blink_MediaStreamType type, OpenDeviceCallback callback, MediaDeviceSaltAndOrigin salt_and_origin) {
    if (!IsOriginAllowed(salt_and_origin.origin)) {
        callback(false, NULL);
        return;
    }

    MediaStreamManager_OpenDevice(page_request_id, device_id, type, callback, OnDeviceStopped);
}
