void xfs_da_fix_hashpath(struct xfs_da_state *state, struct xfs_da_state_path *path) {
    struct xfs_da_intnode *node;
    struct xfs_dir2_leaf *leaf;
    struct xfs_dir2_leaf_entry *entries;
    int i, j, level;

    for (level = path->path_level - 1; level >= 0; level--) {
        struct xfs_buf *bp = path->path_entry[level].bp;
        uint32_t magic = be16_to_cpu(bp->b_addr->magic);

        if (magic == XFS_DIR2_LEAF1_MAGIC || magic == XFS_DIR2_LEAFN_MAGIC) {
            leaf = (struct xfs_dir2_leaf *)bp->b_addr;
            entries = xfs_dir2_leafN_ents(leaf);
            for (i = be16_to_cpu(leaf->count) - 1; i >= 0; i--) {
                if (entries[i].hashval == state->hashval) {
                    for (j = i + 1; j < be16_to_cpu(leaf->count); j++) {
                        entries[j].hashval = entries[j - 1].hashval;
                    }
                    entries[i].hashval = state->hashval;
                    xfs_trans_log_buf(state->args->trans, bp, 0, XFS_DIR2_LEAF_SIZE - 1);
                    break;
                }
            }
        } else if (magic == XFS_DA_NODE_MAGIC) {
            node = (struct xfs_da_intnode *)bp->b_addr;
            for (i = node->count - 1; i >= 0; i--) {
                if (node->hashval[i] == state->hashval) {
                    memmove(&node->hashval[i + 1], &node->hashval[i], sizeof(uint32_t) * (node->count - i));
                    node->hashval[i] = state->hashval;
                    xfs_trans_log_buf(state->args->trans, bp, 0, XFS_DA_NODE_SIZE - 1);
                    break;
                }
            }
        }
    }
}
