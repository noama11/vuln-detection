int decodeADPCMAudioBlock(const uint8_t* encoded, int16_t* decoded) {
    int channelCount = getTrackChannelCount();
    int sampleCount = 0;

    for (int ch = 0; ch < channelCount; ++ch) {
        ADPCMState state;
        initADPCMState(&state);

        uint8_t predictor = *encoded++;
        uint8_t index = *encoded++;

        for (int i = 0; i < FRAME_COUNT; ++i) {
            int16_t sample1 = decodeSample(predictor, index, &state);
            int16_t sample2 = decodeSample(predictor, index, &state);

            decoded[sampleCount++] = sample1;
            decoded[sampleCount++] = sample2;
        }
    }

    return sampleCount * sizeof(int16_t);
}
