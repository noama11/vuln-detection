struct sk_buff *fragment_udp_packet(struct sk_buff *skb, u32 features) {
    int mss = skb->dev->mtu - sizeof(struct ipv6hdr) - sizeof(struct udphdr);
    struct sk_buff *frag_skb;

    if (skb->len <= mss)
        return skb;

    if (skb_is_gso(skb)) {
        skb_gso_validate(skb, mss);
        skb->csum_start = 0;
        skb->csum_offset = offsetof(struct udphdr, check);
    }

    skb_push(skb, sizeof(struct udphdr));
    skb_push(skb, sizeof(struct ipv6hdr));

    frag_skb = skb_segment(skb, features);
    skb_pop(skb, sizeof(struct ipv6hdr) + sizeof(struct udphdr));

    if (IS_ERR(frag_skb))
        return frag_skb;

    return frag_skb;
}
