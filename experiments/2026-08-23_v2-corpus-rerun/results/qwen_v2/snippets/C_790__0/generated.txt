void js_EmitTryCatchFinally(JSTranslator *J, js_Ast *trystm, js_Ast *catchvar, js_Ast *catchstm, js_Ast *finallystm) {
    int tryLabel = J->labelCount++;
    int catchLabel = J->labelCount++;
    int finallyLabel = J->labelCount++;
    int endLabel = J->labelCount++;

    J->emit(J, "goto L%d;\n", tryLabel);
    J->emit(J, "L%d:\n", tryLabel);
    js_EmitStatement(J, trystm);

    if (J->exceptionLabel >= 0) {
        J->emit(J, "if (J->exception) goto L%d;\n", catchLabel);
    }

    J->emit(J, "L%d:\n", catchLabel);
    if (catchvar && J->strictMode) {
        J->emit(J, "if (strcmp(\"%s\", \"%s\") == 0) {\n", catchvar->name, "eval");
        J->emit(J, "    J->error = \"Invalid strict mode variable %s\";\n", catchvar->name);
        J->emit(J, "}\n");
    }

    if (catchvar) {
        J->emit(J, "js_SetVariable(J, \"%s\", J->exception);\n", catchvar->name);
    }

    js_EmitStatement(J, catchstm);

    if (J->exception) {
        J->emit(J, "J->exception = 1;\n");
    }

    J->emit(J, "goto L%d;\n", finallyLabel);
    J->emit(J, "L%d:\n", finallyLabel);
    js_EmitStatement(J, finallystm);
    J->emit(J, "L%d:\n", endLabel);
}
