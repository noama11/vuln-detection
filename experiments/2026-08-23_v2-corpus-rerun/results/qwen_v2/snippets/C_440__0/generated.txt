void mrb_gc_mark_context_stack(mrb_state *mrb, struct mrb_context *c) {
  mrb_value *sp = c->stack;
  mrb_value *stack = c->stack;
  mrb_value *stack_end = c->stack + MRB_STACK_SIZE;

  if (!stack) return;

  while (sp < stack_end && *sp != mrb_nil_value()) {
    mrb_gc_mark(mrb, *sp);
    sp++;
  }

  while (sp < stack_end) {
    *sp++ = mrb_nil_value();
  }
}
