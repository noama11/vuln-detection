int read_multimedia_header(AVFormatContext *avctx) {
    int version;

    // Skip initial bytes and read the version
    if (avio_skip(avctx->pb, 4) < 0)
        return AVERROR(EIO);

    version = avio_r8(avctx->pb);

    if (version == 2) {
        AVStream *astream = avformat_new_stream(avctx, NULL);
        AVStream *vstream = avformat_new_stream(avctx, NULL);

        if (!astream || !vstream)
            return AVERROR(ENOMEM);

        // Set audio stream parameters
        astream->codecpar->codec_type = AVMEDIA_TYPE_AUDIO;
        astream->codecpar->sample_rate = avio_rl32(avctx->pb);
        astream->codecpar->channels = avio_r8(avctx->pb);
        astream->codecpar->format = avio_r8(avctx->pb);

        // Set video stream parameters
        vstream->codecpar->codec_type = AVMEDIA_TYPE_VIDEO;
        vstream->codecpar->width = avio_rl32(avctx->pb);
        vstream->codecpar->height = avio_rl32(avctx->pb);
        vstream->codecpar->format = avio_r8(avctx->pb);

        // Read metadata and frame information
        av_dict_set(&avctx->metadata, "title", avio_get_str(avctx->pb, 1024), 0);
        av_dict_set(&avctx->metadata, "artist", avio_get_str(avctx->pb, 1024), 0);

        // Read frame info
        int num_frames = avio_rl32(avctx->pb);
        for (int i = 0; i < num_frames; i++) {
            avio_skip(avctx->pb, 8); // Skip frame data
        }

        return 0;
    } else if (version == 0) {
        int next_val = avio_r8(avctx->pb);
        if (next_val == 3) {
            int num_audio_tracks = avio_rl32(avctx->pb);
            int num_video_tracks = avio_rl32(avctx->pb);

            for (int i = 0; i < num_audio_tracks; i++) {
                AVStream *astream = avformat_new_stream(avctx, NULL);
                if (!astream)
                    return AVERROR(ENOMEM);

                astream->codecpar->codec_type = AVMEDIA_TYPE_AUDIO;
                astream->codecpar->sample_rate = avio_rl32(avctx->pb);
                astream->codecpar->channels = avio_r8(avctx->pb);
            }

            for (int i = 0; i < num_video_tracks; i++) {
                AVStream *vstream = avformat_new_stream(avctx, NULL);
                if (!vstream)
                    return AVERROR(ENOMEM);

                vstream->codecpar->codec_type = AVMEDIA_TYPE_VIDEO;
                vstream->codecpar->width = avio_rl32(avctx->pb);
                vstream->codecpar->height = avio_rl32(avctx->pb);
                vstream->codecpar->format = avio_r8(avctx->pb);
            }

            return 0;
        }
    }

    // Request sample for unrecognized version
    av_log(avctx, AV_LOG_ERROR, "Unrecognized version, sample required.\n");
    return AVERROR_PATCHWELCOME;
}
