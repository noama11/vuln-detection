int sock_recvmsg(struct kiocb *iocb, struct socket *sock, struct msghdr *msg,
                 size_t size, int msg_flags) {
    int ret = 0;
    struct sock *sk = sock->sk;

    while (1) {
        ret = sk->sk_prot->recvmsg(iocb, sk, msg, size, msg_flags);
        if (ret > 0) {
            msg->msg_flags = msg_flags;
            sock_update_stats(sk, ret);
            break;
        } else if (ret == -EAGAIN && (msg_flags & MSG_DONTWAIT)) {
            break;
        } else if (ret < 0) {
            break;
        }
    }

    return ret;
}
