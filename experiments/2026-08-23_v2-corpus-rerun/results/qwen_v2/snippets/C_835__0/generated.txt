bool extract_socket_address_and_port(struct pool *pool, char *url) {
    char *start = url;
    char *end = NULL;
    char *port_str = NULL;
    int port = 0;
    size_t address_len = 0;
    char address_buffer[1024];

    // Check for IPv6 address in brackets
    if (*start == '[') {
        start++;
        end = strchr(start, ']');
        if (!end) {
            return false;
        }
        port_str = strchr(end, ':');
        if (port_str) {
            *port_str++ = '\0';
        }
    } else {
        end = strchr(start, ':');
        if (end) {
            *end++ = '\0';
            port_str = end;
            port_str = strchr(port_str, ':');
            if (port_str) {
                *port_str++ = '\0';
            } else {
                port_str = NULL;
            }
        }
    }

    if (!end) {
        end = start + strlen(start);
    }

    if (end - start >= sizeof(address_buffer)) {
        return false;
    }

    strncpy(address_buffer, start, sizeof(address_buffer) - 1);
    address_buffer[sizeof(address_buffer) - 1] = '\0';

    if (!port_str) {
        port = 80; // default port
    } else {
        port = atoi(port_str);
    }

    // Update the pool structure with the extracted values
    strncpy(pool->address, address_buffer, sizeof(pool->address) - 1);
    pool->port = port;

    return true;
}
