unsigned process_audio_data(SF_PRIVATE *psf, unsigned len)
{
    unsigned bytes_read = 0;
    unsigned remaining = len;
    FLAC__StreamDecoder *decoder = psf->decoder;

    if (!decoder || !FLAC__stream_decoder_get_state(decoder)) {
        psf_log_printf(psf, "FLAC decoder is not initialized or in an invalid state.\n");
        return 0;
    }

    if (psf->frame && psf->frame_remaining > 0) {
        unsigned copy_len = SF_MIN(psf->frame_remaining, remaining);
        memcpy(psf->current_buffer + bytes_read, psf->frame, copy_len);
        bytes_read += copy_len;
        psf->frame += copy_len;
        psf->frame_remaining -= copy_len;
        remaining -= copy_len;
    }

    while (remaining > 0 && !FLAC__stream_decoder_get_state(decoder)) {
        FLAC__StreamDecoderState state = FLAC__stream_decoder_get_state(decoder);
        if (state != FLAC__STREAM_DECODER_SEARCHING_FOR_METADATA && state != FLAC__STREAM_DECODER_READING_METADATA) {
            break;
        }

        if (FLAC__stream_decoder_process_single(decoder)) {
            unsigned decoded = psf->frame_remaining;
            unsigned copy_len = SF_MIN(decoded, remaining);
            memcpy(psf->current_buffer + bytes_read, psf->frame, copy_len);
            bytes_read += copy_len;
            psf->frame += copy_len;
            psf->frame_remaining -= copy_len;
            remaining -= copy_len;
        } else {
            psf_log_printf(psf, "Error decoding FLAC audio frame.\n");
            psf->frame = NULL;
            break;
        }
    }

    return bytes_read;
}
