struct net *get_net_ns_by_id(struct net *net, int id) {
    struct net *result = NULL;

    if (id < 0)
        return NULL;

    read_lock(&net->ns_id_lock);
    spin_lock(&net->ns_id_spinlock);

    result = idr_find(&net->ns_idr, id);
    if (result) {
        get_net(result);
    }

    spin_unlock(&net->ns_id_spinlock);
    read_unlock(&net->ns_id_lock);

    return result;
}
