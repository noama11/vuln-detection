int parse_imap_fetch_response(struct ImapHeader *h, char *s) {
    char *ptr = s;
    int in_field = 0;
    char field_name[64];
    int field_name_len = 0;
    int in_value = 0;
    char value[1024];
    int value_len = 0;

    while (*ptr) {
        if (isspace(*ptr)) {
            if (in_field && in_value) {
                h->flags = 0;
                h->uid = 0;
                h->internal_date[0] = '\0';
                h->size = 0;
                return -1;
            }
            ptr++;
            continue;
        }

        if (!in_field) {
            if (isalpha(*ptr)) {
                field_name[field_name_len++] = *ptr;
                in_field = 1;
            } else {
                h->flags = 0;
                h->uid = 0;
                h->internal_date[0] = '\0';
                h->size = 0;
                return -1;
            }
        } else {
            if (*ptr == '(') {
                if (field_name_len == 5 && !strncmp(field_name, "FLAGS", 5)) {
                    in_value = 1;
                } else if (field_name_len == 3 && !strncmp(field_name, "UID", 3)) {
                    in_value = 1;
                } else if (field_name_len == 10 && !strncmp(field_name, "INTERNALDA", 10)) {
                    in_value = 1;
                } else if (field_name_len == 9 && !strncmp(field_name, "RFC822.SIZE", 9)) {
                    in_value = 1;
                } else if (field_name_len == 9 && !strncmp(field_name, "BODY", 4)) {
                    return -2;
                } else if (field_name_len == 11 && !strncmp(field_name, "RFC822.HEAD", 11)) {
                    return -2;
                } else {
                    h->flags = 0;
                    h->uid = 0;
                    h->internal_date[0] = '\0';
                    h->size = 0;
                    return -1;
                }
            } else if (isalnum(*ptr)) {
                field_name[field_name_len++] = *ptr;
            } else {
                h->flags = 0;
                h->uid = 0;
                h->internal_date[0] = '\0';
                h->size = 0;
                return -1;
            }
        }

        ptr++;
    }

    if (in_field && in_value) {
        h->flags = 0;
        h->uid = 0;
        h->internal_date[0] = '\0';
        h->size = 0;
        return -1;
    }

    h->flags = 0;
    h->uid = 0;
    h->internal_date[0] = '\0';
    h->size = 0;
    return -1;
}
