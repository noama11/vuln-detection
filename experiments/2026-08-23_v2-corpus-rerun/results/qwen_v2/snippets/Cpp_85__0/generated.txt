int subtract_black_level() {
  if (!is_phaseone_compressed && (black_level[0] || black_level[1] || black_level[2] || black_level[3])) {
    int maxval = 0;
    int bl[4];
    for (int i = 0; i < 4; ++i)
      bl[i] = black_level[i];

    for (int c = 0; c < 4; ++c) {
      for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
          int val = image[y][x][c] - bl[c];
          if (val < 0)
            val = 0;
          image[y][x][c] = val;
          if (val > maxval)
            maxval = val;
        }
      }
    }
    return maxval;
  } else {
    int maxval = 0;
    for (int c = 0; c < 4; ++c) {
      for (int y = 0; y < height; ++y) {
        for (int x = 0; x < width; ++x) {
          if (image[y][x][c] > maxval)
            maxval = image[y][x][c];
        }
      }
    }
    return maxval;
  }
}
