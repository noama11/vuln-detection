void handle_media_stream_request(bool update_content_setting) {
    if (!are_content_settings_available() || !allows_media_stream_access()) {
        return;
    }

    MediaDevice* devices = get_default_media_devices();
    if (!devices) {
        return;
    }

    if (update_content_setting) {
        set_permission_for_secure_schemes();
    }

    register_media_stream_with_ui(devices);

    execute_callback(devices, get_ui_information());
}
