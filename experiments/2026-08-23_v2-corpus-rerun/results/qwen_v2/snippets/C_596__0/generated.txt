#include <linux/printk.h>
#include <linux/vmalloc.h>
#include <linux/kref.h>

#define dprintk(level, fmt, args...) \
    do { \
        if (current_debug_level >= level) \
            printk("vbuf-sg: " fmt, ##args); \
    } while (0)

struct videobuf_mapping {
    struct kref ref;
    // Other fields...
};

void videobuf_vm_open(struct vm_area_struct *vma)
{
    struct videobuf_mapping *mapping = vma->vm_private_data;

    dprintk(1, "Opening vma at %p, mapping %p, refcount %d\n",
            vma, mapping, kref_read(&mapping->ref));
    kref_get(&mapping->ref);
}
