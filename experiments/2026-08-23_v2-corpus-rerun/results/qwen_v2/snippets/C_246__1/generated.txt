int netlink_dump(struct sock *sk)
{
    struct sk_buff *skb = NULL;
    int ret = 0;

    mutex_lock(&sk->sk_mutex);
    if (sk->sk_callback_running) {
        mutex_unlock(&sk->sk_mutex);
        return -EBUSY;
    }

    if (!sk->sk_rcvbuf) {
        ret = -ENOBUFS;
        goto out;
    }

    skb = alloc_skb(sk->sk_rcvbuf, GFP_KERNEL);
    if (!skb) {
        ret = -ENOMEM;
        goto out;
    }

    if (!sk->dump_callback || sk->dump_callback(sk, skb)) {
        ret = -EINVAL;
        goto out;
    }

    if (sk->filter && sk_filter(skb, sk->filter)) {
        ret = -EPERM;
        goto out;
    }

    if (skb->len > 0) {
        netlink_unicast(sk, skb, sk->sk_protocol, MSG_DONTWAIT);
    } else {
        kfree_skb(skb);
    }

out:
    mutex_unlock(&sk->sk_mutex);
    return ret;
}
