sctp_disposition_t sctp_handle_cookie_echo(struct net *net,
                                           const struct sctp_endpoint *ep,
                                           const struct sctp_association *asoc,
                                           sctp_subtype_t type,
                                           void *arg,
                                           sctp_cmd_seq_t *commands)
{
    sctp_chunk_t *chunk = (sctp_chunk_t *)arg;
    sctp_disposition_t disposition = SCTP_DISPOSITION_CONSUME;
    sctp_cookie_param_t *cookie;
    int len;

    if (!chunk || chunk->length < sizeof(sctp_cookie_param_t)) {
        return SCTP_DISPOSITION_BAD;
    }

    len = ntohs(chunk->length);
    cookie = (sctp_cookie_param_t *)chunk->skb->data;

    if (len < sizeof(sctp_cookie_param_t)) {
        return SCTP_DISPOSITION_BAD;
    }

    if (sctp_unpack_cookie(net, ep, asoc, cookie, len)) {
        sctp_cmd_seq_add(commands, SCTP_CMD_COOKIE_ECHOED);
        if (ntohl(asoc->c.my_vtag) != ntohs(cookie->vtag)) {
            sctp_cmd_seq_add(commands, SCTP_CMD_COOKIE_ECHO_COLLISION);
        } else {
            sctp_cmd_seq_add(commands, SCTP_CMD_COOKIE_ECHO_RESTART);
        }
    } else {
        disposition = SCTP_DISPOSITION_NOMEM;
    }

    return disposition;
}
