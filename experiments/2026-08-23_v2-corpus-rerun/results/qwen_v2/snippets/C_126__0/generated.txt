int rock_ridge_continue(struct rock_state *rs) {
    if (!rs || rs->offset + rs->size > rs->block_size) {
        return -1;
    }

    void *buffer = malloc(rs->size);
    if (!buffer) {
        return -1;
    }

    int result = read_extent(rs->extent, buffer, rs->offset, rs->size);
    if (result < 0) {
        free(buffer);
        return result;
    }

    rs->buffer = buffer;
    rs->offset += rs->size;

    return 0;
}
