void unregister_ppp_channel(struct ppp_channel *chan) {
    if (!chan)
        return;

    chan->ppp = NULL;

    write_lock(&chan->lock);

    if (chan->file) {
        ppp_disconnect_channel(chan);
        list_del(&chan->list);
        fput_light(chan->file, 1);

        if (atomic_dec_and_test(&chan->file->f_count))
            ppp_destroy_channel(chan);

        wake_up_all(&chan->file->wait);
    }

    write_unlock(&chan->lock);
}
