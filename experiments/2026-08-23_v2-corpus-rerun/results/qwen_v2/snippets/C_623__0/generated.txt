ExprDef *append_keysym_list(ExprDef *expr, ExprDef *append) {
    size_t original_size = expr->keysym_count;
    size_t append_count = append->keysym_count;

    // Resize the keysym array of the first expression to accommodate the new entries
    expr->keysym = realloc(expr->keysym, (original_size + append_count) * sizeof(*expr->keysym));
    if (!expr->keysym) {
        // Handle memory allocation failure
        free(append);
        return NULL;
    }

    // Copy the keysym list from the second expression to the first
    for (size_t i = 0; i < append_count; ++i) {
        expr->keysym[original_size + i] = append->keysym[i];
    }

    // Update the count of keysyms in the first expression
    expr->keysym_count += append_count;

    // Free the memory associated with the second expression
    free(append->keysym);
    free(append);

    return expr;
}
