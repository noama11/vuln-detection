int snd_seq_remove_events(struct snd_seq_client *client, void __user *arg)
{
    struct {
        int mode;
    } info;

    if (copy_from_user(&info, arg, sizeof(info)))
        return -EFAULT;

    if (info.mode & SND_SEQ_REMOVE_INPUT) {
        if (client->type == SND_SEQ_CLIENT_TYPE_USER)
            snd_seq_fifo_clear(&client->input_fifo);
    }

    if (info.mode & SND_SEQ_REMOVE_OUTPUT) {
        snd_seq_cell_t *cell;
        while ((cell = snd_seq_queue_remove_cell(client->output_queue)) != NULL)
            snd_seq_cell_free(cell);
    }

    return 0;
}
