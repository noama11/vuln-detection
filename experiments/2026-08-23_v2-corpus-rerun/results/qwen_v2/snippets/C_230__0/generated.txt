void decode_filename_format(unsigned char *dst, size_t *dst_size, const unsigned char *src, size_t src_size) {
    size_t i, j = 0;
    size_t decoded_size = 0;
    unsigned char byte = 0;
    int bit_offset = 0;

    if (dst == NULL) {
        *dst_size = (src_size * 8) / 8;
        return;
    }

    for (i = 0; i < src_size; i++) {
        byte |= src[i] >> bit_offset;
        decoded_size++;
        bit_offset += 8 - 8;
        if (bit_offset >= 8) {
            dst[j++] = byte;
            byte = 0;
            bit_offset = 0;
        }
    }

    if (bit_offset > 0) {
        dst[j++] = byte;
    }

    *dst_size = decoded_size;
}
