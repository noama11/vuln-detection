OPJ_BOOL read_and_decode_rle4_bitmap(FILE* IN, OPJ_UINT8* pData, OPJ_UINT32 stride, OPJ_UINT32 width, OPJ_UINT32 height) {
    OPJ_UINT32 row, col;
    OPJ_UINT32 expected_pixel_count = width * height;
    OPJ_UINT32 pixel_count = 0;
    OPJ_UINT8 byte;
    OPJ_UINT8* pRow = pData;
    OPJ_UINT8* pPixel = pData;

    for (row = 0; row < height; ++row) {
        OPJ_UINT32 cols_remaining = width;
        OPJ_UINT8* pEndRow = pRow + stride;

        while (cols_remaining > 0) {
            if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                return OPJ_FALSE;
            }

            if ((byte & 0x80) != 0) {
                /* Encoded mode */
                OPJ_UINT8 count = (byte & 0x7F) + 1;
                OPJ_UINT8 value = 0;

                if (fread(&value, sizeof(OPJ_UINT8), 1, IN) != 1) {
                    return OPJ_FALSE;
                }

                while (count > 0 && cols_remaining > 0) {
                    *pPixel++ = (value >> 4) & 0x0F;
                    *pPixel++ = value & 0x0F;
                    count -= 2;
                    cols_remaining -= 2;
                    pixel_count += 2;
                }
            } else {
                /* Absolute mode */
                OPJ_UINT8 count = byte + 1;

                if (count > cols_remaining) {
                    count = cols_remaining;
                }

                OPJ_UINT8* pDest = pPixel;
                OPJ_UINT8* pEnd = pDest + count;

                while (pDest < pEnd) {
                    if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                        return OPJ_FALSE;
                    }

                    *pDest++ = (byte >> 4) & 0x0F;
                    *pDest++ = byte & 0x0F;
                    pixel_count += 2;
                }

                pPixel += count;
                cols_remaining -= count;
            }

            if (pPixel >= pEndRow) {
                break;
            }
        }

        pRow += stride;
        pPixel = pRow;
    }

    if (pixel_count != expected_pixel_count) {
        /* Warning: pixel count mismatch */
    }

    return (pixel_count == expected_pixel_count) ? OPJ_TRUE : OPJ_FALSE;
}
