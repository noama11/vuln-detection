int kvm_set_one_core_register(struct kvm_vcpu *vcpu, const struct kvm_one_reg *reg)
{
    int ret = 0;
    unsigned int offset;
    void *dst;

    offset = kvm_register_offset(reg->id);
    if (offset == KVM_REG_INVALID)
        return -ENOENT;

    if (offset + sizeof(u64) > sizeof(vcpu->arch.regs))
        return -EINVAL;

    dst = (void *)&vcpu->arch.regs + offset;
    if (copy_from_user(&vcpu->arch.regs, reg->addr, sizeof(u64)))
        return -EFAULT;

    if (reg->id == KVM_REG_ARM_CORE_REG) {
        u64 val = *(u64 *)dst;
        if ((val & KVM_REG_MODE_MASK) != KVM_REG_MODE_EL1 &&
            (val & KVM_REG_MODE_MASK) != KVM_REG_MODE_EL2) {
            return -EINVAL;
        }
    }

    return ret;
}
