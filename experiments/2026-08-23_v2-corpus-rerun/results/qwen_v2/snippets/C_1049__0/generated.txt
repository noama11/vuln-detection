void FFmpegVideoDecodeEngine_Initialize(MessageLoop* message_loop, VideoDecodeEngine::EventHandler* event_handler, VideoDecodeContext* context, const VideoDecoderConfig& config) {
    AVCodecContext* codec_context = avcodec_alloc_context3(NULL);
    if (!codec_context) {
        // Handle error
        return;
    }

    // Configure codec context based on config
    codec_context->thread_count = 4; // Example default value

    int ret = avcodec_open2(codec_context, avcodec_find_decoder(AV_CODEC_ID_H264), NULL);
    if (ret < 0) {
        // Handle error
        avcodec_free_context(&codec_context);
        return;
    }

    // Setup buffer pool for output frames
    AVBufferPool* buffer_pool = av_buffer_pool_init(1024, NULL);
    if (!buffer_pool) {
        // Handle error
        avcodec_free_context(&codec_context);
        return;
    }

    // Handle codec-specific configurations
    codec_context->flags |= AV_CODEC_FLAG_LOW_DELAY;

    // Error detection settings
    codec_context->error_detection = AV_EF_CAREFUL;

    // Notify event handler upon completion
    event_handler->on_initialized(message_loop, context, codec_context);
}
