int sock_recv(int fd, void __user *ubuf, size_t size, unsigned int flags,
              struct sockaddr __user *addr, int __user *addr_len) {
    struct socket *sock;
    struct msghdr msg;
    struct iovec iov;
    int ret;

    if (size == 0)
        return -EINVAL;

    sock = sockfd_lookup(fd, &ret);
    if (!sock)
        return ret;

    iov.iov_base = ubuf;
    iov.iov_len = size;

    msg.msg_name = addr;
    msg.msg_namelen = addr ? sizeof(*addr) : 0;
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;
    msg.msg_control = NULL;
    msg.msg_controllen = 0;
    msg.msg_flags = flags;

    ret = sock->ops->recvmsg(sock, &msg, size, flags);

    if (addr && msg.msg_namelen) {
        if (copy_to_user(addr_len, &msg.msg_namelen, sizeof(*addr_len)))
            ret = -EFAULT;
    }

    sockfd_put(sock);
    return ret;
}
