int ext4_orphan_add(handle_t *handle, struct inode *inode) {
    int err = 0;
    struct super_block *sb = inode->i_sb;
    struct ext4_super_block *es = EXT4_SB(sb)->s_es;
    struct ext4_inode_info *ei = EXT4_I(inode);

    if (!EXT4_HAS_JOURNAL(sb)) {
        return -EINVAL;
    }

    spin_lock(&EXT4_SB(sb)->s_orphan_lock);

    if (ei->i_orphan) {
        spin_unlock(&EXT4_SB(sb)->s_orphan_lock);
        return -EEXIST;
    }

    if (!S_ISREG(inode->i_mode) && !S_ISDIR(inode->i_mode) &&
        !S_ISLNK(inode->i_mode)) {
        spin_unlock(&EXT4_SB(sb)->s_orphan_lock);
        return -EINVAL;
    }

    err = journal_get_write_access(handle, sb->s_jdev);
    if (err) {
        spin_unlock(&EXT4_SB(sb)->s_orphan_lock);
        return err;
    }

    err = journal_space_reservation(handle, sb->s_jdev, 1);
    if (err) {
        spin_unlock(&EXT4_SB(sb)->s_orphan_lock);
        return err;
    }

    if (es->s_orphan_count == 0) {
        es->s_orphan_count = cpu_to_le32(1);
        es->s_last_orphan = cpu_to_le32(inode->i_ino);
    } else {
        ei->i_orphan = le32_to_cpu(es->s_last_orphan);
        es->s_last_orphan = cpu_to_le32(inode->i_ino);
    }

    mark_inode_dirty(inode);

    list_add(&ei->i_orphan_list, &EXT4_SB(sb)->s_orphan);
    spin_unlock(&EXT4_SB(sb)->s_orphan_lock);

    return err;
}
