int lego_usb_tower_init(struct usb_interface *interface, const struct usb_device_id *id) {
    struct usb_device *udev = interface_to_usbdev(interface);
    struct lego_usb_tower_dev *dev;
    int retval = -ENOMEM;

    dev = kzalloc(sizeof(*dev), GFP_KERNEL);
    if (!dev)
        goto error;

    spin_lock_init(&dev->lock);
    mutex_init(&dev->data_mutex);

    dev->interface = interface;
    dev->udev = usb_get_dev(udev);
    usb_set_intfdata(interface, dev);

    if (usb_find_common_endpoint(udev, &dev->int_in, NULL, NULL, NULL, USB_DIR_IN, USB_ENDPOINT_XFER_INT, NULL)) {
        dev_err(&udev->dev, "Could not find interrupt in endpoint\n");
        goto error;
    }

    dev->read_buffer = kmalloc(LEGOWARE_READ_BUFFER_SIZE, GFP_KERNEL);
    if (!dev->read_buffer)
        goto error;

    dev->write_buffer = kmalloc(LEGOWARE_WRITE_BUFFER_SIZE, GFP_KERNEL);
    if (!dev->write_buffer)
        goto error;

    retval = usb_submit_urb(dev->int_in, GFP_KERNEL);
    if (retval) {
        dev_err(&udev->dev, "Failed to submit interrupt urb: %d\n", retval);
        goto error;
    }

    /* Retrieve and log firmware version */
    if (lego_usb_get_firmware_version(dev)) {
        dev_err(&udev->dev, "Failed to get firmware version\n");
        goto error;
    }

    return 0;

error:
    if (dev) {
        usb_set_intfdata(interface, NULL);
        usb_put_dev(dev->udev);
        kfree(dev->read_buffer);
        kfree(dev->write_buffer);
        kfree(dev);
    }
    return retval;
}
