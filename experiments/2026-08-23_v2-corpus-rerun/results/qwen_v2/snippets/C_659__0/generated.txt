int get_jpeg_dimensions(unsigned char* data, unsigned int data_size, int* width, int* height) {
    if (data_size < 2 || data[0] != 0xFF || data[1] != 0xD8) {
        return -1; // Invalid SOI marker
    }

    unsigned int i = 2;
    while (i < data_size - 1) {
        if (data[i] == 0xFF && data[i + 1] == 0xC0) {
            // Found the Start of Frame (SOF) marker
            if (data_size < i + 7) {
                return -1; // Not enough data for SOF header
            }

            // Skip the marker length (2 bytes)
            i += 2;

            // Extract height (2 bytes) and width (2 bytes)
            *height = (data[i + 1] << 8) | data[i];
            *width = (data[i + 3] << 8) | data[i + 2];

            return 0; // Success
        } else if (data[i] == 0xFF) {
            // Skip to the next marker
            while (i < data_size && data[i] == 0xFF) {
                i++;
            }

            if (i < data_size) {
                i++; // Skip the marker byte
            }
        } else {
            i++;
        }
    }

    return -1; // Could not find SOF marker
}
