int get_interface_settings(struct fst_card_info *card, struct fst_port_info *port, struct ifreq *ifr) {
    struct sync_config sync;
    int size;

    if (!card || !port || !ifr)
        return -EFAULT;

    switch (port->hw_if) {
        case IF_TYPE_E1:
            ifr->ifr_type = IF_TYPE_E1;
            break;
        case IF_TYPE_T1:
            ifr->ifr_type = IF_TYPE_T1;
            break;
        default:
            return -EFAULT;
    }

    size = ifr->ifr_size;
    if (size < sizeof(sync))
        return -ENOMEM;

    sync.clock_rate = port->clock_rate;
    sync.clock_type = port->clock_type;

    if (copy_to_user(ifr->ifr_data, &sync, sizeof(sync)))
        return -EFAULT;

    ifr->ifr_size = sizeof(sync);
    return 0;
}
