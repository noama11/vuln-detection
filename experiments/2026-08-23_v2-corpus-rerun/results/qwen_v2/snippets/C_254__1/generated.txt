int process_sysctl_dir_read(struct file *file, struct dir_context *ctx) {
    struct sysctl_dir_header *header;
    int ret;

    header = get_sysctl_dir_header(file);
    if (!header)
        return -ENOENT;

    ret = dir_emit_dot(ctx);
    if (ret)
        return ret;

    ret = dir_emit_dotdot(ctx);
    if (ret)
        return ret;

    while (header->entries && !dir_emit_done(ctx)) {
        struct sysctl_entry *entry = header->entries;
        ret = scan_sysctl_entry(entry, ctx);
        if (ret)
            break;
        header->entries = entry->next;
    }

    finish_sysctl_header(header);
    return 0;
}
