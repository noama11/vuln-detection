int flush_key_notify_policy(const struct km_event *c)
{
    struct sk_buff *skb;
    int err;

    skb = alloc_skb(KEY_NOTIFY_MSG_SIZE, GFP_KERNEL);
    if (!skb) {
        return -ENOBUFS;
    }

    /* Initialize the message header with information from the event */
    skb->data[0] = c->sequence;
    skb->data[1] = c->port_id;

    /* Broadcast the message */
    err = broadcast_key_message(skb);
    if (err < 0) {
        kfree_skb(skb);
        return err;
    }

    return 0;
}
