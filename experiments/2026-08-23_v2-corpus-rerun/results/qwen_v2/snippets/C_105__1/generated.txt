krb5_error_code prepare_kdc_request(krb5_kdc_req *request, krb5_principal *krbtgt_princ) {
    krb5_error_code ret = 0;
    krb5_data *realm = NULL;

    if ((request->kdc_options & KDC_OPT_CANONICALIZE) && krbtgt_princ) {
        /* Attempt to map the FQDN to a realm */
        ret = krb5_kntvname_to_realm(krb5_context, &request->client, &realm);
        if (ret != 0) {
            return ret;
        }

        /* Modify the request to construct a cross-realm TGT */
        if (realm != NULL) {
            krb5_free_data(krb5_context, realm);
            realm = NULL;
        }

        ret = krb5_build_principal_ext(krb5_context, krbtgt_princ, realm->length, realm->data, 0, NULL);
        if (ret != 0) {
            krb5_free_data(krb5_context, realm);
            return ret;
        }
    }

    return ret;
}
