#include <sys/inotify.h>
#include <unistd.h>
#include <errno.h>
#include <limits.h>

int create_inotify_instance(int flags) {
    // Check if the flags are valid
    if ((flags & ~(IN_CLOEXEC | IN_NONBLOCK)) != 0) {
        errno = EINVAL;
        return -1;
    }

    // Check if the user has exceeded the maximum allowed inotify instances
    // This is a simplified check and may need to be adapted to the system's actual limits
    if (RLIM_INFINITY == RLIM_INFINITY) {
        errno = ENFILE;
        return -1;
    }

    // Create a new inotify instance
    int fd = inotify_init1(flags);
    if (fd < 0) {
        return -1;
    }

    return fd;
}
