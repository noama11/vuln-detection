void prepare_upnp_event_notify(struct upnp_event_notify *obj) {
    if (!obj->subscription_valid) {
        obj->state = UPNP_EVENT_ERROR;
        return;
    }

    const char *service_type = obj->subscription.service_type;
    const char *xml_vars = get_xml_vars_for_service(service_type);

    if (!xml_vars) {
        obj->state = UPNP_EVENT_ERROR;
        return;
    }

    size_t buffer_size = 256;
    char *buffer = malloc(buffer_size);
    if (!buffer) {
        obj->state = UPNP_EVENT_ERROR;
        return;
    }

    while (1) {
        int result = snprintf(buffer, buffer_size, "NOTIFY * HTTP/1.1\r\n"
                              "Content-Type: text/xml; charset=\"utf-8\"\r\n"
                              "NT: upnp:event\r\n"
                              "NTS: ss:ev\r\n"
                              "SID: %s\r\n"
                              "SEQ: %d\r\n"
                              "Content-Length: %zu\r\n"
                              "\r\n"
                              "%s",
                              obj->subscription.sid,
                              obj->sequence_number,
                              strlen(xml_vars),
                              xml_vars);

        if (result < 0) {
            free(buffer);
            obj->state = UPNP_EVENT_ERROR;
            return;
        }

        if ((size_t)result < buffer_size) {
            obj->message = buffer;
            obj->message_length = result;
            obj->state = UPNP_EVENT_READY;
            return;
        }

        buffer_size *= 2;
        char *new_buffer = realloc(buffer, buffer_size);
        if (!new_buffer) {
            free(buffer);
            obj->state = UPNP_EVENT_ERROR;
            return;
        }
        buffer = new_buffer;
    }
}
