void rds_info_flip_copy(struct rds_incoming *inc, struct rds_info_iterator *iter, __be32 saddr, __be32 daddr, int flip)
{
    struct rds_info_message msg;

    msg.seq = inc->seq;
    msg.len = inc->len;

    if (flip) {
        msg.saddr = daddr;
        msg.daddr = saddr;
        msg.sport = inc->dport;
        msg.dport = inc->sport;
    } else {
        msg.saddr = saddr;
        msg.daddr = daddr;
        msg.sport = inc->sport;
        msg.dport = inc->dport;
    }

    rds_info_copy(iter, &msg, sizeof(msg));
}
