void test_gimp_image_save_load(Gimp *gimp, gboolean with_unusual_stuff, gboolean compat_paths, gboolean use_gimp_2_8_features)
{
  GimpImage *image;
  gchar *temp_file;
  GimpImage *loaded_image;
  GError *error = NULL;

  image = create_test_gimp_image(gimp, with_unusual_stuff, compat_paths, use_gimp_2_8_features);
  g_assert(image != NULL);

  temp_file = g_strdup_printf("/tmp/gimp_test_image_%d.xcf", getpid());

  gimp_xcf_save(image, temp_file, NULL, NULL, &error);
  g_assert_no_error(error);

  loaded_image = gimp_xcf_load(temp_file, NULL, &error);
  g_assert_no_error(error);
  g_assert(loaded_image != NULL);

  gimp_image_delete(loaded_image);

  g_unlink(temp_file);
  g_free(temp_file);
}
