int ppp_register_channel(struct net *net, struct ppp_channel *chan)
{
    struct ppp_channel *new_chan;
    int ret = -ENOMEM;

    new_chan = kmalloc(sizeof(*new_chan), GFP_KERNEL);
    if (!new_chan)
        goto out;

    memset(new_chan, 0, sizeof(*new_chan));
    new_chan->chan = chan;

    spin_lock_bh(&ppp_channel_lock);
    list_add(&new_chan->list, &net->ppp.channels);
    net->ppp.num_channels++;
    spin_unlock_bh(&ppp_channel_lock);

    ret = 0;

out:
    return ret;
}
