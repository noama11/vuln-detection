AP4_Result ReadVisualSampleEntry(AP4_ByteStream& stream) {
    AP4_Result result = ReadSampleEntry(stream);
    if (AP4_FAILED(result)) return result;

    AP4_UI32 predefined1;
    AP4_UI16 width;
    AP4_UI16 height;
    AP4_UI32 horizResolution;
    AP4_UI32 vertResolution;
    AP4_UI32 dataSize;
    AP4_UI16 frameCount;
    AP4_UI16 compressorNameLength = 32;
    char compressorName[32];

    result = stream.ReadUI32(predefined1);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI16(width);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI16(height);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI32(horizResolution);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI32(vertResolution);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI32(dataSize);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI16(frameCount);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI16(compressorNameLength);
    if (AP4_FAILED(result)) return result;

    if (compressorNameLength > sizeof(compressorName)) {
        compressorNameLength = sizeof(compressorName);
    }

    result = stream.Read(compressorName, compressorNameLength);
    if (AP4_FAILED(result)) return result;

    compressorName[compressorNameLength] = '\0';

    AP4_UI16 depth;
    AP4_UI16 predefined2;

    result = stream.ReadUI16(depth);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI16(predefined2);
    if (AP4_FAILED(result)) return result;

    return AP4_SUCCESS;
}
