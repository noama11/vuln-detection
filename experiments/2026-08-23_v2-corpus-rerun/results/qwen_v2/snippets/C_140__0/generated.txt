int cifs_tree_connect(unsigned int xid, struct cifs_ses *ses, const char *tree,
                      struct cifs_tcon *tcon, const struct nls_table *cp) {
    int rc = 0;
    char *utf16_path = NULL;
    int utf16_path_len = 0;
    struct smb2_tree_connect_req *req = NULL;
    struct smb2_tree_connect_rsp *rsp = NULL;
    struct kvec iov[1];
    struct kvec rsp_iov;
    int resp_buftype;
    void *resp_buf;

    if (!ses || !tree || !tcon || !cp) {
        rc = -EINVAL;
        goto out;
    }

    utf16_path_len = cifs_str_to_utf16le(tree, NULL, 0, cp);
    if (utf16_path_len < 0) {
        rc = utf16_path_len;
        goto out;
    }

    utf16_path = kmalloc(utf16_path_len + 2, GFP_KERNEL);
    if (!utf16_path) {
        rc = -ENOMEM;
        goto out;
    }

    utf16_path_len = cifs_str_to_utf16le(tree, utf16_path, utf16_path_len + 2, cp);
    if (utf16_path_len < 0) {
        rc = utf16_path_len;
        goto out;
    }

    req = cifs_smb2_init(SMB2_TREE_CONNECT, NULL, 0, &iov, 1);
    if (!req) {
        rc = -ENOMEM;
        goto out;
    }

    req->path_offset = cpu_to_le16(offsetof(struct smb2_tree_connect_req, path));
    req->path_length = cpu_to_le16(utf16_path_len);
    memcpy(req->path, utf16_path, utf16_path_len);

    rc = cifs_call_async(ses, tcon, iov, 1, &rsp_iov, &resp_buftype, 0, NULL, NULL);
    if (rc < 0)
        goto out;

    rsp = (struct smb2_tree_connect_rsp *)rsp_iov.iov_base;
    if (rsp->Capabilities && rsp->Capabilities != tcon->capabilities) {
        tcon->capabilities = le32_to_cpu(rsp->Capabilities);
    }

    tcon->share_type = le32_to_cpu(rsp->ShareType);
    tcon->maximal_access = le32_to_cpu(rsp->MaximalAccess);

    if (resp_buftype == CIFS_SMALL_BUFFER) {
        cifs_small_buf_release(rsp_iov.iov_base);
    } else if (resp_buftype == CIFS_LARGE_BUFFER) {
        cifs_large_buf_release(rsp_iov.iov_base);
    }

    kfree(utf16_path);
    return 0;

out:
    kfree(utf16_path);
    if (req)
        cifs_smb2_free(req);
    if (resp_buftype == CIFS_SMALL_BUFFER)
        cifs_small_buf_release(rsp_iov.iov_base);
    else if (resp_buftype == CIFS_LARGE_BUFFER)
        cifs_large_buf_release(rsp_iov.iov_base);
    return rc;
}
