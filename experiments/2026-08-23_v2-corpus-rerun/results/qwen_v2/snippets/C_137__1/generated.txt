void handle_syscall_exit(void *data, struct pt_regs *regs, long ret) {
    struct trace_array *tr = data;
    int syscall_nr = (int) regs->ax;
    struct ftrace_event_file *file;
    struct ring_buffer *buffer;
    struct ring_buffer_event *event;
    struct syscall_exit_event *entry;

    if (syscall_nr < 0 || syscall_nr >= __NR_syscall_max)
        return;

    file = tr->events[syscall_exit_event_id];
    if (!file || !file->enabled)
        return;

    buffer = tr->buffer;
    event = ring_buffer_lock_reserve(buffer, sizeof(*entry));
    if (!event)
        return;

    entry = ring_buffer_event_data(event);
    entry->syscall_nr = syscall_nr;
    entry->ret = ret;
    ring_buffer_unlock_commit(buffer, event);
}
