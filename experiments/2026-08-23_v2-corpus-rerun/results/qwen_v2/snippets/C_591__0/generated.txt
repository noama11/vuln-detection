#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <errno.h>

// Assuming RSFilter is a user-defined type
typedef struct RSFilter RSFilter;

// Assuming this is a helper function that generates the DOT representation
char* generate_dot_representation(RSFilter *filter);

void visualize_filter_graph(RSFilter *filter) {
    if (!filter) {
        fprintf(stderr, "Warning: Invalid filter provided.\n");
        return;
    }

    char *dot_content = generate_dot_representation(filter);
    if (!dot_content) {
        fprintf(stderr, "Warning: Failed to generate DOT content.\n");
        return;
    }

    char dot_filename[] = "/tmp/filter_graph_XXXXXX.dot";
    int fd = mkstemp(dot_filename);
    if (fd == -1) {
        perror("Warning: Failed to create temporary DOT file");
        free(dot_content);
        return;
    }

    FILE *dot_file = fdopen(fd, "w");
    if (!dot_file) {
        perror("Warning: Failed to open temporary DOT file");
        close(fd);
        free(dot_content);
        return;
    }

    if (fprintf(dot_file, "%s", dot_content) < 0) {
        perror("Warning: Failed to write to DOT file");
    }

    fclose(dot_file);
    free(dot_content);

    char png_filename[] = "/tmp/filter_graph_XXXXXX.png";
    if (mktemp(png_filename) == NULL) {
        perror("Warning: Failed to generate temporary PNG filename");
        unlink(dot_filename);
        return;
    }

    char command[256];
    snprintf(command, sizeof(command), "dot -Tpng %s -o %s", dot_filename, png_filename);
    int result = system(command);

    if (result != 0) {
        fprintf(stderr, "Warning: Failed to convert DOT to PNG.\n");
        unlink(dot_filename);
        return;
    }

    unlink(dot_filename);

    snprintf(command, sizeof(command), "gnome-open %s", png_filename);
    result = system(command);

    if (result != 0) {
        fprintf(stderr, "Warning: Failed to open PNG file.\n");
    }
}
