krb5_error_code check_kdc_public_key_identifier(krb5_context context,
                                                pkinit_plg_crypto_context plg_cryptoctx,
                                                pkinit_req_crypto_context req_cryptoctx,
                                                pkinit_identity_crypto_context id_cryptoctx,
                                                unsigned char *pdid_buf,
                                                unsigned int pdid_len,
                                                int *valid_kdcPkId) {
    krb5_error_code retval = 0;
    krb5_data pdid_data;
    pkinit_pkeyid pkeyid = NULL;
    krb5_principal kdc_princ = NULL;
    krb5_principal client_princ = NULL;
    krb5_principal server_princ = NULL;
    krb5_principal princ = NULL;
    krb5_boolean match = FALSE;
    krb5_principal issuer = NULL;
    krb5_data serial_number = {0, 0};
    krb5_principal kdc_issuer = NULL;
    krb5_data kdc_serial = {0, 0};

    *valid_kdcPkId = 0;

    pdid_data.data = (char *)pdid_buf;
    pdid_data.length = pdid_len;

    if ((retval = pkinit_decode_pkeyid(context, pdid_data, &pkeyid))) {
        goto cleanup;
    }

    if ((retval = krb5_unparse_name(context, plg_cryptoctx->kdc_princ, &princ))) {
        goto cleanup;
    }

    if ((retval = krb5_unparse_name(context, pkeyid->subject, &issuer))) {
        goto cleanup;
    }

    if ((retval = krb5_unparse_name(context, plg_cryptoctx->kdc_princ, &kdc_issuer))) {
        goto cleanup;
    }

    if (strcmp(issuer, kdc_issuer) != 0) {
        goto cleanup;
    }

    serial_number = pkeyid->serial_number;
    kdc_serial = plg_cryptoctx->kdc_serial;

    if (serial_number.length != kdc_serial.length ||
        memcmp(serial_number.data, kdc_serial.data, serial_number.length) != 0) {
        goto cleanup;
    }

    *valid_kdcPkId = 1;

cleanup:
    if (pkeyid) {
        pkinit_free_pkeyid(context, pkeyid);
    }

    if (princ) {
        krb5_free_unparsed_name(context, princ);
    }

    if (issuer) {
        krb5_free_unparsed_name(context, issuer);
    }

    if (kdc_issuer) {
        krb5_free_unparsed_name(context, kdc_issuer);
    }

    if (retval) {
        *valid_kdcPkId = 0;
        return KRB5KDC_ERR_PREAUTH_FAILED;
    }

    return 0;
}
