void dtls_clear_queues(SSL *s) {
    DTLS1_RECORD_DATA *rdata;
    CRYPTO_BUFFER *buf;

    /* Clear unprocessed records */
    while ((rdata = sk_DTLS1_RECORD_DATA_pop(s->d1->unprocessed_rcds)) != NULL) {
        if (rdata->buf != NULL) {
            OPENSSL_free(rdata->buf);
        }
        OPENSSL_free(rdata);
    }

    /* Clear processed records */
    while ((rdata = sk_DTLS1_RECORD_DATA_pop(s->d1->processed_rcds)) != NULL) {
        if (rdata->buf != NULL) {
            OPENSSL_free(rdata->buf);
        }
        OPENSSL_free(rdata);
    }

    /* Clear buffered messages */
    while ((buf = sk_CRYPTO_BUFFER_pop(s->d1->buffered_messages)) != NULL) {
        CRYPTO_BUFFER_free(buf);
    }

    /* Clear sent messages */
    while ((buf = sk_CRYPTO_BUFFER_pop(s->d1->sent_messages)) != NULL) {
        CRYPTO_BUFFER_free(buf);
    }

    /* Clear buffered application data */
    while ((buf = sk_CRYPTO_BUFFER_pop(s->d1->buffered_app_data)) != NULL) {
        CRYPTO_BUFFER_free(buf);
    }
}
