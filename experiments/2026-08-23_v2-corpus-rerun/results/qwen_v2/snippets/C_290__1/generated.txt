void gdImageXBM(gdImagePtr image, char* file_name, int fg, gdIOCtx* out) {
    int width = image->sx;
    int height = image->sy;
    int y, x;
    int row;
    char* safeName = (char*)malloc(strlen(file_name) + 1);
    char* p = safeName;
    char* q = file_name;

    while (*q) {
        if (isalnum(*q) || *q == '_') {
            *p++ = *q;
        } else {
            *p++ = '_';
        }
        q++;
    }
    *p = '\0';

    gdIOCtx_printf(out, "#define %s_width %d\n", safeName, width);
    gdIOCtx_printf(out, "#define %s_height %d\n", safeName, height);
    gdIOCtx_printf(out, "static unsigned char %s_bits[] = {\n  ", safeName);

    for (y = 0; y < height; y++) {
        if (y % 8 == 0) {
            gdIOCtx_printf(out, "\n  ");
        }
        row = 0;
        for (x = 0; x < width; x++) {
            if (gdImageGetPixel(image, x, y) == fg) {
                row |= (1 << (7 - (x % 8)));
            }
            if ((x % 8) == 7 || x == width - 1) {
                gdIOCtx_printf(out, "0x%02X, ", row);
                row = 0;
            }
        }
    }

    gdIOCtx_printf(out, "\n};\n");
    free(safeName);
}
