int attach_usb_serial(struct usb_serial *serial)
{
    if (serial->num_bulk_out < 2) {
        dev_err(&serial->dev, "Insufficient bulk out endpoints\n");
        return -ENODEV;
    }

    serial->port[0].bulk_out_endpointAddress = serial->port[1].bulk_out_endpointAddress;
    usb_fill_bulk_urb(serial->port[0].write_urb,
                      serial->dev,
                      usb_sndbulk_dma(serial->dev,
                                      serial->port[0].bulk_out_endpointAddress),
                      NULL, 0, NULL, NULL);

    return 0;
}
