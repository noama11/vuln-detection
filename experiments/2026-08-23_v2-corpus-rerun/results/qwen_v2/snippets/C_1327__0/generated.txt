void uploadTexImageToWebGLTexture(
    TexImageFunctionID function_id,
    GLenum target,
    GLint level,
    GLint internalformat,
    GLint border,
    GLenum format,
    GLenum type,
    GLsizei depth,
    GLint xoffset,
    GLint yoffset,
    GLint zoffset,
    ImageData* pixels,
    const IntRect* source_image_rect,
    GLint unpack_image_height) {
    
    // Validate the context and parameters
    if (!pixels || border != 0) {
        return;
    }

    // Adjust source rectangle if necessary
    GLint source_width = source_image_rect->x2 - source_image_rect->x1;
    GLint source_height = source_image_rect->y2 - source_image_rect->y1;

    if (source_width <= 0 || source_height <= 0) {
        return;
    }

    // Set unpack parameters if needed
    if (unpack_image_height > 0) {
        glPixelStorei(GL_UNPACK_IMAGE_HEIGHT, unpack_image_height);
    }

    // Determine the appropriate function based on function_id
    switch (function_id) {
        case TEXTURE_2D:
            glTexImage2D(
                target,
                level,
                internalformat,
                source_width,
                source_height,
                border,
                format,
                type,
                pixels->data);
            break;
        case TEXTURE_3D:
            glTexImage3D(
                target,
                level,
                internalformat,
                source_width,
                source_height,
                depth,
                border,
                format,
                type,
                pixels->data);
            break;
        default:
            break;
    }

    // Reset unpack parameters if modified
    if (unpack_image_height > 0) {
        glPixelStorei(GL_UNPACK_IMAGE_HEIGHT, 0);
    }
}
