void process_hid_input(struct hid_device *hid, struct hid_field *field, __u8 *data, int interrupt)
{
    int i, j;
    int *values;
    int rc;

    values = kmalloc(field->report_count * sizeof(int), GFP_KERNEL);
    if (!values)
        return;

    for (i = 0; i < field->report_count; i++) {
        rc = hid_input_get_value(field, i, values + i);
        if (rc < 0) {
            kfree(values);
            return;
        }

        if (values[i] < field->logical_minimum || values[i] > field->logical_maximum) {
            hid_process_event(hid, field, values[i], interrupt);
        }
    }

    for (j = 0; j < field->report_count; j++) {
        field->value[j] = values[j];
    }

    kfree(values);
}
