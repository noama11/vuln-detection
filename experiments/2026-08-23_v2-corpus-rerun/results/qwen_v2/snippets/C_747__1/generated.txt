void format_connection_address(const conn *c, const int af, char *addr, struct sockaddr *sock_addr) {
    char ip_str[INET6_ADDRSTRLEN] = "";
    int port = 0;

    if (af == AF_INET) {
        struct sockaddr_in *s = (struct sockaddr_in *)sock_addr;
        inet_ntop(AF_INET, &(s->sin_addr), ip_str, INET6_ADDRSTRLEN);
        port = ntohs(s->sin_port);
    } else if (af == AF_INET6) {
        struct sockaddr_in6 *s = (struct sockaddr_in6 *)sock_addr;
        inet_ntop(AF_INET6, &(s->sin6_addr), ip_str, INET6_ADDRSTRLEN);
        port = ntohs(s->sin6_port);
    } else if (af == AF_UNIX) {
        struct sockaddr_un *s = (struct sockaddr_un *)sock_addr;
        strncpy(ip_str, s->sun_path, sizeof(ip_str) - 1);
        ip_str[sizeof(ip_str) - 1] = '\0';
    }

    if (ip_str[0] == '\0') {
        strncpy(ip_str, "[unknown]", sizeof(ip_str) - 1);
    }

    if (port != 0) {
        snprintf(addr, MAX_ADDRSTRLEN, "%s://%s:%d", c->transport, ip_str, port);
    } else {
        snprintf(addr, MAX_ADDRSTRLEN, "%s://%s", c->transport, ip_str);
    }
}
