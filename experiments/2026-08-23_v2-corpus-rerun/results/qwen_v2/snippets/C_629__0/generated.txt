gboolean pango_emoji_iter_next_emoji(PangoEmojiIter *iter) {
  if (iter->index >= iter->text_len)
    return FALSE;

  gunichar c = g_utf8_get_char(iter->text + iter->index);
  int char_len = g_utf8_next_char(iter->text + iter->index) - (iter->text + iter->index);

  iter->char_len = char_len;
  iter->index += char_len;

  gboolean is_emoji = FALSE;

  if (pango_is_emoji(c)) {
    iter->emoji = c;
    is_emoji = TRUE;
  } else if (c == 0x200D && iter->prev_is_emoji) {
    iter->emoji = c;
    is_emoji = TRUE;
  } else if ((c == 0xFE0F || c == 0xFE0E) && iter->prev_is_emoji) {
    iter->emoji = c;
    is_emoji = TRUE;
  }

  iter->prev_is_emoji = is_emoji;

  return is_emoji;
}
