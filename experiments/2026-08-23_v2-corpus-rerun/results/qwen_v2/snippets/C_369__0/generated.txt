Sdb* process_version_definitions(ELFOBJ* bin, Elf_(Shdr)* shdr, int sz) {
    if (!bin || !shdr || sz <= 0) {
        return NULL;
    }

    void* data = elf_getdata(bin, shdr);
    if (!data) {
        return NULL;
    }

    Sdb* sdb = sdb_new0();
    if (!sdb) {
        return NULL;
    }

    const char* strtab = elf_strptr(bin, shdr->sh_link, 0);
    if (!strtab) {
        sdb_free(sdb);
        return NULL;
    }

    const Elf_(Half)* p = data;
    while ((char*)p < (char*)data + sz) {
        Elf_(Half) version = *p++;
        if (version == 0) {
            break;
        }

        Elf_(Word) ndx = *p++;
        Elf_(Word) len = *p++;
        const char* name = &strtab[*p++];
        char* key = sdb_ksprintf("version.%s", name);
        if (!sdb_set(sdb, key, "ndx", ndx)) {
            free(key);
            sdb_free(sdb);
            return NULL;
        }
        free(key);

        while ((char*)p < (char*)data + sz && *p != 0) {
            Elf_(Half) aux_version = *p++;
            Elf_(Word) aux_ndx = *p++;
            Elf_(Word) aux_len = *p++;
            const char* aux_name = &strtab[*p++];
            char* aux_key = sdb_ksprintf("version.%s.aux", name);
            if (!sdb_set(sdb, aux_key, "ndx", aux_ndx)) {
                free(aux_key);
                sdb_free(sdb);
                return NULL;
            }
            free(aux_key);
        }
    }

    return sdb;
}
