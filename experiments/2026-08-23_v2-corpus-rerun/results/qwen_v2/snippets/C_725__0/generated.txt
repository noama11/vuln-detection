TEE_Result verify_asymmetric_signature(unsigned long state,
                                       const struct utee_attribute *usr_params,
                                       size_t num_params,
                                       const void *data,
                                       size_t data_len,
                                       const void *sig,
                                       size_t sig_len) {
    TEE_Result res = TEE_ERROR_GENERIC;
    void *session = NULL;
    struct utee_attribute *attrs = NULL;

    /* Retrieve the current session */
    session = get_current_session(state);
    if (!session)
        return TEE_ERROR_BAD_STATE;

    /* Check access rights for data and signature */
    if (!check_access_rights(session, data, data_len, sig, sig_len))
        return TEE_ERROR_ACCESS_DENIED;

    /* Allocate memory for attributes */
    attrs = calloc(num_params, sizeof(struct utee_attribute));
    if (!attrs)
        return TEE_ERROR_OUT_OF_MEMORY;

    memcpy(attrs, usr_params, num_params * sizeof(struct utee_attribute));

    /* Verify key initialization */
    if (!is_key_initialized(attrs))
        goto error_cleanup;

    /* Perform algorithm-specific verification */
    switch (attrs->algorithm) {
        case TEE_ALG_RSA_VERIFY:
            res = verify_rsa_signature(attrs, data, data_len, sig, sig_len);
            break;
        case TEE_ALG_DSA_VERIFY:
            res = verify_dsa_signature(attrs, data, data_len, sig, sig_len);
            break;
        case TEE_ALG_ECDSA_VERIFY:
            res = verify_ecdsa_signature(attrs, data, data_len, sig, sig_len);
            break;
        default:
            res = TEE_ERROR_NOT_SUPPORTED;
            break;
    }

error_cleanup:
    free(attrs);
    return res;
}
