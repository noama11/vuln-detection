int snd_timer_user_select(struct file *file, struct snd_timer_select __user *_tselect)
{
    struct snd_timer_user *tun = file->private_data;
    struct snd_timer *timer;
    struct snd_timer_select tselect;
    int err;

    if (tun->timer)
        snd_timer_close(tun->timer);

    if (copy_from_user(&tselect, _tselect, sizeof(tselect)))
        return -EFAULT;

    timer = snd_timer_new(tselect.card, tselect.id, &tselect, NULL);
    if (!timer)
        return -ENOMEM;

    tun->timer = timer;
    timer->private_data = tun;

    if (tselect.flags & SNDRV_TIMER_SELECT_FAST)
        timer->flags |= SNDRV_TIMER_FLG_FAST;

    timer->callback = snd_timer_user_callback;
    timer->private_free = snd_timer_user_free;

    return 0;
}
