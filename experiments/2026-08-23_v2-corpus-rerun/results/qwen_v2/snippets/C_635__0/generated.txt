int process_unique_identifier(char *line, void *data) {
    struct context *ctx = (struct context *)data;
    int index = 0;
    char *endptr = NULL;

    if (line == NULL || ctx == NULL) {
        return -1;
    }

    // Convert line to integer index
    index = strtol(line, &endptr, 10);
    if (endptr == line || *endptr != '\0') {
        return -1;
    }

    // Trim whitespace from line
    while (isspace((unsigned char)*line)) {
        line++;
    }

    // Check for valid unique identifier
    if (strlen(line) == 0) {
        return -1;
    }

    // Search for identifier in headers
    int i;
    int found = 0;
    for (i = 0; i < ctx->header_count; i++) {
        if (strcmp(ctx->headers[i].identifier, line) == 0) {
            found = 1;
            if (ctx->headers[i].index != index) {
                ctx->clear_cache = 1;
            }
            ctx->headers[i].reference_number = index;
            ctx->headers[i].index = i;
            break;
        }
    }

    // If identifier not found, add new header
    if (!found) {
        if (ctx->header_count >= ctx->header_capacity) {
            // Resize headers array if needed
            ctx->header_capacity += 10;
            struct header *new_headers = realloc(ctx->headers, ctx->header_capacity * sizeof(struct header));
            if (new_headers == NULL) {
                return -1;
            }
            ctx->headers = new_headers;
        }
        struct header *new_header = &ctx->headers[ctx->header_count];
        new_header->identifier = strdup(line);
        if (new_header->identifier == NULL) {
            return -1;
        }
        new_header->reference_number = index;
        new_header->index = ctx->header_count;
        ctx->header_count++;
    }

    return 0;
}
