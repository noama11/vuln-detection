size_t RegisterMPCImage(void) {
  MagickInfo
    *magick_info;

  magick_info = AcquireMagickInfo("MPC", "Magick Persistent Cache", "Register MPC image format");
  magick_info->decoder = (MagickDecoder) ReadMPCImage;
  magick_info->encoder = (MagickEncoder) WriteMPCImage;
  magick_info->seekable = MagickTrue;
  magick_info->stealth = MagickTrue;
  return (RegisterMagickInfo(magick_info));
}
