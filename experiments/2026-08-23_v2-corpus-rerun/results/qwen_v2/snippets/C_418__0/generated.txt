struct bio *bio_map_user_iov(struct request_queue *q, const struct iov_iter *iter, gfp_t gfp_mask)
{
    struct bio *bio;
    struct page **pages;
    size_t nr_pages;
    ssize_t ret;

    ret = iov_iter_get_pages_alloc(iter, &pages, iov_iter_count(iter), BIO_MAX_PAGES, &nr_pages);
    if (ret < 0)
        return ERR_PTR(ret);

    bio = bio_alloc(gfp_mask, nr_pages);
    if (!bio) {
        kfree(pages);
        return ERR_PTR(-ENOMEM);
    }

    bio->bi_iter.bi_size = ret;
    bio_set_pages(bio, pages, nr_pages);
    bio->bi_flags |= BIO_USER_MAPPED;

    return bio;
}
