int ceph_auth_client_process_ticket(struct ceph_auth_client *ac,
                                    struct ceph_crypto_key *secret,
                                    void **p, void *end)
{
    int ret = -EINVAL;
    struct ceph_auth_ticket_handler *handler = NULL;
    void *ticket_data = NULL;
    size_t ticket_len = 0;
    struct ceph_auth_ticket ticket;

    // Decode ticket type and version
    if ((end - *p) < sizeof(ticket)) {
        goto out;
    }
    ceph_decode_copy(&ticket, p, sizeof(ticket), end);
    ceph_decode_advance(p, sizeof(ticket));

    handler = ceph_get_ticket_handler(ac, ticket.type);
    if (!handler) {
        goto out;
    }

    // Decrypt ticket data
    if (ticket.encrypted) {
        ret = ceph_decrypt_ticket(&ticket_data, &ticket_len, secret, *p, end);
    } else {
        ticket_data = *p;
        ticket_len = end - *p;
    }

    if (ret < 0) {
        goto out;
    }

    // Update session key and validity
    ret = ceph_update_session_key(ac, ticket_data, ticket_len);
    if (ret < 0) {
        goto out;
    }

    ret = ceph_update_validity(ac, &ticket);
    if (ret < 0) {
        goto out;
    }

    // Update ticket blob and validity in handler
    handler->ticket_blob = ticket_data;
    handler->valid_until = ticket.valid_until;

    ret = 0;

out:
    if (ret < 0) {
        ceph_free(ticket_data);
    }
    return ret;
}
