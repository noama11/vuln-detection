void generate_download_path(uint32_t download_id, const content_DownloadTargetCallback* callback, const base_FilePath* suggested_path) {
    // Ensure this function is called on the UI thread
    #ifdef DCHECK_CURRENTLY_ON
    DCHECK_CURRENTLY_ON(BrowserThread::UI);
    #endif

    // Create a copy of the suggested path
    base_FilePath path = *suggested_path;

    // Append ".crdownload" to the file name to indicate the download is in progress
    base_FilePath file_name = base_FilePath_GetFileName(&path);
    base_FilePath new_file_name = base_FilePath_Append(&file_name, base_FilePath_FromString(".crdownload"));
    base_FilePath_SetFileName(&path, &new_file_name);

    // Prepare the callback parameters
    content_DownloadTargetCallback_Params params;
    params.path = path;
    params.disposition = TARGET_DISPOSITION_OVERWRITE;
    params.danger_type = DANGEROUS_FILE_TYPE_NONE;

    // Invoke the callback with the generated path and parameters
    content_DownloadTargetCallback_Invoke(callback, &params);
}
