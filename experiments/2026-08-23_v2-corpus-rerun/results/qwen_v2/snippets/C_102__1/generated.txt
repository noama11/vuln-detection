int remount_shmem(struct super_block *sb, int *flags, char *data) {
    struct shmem_config config;
    int ret = parse_shmem_options(data, &config);
    if (ret < 0)
        return ret;

    spin_lock(&sb->s_lock);
    if (sb->s_blocks_used > config.max_blocks || sb->s_inodes_used > config.max_inodes) {
        spin_unlock(&sb->s_lock);
        return -EINVAL;
    }

    sb->s_max_blocks = config.max_blocks;
    sb->s_max_inodes = config.max_inodes;
    sb->s_free_inodes = config.max_inodes - sb->s_inodes_used;

    if (config.policy)
        update_memory_policy(sb, config.policy);

    spin_unlock(&sb->s_lock);
    return 0;
}
