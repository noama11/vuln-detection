size_t RegisterJPEGImage(void) {
  MagickInfo *magick_info;
  size_t length;

  static const char *jpeg_version = "6b";
  static const char *description = "JPEG image format";

  magick_info = (MagickInfo *) AcquireMagickMemory(sizeof(*magick_info));
  if (magick_info == (MagickInfo *) NULL)
    return 0;

  (void) memset(magick_info, 0, sizeof(*magick_info));
  magick_info->name = ConstantString("JPEG");
  magick_info->description = ConstantString(description);
  magick_info->decoder = (MagickDecoderHandler) JPEGReadImage;
  magick_info->encoder = (MagickEncoderHandler) JPEGWriteImage;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = ConstantString("image/jpeg");

  if (strcmp(jpeg_version, "6b") == 0)
    magick_info->thread_support = MagickTrue;
  else
    magick_info->thread_support = MagickFalse;

  (void) RegisterMagickInfo("JPE", magick_info);
  (void) RegisterMagickInfo("JPEG", magick_info);
  (void) RegisterMagickInfo("JPG", magick_info);
  (void) RegisterMagickInfo("JPS", magick_info);
  (void) RegisterMagickInfo("PJPEG", magick_info);

  length = strlen(magick_info->name) + 1;
  magick_info->signature = MagickSignature;
  magick_info->next = (MagickInfo *) NULL;

  return length;
}
