void initialize_plugin_on_io_thread(const char* mime_type) {
    // Retrieve plugin information for the specified MIME type
    PluginInfo* plugin_info = get_plugin_info_by_mime(mime_type);
    if (!plugin_info) {
        signal_event("no_plugin_found");
        return;
    }

    const char* plugin_path = get_plugin_path(plugin_info);
    if (!plugin_path) {
        signal_event("plugin_path_not_found");
        return;
    }

    if (is_pepper_plugin(plugin_path)) {
        open_pepper_communication_channel(plugin_path);
    } else if (is_npapi_plugin(plugin_path)) {
        open_npapi_communication_channel(plugin_path);
    } else {
        signal_event("unsupported_plugin_type");
    }

    // Ensure the function runs on the IO thread and manages reference count
    manage_reference_count(plugin_info, plugin_path);
}
