void reset_signal_handlers(struct task_struct *t, int force_default) {
    int sig;

    for (sig = 1; sig < _NSIG; sig++) {
        if (force_default || !sigismember(&t->blocked, sig)) {
            t->sighand->action[sig].sa_handler = SIG_DFL;
            sigemptyset(&t->sighand->action[sig].sa_mask);
            t->sighand->action[sig].sa_flags = 0;
        }
    }
}
