int update_or_add_device(const struct header *headers, time_t t) {
    struct device *current = device_list;

    while (current != NULL) {
        if (headers_match(current->headers, headers)) {
            // Update existing device
            current->timestamp = t;

            // Reallocate memory if needed for the new location header
            if (headers->location != NULL && strlen(headers->location) > strlen(current->headers->location)) {
                char *new_location = strdup(headers->location);
                if (new_location == NULL) {
                    return -1;
                }
                free(current->headers->location);
                current->headers->location = new_location;
            }

            return 0;
        }
        current = current->next;
    }

    // No matching device found, create a new one
    struct device *new_device = malloc(sizeof(struct device));
    if (new_device == NULL) {
        return -1;
    }

    new_device->headers = duplicate_headers(headers);
    if (new_device->headers == NULL) {
        free(new_device);
        return -1;
    }

    new_device->timestamp = t;
    new_device->next = device_list;
    device_list = new_device;

    send_notification(new_device);

    return 1;
}
