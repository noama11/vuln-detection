int iwgif_read_and_process_image(struct iwgifrcontext *rctx) {
    if (!rctx) {
        return 0;
    }

    // Read image header
    uint8_t header[6];
    if (fread(header, 1, 6, rctx->file) != 6) {
        return 0;
    }

    rctx->width = (header[0] << 8) | header[1];
    rctx->height = (header[2] << 8) | header[3];
    uint8_t packed_fields = header[4];
    rctx->interlaced = (packed_fields & 0x40) != 0;

    // Validate image dimensions
    if (rctx->width <= 0 || rctx->height <= 0) {
        return 0;
    }

    // Initialize color tables if present
    if (packed_fields & 0x80) {
        int color_table_size = 1 << ((packed_fields & 0x07) + 1);
        rctx->local_color_table = malloc(color_table_size * 3);
        if (!rctx->local_color_table) {
            return 0;
        }

        if (fread(rctx->local_color_table, 3, color_table_size, rctx->file) != color_table_size) {
            free(rctx->local_color_table);
            rctx->local_color_table = NULL;
            return 0;
        }

        // Handle transparency
        rctx->transparent_index = -1;
        if (rctx->transparent_flag) {
            rctx->transparent_index = rctx->transparent_color_index;
        }
    }

    // Read LZW code size
    uint8_t lzw_code_size;
    if (fread(&lzw_code_size, 1, 1, rctx->file) != 1) {
        return 0;
    }

    if (lzw_code_size < 1 || lzw_code_size > 8) {
        return 0;
    }

    // Initialize screen context
    rctx->current_line = 0;
    rctx->current_pixel = 0;

    // Read pixel data in subblocks
    uint8_t subblock_size;
    while (fread(&subblock_size, 1, 1, rctx->file) == 1) {
        if (subblock_size == 0) {
            break;
        }

        uint8_t *subblock_data = malloc(subblock_size);
        if (!subblock_data) {
            return 0;
        }

        if (fread(subblock_data, 1, subblock_size, rctx->file) != subblock_size) {
            free(subblock_data);
            return 0;
        }

        // Process subblock with LZW decoding
        if (!iwgif_lzw_decode(rctx, subblock_data, subblock_size)) {
            free(subblock_data);
            return 0;
        }

        free(subblock_data);
    }

    return 1;
}
