int ext4_orphan_del(handle_t *handle, struct inode *inode) {
    struct super_block *sb = inode->i_sb;
    struct ext4_super_block *es = EXT4_SB(sb);
    struct ext4_inode_info *ei = EXT4_I(inode);
    struct ext4_iloc iloc;
    int err = 0;

    if (!EXT4_HAS_JOURNAL(sb))
        return 0;

    mutex_lock(&EXT4_SB(sb)->s_orphan_lock);

    if (!ei->i_dtime) {
        err = -ENOENT;
        goto out;
    }

    if (handle) {
        err = ext4_journal_get_write_access(handle, sb, ei->i_bh, 0);
        if (err)
            goto out;
    }

    if (EXT4_SB(sb)->s_orphan == inode) {
        EXT4_SB(sb)->s_orphan = NULL;
        if (handle) {
            err = ext4_handle_dirty_metadata(handle, sb, ei->i_bh);
            if (err)
                goto out;
        }
    } else {
        struct inode *prev_inode = NULL;
        struct inode *tmp_inode = EXT4_SB(sb)->s_orphan;

        while (tmp_inode) {
            if (tmp_inode->i_private == inode) {
                tmp_inode->i_private = inode->i_private;
                if (handle) {
                    err = ext4_handle_dirty_metadata(handle, sb, ei->i_bh);
                    if (err)
                        goto out;
                }
                break;
            }
            prev_inode = tmp_inode;
            tmp_inode = tmp_inode->i_private;
        }
    }

    ei->i_dtime = 0;
    mark_inode_dirty(inode);

out:
    mutex_unlock(&EXT4_SB(sb)->s_orphan_lock);
    return err;
}
