bool resolve_lhs(struct xkb_context *ctx, const ExprDef *expr,
                 const char **elem_rtrn, const char **field_rtrn,
                 ExprDef **index_rtrn) {
    switch (expr->type) {
        case EXPR_IDENT:
            *elem_rtrn = expr->ident.name;
            *field_rtrn = NULL;
            *index_rtrn = NULL;
            return true;

        case EXPR_FIELD:
            *elem_rtrn = expr->field.base->ident.name;
            *field_rtrn = expr->field.name;
            *index_rtrn = NULL;
            return true;

        case EXPR_ARRAY:
            *elem_rtrn = expr->array.base->ident.name;
            *field_rtrn = NULL;
            *index_rtrn = expr->array.index;
            return true;

        default:
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR,
                            "Unexpected operator in LHS resolution");
            return false;
    }
}
