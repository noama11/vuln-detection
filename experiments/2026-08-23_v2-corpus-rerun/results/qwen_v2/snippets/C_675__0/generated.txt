int hwsim_new_radio(struct sk_buff *msg, struct genl_info *info)
{
    struct hwsim_new_radio_params params = { 0 };
    int ret = 0;
    char *custom_name = NULL;

    if (info->attrs[HWSIM_ATTR_REGULATORY_STRICTNESS]) {
        params.regulatory_strictness = nla_get_u32(info->attrs[HWSIM_ATTR_REGULATORY_STRICTNESS]);
    }

    if (info->attrs[HWSIM_ATTR_SUPPORT_P2P_DEVICE]) {
        params.support_p2p_device = true;
    }

    if (info->attrs[HWSIM_ATTR_CHANNELS]) {
        params.channels = nla_get_u32(info->attrs[HWSIM_ATTR_CHANNELS]);
    }

    if (info->attrs[HWSIM_ATTR_RADIO_NAME]) {
        custom_name = nla_data(info->attrs[HWSIM_ATTR_RADIO_NAME]);
        params.name = kstrdup(custom_name, GFP_KERNEL);
        if (!params.name) {
            ret = -ENOMEM;
            goto out;
        }
    }

    if (info->attrs[HWSIM_ATTR_CUSTOM_REGULATORY]) {
        params.custom_regulatory = nla_get_u8(info->attrs[HWSIM_ATTR_CUSTOM_REGULATORY]);
        if (params.custom_regulatory >= MAX_CUSTOM_REGULATORY_DOMAINS) {
            ret = -EINVAL;
            goto out;
        }
    }

    ret = mac80211_hwsim_new_radio(&params);
    if (ret < 0) {
        goto out;
    }

out:
    if (params.name && params.name != custom_name) {
        kfree(params.name);
    }
    return ret;
}
