bool check_new_id_mapping(const struct file *file, struct user_namespace *ns,
                          int cap_setid, struct uid_gid_map *new_map) {
    struct user_namespace *old_ns;
    kuid_t kuid;

    if (new_map->nr_extents == 1 && new_map->extent[0].lower == new_map->extent[0].upper &&
        new_map->extent[0].count == 1) {
        if (cap_setid == CAP_SETUID) {
            kuid = current_fsuid();
        } else {
            kgid_t kgid = current_fsgid();
            return kgid_eq(kgid, new_map->extent[0].lower);
        }
        return kuid_eq(kuid, new_map->extent[0].lower);
    }

    if (ns == &init_user_ns)
        return false;

    old_ns = get_user_ns(ns->parent);
    if (file->f_cred->user_ns == old_ns) {
        put_user_ns(old_ns);
        return ns_capable(old_ns, cap_setid);
    }

    put_user_ns(old_ns);
    return false;
}
