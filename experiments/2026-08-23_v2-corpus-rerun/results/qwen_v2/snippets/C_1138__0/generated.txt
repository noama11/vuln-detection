void ShowOneClickSignInInfoBar(net::URLRequest* request, ProfileIOData* io_data, int child_id, int route_id) {
  const std::string& sign_in_values = request->extra_request_headers().FindHeader("X-Chrome-Identity");
  if (sign_in_values.empty())
    return;

  std::string email;
  std::string session_index;
  bool auto_accept = false;

  // Parse sign-in values.
  std::vector<std::string> parts = base::SplitString(sign_in_values, ";", base::TRIM_WHITESPACE, base::SPLIT_WANT_ALL);
  for (const std::string& part : parts) {
    if (base::StartsWith(part, "email=")) {
      email = part.substr(6);
    } else if (base::StartsWith(part, "session_index=")) {
      session_index = part.substr(14);
    } else if (base::StartsWith(part, "auto_accept=true")) {
      auto_accept = true;
    }
  }

  if (email.empty() && session_index.empty() && !auto_accept)
    return;

  GURL url = request->url();
  if (!url.SchemeIs("https") || !url.DomainIs("accounts.google.com"))
    return;

  base::PostTaskWithTraits(
      FROM_HERE, {base::MayBlock(), base::TaskPriority::BACKGROUND},
      base::BindOnce(&ShowOneClickSignInInfoBarOnUIThread, email, session_index, auto_accept, child_id, route_id));
}
