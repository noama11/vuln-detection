void parse_pdf_object(struct pdf_struct *pdf, struct pdf_obj *obj) {
    size_t start = 0;
    size_t end = 0;
    size_t dict_start = 0;
    size_t dict_end = 0;

    // Check the size of the object and locate the start of its dictionary
    if (obj->size <= 0) {
        return;
    }

    // Find the start of the dictionary
    dict_start = find_dict_start(pdf, obj->data, obj->size);
    if (dict_start == obj->size) {
        return;
    }

    // Find the end of the dictionary
    dict_end = find_dict_end(pdf, obj->data + dict_start, obj->size - dict_start);
    if (dict_end == obj->size) {
        return;
    }

    // Process names in the dictionary and normalize them
    for (size_t i = dict_start; i < dict_start + dict_end; ++i) {
        if (is_pdf_name(pdf, obj->data + i)) {
            normalize_name(pdf, obj->data + i);
        }
    }

    // Handle different states such as linearization and actions
    if (is_linearized(pdf, obj->data + dict_start, dict_end)) {
        obj->flags |= PDF_OBJ_FLAG_LINEARIZED;
    }

    if (is_action(pdf, obj->data + dict_start, dict_end)) {
        obj->flags |= PDF_OBJ_FLAG_ACTION;
    }

    // Check for filters and update the object's status based on the number of filters detected
    int filter_count = count_filters(pdf, obj->data + dict_start, dict_end);
    if (filter_count > 0) {
        obj->flags |= PDF_OBJ_FLAG_FILTERED;
    }

    // Log the object's flags for debugging purposes
    log_flags(obj->flags);
}
