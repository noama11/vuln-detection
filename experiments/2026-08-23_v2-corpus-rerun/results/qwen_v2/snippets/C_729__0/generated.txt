TEE_Result set_mem_parameters(const struct optee_msg_param_rmem *rmem,
                              struct param_mem *mem) {
    struct TEE_SharedMemory *shm = TEE_GetSharedMemoryByRef(rmem->shm_ref);

    if (!shm) {
        return TEE_ERROR_ITEM_NOT_FOUND;
    }

    mem->offset = rmem->offset;
    mem->size = rmem->size;

    if (mem->offset + mem->size > shm->size) {
        return TEE_ERROR_BAD_PARAMETERS;
    }

    return TEE_SUCCESS;
}
