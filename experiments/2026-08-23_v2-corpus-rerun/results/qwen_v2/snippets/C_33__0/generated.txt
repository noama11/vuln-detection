int set_ext_regs(struct task_struct *child, void __user *uregs)
{
    if (!access_ok(uregs, sizeof(struct user_i387_struct)))
        return -EFAULT;

    flush_and_enable_fpu();
    release_fpu();

    if (copy_from_user(&child->thread.i387, uregs, sizeof(struct user_i387_struct))) {
        return -EFAULT;
    }

    return 0;
}
