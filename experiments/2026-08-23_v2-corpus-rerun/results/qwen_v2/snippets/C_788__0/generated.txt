int archive_read_support_zip_cleanup(struct archive_read *a) {
  struct zip *zip = (struct zip *)a->format->data;

  if (zip == NULL)
    return ARCHIVE_OK;

  if (zip->zlib_stream != NULL)
    inflateEnd(zip->zlib_stream);
  if (zip->lzma_stream != NULL)
    lzma_end(zip->lzma_stream);
  if (zip->bz2_stream != NULL)
    BZ2_bzDecompressEnd(zip->bz2_stream);

  struct zip_entry *entry = zip->entries;
  struct zip_entry *next;
  while (entry != NULL) {
    next = entry->next;
    free(entry->filename);
    free(entry->extra);
    free(entry->comment);
    free(entry);
    entry = next;
  }

  free(zip->central_directory);
  free(zip->local_file_header);
  free(zip);

  a->format->data = NULL;

  return ARCHIVE_OK;
}
