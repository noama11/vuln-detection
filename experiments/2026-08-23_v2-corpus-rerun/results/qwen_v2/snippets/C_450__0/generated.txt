struct sock *sctp6_sock_create(struct sock *sk, struct sctp_association *asoc, bool kern) {
    struct sock *new_sk;
    struct sctp6_sock *new_sp;
    struct sctp_sock *sp = sctp_sk(sk);

    new_sk = sk_clone(sk, GFP_KERNEL);
    if (!new_sk)
        return NULL;

    new_sp = (struct sctp6_sock *)new_sk;
    new_sp->asoc = *asoc;
    new_sp->asoc.sk = new_sk;

    new_sp->sctp = *sp;
    new_sp->sctp.sk = new_sk;

    sctp_sock_init(new_sk, kern);

    sctp6_init_sock(new_sk);

    return new_sk;
}
