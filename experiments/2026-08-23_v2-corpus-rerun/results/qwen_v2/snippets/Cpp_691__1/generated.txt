std::map<std::string, std::string> get_extended_memcache_stats(const std::string& type, int slabid, int limit) {
    std::map<std::string, std::string> result;
    std::vector<MemcacheStats> stats = get_memcache_data(type, slabid, limit);

    for (const auto& server_stats : stats) {
        std::string key = server_stats.hostname + ":" + std::to_string(server_stats.port);
        for (const auto& stat : server_stats.stats) {
            result[key + ":" + stat.first] = stat.second;
        }
    }

    free_memcache_stats(stats);
    return result;
}
