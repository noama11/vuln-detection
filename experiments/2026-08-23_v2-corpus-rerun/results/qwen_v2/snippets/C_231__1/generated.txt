size_t process_note_headers(struct magic_set *ms, void *vbuf, size_t offset, size_t size, int clazz, int swap, size_t align, int *flags, uint16_t *notecount) {
    uint16_t note_type;
    size_t note_name_size, note_desc_size, note_total_size;
    uint16_t *remaining_notes = notecount;
    uint8_t *buf = (uint8_t *)vbuf;

    while (*remaining_notes > 0 && offset < size) {
        if (offset + align > size) {
            return size;
        }

        offset = (offset + (align - 1)) & ~(align - 1);

        if (offset + sizeof(note_type) > size) {
            return size;
        }

        memcpy(&note_type, buf + offset, sizeof(note_type));
        if (swap) {
            note_type = (note_type >> 8) | (note_type << 8);
        }

        offset += sizeof(note_type);

        if (offset + sizeof(note_name_size) > size) {
            return size;
        }

        memcpy(&note_name_size, buf + offset, sizeof(note_name_size));
        if (swap) {
            note_name_size = (note_name_size >> 8) | (note_name_size << 8);
        }

        offset += sizeof(note_name_size);

        if (offset + sizeof(note_desc_size) > size) {
            return size;
        }

        memcpy(&note_desc_size, buf + offset, sizeof(note_desc_size));
        if (swap) {
            note_desc_size = (note_desc_size >> 8) | (note_desc_size << 8);
        }

        offset += sizeof(note_desc_size);

        note_total_size = sizeof(note_type) + sizeof(note_name_size) + sizeof(note_desc_size) + note_name_size + note_desc_size;

        if (offset + note_total_size > size) {
            return size;
        }

        if (note_type == NT_NETBSD_CORE) {
            *flags |= NOTE_NETBSD_CORE;
        }

        offset += note_name_size + note_desc_size;
        (*remaining_notes)--;
    }

    return offset;
}
