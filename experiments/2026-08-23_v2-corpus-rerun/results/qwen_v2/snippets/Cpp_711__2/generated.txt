#include <string>
#include <openssl/md5.h>
#include <openssl/rc4.h>

std::string computeUValue(std::string const& user_password, QPDF::EncryptionData const& data) {
    // Step 1: Generate encryption key from password and encryption data
    // (Assuming key generation is handled by a helper function)
    std::string encryption_key = generateEncryptionKey(user_password, data);

    // Step 2: Pad the password
    std::string padded_password = padPassword(user_password, data);

    // Step 3: Compute MD5 digest of padded password and encryption data's ID
    std::string input = padded_password + data.getID();
    unsigned char md5_digest[MD5_DIGEST_LENGTH];
    MD5(reinterpret_cast<const unsigned char*>(input.c_str()), input.size(), md5_digest);

    // Step 4: Apply RC4 iteration using the encryption key
    RC4_KEY rc4_key;
    RC4_set_key(&rc4_key, encryption_key.size(), reinterpret_cast<const unsigned char*>(encryption_key.c_str()));
    unsigned char rc4_output[MD5_DIGEST_LENGTH];
    RC4(&rc4_key, MD5_DIGEST_LENGTH, md5_digest, rc4_output);

    // Step 5: Pad the result with arbitrary data to ensure consistency
    std::string u_value(reinterpret_cast<char*>(rc4_output), MD5_DIGEST_LENGTH);
    u_value = padResult(u_value, data.getPaddingLength());

    // Step 6: Return the U value as a string of a specified length
    return u_value.substr(0, data.getUValueLength());
}
