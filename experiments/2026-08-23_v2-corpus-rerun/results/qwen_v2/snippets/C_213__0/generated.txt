void bpf_jit_compile(struct bpf_prog *prog) {
    if (!prog || !jit_enabled)
        return;

    size_t pass, size_estimate = 0;
    void *jit_context = NULL;
    void *jit_image = NULL;

    for (pass = 0; pass < JIT_PASS_COUNT; pass++) {
        size_estimate = estimate_jit_size(prog, pass);
        if (pass == 0)
            jit_context = allocate_jit_context(prog, size_estimate);
        else {
            void *new_jit_image = allocate_jit_image(size_estimate);
            if (!new_jit_image)
                goto cleanup;

            if (size_estimate != estimate_jit_size(prog, pass - 1)) {
                free(jit_image);
                jit_image = new_jit_image;
            }
        }
    }

    if (!jit_image)
        goto cleanup;

    generate_jit_code(prog, jit_context, jit_image);
    prog->bpf_func = jit_image;
    prog->jited = 1;

cleanup:
    free(jit_context);
    if (jit_image && !prog->jited)
        free(jit_image);
}
