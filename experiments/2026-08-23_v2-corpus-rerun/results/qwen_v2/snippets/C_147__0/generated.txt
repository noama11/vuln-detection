void sctp_assoc_update(struct sctp_association *asoc, struct sctp_association *new) {
    /* Copy new parameters from 'new' to 'asoc' */
    asoc->rwnd = new->rwnd;
    asoc->ssthresh = new->ssthresh;
    asoc->cwnd = new->cwnd;
    asoc->partial_failure_point = new->partial_failure_point;
    asoc->flight_size = new->flight_size;
    asoc->last_out_seq = new->last_out_seq;
    asoc->last_in_seq = new->last_in_seq;
    asoc->last_in_tsn = new->last_in_tsn;
    asoc->last_out_tsn = new->last_out_tsn;
    asoc->next_tsn = new->next_tsn;
    asoc->peer_supports_asconf = new->peer_supports_asconf;
    asoc->peer_supports_auth = new->peer_supports_auth;
    asoc->peer_supports_ecn = new->peer_supports_ecn;
    asoc->peer_supports_nat = new->peer_supports_nat;

    /* Remove peer addresses from 'asoc' that are not in 'new' */
    struct sctp_transport *t, *tmp;
    list_for_each_entry_safe(t, tmp, &asoc->peer.transport_addr_list, transport_list) {
        int found = 0;
        struct sctp_transport *t2;
        list_for_each_entry(t2, &new->peer.transport_addr_list, transport_list) {
            if (sctp_cmp_addr(&t->ip, &t2->ip)) {
                found = 1;
                break;
            }
        }
        if (!found) {
            list_del(&t->transport_list);
            kfree(t);
        }
    }

    /* If association is established, update next TSN and clear SSN map */
    if (asoc->state == SCTP_STATE_ESTABLISHED) {
        asoc->next_tsn = new->next_tsn;
        sctp_ssnmap_clear(&asoc->ssnmap);
        sctp_ulpq_flush(&asoc->ulpq);
        asoc->error_count = 0;
    } else {
        /* Add new peer addresses */
        struct sctp_transport *t2;
        list_for_each_entry(t2, &new->peer.transport_addr_list, transport_list) {
            int found = 0;
            struct sctp_transport *t;
            list_for_each_entry(t, &asoc->peer.transport_addr_list, transport_list) {
                if (sctp_cmp_addr(&t->ip, &t2->ip)) {
                    found = 1;
                    break;
                }
            }
            if (!found) {
                t = kmemdup(t2, sizeof(*t), GFP_KERNEL);
                list_add(&t->transport_list, &asoc->peer.transport_addr_list);
            }
        }

        /* Manage SSN map and association ID */
        if (asoc->state == SCTP_STATE_COOKIE_ECHOED) {
            sctp_ssnmap_init(&asoc->ssnmap, new->ssnmap.max_tsn);
            asoc->assoc_id = new->assoc_id;
        }
    }

    /* Update SCTP-AUTH related parameters */
    if (new->peer_supports_auth) {
        asoc->auth_support = new->auth_support;
        asoc->auth_num_keys = new->auth_num_keys;
        asoc->auth_keys = new->auth_keys;
        asoc->auth_active_key = new->auth_active_key;
        asoc->auth_chunk_list = new->auth_chunk_list;
        asoc->auth_hmac_alg = new->auth_hmac_alg;
    }
}
