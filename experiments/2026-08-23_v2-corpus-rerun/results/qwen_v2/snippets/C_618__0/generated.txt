int get_cdrom_drive_status(struct cdrom_device_info *cdi, unsigned long arg) {
    if (!(cdi->flags & CDF_DRIVE_STATUS)) {
        return -EINVAL;
    }

    if (arg > CDROM_MAX_SLOT) {
        return -EINVAL;
    }

    if (arg == CDROM_CURRENT_SLOT) {
        return cdi->current_slot;
    } else {
        return cdi->slots[arg];
    }
}
