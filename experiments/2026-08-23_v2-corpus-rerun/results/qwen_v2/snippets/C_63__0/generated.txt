char *build_unc_path(const struct smb_vol *vol, const struct cifs_sb_info *cifs_sb)
{
    int unc_len = strlen(vol->unc);
    int prepath_len = cifs_sb->prepath ? strlen(cifs_sb->prepath) : 0;
    char *full_path;
    int total_len = unc_len + prepath_len + 1; // +1 for null terminator

    full_path = kmalloc(total_len, GFP_KERNEL);
    if (!full_path)
        return ERR_PTR(-ENOMEM);

    strcpy(full_path, vol->unc);

    if (prepath_len > 0) {
        strcat(full_path, cifs_sb->prepath);
    }

    // Convert directory separators if needed
    for (int i = 0; i < total_len; i++) {
        if (full_path[i] == '/') {
            full_path[i] = '\\';
        }
    }

    cifs_dbg(VFS, "UNC path constructed: %s\n", full_path);

    return full_path;
}
