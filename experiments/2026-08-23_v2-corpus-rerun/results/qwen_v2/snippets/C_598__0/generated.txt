int read_rm_streams(AVFormatContext *s, AVIOContext *pb, AVStream *st, char *mime) {
    int num_streams;
    int num_mdprs;
    int i;
    int ret;

    num_streams = avio_r8(pb);
    num_mdprs = avio_r8(pb);

    if (num_mdprs > 1) {
        // Request a sample
        avio_skip(pb, 1);
    }

    for (i = 0; i < num_mdprs; i++) {
        AVStream *new_st = avformat_new_stream(s, NULL);
        if (!new_st) {
            return AVERROR(ENOMEM);
        }

        ret = avcodec_parameters_copy(new_st->codecpar, st->codecpar);
        if (ret < 0) {
            return ret;
        }

        new_st->codecpar->codec_tag = avio_rl32(pb);
        avio_skip(pb, 4); // Skip some data

        if (avio_feof(pb)) {
            return AVERROR_EOF;
        }
    }

    return 0;
}
