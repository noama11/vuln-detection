int tipc_nl_compat_dump_link(struct tipc_nl_compat_msg *msg, struct nlattr **attrs)
{
    struct nlattr *link_attr;
    struct nlattr *tb[LINK_ATTR_MAX + 1];
    int err;

    if (!attrs[LINK_ATTR_INFO])
        return -EINVAL;

    link_attr = attrs[LINK_ATTR_INFO];
    err = nla_parse(tb, LINK_ATTR_MAX, nla_data(link_attr), nla_len(link_attr), NULL);
    if (err < 0)
        return err;

    if (!tb[LINK_ATTR_DEST] || !tb[LINK_ATTR_STATUS] || !tb[LINK_ATTR_NAME])
        return -EINVAL;

    msg->type = TIPC_TLV_LINK_INFO;
    msg->value = nla_get_u32(tb[LINK_ATTR_DEST]);
    msg->length = strlen(nla_get_string(tb[LINK_ATTR_NAME])) + 1;
    memcpy(msg->data, nla_get_string(tb[LINK_ATTR_NAME]), msg->length);

    if (nla_get_u8(tb[LINK_ATTR_STATUS]) == LINK_STATUS_UP)
        msg->flags |= TIPC_LINK_UP;

    return 0;
}
