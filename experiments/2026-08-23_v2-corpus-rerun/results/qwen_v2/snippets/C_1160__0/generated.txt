#include "base/process.h"
#include "base/platform_file.h"
#include "ipc/ipc_platform_file.h"

IPC::PlatformFileForTransit DuplicatePlatformFileForProcess(base::PlatformFile handle, base::ProcessHandle process, bool close_source_handle) {
#if defined(OS_WIN)
    HANDLE target_handle = INVALID_HANDLE_VALUE;
    if (DuplicateHandle(GetCurrentProcess(), handle, process, &target_handle, 0, FALSE, close_source_handle ? DUPLICATE_CLOSE_SOURCE : 0)) {
        return IPC::PlatformFileForTransit(target_handle);
    }
#else
    int target_fd = handle;
    if (close_source_handle) {
        int flags = fcntl(handle, F_GETFD);
        if (flags == -1 || fcntl(handle, F_SETFD, flags & ~FD_CLOEXEC) == -1) {
            return IPC::PlatformFileForTransit();
        }
    }
    return IPC::PlatformFileForTransit(target_fd);
#endif

    return IPC::PlatformFileForTransit();
}
