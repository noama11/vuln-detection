#include <linux/slab.h>
#include <linux/workqueue.h>
#include <linux/coresight.h>

void *setup_aux_event_data(int event_cpu, void **pages, int nr_pages, bool overwrite)
{
    struct aux_event_data *event_data;
    struct coresight_device *sink;
    int cpu;

    event_data = kzalloc(sizeof(*event_data), GFP_KERNEL);
    if (!event_data)
        return NULL;

    sink = coresight_get_enabled_sink();
    if (!sink || !sink->alloc_buffer) {
        kfree(event_data);
        return NULL;
    }

    for_each_cpu(cpu, &event_data->cpus) {
        struct aux_event_path *path;

        path = kzalloc(sizeof(*path), GFP_KERNEL);
        if (!path) {
            cleanup_event_data(event_data);
            kfree(event_data);
            return NULL;
        }

        path->cpu = cpu;
        path->pages = pages;
        path->nr_pages = nr_pages;
        path->overwrite = overwrite;

        INIT_WORK(&path->free_work, free_event_path_work);
        list_add(&path->list, &event_data->paths);
    }

    event_data->sink = sink;
    INIT_WORK(&event_data->free_work, free_event_data_work);

    return event_data;
}
