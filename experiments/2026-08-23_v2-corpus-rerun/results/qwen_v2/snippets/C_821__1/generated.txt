int register_vlan(struct mlx4_dev *dev, u8 port, u16 vlan, int *index)
{
	int err = -ENOSPC;
	int i;

	mutex_lock(&dev->port_vlans[port].vlan_table_mutex);

	for (i = 0; i < MLX4_MAX_VLAN_ENTRIES; ++i) {
		struct mlx4_vlan_entry *entry = &dev->port_vlans[port].vlan_table[i];

		if (entry->vlan == vlan) {
			/* VLAN already exists, increment reference count */
			entry->ref_count++;
			*index = i;
			err = 0;
			goto out;
		}
	}

	/* Find an empty slot */
	for (i = 0; i < MLX4_MAX_VLAN_ENTRIES; ++i) {
		struct mlx4_vlan_entry *entry = &dev->port_vlans[port].vlan_table[i];

		if (entry->vlan == 0) {
			entry->vlan = vlan;
			entry->ref_count = 1;
			*index = i;
			err = 0;
			break;
		}
	}

out:
	mutex_unlock(&dev->port_vlans[port].vlan_table_mutex);
	return err;
}
