int subtractBlackLevelAndUpdateMax() {
    if (!isPhaseOneCompressed && blackLevel[0] != 0 || blackLevel[1] != 0 || blackLevel[2] != 0) {
        #define GET_PIXEL(data, row, col, channel) data[(row) * width + (col)][channel]
        #define CLIP(value, min, max) ((value) < (min) ? (min) : ((value) > (max) ? (max) : (value)))

        maxDataValue = 0;

        for (int row = 0; row < height; ++row) {
            for (int col = 0; col < width; ++col) {
                for (int channel = 0; channel < 3; ++channel) {
                    int pixelValue = GET_PIXEL(imageData, row, col, channel) - blackLevel[channel];
                    pixelValue = CLIP(pixelValue, 0, maxValidValue);
                    GET_PIXEL(imageData, row, col, channel) = pixelValue;
                    if (pixelValue > maxDataValue) {
                        maxDataValue = pixelValue;
                    }
                }
            }
        }

        #undef GET_PIXEL
        #undef CLIP
    } else {
        maxDataValue = 0;
        for (int row = 0; row < height; ++row) {
            for (int col = 0; col < width; ++col) {
                for (int channel = 0; channel < 3; ++channel) {
                    int pixelValue = GET_PIXEL(imageData, row, col, channel);
                    if (pixelValue > maxDataValue) {
                        maxDataValue = pixelValue;
                    }
                }
            }
        }
    }

    return 0;
}
