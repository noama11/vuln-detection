xfs_buf_t*
xfs_buf_find(
	struct xfs_buftarg	*btp,
	struct xfs_buf_map	*map,
	int			nmaps,
	xfs_buf_flags_t		flags,
	xfs_buf_t		*new_bp)
{
	struct xfs_buf		*bp;
	struct xfs_buf		*newbuf = NULL;
	struct xfs_buf		*tmp;
	struct xfs_buf		*newest;
	struct xfs_buf		*oldest;
	struct xfs_buf		*next;
	struct xfs_buf		*prev;
	struct xfs_buf		*match = NULL;
	xfs_daddr_t		d;
	int			i;
	int			error = 0;
	int			needverify = 0;

	/*
	 * Check for invalid parameters.
	 */
	if (!btp || !map || nmaps <= 0)
		return NULL;

	/*
	 * Traverse the red-black tree to find a matching buffer.
	 */
	for (i = 0; i < nmaps; i++) {
		d = map[i].bm_bn;
		bp = xfs_buf_find_by_block(btp, d);
		if (bp) {
			match = bp;
			break;
		}
	}

	if (match) {
		/*
		 * Handle stale buffer.
		 */
		if (match->b_flags & XBF_STALE) {
			xfs_buf_stale_clear(match);
			match->b_flags |= XBF_DONE;
		}

		/*
		 * Lock the buffer.
		 */
		xfs_buf_lock(match);
		return match;
	}

	/*
	 * No match found. Insert a new buffer if requested.
	 */
	if (new_bp) {
		newbuf = new_bp;
		xfs_buf_insert(btp, newbuf);
		return newbuf;
	}

	return NULL;
}
