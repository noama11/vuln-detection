int handle_ip4_route_info(struct sk_buff *in_skb, struct nlmsghdr *nlh, struct netlink_ext_ack *extack)
{
    struct rtmsg *rtm = NLMSG_DATA(nlh);
    struct net *net = sock_net(SK_CAST(in_skb->sk));
    struct sk_buff *skb;
    struct nl_info info = {
        .nlhdr = nlh,
        .extack = extack,
    };
    int err;

    if (nlh->nlmsg_len < NLMSG_LENGTH(sizeof(*rtm))) {
        return -EINVAL;
    }

    if (rtm->rtm_dst_len && rtm->rtm_src_len) {
        return -EINVAL;
    }

    skb = nlmsg_new(NLMSG_DEFAULT_SIZE, GFP_KERNEL);
    if (!skb) {
        return -ENOMEM;
    }

    err = rtnl_fill_info(skb, 0, RTM_NEWROUTE, 0, net, rtm, NULL, 0, &info);
    if (err < 0) {
        kfree_skb(skb);
        return err;
    }

    return nlmsg_unicast(net->genl_sock, skb, NETLINK_CB(in_skb).portid);
}
