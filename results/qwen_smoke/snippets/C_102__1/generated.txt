int remount_shmem(struct super_block *sb, int *flags, char *data) {
    struct shmem_sb_info *info = sb->s_fs_info;
    struct shmem_options options;
    int error;

    error = parse_shmem_options(data, &options);
    if (error)
        return error;

    down_write(&sb->s_umount);
    if (info->max_blocks && info->used_blocks > options.max_blocks) {
        up_write(&sb->s_umount);
        return -EBUSY;
    }
    if (info->max_inodes && info->used_inodes > options.max_inodes) {
        up_write(&sb->s_umount);
        return -EBUSY;
    }

    info->max_blocks = options.max_blocks;
    info->max_inodes = options.max_inodes;
    info->free_inodes = options.max_inodes - info->used_inodes;

    if (options.policy)
        set_shmem_policy(sb, options.policy);

    up_write(&sb->s_umount);
    return 0;
}
