int process_diag_message(struct sk_buff *skb, struct nlmsghdr *nlh) {
    int err = -EINVAL;
    struct sock_diag_handler *handler;

    if (nlh->nlmsg_len < NLMSG_HDRLEN)
        return -EINVAL;

    handler = sock_diag_lock_handler(nlh->nlmsg_type);
    if (!handler)
        return -ENOENT;

    err = handler->dump(skb, nlh);
    sock_diag_unlock_handler(handler);

    return err;
}
