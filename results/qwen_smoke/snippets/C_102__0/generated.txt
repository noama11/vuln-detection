int remount_shmem(struct super_block *sb, int *flags, char *data) {
    struct shmem_config new_config;
    int ret = 0;

    if (parse_shmem_options(data, &new_config) < 0)
        return -EINVAL;

    spin_lock(&sb->s_lock);

    if (new_config.max_blocks < sb->s_blocks || new_config.max_inodes < sb->s_inodes) {
        spin_unlock(&sb->s_lock);
        return -EINVAL;
    }

    sb->s_max_blocks = new_config.max_blocks;
    sb->s_max_inodes = new_config.max_inodes;
    sb->s_policy = new_config.policy;

    spin_unlock(&sb->s_lock);

    return ret;
}
