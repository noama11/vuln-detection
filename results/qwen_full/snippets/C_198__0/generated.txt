#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <errno.h>
#include <unistd.h>

int set_interface_variable(const char *iface, const char *var, const char *name, uint32_t val) {
    char path[256];
    FILE *fp;

    // Construct the file path using the provided interface name and variable format
    if (snprintf(path, sizeof(path), var, iface) < 0) {
        return -1;
    }

    // Check if the file exists
    if (access(path, F_OK) != 0) {
        return -1;
    }

    // Open the file for writing
    fp = fopen(path, "w");
    if (!fp) {
        return -1;
    }

    // Write the value to the file
    if (fprintf(fp, "%u\n", val) < 0) {
        fclose(fp);
        return -1;
    }

    // Close the file
    if (fclose(fp) != 0) {
        return -1;
    }

    return 0;
}
