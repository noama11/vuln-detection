CURLcode smtp_escape_eob(struct connectdata *conn, ssize_t nread) {
  char *scratch = NULL;
  char *src, *dst;
  size_t i, len;
  bool replaced = FALSE;
  CURLcode result = CURLE_OK;

  if (nread <= 0)
    return CURLE_OK;

  scratch = malloc(nread + 2); /* +2 for possible CRLF addition */
  if (!scratch) {
    return CURLE_OUT_OF_MEMORY;
  }

  src = conn->upload.upload_buffer;
  dst = scratch;

  for (i = 0; i < nread; i++) {
    if (src[i] == '\r' && i + 1 < nread && src[i + 1] == '\n') {
      *dst++ = '\r';
      *dst++ = '\n';
      *dst++ = '.';
      i++; /* Skip the next '\n' */
      replaced = TRUE;
    } else {
      *dst++ = src[i];
    }
  }

  if (replaced) {
    /* If the last characters are not CRLF, add them */
    if (nread > 0 && src[nread - 1] != '\n') {
      *dst++ = '\r';
      *dst++ = '\n';
    }

    len = dst - scratch;
    if (conn->upload.upload_buffer) {
      free(conn->upload.upload_buffer);
    }
    conn->upload.upload_buffer = scratch;
    conn->upload.size = len;
  } else {
    free(scratch);
  }

  return result;
}
