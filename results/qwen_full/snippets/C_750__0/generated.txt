unsigned char *detach_blob(BlobInfo *blob_info) {
    if (!blob_info) {
        return NULL;
    }

    if (blob_info->data && blob_info->is_mapped) {
        munmap(blob_info->data, blob_info->size);
        blob_info->data = NULL;
        blob_info->is_mapped = 0;
    }

    unsigned char *detached_data = blob_info->data;
    blob_info->data = NULL;
    blob_info->size = 0;

    return detached_data;
}
