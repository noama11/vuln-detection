int handle_siocgstampns_ioctl(struct net *net, struct socket *sock, unsigned int cmd, void __user *up)
{
    int err;
    struct timespec ts;
    mm_segment_t old_fs = get_fs();

    set_fs(KERNEL_DS);
    err = sock_do_ioctl(net, sock, cmd, (unsigned long)&ts);
    set_fs(old_fs);

    if (err < 0)
        return err;

    err = compat_put_timespec(&ts, up);

    return err;
}
