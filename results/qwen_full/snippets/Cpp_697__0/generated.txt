int process_data_stream(int len, const u_char* data) {
    if (len <= 0 || !data) {
        return 0;
    }

    int processed = 0;
    u_char buffer[1024];
    int buffer_index = 0;
    u_char last_char = '\0';

    for (int i = 0; i < len; ++i) {
        u_char current_char = data[i];
        ++processed;

        if (current_char == '\r' || current_char == '\n') {
            if (buffer_index > 0) {
                // Emit the line stored in the buffer
                // (Implementation of emission is omitted as it is not specified)
                buffer_index = 0;
            }

            if (current_char == '\r') {
                // Check for CRLF
                if (i + 1 < len && data[i + 1] == '\n') {
                    ++i;
                    ++processed;
                }
            }
            last_char = current_char;
        } else {
            if (buffer_index < sizeof(buffer) - 1) {
                buffer[buffer_index++] = current_char;
            }
        }
    }

    if (buffer_index > 0) {
        // Emit the remaining line if not completed
        // (Implementation of emission is omitted as it is not specified)
    }

    return processed;
}
