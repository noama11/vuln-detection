struct sock *sctp6_sock_create(struct sock *sk, struct sctp_association *asoc, bool kern) {
    struct sock *new_sk;
    struct sctp6_sock *new_sctp6_sk;

    new_sk = sk_clone(sk, GFP_KERNEL);
    if (!new_sk)
        return NULL;

    new_sctp6_sk = (struct sctp6_sock *)new_sk;

    new_sctp6_sk->asoc = *asoc;
    new_sctp6_sk->kern = kern;

    if (ipv6_sock_init(new_sk) < 0) {
        sock_put(new_sk);
        return NULL;
    }

    return new_sk;
}
