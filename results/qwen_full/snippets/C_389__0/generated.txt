int
print_ipv6_routing_header(netdissect_options *ndo, const u_char *bp, const u_char *bp2)
{
    const struct ip6_rthdr *rthdr;
    int len, i;
    u_int offset;

    rthdr = (const struct ip6_rthdr *)bp;
    len = rthdr->ip6r_len * 8;
    ND_PRINT((ndo, "Routing Header: Length = %u, Type = %u, Segments Left = %u\n",
              rthdr->ip6r_len * 8, rthdr->ip6r_type, rthdr->ip6r_segleft));

    if (rthdr->ip6r_type == 0) {
        ND_PRINT((ndo, "  Type 0 (IPv6 Routing Header for Source Routes with IPv6)\n"));
        offset = sizeof(struct ip6_rthdr);
        for (i = 0; offset < len; i++) {
            if (offset + sizeof(struct in6_addr) > len) {
                ND_PRINT((ndo, "[|truncated]"));
                return -1;
            }
            ND_PRINT((ndo, "    Address[%d]: %s\n", i, ip6addr_string(ndo, (const u_char *)&rthdr->ip6r_data[i * sizeof(struct in6_addr)])));
            offset += sizeof(struct in6_addr);
        }
    } else {
        ND_PRINT((ndo, "  Unknown routing header type\n"));
    }

    return len;
}
