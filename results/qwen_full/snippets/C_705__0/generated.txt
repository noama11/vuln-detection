void pagevec_unlock(struct pagevec *pvec, struct zone *zone) {
    int i;
    struct page *page;
    int unlocked = 0;

    for (i = 0; i < pvec->nr_pages; i++) {
        page = pvec->pages[i];
        if (PageLocked(page)) {
            unlock_page(page);
            unlocked++;
        }
    }

    for (i = 0; i < pvec->nr_pages; i++) {
        page = pvec->pages[i];
        if (!PageLocked(page)) {
            put_page(page);
        }
    }

    if (unlocked > 0) {
        spin_lock(&zone->lock);
        for (i = 0; i < pvec->nr_pages; i++) {
            page = pvec->pages[i];
            if (PageLocked(page)) {
                __pagevec_putback(page, zone);
            }
        }
        spin_unlock(&zone->lock);
    }
}
