int process_ticket(struct ceph_auth_client *ac, struct ceph_crypto_key *secret, void **p, void *end) {
    int ret = -EINVAL;
    struct ceph_auth_ticket_handler *handler = NULL;
    void *ticket_data = NULL;
    size_t ticket_len;
    struct ceph_crypto_key new_key;
    struct ceph_timespec valid_since, valid_until;
    void *blob = NULL;
    int encrypted;

    if (!ac || !secret || !p || !*p || !end || *p > end) {
        goto out;
    }

    /* decode ticket type and structure version */
    uint32_t ticket_type = ceph_decode_32(p);
    uint32_t version = ceph_decode_32(p);

    handler = get_ticket_handler(ac, ticket_type);
    if (!handler) {
        goto out;
    }

    ticket_len = ceph_decode_32(p);
    if (*p + ticket_len > end) {
        goto out;
    }

    ticket_data = *p;
    *p += ticket_len;

    encrypted = ceph_decode_8(p);

    if (encrypted) {
        ret = decrypt_ticket(secret, ticket_data, ticket_len, &new_key, &valid_since, &valid_until, &blob);
        if (ret < 0) {
            goto out;
        }
    } else {
        /* handle unencrypted ticket */
        new_key.key = *p;
        new_key.length = ceph_decode_32(p);
        *p += new_key.length;

        valid_since.tv_sec = ceph_decode_64(p);
        valid_since.tv_nsec = ceph_decode_32(p);
        valid_until.tv_sec = ceph_decode_64(p);
        valid_until.tv_nsec = ceph_decode_32(p);

        blob = *p;
        ticket_len = ceph_decode_32(p);
        *p += ticket_len;
    }

    ret = update_session_key(ac, &new_key, &valid_since, &valid_until, blob);
    if (ret < 0) {
        goto out;
    }

    ret = 0;

out:
    if (handler) {
        put_ticket_handler(handler);
    }
    if (blob) {
        free(blob);
    }
    return ret;
}
