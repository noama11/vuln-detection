int bnep_connadd(struct bnep_connadd_req *req, struct socket *sock)
{
    struct bnep_session *session;
    struct net_device *dev;
    int err;

    dev = alloc_netdev(sizeof(struct bnep_session), "bnep%d",
                       NET_NAME_UNKNOWN, ether_setup);
    if (!dev)
        return -ENOMEM;

    session = netdev_priv(dev);
    session->dev = dev;
    session->sock = sock;
    memcpy(&session->src_lsap, &req->src_lsap, sizeof(session->src_lsap));
    memcpy(&session->dst_lsap, &req->dst_lsap, sizeof(session->dst_lsap));

    if (bnep_session_exists(session->dst_lsap)) {
        err = -EEXIST;
        goto error;
    }

    if (req->flags & BNEP_CONNADD_FLAG_MULTICAST)
        dev->flags |= IFF_MULTICAST;

    if (req->flags & BNEP_CONNADD_FLAG_PROTO_FILTER) {
        session->proto_filters = req->proto_filters;
        session->proto_filters_len = req->proto_filters_len;
    }

    err = register_netdev(dev);
    if (err)
        goto error;

    session->thread = kthread_run(bnep_session_handler, session, "bnepd");
    if (IS_ERR(session->thread)) {
        err = PTR_ERR(session->thread);
        goto error_unregister;
    }

    memcpy(req->dev_name, dev->name, IFNAMSIZ);

    return 0;

error_unregister:
    unregister_netdev(dev);
error:
    free_netdev(dev);
    return err;
}
