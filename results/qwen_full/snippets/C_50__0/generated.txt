#include <stdint.h>

#define LVT_MASK 0x0000ffff
#define LINT_MASK 0x0000ffff

typedef struct kvm_lapic {
    uint32_t regs[128];
} kvm_lapic_t;

uint32_t kvm_apic_id(kvm_lapic_t *apic) {
    return (apic->regs[0x20 >> 2] >> 24) & 0xff;
}

void recalculate_apic_map(void *kvm) {
    kvm_lapic_t **new_apic_map;
    kvm_lapic_t **old_apic_map;
    kvm_lapic_t *vcpu;

    new_apic_map = (kvm_lapic_t **)malloc(sizeof(kvm_lapic_t *));
    old_apic_map = (kvm_lapic_t **)malloc(sizeof(kvm_lapic_t *));
    vcpu = (kvm_lapic_t *)kvm;
}
