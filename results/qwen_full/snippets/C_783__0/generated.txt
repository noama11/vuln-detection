int megasas_alloc_cmds(struct megasas_instance *instance) {
    int i;
    struct megasas_cmd **cmd_list;

    cmd_list = kzalloc(instance->max_num_cmd * sizeof(struct megasas_cmd *), GFP_KERNEL);
    if (!cmd_list) {
        dev_err(&instance->pdev->dev, "Failed to allocate command list\n");
        return -ENOMEM;
    }

    for (i = 0; i < instance->max_num_cmd; i++) {
        struct megasas_cmd *cmd = &instance->cmd_list[i];
        cmd->index = i;
        cmd->instance = instance;
        list_add_tail(&cmd->list, &instance->cmd_pool);
    }

    if (megasas_create_frame_pool(instance)) {
        dev_err(&instance->pdev->dev, "Failed to create frame pool\n");
        kfree(cmd_list);
        return -ENOMEM;
    }

    return 0;
}
