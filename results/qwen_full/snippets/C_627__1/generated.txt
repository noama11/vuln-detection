int git_pkt_parse_ref(out git_pkt **pkt, const char *line, size_t len) {
    git_pkt_ng *ng_pkt = NULL;
    const char *ref_start = NULL;
    const char *msg_start = NULL;
    size_t ref_len = 0;
    size_t msg_len = 0;
    int ret = -1;

    if (!line || len < 5) {
        return -1;
    }

    if (strncmp(line, "ref ", 4) != 0) {
        return -1;
    }

    line += 4;
    len -= 4;

    if (len == 0 || *line != ' ') {
        return -1;
    }

    line++;
    len--;

    ref_start = line;

    while (len > 0 && *line != '\x00') {
        line++;
        len--;
    }

    if (len == 0 || *line != '\x00') {
        return -1;
    }

    ref_len = line - ref_start;
    line++;
    len--;

    msg_start = line;

    while (len > 0 && *line != '\x00') {
        line++;
        len--;
    }

    if (len == 0 || *line != '\x00') {
        return -1;
    }

    msg_len = line - msg_start;

    ng_pkt = malloc(sizeof(git_pkt_ng));
    if (!ng_pkt) {
        return -1;
    }

    ng_pkt->ref = malloc(ref_len + 1);
    if (!ng_pkt->ref) {
        free(ng_pkt);
        return -1;
    }

    ng_pkt->msg = malloc(msg_len + 1);
    if (!ng_pkt->msg) {
        free(ng_pkt->ref);
        free(ng_pkt);
        return -1;
    }

    memcpy(ng_pkt->ref, ref_start, ref_len);
    ng_pkt->ref[ref_len] = '\0';

    memcpy(ng_pkt->msg, msg_start, msg_len);
    ng_pkt->msg[msg_len] = '\0';

    *pkt = (git_pkt **)ng_pkt;
    return 0;
}
