int netlink_dump(struct sock *sk)
{
    struct sk_buff *skb = NULL;
    int err = -ENOMEM;

    mutex_lock(&sk->sk_lock);
    if (sk->sk_callback_running) {
        mutex_unlock(&sk->sk_lock);
        return -EBUSY;
    }

    if (!sk->sk_rcvbuf)
        goto out;

    skb = alloc_skb(sk->sk_rcvbuf, GFP_KERNEL);
    if (!skb)
        goto out;

    err = sk->sk_data_ready(sk, skb);
    if (err < 0)
        goto out;

    if (sk->sk_filter) {
        if (sk_filter(skb)) {
            netlink_unicast(sk, skb, sk->sk_filter);
        } else {
            kfree_skb(skb);
        }
    } else {
        kfree_skb(skb);
    }

    err = 0;

out:
    mutex_unlock(&sk->sk_lock);
    return err;
}
