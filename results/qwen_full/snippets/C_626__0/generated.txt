unsigned patch_call_insn(void *insnbuf, const void *target, u16 tgt_clobbers,
                         unsigned long addr, u16 site_clobbers, unsigned len)
{
    long delta = (long)target - (long)addr;

    if (len < 5) {
        return len;
    }

    *(u8 *)insnbuf = 0xe8; /* Opcode for relative call */
    *(long *)((u8 *)insnbuf + 1) = delta;

    return 5;
}
