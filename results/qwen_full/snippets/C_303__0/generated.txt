int cypress_probe(struct usb_serial_port *port)
{
    struct cypress_private *priv;
    int ret = -ENOMEM;

    priv = kzalloc(sizeof(*priv), GFP_KERNEL);
    if (!priv)
        return ret;

    spin_lock_init(&priv->lock);
    priv->in_fifo = usb_fifo_alloc(port->serial->dev, 0, GFP_KERNEL);
    priv->out_fifo = usb_fifo_alloc(port->serial->dev, 0, GFP_KERNEL);

    if (!priv->in_fifo || !priv->out_fifo) {
        usb_fifo_free(priv->in_fifo);
        usb_fifo_free(priv->out_fifo);
        kfree(priv);
        return -ENOMEM;
    }

    if (port->serial->num_ports > 1)
        priv->packet_format = PACKET_FORMAT_BULK;
    else
        priv->packet_format = PACKET_FORMAT_INTERRUPT;

    priv->read_interval = port->read_urb->interval;
    priv->write_interval = port->write_urb->interval;

    /* Set drain delay for the device */
    priv->drain_delay = 10;

    usb_set_serial_port_data(port, priv);

    return 0;
}
