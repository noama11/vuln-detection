int online_pages(unsigned long pfn, unsigned long nr_pages) {
    int ret = 0;
    struct memory_notify notification;

    lock_hotplug();

    notification.start_pfn = pfn;
    notification.nr_pages = nr_pages;

    if (!node_online(pfn_to_nid(pfn))) {
        notification.status_change_nid = pfn_to_nid(pfn);
    }

    ret = send_memory_notify(MEM_GOING_ONLINE, &notification, NULL);
    if (ret) {
        unlock_hotplug();
        return ret;
    }

    if (!zone_populated(pfn_to_zone(pfn))) {
        rebuild_zonelist();
    }

    ret = online_page_range(pfn, nr_pages);
    if (ret) {
        unlock_hotplug();
        return ret;
    }

    update_zone_statistics(pfn, nr_pages);
    rebuild_zonelist_if_needed();

    send_memory_notify(MEM_ONLINE, &notification, NULL);
    unlock_hotplug();

    return 0;
}
