bool PP_ImageData_Init(PP_ImageDataFormat format, int width, int height, bool init_to_zero) {
    // Check if the provided image format is supported
    if (format < PP_IMAGEDATAFORMAT_BGRA || format > PP_IMAGEDATAFORMAT_RGBA) {
        return false;
    }

    // Validate width and height
    if (width <= 0 || height <= 0) {
        return false;
    }

    // Set member variables
    imageData.format = format;
    imageData.width = width;
    imageData.height = height;
    imageData.init_to_zero = init_to_zero;

    // Call the backend's Init method to complete the initialization
    return backend_Init(&imageData);
}
