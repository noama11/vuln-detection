int handle_scsi_command(SCSIDiskReq *r) {
    if (r == NULL || r->cmd == NULL || r->cmd_len > MAX_COMMAND_LENGTH) {
        return -1;
    }

    if (r->device_type == ROM_DEVICE && (r->cmd[0] == SCSI_EJECT || r->cmd[0] == SCSI_START_STOP)) {
        if (r->cmd[4] & EJECT_TRAY_BIT) {
            if (r->tray_locked) {
                return -1; // Tray is locked, cannot eject
            }
            r->tray_open = 1;
        } else if (r->cmd[4] & LOAD_TRAY_BIT) {
            r->tray_open = 0;
        }
    }

    if (r->cmd[0] == SCSI_TEST_UNIT_READY) {
        if (r->tray_open) {
            return -1; // Unit not ready if tray is open
        }
        return 0;
    }

    if (r->cmd[0] == SCSI_INQUIRY) {
        if (r->data == NULL || r->data_len < MIN_INQUIRY_DATA_SIZE) {
            return -1;
        }
        // Fill in the inquiry data here if needed
        return 0;
    }

    if (r->cmd[0] == SCSI_REQUEST_SENSE) {
        if (r->data == NULL || r->data_len < REQUEST_SENSE_DATA_SIZE) {
            return -1;
        }
        // Fill in the sense data here if needed
        return 0;
    }

    return -1; // Unknown or illegal request
}
