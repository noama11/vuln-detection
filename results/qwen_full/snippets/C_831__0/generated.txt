int get_bluetooth_socket_name(struct socket *sock, struct sockaddr *addr, int *addr_len, int peer)
{
    struct sockaddr_hci *hci_addr;

    if (!sock || !addr || !addr_len)
        return -EBADFD;

    if (!sock->sk || !sock->sk->sk_prot || !sock->sk->sk_prot->get_name)
        return -EBADFD;

    hci_addr = (struct sockaddr_hci *)addr;
    hci_addr->hci_family = AF_BLUETOOTH;

    if (peer) {
        hci_addr->hci_dev = sock->sk->sk_bound_dev;
        hci_addr->hci_channel = BT_CHANNEL_ACL;
    } else {
        hci_addr->hci_dev = sock->sk->sk_rcv_dev;
        hci_addr->hci_channel = BT_CHANNEL_RAW;
    }

    *addr_len = sizeof(struct sockaddr_hci);
    return 0;
}
