struct net *get_net_ns_by_id(struct net *net, int id) {
    struct net *result = NULL;

    if (id < 0)
        return NULL;

    read_lock(&net_namespace_lock);
    spin_lock(&net_idr_lock);
    result = idr_find(&net_idr, id);
    if (result)
        get_net(result);
    spin_unlock(&net_idr_lock);
    read_unlock(&net_namespace_lock);

    return result;
}
