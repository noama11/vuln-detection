int br_add_multicast_group(struct net_bridge *br, struct net_bridge_port *port, struct br_ip *group) {
    int result = 0;

    spin_lock(&br->multicast_lock);

    if (!netif_running(br->dev)) {
        result = -EINVAL;
        goto unlock;
    }

    if (port && !port->enabled) {
        result = -EINVAL;
        goto unlock;
    }

    if (!port) {
        struct net_bridge_mcast *mcast;

        mcast = kmalloc(sizeof(*mcast), GFP_KERNEL);
        if (!mcast) {
            result = -ENOMEM;
            goto unlock;
        }

        br_ip_copy(&mcast->group, group);
        hlist_add_head(&mcast->hlist, &br->multicast_list);
    } else {
        struct net_bridge_port_group *pgrp;

        pgrp = kzalloc(sizeof(*pgrp), GFP_KERNEL);
        if (!pgrp) {
            result = -ENOMEM;
            goto unlock;
        }

        br_ip_copy(&pgrp->group, group);
        pgrp->port = port;
        hlist_add_head(&pgrp->hlist, &port->multicast_list);

        pgrp->timer.expires = jiffies + BR_MULTICAST_TIMER_INTERVAL;
        add_timer(&pgrp->timer);
    }

unlock:
    spin_unlock(&br->multicast_lock);
    return result;
}
