int remount_shmem(struct super_block *sb, int *flags, char *data) {
    struct shmem_config new_config;
    int result = 0;

    if (parse_shmem_options(data, &new_config) < 0) {
        return -EINVAL;
    }

    spin_lock(&sb->s_lock);

    if (sb->s_blocks_used > new_config.max_blocks || sb->s_inodes_used > new_config.max_inodes) {
        spin_unlock(&sb->s_lock);
        return -ENOSPC;
    }

    sb->s_max_blocks = new_config.max_blocks;
    sb->s_max_inodes = new_config.max_inodes;
    sb->s_free_inodes = new_config.max_inodes - sb->s_inodes_used;

    if (new_config.policy) {
        setup_memory_policy(sb, new_config.policy);
    }

    spin_unlock(&sb->s_lock);

    return result;
}
