struct mm_struct *init_mm(struct mm_struct *mm, struct task_struct *p, struct user_namespace *user_ns) {
    int err;

    /* Initialize memory mappings */
    mm->mmap = NULL;
    mm->mmap_cache = NULL;
    mm->get_unmapped_area = NULL;
    mm->unmap_area = NULL;

    /* Initialize reference counts */
    atomic_set(&mm->count, 1);
    mm->task_size = TASK_SIZE;
    mm->highest_vm_end = TASK_SIZE;

    /* Initialize locks */
    spin_lock_init(&mm->page_table_lock);
    mutex_init(&mm->mmap_lock);

    /* Configure flags based on current process's memory management state */
    if (current->mm) {
        mm->flags = current->mm->flags;
    } else {
        mm->flags = 0;
    }

    /* Allocate page tables */
    mm->pgd = get_pgd();
    if (!mm->pgd) {
        return NULL;
    }

    /* Initialize new context for the task */
    err = init_new_context(p, mm);
    if (err) {
        free_pgd(mm->pgd);
        return NULL;
    }

    return mm;
}
