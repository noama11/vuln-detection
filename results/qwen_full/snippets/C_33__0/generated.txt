int set_ext_regs(struct task_struct *child, void __user *uregs)
{
    int ret;

    /* Flush and release coprocessors if applicable */
    flush_and_release_coprocessors();

    /* Copy extended register values from user space */
    ret = copy_from_user(&child->thread.ext_regs, uregs, sizeof(child->thread.ext_regs));
    if (ret) {
        return -EFAULT;
    }

    return 0;
}
