int flush_key_notification_policy(const struct km_event *c) {
    struct sk_buff *skb;
    struct sadb_msg *msg;
    int err;

    skb = alloc_skb(sizeof(*msg), GFP_KERNEL);
    if (!skb)
        return -ENOBUFS;

    msg = skb_put(skb, sizeof(*msg));
    memset(msg, 0, sizeof(*msg));
    msg->sadb_msg_version = PF_KEY_V2;
    msg->sadb_msg_type = SADB_GETSPDINFO;
    msg->sadb_msg_errno = 0;
    msg->sadb_msg_satype = 0;
    msg->sadb_msg_len = sizeof(*msg) / sizeof(uint64_t);
    msg->sadb_msg_reserved = 0;

    err = nf_key_queue_msg(NULL, skb, c->seq, c->portid, 0);
    if (err < 0) {
        kfree_skb(skb);
        return err;
    }

    return 0;
}
