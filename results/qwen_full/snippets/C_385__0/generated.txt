int tcp_disconnect(struct sock *sk, int flags) {
    int err = 0;

    if (sk->sk_state != TCP_CLOSE) {
        switch (sk->sk_state) {
            case TCP_LISTEN:
                tcp_stop_listen(sk);
                break;
            case TCP_SYN_SENT:
            case TCP_SYN_RECV:
                /* Send a reset if needed */
                if (tcp_send_reset(sk))
                    err = -1;
                break;
            default:
                break;
        }

        sk->sk_state = TCP_CLOSE;
    }

    tcp_purge_tx_queue(sk);
    tcp_purge_rx_queue(sk);

    sk->sk_wmem_queued = 0;
    sk->sk_forward_alloc = 0;
    sk->sk_wnd_end_seq = 0;
    sk->sk_rcv_wnd = 0;
    sk->sk_snd_wnd = 0;

    tcp_cleanup_fastopen(sk);

    return err;
}
