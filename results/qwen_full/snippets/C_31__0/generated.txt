int cifs_search_next(int xid, struct cifs_tcon *tcon, __u16 searchHandle, struct cifs_search_info *psrch_inf) {
    struct smb_hdr *smb;
    struct smb_composite_search_req *req;
    struct smb_composite_search_rsp *rsp;
    int rc = 0;
    int bytes_returned = 0;

    smb = cifs_composite_init(xid, tcon, CIFS_SEARCH, 0, NULL);
    if (!smb) {
        return -ENOMEM;
    }

    req = (struct smb_composite_search_req *)((char *)smb + sizeof(struct smb_hdr));
    req->SearchHandle = searchHandle;
    req->SearchCount = psrch_inf->SearchCount;
    req->SearchFlags = psrch_inf->SearchFlags;

    rc = SendReceive2(xid, tcon, (char *)smb, &bytes_returned, 0);
    if (rc != 0) {
        cifs_composite_free(smb);
        return rc;
    }

    rsp = (struct smb_composite_search_rsp *)((char *)smb + sizeof(struct smb_hdr));
    if (rsp->ByteCount == 0) {
        psrch_inf->EndOfSearch = true;
    } else {
        psrch_inf->SearchCount = rsp->SearchCount;
        psrch_inf->SearchFlags = rsp->SearchFlags;
        psrch_inf->EndOfSearch = (rsp->EndOfSearch != 0);
    }

    cifs_composite_free(smb);
    return 0;
}
