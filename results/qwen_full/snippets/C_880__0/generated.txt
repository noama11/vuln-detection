#include <stdio.h>
#include <stdlib.h>
#include <sys/stat.h>
#include <limits.h>

// Global flag to indicate if umask was set from command line
int umask_set_from_cmdline = 0;

mode_t set_umask_from_string(const char *optarg) {
    char *endptr;
    long umask_val = strtol(optarg, &endptr, 8);

    // Check for conversion failure or invalid input
    if (optarg == endptr || *endptr != '\0' || umask_val < 0 || umask_val > 0777) {
        fprintf(stderr, "Invalid umask value: %s\n", optarg);
        return 0;
    }

    mode_t new_umask = (mode_t)umask_val;
    umask(new_umask);
    umask_set_from_cmdline = 1;

    return new_umask;
}
