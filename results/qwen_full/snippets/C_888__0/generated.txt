#include <assert.h>
#include <stdlib.h>

CustomStreamInfo *destroy_custom_stream(CustomStreamInfo *custom_stream) {
    if (custom_stream == NULL) {
        return NULL;
    }

    assert(custom_stream->blob_info != NULL);
    // Log event if debugging is enabled
    // Handle any mapped blob resources
    if (custom_stream->blob_data != NULL) {
        // Clear the blob data pointer
        free(custom_stream->blob_data);
        custom_stream->blob_data = NULL;
    }

    // Relinquish the mapped resource
    free(custom_stream);
    return NULL;
}
