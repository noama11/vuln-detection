long enumerate_media_entities(struct media_device *mdev, struct media_entity_desc __user *uent)
{
    struct media_entity_desc kent;
    struct media_entity *entity;
    int ret;

    if (!mdev || !uent)
        return -EINVAL;

    entity = media_entity_get_by_id(mdev, mdev->current_id);
    if (!entity)
        return -ENOENT;

    memset(&kent, 0, sizeof(kent));
    kent.id = entity->id;
    kent.name = entity->name;
    kent.type = entity->type;
    kent.flags = entity->flags;
    kent.group_id = entity->group_id;
    kent.pad_num = entity->num_pads;
    kent.link_num = entity->num_links;

    if (copy_to_user(uent, &kent, sizeof(kent)))
        return -EFAULT;

    return 0;
}
