#include <string.h>
#include <stdint.h>

// Assuming szLine is a global or static buffer of type uint16_t* and unicode_type is a global variable
// indicating endianness (0 for little-endian, 1 for big-endian)

void process_utf16_to_utf8() {
    uint16_t *szLine = ...; // Assume szLine is initialized elsewhere
    int unicode_type = ...; // Assume unicode_type is initialized elsewhere
    int i = 0;

    if (unicode_type) {
        // Big-endian processing
        while (szLine[i] != 0) {
            uint16_t temp = szLine[i];
            szLine[i] = (temp >> 8) | (temp << 8);
            i++;
        }
    } else {
        // Little-endian processing (no swap needed)
        while (szLine[i] != 0) {
            i++;
        }
    }
}
