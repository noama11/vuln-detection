int attach_usb_serial(struct usb_serial *serial) {
    if (serial->num_ports < 2 || serial->port[0].bulk_out_endpointAddress == 0 ||
        serial->port[1].bulk_out_endpointAddress == 0) {
        return -ENODEV;
    }

    serial->port[0].bulk_out_endpointAddress = serial->port[1].bulk_out_endpointAddress;
    serial->port[0].write_urb->pipe = usb_sndbulkpipe(serial->dev,
                                                      serial->port[0].bulk_out_endpointAddress);

    return 0;
}
