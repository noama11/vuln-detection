int print_asn1_object_to_bio(BIO *bio, const ASN1_OBJECT *obj) {
    char buffer[256];
    int len = OBJ_obj2txt(buffer, sizeof(buffer), obj, 0);
    if (len <= 0) {
        return 0;
    }
    if (BIO_write(bio, buffer, len) != len) {
        return 0;
    }
    if (BIO_write(bio, "\n", 1) != 1) {
        return 0;
    }
    return 1;
}
