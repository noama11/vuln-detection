ret_t validate_credentials_ldap(cherokee_validator_ldap_t *ldap, cherokee_connection_t *conn) {
    ret_t ret = ret_error;
    LDAPMessage *result = NULL;
    char *search_filter = NULL;
    char *user_dn = NULL;

    if (!ldap || !conn || !conn->user || !conn->password) {
        return ret_error;
    }

    if (ldap_build_search_filter(ldap, conn->user, &search_filter) != ret_ok) {
        return ret_error;
    }

    if (ldap_perform_search(ldap, search_filter, &result) != ret_ok) {
        goto cleanup;
    }

    if (ldap_count_entries(ldap->ld, result) != 1) {
        ret = ret_not_found;
        goto cleanup;
    }

    user_dn = ldap_get_dn(ldap->ld, ldap_first_entry(ldap->ld, result));
    if (!user_dn) {
        goto cleanup;
    }

    if (ldap_bind_s(ldap->ld, user_dn, conn->password, LDAP_AUTH_SIMPLE) != LDAP_SUCCESS) {
        ret = ret_not_found;
    } else {
        ret = ret_ok;
    }

cleanup:
    if (search_filter) {
        free(search_filter);
    }

    if (result) {
        ldap_msgfree(result);
    }

    if (user_dn) {
        ldap_memfree(user_dn);
    }

    ldap_unbind(ldap->ld);
    return ret;
}
