bool extract_socket_address_and_port(struct pool *pool, char *url) {
    char *start = url;
    char *end = NULL;
    char *colon = NULL;
    char *address = NULL;
    char *port = NULL;
    size_t address_len = 0;
    int default_port = 80;

    // Check for IPv6 address (enclosed in brackets)
    if (*start == '[') {
        start++;
        end = strchr(start, ']');
        if (!end) {
            return false;
        }
        // Move past the closing bracket
        end++;
        if (*end == ':') {
            colon = end;
        }
    } else {
        // Find the colon for the port
        colon = strchr(start, ':');
    }

    if (colon) {
        // Port is present
        port = colon + 1;
        *colon = '\0';
        address_len = colon - start;
    } else {
        // No port specified, use default
        port = NULL;
        address_len = strlen(start);
    }

    // Validate URL length
    if (address_len == 0 || address_len > 1024) {
        return false;
    }

    // Copy the address into a buffer
    address = (char *)malloc(address_len + 1);
    if (!address) {
        return false;
    }
    strncpy(address, start, address_len);
    address[address_len] = '\0';

    // Set the default port if none is specified
    if (!port) {
        port = (char *)malloc(6); // Enough for 5-digit port
        if (!port) {
            free(address);
            return false;
        }
        snprintf(port, 6, "%d", default_port);
    }

    // Store the extracted values in the pool
    pool->socket_address = address;
    pool->port = port;

    return true;
}
