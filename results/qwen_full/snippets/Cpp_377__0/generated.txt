AP4_Result ReadVisualSampleEntry(AP4_ByteStream& stream) {
    AP4_Result result = ReadSampleEntry(stream);
    if (AP4_FAILED(result)) return result;

    AP4_UI32 predefined1;
    AP4_UI16 width;
    AP4_UI16 height;
    AP4_UI16 horizResolution;
    AP4_UI16 vertResolution;
    AP4_UI16 dataOffset;
    AP4_UI16 frameCount;
    char compressorName[32];

    result = stream.ReadUI32(predefined1);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(width);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(height);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(horizResolution);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(vertResolution);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(dataOffset);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(frameCount);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadBytes(compressorName, sizeof(compressorName));
    if (AP4_FAILED(result)) return result;

    // Ensure null termination
    compressorName[sizeof(compressorName) - 1] = '\0';

    AP4_UI16 depth;
    AP4_UI16 predefined2;

    result = stream.ReadUI16(depth);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(predefined2);
    if (AP4_FAILED(result)) return result;

    return AP4_SUCCESS;
}
