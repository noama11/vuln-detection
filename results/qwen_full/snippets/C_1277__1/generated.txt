#include <pthread.h>
#include <stdint.h>
#include <stdlib.h>

// Assuming a simplified version of ThreadSafeRefCounted for C
typedef struct ThreadSafeRefCounted {
    pthread_mutex_t ref_count_mutex;
    int ref_count;
} ThreadSafeRefCounted;

typedef struct Encoder Encoder;

struct Encoder {
    ThreadSafeRefCounted base;
    pthread_mutex_t paused_mutex;
    int paused;
    // Task runners are abstracted here as function pointers
    void (*main_task_runner)();
    void (*encode_task_runner)();
};

Encoder* encoder_create() {
    Encoder* encoder = (Encoder*)malloc(sizeof(Encoder));
    pthread_mutex_init(&encoder->base.ref_count_mutex, NULL);
    encoder->base.ref_count = 1;
    pthread_mutex_init(&encoder->paused_mutex, NULL);
    encoder->paused = 0;
    encoder->main_task_runner = NULL;
    encoder->encode_task_runner = NULL;
    return encoder;
}

void encoder_start_frame_encode(Encoder* encoder) {
    if (!encoder) return;

    // Start the encoding process
    // Placeholder for actual encoding logic
    // This could involve configuring the encoder and starting the encoding task
    if (encoder->encode_task_runner) {
        encoder->encode_task_runner();
    }
}

void encoder_pause(Encoder* encoder) {
    if (!encoder) return;
    pthread_mutex_lock(&encoder->paused_mutex);
    encoder->paused = 1;
    pthread_mutex_unlock(&encoder->paused_mutex);
}

void encoder_resume(Encoder* encoder) {
    if (!encoder) return;
    pthread_mutex_lock(&encoder->paused_mutex);
    encoder->paused = 0;
    pthread_mutex_unlock(&encoder->paused_mutex);
}

void encoder_destroy(Encoder* encoder) {
    if (!encoder) return;
    pthread_mutex_destroy(&encoder->base.ref_count_mutex);
    pthread_mutex_destroy(&encoder->paused_mutex);
    free(encoder);
}
