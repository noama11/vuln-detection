#include <wolfssl/wolfcrypt/integer.h>
#include <wolfssl/wolfcrypt/random.h>

int generate_ec_k(WC_RNG* rng, int size, mp_int* k, mp_int* order) {
    int ret = 0;
    byte* randBytes = NULL;
    mp_int temp;

    if (rng == NULL || k == NULL || order == NULL || size <= 0) {
        return BAD_FUNC_ARG;
    }

    randBytes = (byte*)XMALLOC(size, NULL, DYNAMIC_TYPE_TMP_BUFFER);
    if (randBytes == NULL) {
        return MEMORY_E;
    }

    if ((ret = wc_RNG_GenerateBlock(rng, randBytes, size)) != 0) {
        XFREE(randBytes, NULL, DYNAMIC_TYPE_TMP_BUFFER);
        return ret;
    }

    if ((ret = mp_init(&temp)) != MP_OKAY) {
        XFREE(randBytes, NULL, DYNAMIC_TYPE_TMP_BUFFER);
        return ret;
    }

    if ((ret = mp_read_unsigned_bin(&temp, randBytes, size)) != MP_OKAY) {
        XFREE(randBytes, NULL, DYNAMIC_TYPE_TMP_BUFFER);
        mp_clear(&temp);
        return ret;
    }

    if ((ret = mp_mod(&temp, order, k)) != MP_OKAY) {
        XFREE(randBytes, NULL, DYNAMIC_TYPE_TMP_BUFFER);
        mp_clear(&temp);
        return ret;
    }

    if (mp_cmp_d(k, 0) == MP_EQ) {
        if ((ret = mp_set_int(k, 1)) != MP_OKAY) {
            XFREE(randBytes, NULL, DYNAMIC_TYPE_TMP_BUFFER);
            mp_clear(&temp);
            return ret;
        }
    }

    XFREE(randBytes, NULL, DYNAMIC_TYPE_TMP_BUFFER);
    mp_clear(&temp);
    return 0;
}
