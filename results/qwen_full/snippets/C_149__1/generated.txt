int get_pppol2tp_sockopts(struct socket *sock, int level, int optname,
                          char __user *optval, int __user *optlen)
{
    int ret = 0;
    struct sock *sk = sock->sk;
    struct pppol2tp_session *session;
    int len;

    if (level != SOL_PPPOL2TP)
        return -EINVAL;

    if (get_user(len, optlen))
        return -EFAULT;

    if (len < 0)
        return -EINVAL;

    if (!sk || !sk->sk_bound_dev_if)
        return -ENOTCONN;

    session = pppol2tp_get_session(sk);
    if (!session)
        return -ENOENT;

    if (session->session_id == 0) {
        /* Tunnel options */
        if (optname == PPCP_OPTION) {
            if (len < sizeof(int))
                return -EINVAL;
            if (put_user(PPCP_OPTION_DEFAULT, (int __user *)optval))
                return -EFAULT;
        } else {
            return -ENOPROTOOPT;
        }
    } else {
        /* Session options */
        if (optname == PPPOL2TP_SO_RECVSEQ) {
            if (len < sizeof(int))
                return -EINVAL;
            if (put_user(session->recv_seq, (int __user *)optval))
                return -EFAULT;
        } else if (optname == PPPOL2TP_SO_SENDSEQ) {
            if (len < sizeof(int))
                return -EINVAL;
            if (put_user(session->send_seq, (int __user *)optval))
                return -EFAULT;
        } else {
            return -ENOPROTOOPT;
        }
    }

    if (put_user(len, optlen))
        return -EFAULT;

    return ret;
}
