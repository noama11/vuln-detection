long compat_recvmmsg_time32(int fd, struct compat_mmsghdr __user *mmsg,
                            unsigned int vlen, unsigned int flags,
                            struct compat_timespec __user *timeout)
{
    long ret;
    struct timespec ts;
    struct timespec __user *ktimeout = NULL;

    if (timeout) {
        if (get_user(ts.tv_sec, &timeout->tv_sec) ||
            get_user(ts.tv_nsec, &timeout->tv_nsec))
            return -EFAULT;
        ktimeout = &ts;
    }

    ret = recvmmsg(fd, (struct mmsghdr __user *)mmsg, vlen, flags, ktimeout);

    if (ret >= 0)
        return ret;

    return ret;
}
