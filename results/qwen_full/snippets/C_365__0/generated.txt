int configure_usb_console(struct console *co, char *options) {
    int baud_rate = 0;
    int data_bits = 0;
    char parity = '\0';
    int flow_control = 0;
    char *token;
    char *saveptr;
    char *options_copy = strdup(options);
    int ret = 0;
    struct usb_serial_port *port;

    if (!options_copy) {
        return -ENOMEM;
    }

    token = strtok_r(options_copy, ",", &saveptr);
    while (token) {
        if (strncmp(token, "baud", 4) == 0) {
            baud_rate = atoi(token + 4);
        } else if (strncmp(token, "bits", 4) == 0) {
            data_bits = atoi(token + 4);
        } else if (strncmp(token, "parity", 6) == 0) {
            parity = token[6];
        } else if (strncmp(token, "flow", 4) == 0) {
            flow_control = atoi(token + 4);
        }
        token = strtok_r(NULL, ",", &saveptr);
    }

    port = get_usb_serial_port_by_index(co->index);
    if (!port) {
        ret = -ENODEV;
        goto out;
    }

    if (!port->connected) {
        printk(KERN_ERR "USB console port not connected\n");
        ret = -ENODEV;
        goto out;
    }

    if (usb_serial_open(port, co->tty)) {
        printk(KERN_ERR "Failed to open USB serial port\n");
        ret = -EIO;
        goto out;
    }

    co->tty->termios->c_cflag &= ~CSTOPB;
    co->tty->termios->c_cflag &= ~PARENB;
    co->tty->termios->c_cflag &= ~CSTOPB;
    co->tty->termios->c_cflag &= ~CSIZE;

    switch (data_bits) {
        case 5:
            co->tty->termios->c_cflag |= CS5;
            break;
        case 6:
            co->tty->termios->c_cflag |= CS6;
            break;
        case 7:
            co->tty->termios->c_cflag |= CS7;
            break;
        case 8:
            co->tty->termios->c_cflag |= CS8;
            break;
        default:
            co->tty->termios->c_cflag |= CS8;
            break;
    }

    if (parity == 'o' || parity == 'O') {
        co->tty->termios->c_cflag |= PARENB;
        co->tty->termios->c_cflag |= PARODD;
    } else if (parity == 'e' || parity == 'E') {
        co->tty->termios->c_cflag |= PARENB;
        co->tty->termios->c_cflag &= ~PARODD;
    }

    if (flow_control == 1) {
        co->tty->termios->c_cflag |= CRTSCTS;
    }

    co->tty->termios->c_ispeed = baud_rate;
    co->tty->termios->c_ospeed = baud_rate;

    co->is_console = 1;

out:
    free(options_copy);
    return ret;
}
