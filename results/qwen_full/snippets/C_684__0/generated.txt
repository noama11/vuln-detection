#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <sys/stat.h>
#include <unistd.h>

#include "lept.h"  // Assuming this is the header for l_int32 and related functions

l_int32 generate_html_viewer(const char* dirin, const char* dirout, const char* rootname, l_int32 thumbwidth, l_int32 viewwidth) {
    // Validate input parameters
    if (!dirin || !dirout || !rootname || thumbwidth <= 0 || viewwidth <= 0) {
        return 1;  // Invalid input
    }

    // Create output directory if it doesn't exist
    struct stat st;
    if (stat(dirout, &st) == -1) {
        if (mkdir(dirout, 0777) != 0) {
            return 2;  // Failed to create output directory
        }
    }

    // Open input directory
    DIR* dp = opendir(dirin);
    if (!dp) {
        return 3;  // Failed to open input directory
    }

    // Read image filenames
    struct dirent* entry;
    char** filenames = NULL;
    l_int32 count = 0;
    while ((entry = readdir(dp)) != NULL) {
        if (entry->d_type == DT_REG) {
            const char* ext = strrchr(entry->d_name, '.');
            if (ext && (strcmp(ext, ".jpg") == 0 || strcmp(ext, ".jpeg") == 0 || strcmp(ext, ".png") == 0)) {
                filenames = (char**)realloc(filenames, (count + 1) * sizeof(char*));
                filenames[count] = strdup(entry->d_name);
                count++;
            }
        }
    }
    closedir(dp);

    if (count == 0) {
        free(filenames);
        return 4;  // No image files found
    }

    // Generate HTML content
    char html_main[1024];
    char html_links[1024];
    snprintf(html_main, sizeof(html_main), "%s/%s.html", dirout, rootname);
    snprintf(html_links, sizeof(html_links), "%s/%s_links.html", dirout, rootname);

    FILE* fp_main = fopen(html_main, "w");
    if (!fp_main) {
        free(filenames);
        return 5;  // Failed to open main HTML file
    }

    FILE* fp_links = fopen(html_links, "w");
    if (!fp_links) {
        fclose(fp_main);
        free(filenames);
        return 6;  // Failed to open links HTML file
    }

    // Write HTML headers
    fprintf(fp_main, "<!DOCTYPE html>\n<html>\n<head>\n<title>Image Viewer</title>\n</head>\n<body>\n");
    fprintf(fp_links, "<!DOCTYPE html>\n<html>\n<head>\n<title>Image Links</title>\n</head>\n<body>\n");

    // Generate thumbnails and viewable images
    for (l_int32 i = 0; i < count; i++) {
        char input_path[1024];
        char thumb_path[1024];
        char view_path[1024];
        char link_path[1024];

        snprintf(input_path, sizeof(input_path), "%s/%s", dirin, filenames[i]);

        // Generate thumbnail path
        char* thumb_ext = strrchr(filenames[i], '.');
        char thumb_name[1024];
        snprintf(thumb_name, sizeof(thumb_name), "%s_thumb%s", rootname, thumb_ext);
        snprintf(thumb_path, sizeof(thumb_path), "%s/%s", dirout, thumb_name);

        // Generate viewable image path
        char view_name[1024];
        snprintf(view_name, sizeof(view_name), "%s_view%s", rootname, thumb_ext);
        snprintf(view_path, sizeof(view_path), "%s/%s", dirout, view_name);

        // Generate link path
        snprintf(link_path, sizeof(link_path), "%s.html#%s", rootname, filenames[i]);

        // Generate thumbnail and viewable image (placeholder for actual image processing)
        // In a real implementation, you would use a library like Leptonica or ImageMagick to generate these.

        // Write HTML content
        fprintf(fp_main, "<a href=\"#%s\"><img src=\"%s\" alt=\"%s\" width=\"%d\"></a>\n", filenames[i], thumb_path, filenames[i], thumbwidth);
        fprintf(fp_main, "<div id=\"%s\">\n<img src=\"%s\" alt=\"%s\" width=\"%d\">\n</div>\n", filenames[i], view_path, filenames[i], viewwidth);
        fprintf(fp_links, "<a href=\"%s\">%s</a><br>\n", link_path, filenames[i]);
    }

    // Write HTML footers
    fprintf(fp_main, "</body>\n</html>\n");
    fprintf(fp_links, "</body>\n</html>\n");

    // Clean up
    fclose(fp_main);
    fclose(fp_links);
    for (l_int32 i = 0; i < count; i++) {
        free(filenames[i]);
    }
    free(filenames);

    return 0;  // Success
}
