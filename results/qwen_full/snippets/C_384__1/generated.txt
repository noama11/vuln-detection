OPJ_BOOL write_sot_marker(opj_j2k_t *p_j2k, OPJ_BYTE *p_data, OPJ_UINT32 p_total_data_size, OPJ_UINT32 *p_data_written, const opj_stream_private_t *p_stream, opj_event_mgr_t *p_manager) {
    if (!p_j2k || !p_data || !p_data_written) {
        return OPJ_FALSE;
    }

    /* SOT marker: 0xFF, 0x91 */
    if (*p_data_written + 2 > p_total_data_size) {
        return OPJ_FALSE;
    }

    p_data[(*p_data_written)++] = 0xFF;
    p_data[(*p_data_written)++] = 0x91;

    return OPJ_TRUE;
}
