#include <linux/personality.h>
#include <linux/unistd.h>
#include <linux/version.h>
#include <asm/uaccess.h>

int compat_get_uname(char __user *release, size_t len) {
    int ret = 0;
    char version[32];
    unsigned int version_code = LINUX_VERSION_CODE;

    if (current->personality & UNAME26) {
        if (version_code >= LINUX_VERSION(3, 0, 0)) {
            int major = 2;
            int minor = 6;
            int patch = version_code - LINUX_VERSION(3, 0, 0) + 1;
            int version_len = 0;

            version_len += snprintf(version, sizeof(version), "Linux 2.6.%d", patch);

            if (version_len >= len) {
                return -EINVAL;
            }

            if (copy_to_user(release, version, version_len)) {
                return -EFAULT;
            }

            return version_len;
        }
    }

    return -EINVAL;
}
