#include <ctype.h>
#include <stdio.h>

#define MAX_WORD_SIZE 1024

void process_escape_sequence(char *word, int *index, FILE *input) {
    int escape_active = 0;
    int c = fgetc(input);

    if (c == '\\') {
        escape_active = 1;
        c = fgetc(input);
    }

    if (escape_active && c == '\n') {
        c = fgetc(input);
    }

    if (escape_active) {
        switch (c) {
            case 'a':
                c = '\a';
                break;
            case 'b':
                c = '\b';
                break;
            case 'f':
                c = '\f';
                break;
            case 'n':
                c = '\n';
                break;
            case 'r':
                c = '\r';
                break;
            case 't':
                c = '\t';
                break;
            case 'v':
                c = '\v';
                break;
            case '\\':
                c = '\\';
                break;
            case '\'':
                c = '\'';
                break;
            case '"':
                c = '"';
                break;
            case '?':
                c = '?';
                break;
            default:
                if (isdigit(c)) {
                    int octal_value = c - '0';
                    c = fgetc(input);
                    if (isdigit(c)) {
                        octal_value = octal_value * 8 + (c - '0');
                        c = fgetc(input);
                        if (isdigit(c)) {
                            octal_value = octal_value * 8 + (c - '0');
                        }
                    }
                    c = octal_value;
                } else if (c == 'x' || c == 'X') {
                    int hex_value = 0;
                    c = fgetc(input);
                    while (isxdigit(c)) {
                        hex_value = hex_value * 16 + (isdigit(c) ? c - '0' : toupper(c) - 'A' + 10);
                        c = fgetc(input);
                    }
                    c = hex_value;
                } else {
                    c = fgetc(input);
                }
        }
    }

    if (*index < MAX_WORD_SIZE - 1) {
        word[(*index)++] = c;
    }
}
