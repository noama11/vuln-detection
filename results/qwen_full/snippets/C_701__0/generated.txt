#define arch_timer_reg_read_stable(reg) ({ \
    u64 __val; \
    if (needs_unstable_timer_counter_workaround()) { \
        preempt_disable(); \
        void (*__workaround_func)(u64 *); \
        __workaround_func = __this_cpu_read(timer_workaround_func); \
        if (__workaround_func) { \
            __workaround_func(&__val); \
        } else { \
            __val = read_sysreg(reg); \
        } \
        preempt_enable(); \
    } else { \
        __val = read_sysreg(reg); \
    } \
    __val; \
})
