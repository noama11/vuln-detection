void* AllocateNonPagedPoolWithTag(size_t size)
{
    if (size == 0) {
        return NULL;
    }

    // Add space for the header
    size_t totalSize = sizeof(size_t) + size;
    void* pool = ExAllocatePoolWithTag(NonPagedPool, totalSize, 'MyTag');

    if (pool == NULL) {
        return NULL;
    }

    // Store the size in the header
    size_t* header = (size_t*)pool;
    *header = size;

    // Return a pointer to the data portion
    return (void*)(header + 1);
}
