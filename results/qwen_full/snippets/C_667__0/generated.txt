void scsi_eh_finish_cmd(struct scsi_cmnd *cmd)
{
    struct Scsi_Host *shost = cmd->device->host;

    /* Ensure the command is not racing with completions from the low-level driver */
    spin_lock_irq(shost->host_lock);
    cmd->eh_flags |= SCSI_EH_FINISHING;
    spin_unlock_irq(shost->host_lock);

    /* Complete the command and move it to the error-handler queue */
    scsi_finish_command(cmd);
}
