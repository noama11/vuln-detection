#include <linux/key.h>
#include <linux/errno.h>
#include <linux/uaccess.h>

long read_key_value(key_serial_t keyid, char __user *buffer, size_t buflen) {
    struct key *key;
    long ret = 0;
    void *payload;
    size_t datalen;

    key = key_lookup(keyid);
    if (IS_ERR(key))
        return PTR_ERR(key);

    if (!key_permission(key, KEY_NEED_READ)) {
        key_put(key);
        return -EACCES;
    }

    payload = key->payload.data;
    datalen = key->datalen;

    if (buflen < datalen) {
        ret = -ERANGE;
    } else {
        if (copy_to_user(buffer, payload, datalen)) {
            ret = -EFAULT;
        } else {
            ret = datalen;
        }
    }

    key_put(key);
    return ret;
}
