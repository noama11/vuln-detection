int handle_kex_init(int type, u_int32_t seq, void *ctxt) {
    ssh_kex_t *kex = ((ssh_session_t *)ctxt)->kex;
    int r;

    if (!kex) {
        return SSH_ERR_NO_KEX;
    }

    if ((r = sshpkt_discard_remaining(kex->session)) != 0) {
        return r;
    }

    if (kex->first_kex_follows) {
        if ((r = sshpkt_start(kex->session, SSH2_MSG_KEXINIT)) != 0) {
            return r;
        }
        if ((r = sshpkt_send(kex->session)) != 0) {
            return r;
        }
    }

    if ((r = kex->kex_method->kex(kex)) != 0) {
        return r;
    }

    return 0;
}
