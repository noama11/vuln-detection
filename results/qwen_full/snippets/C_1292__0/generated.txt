#include <pthread.h>
#include <stdint.h>
#include <stdlib.h>

// Assuming ThreadSafeRefCounted is a base class with reference counting
typedef struct ThreadSafeRefCounted ThreadSafeRefCounted;

// Forward declaration for ImageDecoderFactory
typedef struct ImageDecoderFactory ImageDecoderFactory;

// Define the ImageFrameGenerator class
typedef struct ImageFrameGenerator {
    ThreadSafeRefCounted base; // Assuming inheritance via composition
    int full_width;
    int full_height;
    int color_behavior;
    int decoding_status;
    int alpha_channel_info;
    pthread_mutex_t decoding_mutex;
    // Map of client mutexes (simplified as a pointer for now)
    void* client_mutex_map;
    ImageDecoderFactory* decoder_factory;
} ImageFrameGenerator;

// Constructor for ImageFrameGenerator
ImageFrameGenerator* ImageFrameGeneratorCreate(int width, int height, int color_behavior) {
    ImageFrameGenerator* self = (ImageFrameGenerator*)malloc(sizeof(ImageFrameGenerator));
    if (!self) {
        return NULL;
    }

    self->full_width = width;
    self->full_height = height;
    self->color_behavior = color_behavior;
    self->decoding_status = 0; // Assume 0 means decoding not started
    self->alpha_channel_info = 0;
    pthread_mutex_init(&self->decoding_mutex, NULL);
    self->client_mutex_map = NULL;
    self->decoder_factory = NULL;

    return self;
}

// Method to check if decoding has failed
int ImageFrameGeneratorDecodingFailed(ImageFrameGenerator* self) {
    return self->decoding_status < 0;
}

// Method to retrieve the full size of the image
void ImageFrameGeneratorGetFullSize(ImageFrameGenerator* self, int* width, int* height) {
    *width = self->full_width;
    *height = self->full_height;
}

void ImageFrameGeneratorGetSupportedSizes(ImageFrameGenerator* self) {
    // Placeholder for actual logic to retrieve supported sizes
}

// Factory method for creating instances with dependency injection
ImageFrameGenerator* ImageFrameGeneratorCreateWithFactory(int width, int height, int color_behavior, ImageDecoderFactory* factory) {
    ImageFrameGenerator* self = ImageFrameGeneratorCreate(width, height, color_behavior);
    if (!self) {
        return NULL;
    }

    self->decoder_factory = factory;
    return self;
}
