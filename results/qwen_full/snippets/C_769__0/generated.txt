int nfc_deactivate_target(struct sk_buff *skb, struct genl_info *info)
{
    struct nfc_dev *dev;
    int dev_idx, target_idx;
    int rc = 0;

    if (!info->attrs[NFC_ATTR_DEVICE_INDEX] || !info->attrs[NFC_ATTR_TARGET_INDEX])
        return -EINVAL;

    dev_idx = nla_get_u32(info->attrs[NFC_ATTR_DEVICE_INDEX]);
    target_idx = nla_get_u32(info->attrs[NFC_ATTR_TARGET_INDEX]);

    dev = nfc_get_device(dev_idx);
    if (!dev)
        return -ENODEV;

    rc = nfc_deactivate_target_dev(dev, target_idx);
    nfc_put_device(dev);

    return rc;
}
