long send_signal_to_thread(pid_t tgid, pid_t pid, int sig, siginfo_t *info) {
    if (tgid <= 0 || pid <= 0) {
        return -EINVAL;
    }

    if (sig <= 0 || sig > _NSIG) {
        return -EINVAL;
    }

    if (info == NULL) {
        return -EINVAL;
    }

    if (info->si_code < 0 || info->si_code >= _NSIGCODE) {
        return -EINVAL;
    }

    // Assume a helper function `do_send_signal_to_thread` exists in the same context
    return do_send_signal_to_thread(tgid, pid, sig, info);
}
