void readSingleImageFrame(MagickCore::Image *image, MagickCore::ExceptionInfo *exceptionInfo) {
    if (image != (MagickCore::Image *) NULL) {
        MagickCore::Image *nextFrame = image->next;

        while (nextFrame != (MagickCore::Image *) NULL) {
            MagickCore::Image *temp = nextFrame;
            nextFrame = nextFrame->next;
            temp->next = (MagickCore::Image *) NULL;
            DestroyImage(temp);
        }

        ReplaceImage(&image, image);
    }

    if (GetExceptionType(exceptionInfo) == UndefinedException && image == (MagickCore::Image *) NULL) {
        exceptionInfo = DestroyExceptionInfo(exceptionInfo);
        return;
    }

    if (image != (MagickCore::Image *) NULL) {
        ThrowImageException(image->exception);
    }
}
