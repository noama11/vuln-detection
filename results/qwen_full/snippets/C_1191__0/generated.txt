void write_image_to_clipboard(Image* image, const KURL& url, const String& title) {
    if (!image || !image->bitmap) {
        return;
    }

    // Write the image to the clipboard
    clipboard_write_image(image->bitmap);

    // If URL is valid, write additional metadata
    if (url.is_valid) {
        clipboard_write_bookmark(url.value, title.value);
        clipboard_write_html_markup(url.value, title.value);
    }

    // Commit the write operation
    clipboard_commit();
}
