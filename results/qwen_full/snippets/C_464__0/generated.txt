int iwgif_read_and_process_image(struct iwgifrcontext *rctx) {
    if (!rctx || !rctx->file) {
        return 0;
    }

    // Read image header
    char header[6];
    if (fread(header, 1, 6, rctx->file) != 6) {
        return 0;
    }

    rctx->width = (header[3] << 8) | header[2];
    rctx->height = (header[5] << 8) | header[4];
    rctx->interlaced = (header[6] & 0x40) ? 1 : 0;

    // Check for local color table
    if (header[6] & 0x80) {
        int color_table_size = 1 << ((header[6] & 0x07) + 1);
        rctx->local_color_table = malloc(color_table_size * sizeof(uint32_t));
        if (!rctx->local_color_table) {
            return 0;
        }

        for (int i = 0; i < color_table_size; ++i) {
            uint8_t r, g, b;
            if (fread(&r, 1, 1, rctx->file) != 1 ||
                fread(&g, 1, 1, rctx->file) != 1 ||
                fread(&b, 1, 1, rctx->file) != 1) {
                free(rctx->local_color_table);
                rctx->local_color_table = NULL;
                return 0;
            }
            rctx->local_color_table[i] = (r << 16) | (g << 8) | b;
        }

        // Handle transparency
        if (rctx->transparent_index != -1) {
            rctx->local_color_table[rctx->transparent_index] |= 0xFF000000; // Set alpha to 0
        }
    }

    // Read LZW code size
    if (fread(&rctx->lzw_code_size, 1, 1, rctx->file) != 1) {
        return 0;
    }

    if (rctx->lzw_code_size < 2 || rctx->lzw_code_size > 8) {
        return 0; // Invalid LZW code size
    }

    // Initialize screen context
    rctx->current_line = 0;
    rctx->pixels_read = 0;

    // Read subblocks until end of image
    while (!feof(rctx->file)) {
        uint8_t subblock_size;
        if (fread(&subblock_size, 1, 1, rctx->file) != 1) {
            break;
        }

        if (subblock_size == 0) {
            break; // End of image
        }

        uint8_t *subblock_data = malloc(subblock_size);
        if (!subblock_data) {
            return 0;
        }

        if (fread(subblock_data, 1, subblock_size, rctx->file) != subblock_size) {
            free(subblock_data);
            return 0;
        }

        // Process subblock with LZW decoding
        if (!iwgif_process_lzw_subblock(rctx, subblock_data, subblock_size)) {
            free(subblock_data);
            return 0;
        }

        free(subblock_data);
    }

    return 1;
}
