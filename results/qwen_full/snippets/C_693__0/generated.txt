int rds_atomic_op_process(struct rds_sock *rs, struct rds_message *rm, struct cmsghdr *cmsg)
{
    struct rds_atomic_op_args *args;
    struct rds_atomic_op *op;
    struct rds_notifier *notifier;
    int ret = -EINVAL;

    if (cmsg->cmsg_len < CMSG_LEN(sizeof(*args)))
        return ret;

    args = CMSG_DATA(cmsg);
    if (rs->atomic_op_active)
        return -EBUSY;

    op = kzalloc(sizeof(*op), GFP_KERNEL);
    if (!op)
        return -ENOMEM;

    switch (args->type) {
    case RDS_ATOMIC_OP_TYPE_A:
        op->type = RDS_ATOMIC_OP_TYPE_A;
        break;
    case RDS_ATOMIC_OP_TYPE_B:
        op->type = RDS_ATOMIC_OP_TYPE_B;
        break;
    default:
        kfree(op);
        return -EINVAL;
    }

    op->sg = kmalloc_array(args->sg_len, sizeof(*op->sg), GFP_KERNEL);
    if (!op->sg) {
        kfree(op);
        return -ENOMEM;
    }

    if (!IS_ALIGNED((unsigned long)args->local_addr, RDS_ADDR_ALIGNMENT)) {
        kfree(op->sg);
        kfree(op);
        return -EFAULT;
    }

    if (args->notify) {
        notifier = rds_notifier_alloc(rs, args->notify_cb);
        if (!notifier) {
            kfree(op->sg);
            kfree(op);
            return -ENOMEM;
        }
        op->notifier = notifier;
    }

    op->remote_addr = args->remote_addr;
    op->key = args->key;

    rs->atomic_op_active = true;
    rm->atomic_op = op;

    return 0;
}
