static OPJ_BOOL opj_j2k_write_sot(opj_j2k_t *p_j2k, OPJ_BYTE *p_data, OPJ_UINT32 p_total_data_size, OPJ_UINT32 *p_data_written, const opj_stream_private_t *p_stream, opj_event_mgr_t *p_manager) {
    (void)p_j2k;
    (void)p_manager;

    if (!p_data || !p_data_written || !p_stream) {
        return OPJ_FALSE;
    }

    if (p_total_data_size < 11) {
        return OPJ_FALSE;
    }

    p_data[0] = (OPJ_BYTE)(0xFF);
    p_data[1] = (OPJ_BYTE)(0x90); /* SOT marker */
    p_data[2] = (OPJ_BYTE)((p_j2k->current_tile_part_number >> 24) & 0xFF);
    p_data[3] = (OPJ_BYTE)((p_j2k->current_tile_part_number >> 16) & 0xFF);
    p_data[4] = (OPJ_BYTE)((p_j2k->current_tile_part_number >> 8) & 0xFF);
    p_data[5] = (OPJ_BYTE)(p_j2k->current_tile_part_number & 0xFF);
    p_data[6] = (OPJ_BYTE)((p_j2k->current_part_length >> 24) & 0xFF);
    p_data[7] = (OPJ_BYTE)((p_j2k->current_part_length >> 16) & 0xFF);
    p_data[8] = (OPJ_BYTE)((p_j2k->current_part_length >> 8) & 0xFF);
    p_data[9] = (OPJ_BYTE)(p_j2k->current_part_length & 0xFF);
    p_data[10] = (OPJ_BYTE)(p_j2k->number_of_parts);

    *p_data_written = 11;

    if (opj_stream_write(p_stream, p_data, 11) != 11) {
        return OPJ_FALSE;
    }

    return OPJ_TRUE;
}

static OPJ_BOOL opj_j2k_get_sot_values(OPJ_BYTE *p_header_data, OPJ_UINT32 p_header_size, OPJ_UINT32 *p_tile_no, OPJ_UINT32 *p_tot_len, OPJ_UINT32 *p_current_part, OPJ_UINT32 *p_num_parts, opj_event_mgr_t *p_manager) {
    (void)p_manager;

    if (!p_header_data || !p_tile_no || !p_tot_len || !p_current_part || !p_num_parts) {
        return OPJ_FALSE;
    }

    if (p_header_size < 11) {
        return OPJ_FALSE;
    }

    *p_tile_no = ((OPJ_UINT32)p_header_data[2] << 24) |
                 ((OPJ_UINT32)p_header_data[3] << 16) |
                 ((OPJ_UINT32)p_header_data[4] << 8) |
                 (OPJ_UINT32)p_header_data[5];

    *p_tot_len = ((OPJ_UINT32)p_header_data[6] << 24) |
                 ((OPJ_UINT32)p_header_data[7] << 16) |
                 ((OPJ_UINT32)p_header_data[8] << 8) |
                 (OPJ_UINT32)p_header_data[9];

    *p_current_part = (OPJ_UINT32)p_header_data[10];
    *p_num_parts = (OPJ_UINT32)p_header_data[10];

    return OPJ_TRUE;
}
