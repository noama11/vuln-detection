OM_uint32 process_spnego_continuation(OM_uint32 *minstat, gss_buffer_t buf, gss_ctx_id_t *ctx,
                                      gss_buffer_t *responseToken, gss_buffer_t *mechListMIC,
                                      OM_uint32 *negState, send_token_flag *return_token) {
    OM_uint32 status = 0;
    OM_uint32 major_status, minor_status;
    gss_OID mech_type;
    gss_buffer_desc output_token = GSS_C_EMPTY_BUFFER;

    *minstat = 0;

    // Verify token header and process negotiation response
    major_status = gss_unwrap(&minor_status, *ctx, buf, &output_token, NULL, NULL, NULL);
    if (major_status != GSS_S_COMPLETE) {
        status = GSS_S_FAILURE;
        goto cleanup;
    }

    // Check for errors in token format
    if (output_token.length <= 0 || output_token.value == NULL) {
        status = GSS_S_BAD_SIG;
        goto cleanup;
    }

    // Update negotiation state
    *negState = GSS_C_CONTINUE_NEEDED;

    // Prepare response token
    *responseToken = output_token;
    output_token = GSS_C_EMPTY_BUFFER;

    // Prepare mechListMIC if needed
    if (mechListMIC != NULL) {
        *mechListMIC = GSS_C_EMPTY_BUFFER;
    }

    // Indicate whether to send a token
    *return_token = send_token;

    status = GSS_S_CONTINUE_NEEDED;

cleanup:
    if (output_token.value != NULL) {
        gss_release_buffer(&minor_status, &output_token);
    }

    return status;
}
