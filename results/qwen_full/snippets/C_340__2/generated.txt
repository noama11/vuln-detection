int restore_xstate(struct xsave_struct *fx, u64 mask) {
    int ret = 0;
    u64 xcomp_bv = 0;

    if (boot_cpu_has(X86_FEATURE_XSAVES)) {
        xcomp_bv = xgetbv(XCR_XFEATURE_ENABLED_MASK);
        if ((xcomp_bv & mask) == mask) {
            asm volatile("xrstors %0, %1"
                         : /* no outputs */
                         : "m"(*fx), "n"(mask)
                         : "memory");
        } else {
            ret = -EINVAL;
        }
    } else {
        asm volatile("xrstor %0"
                     : /* no outputs */
                     : "m"(*fx)
                     : "memory");
    }

    return ret;
}
