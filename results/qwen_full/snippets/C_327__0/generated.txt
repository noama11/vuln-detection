ssize_t prepare_iovec(struct kiocb *kiocb, int rw, char __user *buf, unsigned long *nr_segs, struct iovec *iovec) {
    ssize_t len = kiocb->ki_nbytes;

    if (rw == READ) {
        if (!access_ok(VERIFY_WRITE, buf, len))
            return -EFAULT;
    } else {
        if (!access_ok(VERIFY_READ, buf, len))
            return -EFAULT;
    }

    iovec->iov_base = buf;
    iovec->iov_len = len;
    *nr_segs = 1;

    return 0;
}
