void set_original_address(struct msghdr *m, struct tipc_msg *msg)
{
    struct sockaddr_tipc *addr = (struct sockaddr_tipc *)m->msg_name;

    if (addr) {
        addr->family = AF_TIPC;
        addr->type = TIPC_ADDR_ID;
        addr->addr.id.port = tipc_msg_orig_port(msg);
        addr->addr.id.node = tipc_msg_orig_node(msg);
        m->msg_namelen = sizeof(struct sockaddr_tipc);
    }
}
