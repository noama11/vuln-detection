unsigned flac_process_audio_data(SF_PRIVATE *psf, unsigned len)
{
    unsigned bytes_read = 0;
    unsigned remaining = len;
    char *buffer = psf->buffer;
    size_t buffer_pos = psf->frame_pos;

    if (buffer_pos < psf->frame_size && psf->frame_remaining > 0) {
        size_t copy_len = psf->frame_remaining;
        if (copy_len > remaining)
            copy_len = remaining;
        memcpy(buffer + buffer_pos, psf->frame_buffer + psf->frame_pos, copy_len);
        buffer_pos += copy_len;
        psf->frame_pos += copy_len;
        bytes_read += copy_len;
        remaining -= copy_len;
    }

    while (remaining > 0 && !psf->eof_reached) {
        size_t decoded = flac_decode_audio(psf, buffer + buffer_pos, remaining);
        if (decoded == 0)
            break;
        buffer_pos += decoded;
        bytes_read += decoded;
        remaining -= decoded;
    }

    psf->frame_pos = buffer_pos;
    return bytes_read;
}
