size_t GetQuantumExtent(const Image *image, const QuantumInfo *quantum_info, const QuantumType quantum_type)
{
  size_t packet_size = 0;

  if (quantum_info == (QuantumInfo *) NULL)
    return(0);

  if (memcmp(quantum_info->signature, MagickSignature, sizeof(quantum_info->signature)) != 0)
    return(0);

  switch (quantum_type)
  {
    case RedQuantum:
    case GreenQuantum:
    case BlueQuantum:
    case OpacityQuantum:
    case IndexQuantum:
      packet_size = image->columns * sizeof(*quantum_info->quantum);
      break;
    case TrueColorQuantum:
    case GrayQuantum:
      packet_size = image->columns * 3 * sizeof(*quantum_info->quantum);
      break;
    case CMYKQuantum:
      packet_size = image->columns * 4 * sizeof(*quantum_info->quantum);
      break;
    case AllQuantum:
      packet_size = image->columns * 4 * sizeof(*quantum_info->quantum);
      break;
    default:
      break;
  }

  return(packet_size);
}
