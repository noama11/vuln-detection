int sock_recvmsg(struct kiocb *iocb, struct socket *sock, struct msghdr *msg,
                 size_t size, int msg_flags) {
    int ret = 0;
    struct sock *sk = sock->sk;

    while (1) {
        ret = sk->ops->recvmsg(iocb, sock, msg, size, msg_flags);
        if (ret > 0) {
            msg->msg_flags = msg_flags;
            sock_update_stats(sk, ret);
            break;
        } else if (ret == -EAGAIN) {
            if (msg_flags & MSG_DONTWAIT) {
                break;
            }
            wait_for_pending_notifications(sk);
            continue;
        } else {
            break;
        }
    }

    return ret;
}
