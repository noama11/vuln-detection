#include <sys/epoll.h>
#include <linux/types.h>
#include <linux/eventpoll.h>
#include <linux/fs.h>
#include <linux/file.h>
#include <linux/uaccess.h>
#include <linux/mutex.h>

int handle_epoll_ctl(int epfd, int op, int fd, struct epoll_event __user *event) {
    struct file *ep_file, *t_file;
    struct eventpoll *ep;
    int error;

    if (op != EPOLL_CTL_ADD && op != EPOLL_CTL_DEL && op != EPOLL_CTL_MOD)
        return -EINVAL;

    if (fd == epfd)
        return -EINVAL;

    ep_file = fget(epfd);
    if (!ep_file)
        return -EBADF;

    if (!is_file_epoll(ep_file)) {
        fput(ep_file);
        return -EINVAL;
    }

    ep = ep_file->private_data;
    t_file = fget(fd);
    if (!t_file) {
        fput(ep_file);
        return -EBADF;
    }

    if (!t_file->f_op || !t_file->f_op->poll) {
        fput(t_file);
        fput(ep_file);
        return -EINVAL;
    }

    mutex_lock(&ep->mtx);

    if (op == EPOLL_CTL_ADD) {
        error = ep_insert(ep, event, t_file, fd);
    } else if (op == EPOLL_CTL_DEL) {
        error = ep_remove(ep, t_file, fd);
    } else { /* EPOLL_CTL_MOD */
        error = ep_modify(ep, t_file, fd, event);
    }

    mutex_unlock(&ep->mtx);

    fput(t_file);
    fput(ep_file);

    return error;
}
