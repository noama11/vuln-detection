int parse_ccitt_facilities(unsigned char *p, struct rose_facilities_struct *facilities, int len) {
    int bytes_processed = 0;

    while (bytes_processed < len) {
        unsigned char byte = p[bytes_processed];

        if (byte == 0x00) {
            break;
        }

        unsigned char first_two_bits = (byte >> 6) & 0x03;

        switch (first_two_bits) {
            case 0x00:
                // Handle case where first two bits are 00
                // Example: source address
                if (bytes_processed + 1 >= len) {
                    return -1;
                }
                facilities->source_address = p[bytes_processed + 1];
                bytes_processed += 2;
                break;
            case 0x01:
                // Handle case where first two bits are 01
                // Example: destination address
                if (bytes_processed + 1 >= len) {
                    return -1;
                }
                facilities->destination_address = p[bytes_processed + 1];
                bytes_processed += 2;
                break;
            case 0x02:
                // Handle case where first two bits are 10
                // Example: source call sign
                if (bytes_processed + 1 >= len) {
                    return -1;
                }
                facilities->source_call_sign = p[bytes_processed + 1];
                bytes_processed += 2;
                break;
            case 0x03:
                // Handle case where first two bits are 11
                // Example: destination call sign
                if (bytes_processed + 1 >= len) {
                    return -1;
                }
                facilities->destination_call_sign = p[bytes_processed + 1];
                bytes_processed += 2;
                break;
            default:
                // Invalid case
                return -1;
        }
    }

    return bytes_processed;
}
