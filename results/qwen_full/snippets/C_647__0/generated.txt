void test_gimp_image_save_load(Gimp *gimp, gboolean with_unusual_stuff, gboolean compat_paths, gboolean use_gimp_2_8_features)
{
  GimpImage *image;
  gchar *temp_file;
  GimpImage *loaded_image;
  GError *error = NULL;

  image = create_test_gimp_image(gimp, with_unusual_stuff, compat_paths, use_gimp_2_8_features);
  g_return_if_fail(image != NULL);

  temp_file = g_strdup_printf("/tmp/gimp-test-save-load-%d.xcf", getpid());

  if (!gimp_file_save(GIMP_FILE_SAVE_EXPORT, NULL, image, NULL, temp_file, "XCF", &error))
    {
      g_critical("Failed to save image: %s", error->message);
      g_clear_error(&error);
      g_free(temp_file);
      return;
    }

  loaded_image = gimp_file_load(GIMP_FILE_LOAD_IMPORTED, NULL, temp_file, "XCF", &error);

  if (!loaded_image || error)
    {
      g_critical("Failed to load image: %s", error ? error->message : "Unknown error");
      g_clear_error(&error);
      g_free(temp_file);
      gimp_image_delete(image);
      return;
    }

  g_return_if_fail(loaded_image != NULL);

  gimp_image_delete(loaded_image);
  gimp_image_delete(image);
  g_unlink(temp_file);
  g_free(temp_file);
}
