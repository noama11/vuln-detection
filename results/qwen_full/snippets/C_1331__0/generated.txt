void generate_download_path(uint32_t download_id, const content_DownloadTargetCallback* callback, const base_FilePath* suggested_path) {
    // Ensure this function is called on the UI thread
    #ifdef DCHECK_CURRENTLY_ON
    DCHECK_CURRENTLY_ON(BrowserThread::UI);
    #endif

    // Create a new file name with ".crdownload" extension
    base_FilePath new_path;
    base_FilePath_ReplaceExtension(suggested_path, &new_path, ".crdownload");

    // Prepare the target disposition and danger type
    int target_disposition = TARGET_DISPOSITION_DEFAULT;
    int danger_type = DANGER_TYPE_NOT_DANGEROUS;

    // Invoke the callback with the generated path and parameters
    content_DownloadTargetCallback_Invoke(callback, new_path, target_disposition, danger_type);
}
