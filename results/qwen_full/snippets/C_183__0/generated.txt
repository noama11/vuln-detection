BOOL ReadScopeList(wStream* s, SCOPE_LIST* scopeList)
{
    UINT32 scopeCount;
    UINT32 i;

    if (!Stream_CheckAndLogRequiredLength(s, 4))
    {
        return FALSE;
    }

    Stream_Read_UINT32(s, scopeCount);

    scopeList->count = scopeCount;
    scopeList->scopes = (LICENSE_BLOB*)malloc(scopeCount * sizeof(LICENSE_BLOB));
    if (!scopeList->scopes)
    {
        return FALSE;
    }

    for (i = 0; i < scopeCount; i++)
    {
        if (!Stream_CheckAndLogRequiredLength(s, 4))
        {
            return FALSE;
        }

        Stream_Read_UINT32(s, scopeList->scopes[i].length);

        if (!Stream_CheckAndLogRequiredLength(s, scopeList->scopes[i].length))
        {
            return FALSE;
        }

        scopeList->scopes[i].data = (BYTE*)malloc(scopeList->scopes[i].length);
        if (!scopeList->scopes[i].data)
        {
            return FALSE;
        }

        Stream_Read(s, scopeList->scopes[i].data, scopeList->scopes[i].length);
    }

    return TRUE;
}
