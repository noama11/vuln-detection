CmdResult handleUnregisteredClientAuthSASL(const std::vector<std::string>& parameters, User* user) {
    if (!user->isUnregistered()) {
        return CMD_FAILURE;
    }

    if (!user->hasCap("sasl")) {
        return CMD_FAILURE;
    }

    if (!user->saslAuthenticator) {
        user->saslAuthenticator = std::make_unique<SASLAuthenticator>(user);
    }

    if (!user->saslAuthenticator->sendClientMessage(parameters)) {
        user->sendAnnounce("SASL authentication failed.");
        user->saslAuthenticator.reset();
        return CMD_FAILURE;
    }

    return CMD_SUCCESS;
}
