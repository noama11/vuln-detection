int set_timer_settings(struct k_itimer *timr, int flags, struct itimerspec64 *new_setting, struct itimerspec64 *old_setting) {
    struct itimerspec64 old;

    if (old_setting)
        *old_setting = timr->it_interval;

    timr->it_interval = (struct itimerspec64){0};

    if (timr->active) {
        int ret = cancel_timer(timr);
        if (ret != 0)
            return TIMER_RETRY;
    }

    timr->it_value = new_setting->it_value;
    timr->it_interval = new_setting->it_interval;

    if (timr->it_value.tv_nsec == 0 && timr->it_value.tv_sec == 0) {
        timr->active = 0;
    } else {
        timr->active = 1;
        arm_timer(timr);
    }

    return 0;
}
