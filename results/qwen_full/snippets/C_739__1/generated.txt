int genl_register_family(struct genl_family *family)
{
    int err = 0;
    struct genl_family *tmp;

    if (!family || !family->name || family->version < 0 || family->maxattr < 0)
        return -EINVAL;

    mutex_lock(&genl_mutex);

    list_for_each_entry(tmp, &genl_families, list) {
        if (strcmp(tmp->name, family->name) == 0) {
            mutex_unlock(&genl_mutex);
            return -EEXIST;
        }
    }

    if (family->id == GENL_ID_GENERATE) {
        family->id = genl_get_free_id();
        if (family->id < 0) {
            mutex_unlock(&genl_mutex);
            return family->id;
        }
    } else if (family->id >= GENL_ID_MIN && family->id < GENL_ID_MAX) {
        if (genl_id_to_family(family->id)) {
            mutex_unlock(&genl_mutex);
            return -EBUSY;
        }
    } else {
        mutex_unlock(&genl_mutex);
        return -EINVAL;
    }

    if (family->n_attrs > 0) {
        family->attrs = kcalloc(family->n_attrs + 1, sizeof(struct nla_policy), GFP_KERNEL);
        if (!family->attrs) {
            mutex_unlock(&genl_mutex);
            return -ENOMEM;
        }
    }

    for (int i = 0; i < family->n_multicast_groups; i++) {
        if (!family->mcgrps[i].name || family->mcgrps[i].id >= GENL_MCGRP_MAX) {
            err = -EINVAL;
            goto error_cleanup;
        }
    }

    list_add_tail(&family->list, &genl_families);

    mutex_unlock(&genl_mutex);

    genl_notify_family_registered(family);
    genl_notify_multicast_groups_registered(family);

    return 0;

error_cleanup:
    if (family->attrs)
        kfree(family->attrs);
    mutex_unlock(&genl_mutex);
    return err;
}
