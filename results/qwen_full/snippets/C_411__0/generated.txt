MagickBooleanType WriteINLINEImage(const ImageInfo *image_info, Image *image)
{
  Image
    *clone_image;

  MagickBooleanType
    status;

  MagickInfo
    *magick_info;

  char
    *base64_data;

  size_t
    base64_length;

  ExceptionInfo
    *exception;

  assert(image_info != (ImageInfo *) NULL);
  assert(image != (Image *) NULL);
  assert(image->signature == MagickSignature);

  if (LogMagickEvent(CoderEvent, __func__, "entering WriteINLINEImage"))
    (void) LogMagickEvent(CoderEvent, __func__, image_info->filename);

  magick_info = GetMagickInfo("INLINE", &image->exception);
  if (magick_info == (MagickInfo *) NULL)
    return MagickFalse;

  clone_image = CloneImage(image, image->columns, image->rows, MagickTrue, &image->exception);
  if (clone_image == (Image *) NULL)
    return MagickFalse;

  status = ConvertImage(clone_image, "BLOB", &clone_image, &image->exception);
  if (status == MagickFalse)
    return MagickFalse;

  base64_length = Base64Encode(clone_image->blob->length, (const unsigned char *) clone_image->blob->data, &base64_data);
  if (base64_data == (char *) NULL)
    return MagickFalse;

  exception = AcquireExceptionInfo();
  (void) OpenBlob(image_info, image, WriteBinaryBlob, exception);
  if (WriteBlobString(image, base64_data) != MagickTrue)
    status = MagickFalse;

  base64_data = (char *) RelinquishMagickMemory(base64_data);
  DestroyImage(clone_image);
  DestroyExceptionInfo(exception);

  return status;
}
