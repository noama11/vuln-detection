int virtio_gpu_object_create(struct virtio_gpu_device *vgdev, unsigned long size, bool kernel,
                             bool pinned, struct virtio_gpu_object **bo_ptr) {
    struct virtio_gpu_object *bo;
    int ret;

    if (!vgdev || !bo_ptr)
        return -EINVAL;

    *bo_ptr = NULL;

    bo = kzalloc(sizeof(*bo), GFP_KERNEL);
    if (!bo)
        return -ENOMEM;

    bo->size = size;
    bo->kernel = kernel;
    bo->pinned = pinned;

    ret = virtio_gpu_alloc_bo(vgdev, bo);
    if (ret)
        goto err_free_bo;

    ret = virtio_gpu_init_placement(bo);
    if (ret)
        goto err_free_bo;

    *bo_ptr = bo;
    return 0;

err_free_bo:
    kfree(bo);
    return ret;
}
