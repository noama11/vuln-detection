bool readDatabaseFromStarZone(StarZone &zone) {
    if (!zone.canReadDatabaseName()) {
        zone.logError("Failed to read database name from zone.");
        return false;
    }

    std::string dbName, tableName;
    if (!zone.read(dbName) || !zone.read(tableName)) {
        zone.logError("Failed to read database or table name.");
        return false;
    }

    setDatabaseName(dbName);
    setTableName(tableName);

    bool success = true;

    if (zone.isCompatibleWith("SQL")) {
        std::string sqlCreate, sqlInsert;
        if (!zone.read(sqlCreate)) {
            zone.logError("Failed to read SQL CREATE statement.");
            success = false;
        }
        if (!zone.read(sqlInsert)) {
            zone.logError("Failed to read SQL INSERT statement.");
            success = false;
        }
        setSqlCreateStatement(sqlCreate);
        setSqlInsertStatement(sqlInsert);
    }

    std::vector<std::string> dataEntries;
    while (zone.hasMoreData()) {
        std::string entry;
        if (!zone.read(entry)) {
            zone.logError("Failed to read data entry.");
            success = false;
            break;
        }
        dataEntries.push_back(entry);
    }

    setDataEntries(dataEntries);

    zone.logState("Database read completed with partial success: " + std::to_string(success));
    zone.closeRecord();

    return success;
}
