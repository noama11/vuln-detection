int dvb_frontend_send_diseqc_msg(struct dvb_frontend *fe, struct dvb_diseqc_master_cmd *d)
{
	int ret;
	struct dvb_frontend_private *state = fe->demodulator_priv;
	struct dvb_diseqc_master_cmd *cmd = &state->diseqc_cmd;
	unsigned int len;

	if (d->msg[0] < 0x04)
		len = 3;
	else
		len = 6;

	memcpy(cmd->msg, d->msg, len);
	cmd->msg_len = len;

	if (d->msg[0] < 0x04)
		cmd->msg[0] |= 0x04;
	else
		cmd->msg[0] &= 0x0f;

	if (d->msg[0] & 0x04)
		cmd->msg[1] = 0x30;

	if (state->diseqc_toneburst)
		cmd->msg[1] |= 0x01;

	if (state->diseqc_toneburst == 2)
		cmd->msg[1] |= 0x02;

	ret = dvb_frontend_wait_for_lnb_ready(fe);
	if (ret)
		return ret;

	msleep(100);

	ret = fe->ops.send_diseqc_master_cmd(fe, cmd);
	if (ret)
		return ret;

	msleep(100);

	return 0;
}
