ByteVector extract_subrange(uint index, uint length) {
    if (index >= this->size()) {
        return ByteVector();
    }

    uint end_index = std::min(index + length, static_cast<uint>(this->size()));
    ByteVector result;
    for (uint i = index; i < end_index; ++i) {
        result.push_back((*this)[i]);
    }
    return result;
}
