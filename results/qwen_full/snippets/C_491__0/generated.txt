gboolean handle_new_file_transfer(gpointer data, gint fd, b_input_condition cond) {
    TransferData *transfer_data = (TransferData *)data;
    Transfer *transfer = transfer_data->transfer;
    Connection *connection = transfer_data->connection;
    Buddy *buddy = connection_find_buddy(connection, transfer->buddy_id);

    if (!buddy) {
        return FALSE;
    }

    transfer->buddy = buddy;
    transfer->fd = fd;

    setup_transfer_callbacks(transfer);
    associate_transfer_data(transfer, transfer_data);

    return FALSE;
}
