CURLcode close_curl_easy_handle(struct Curl_easy *data) {
    if (!data) {
        return CURLE_OK;
    }

    /* Clear timers */
    Curl_cleanup_timer(data);

    /* Detach from multi handle if part of one */
    if (data->multi) {
        Curl_multi_remove_handle(data->multi, data);
    }

    /* Clean up multi handles */
    if (data->multi_easy) {
        Curl_multi_easy_cleanup(data->multi_easy);
    }

    /* Destroy timeout list */
    Curl_llist_destroy(&data->timeouts, NULL);

    /* Free allocated memory for states and configurations */
    Curl_safefree(data->set.str[STRING_COOKIEJAR]);
    Curl_safefree(data->set.str[STRING_COOKIEFILE]);
    Curl_safefree(data->set.str[STRING_USERPWD]);
    Curl_safefree(data->set.str[STRING_PROXY]);
    Curl_safefree(data->set.str[STRING_INTERFACE]);
    Curl_safefree(data->set.str[STRING_NETRC_FILE]);
    Curl_safefree(data->set.str[STRING_SSLCERT]);
    Curl_safefree(data->set.str[STRING_SSLKEY]);
    Curl_safefree(data->set.str[STRING_SSL_CIPHER_LIST]);
    Curl_safefree(data->set.str[STRING_SSL_ECDH_CURVE]);
    Curl_safefree(data->set.str[STRING_SSL_PINNEDPUBLICKEY]);
    Curl_safefree(data->set.str[STRING_CAINFO]);
    Curl_safefree(data->set.str[STRING_CAPATH]);
    Curl_safefree(data->set.str[STRING_RANDOM_FILE]);
    Curl_safefree(data->set.str[STRING_EGDSOCKET]);
    Curl_safefree(data->set.str[STRING_SSL_CRLFILE]);
    Curl_safefree(data->set.str[STRING_SSL_ISSUERCERT]);
    Curl_safefree(data->set.str[STRING_ALTSVC]);
    Curl_safefree(data->set.str[STRING_MAIL_FROM]);
    Curl_safefree(data->set.str[STRING_MAIL Rcpt]);
    Curl_safefree(data->set.str[STRING_TFTP_NO_OPTIONS]);
    Curl_safefree(data->set.str[STRING_TFTP_BLKSIZE]);
    Curl_safefree(data->set.str[STRING_TFTP_MODE]);
    Curl_safefree(data->set.str[STRING_TFTP_TIMEOUT]);
    Curl_safefree(data->set.str[STRING_TFTP_WINDOW_SIZE]);

    /* Close SSL sessions */
    Curl_ssl_close(data);

    /* Clean up redirect info */
    Curl_safefree(data->change.url);
    Curl_safefree(data->change.referer);
    Curl_safefree(data->change.uh);

    /* Clean up share and wildcard structures */
    if (data->share) {
        Curl_share_cleanup(data->share);
    }

    if (data->wildcard) {
        Curl_wildcard_dtor(data->wildcard);
    }

    /* Free the easy handle itself */
    Curl_safefree(data);

    return CURLE_OK;
}
