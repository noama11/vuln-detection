AppProto AppLayerProtoDetectDetect(AppLayerProtoDetectThreadCtx *tctx, Flow *f,
                                   uint8_t *buf, uint32_t buflen, uint8_t ipproto,
                                   uint8_t direction) {
    AppProto alproto = ALPROTO_UNKNOWN;

    if (!FlowGetAppProto(f)) {
        if (tctx->pm) {
            alproto = AppLayerProtoDetectPM(tctx, f, buf, buflen, ipproto, direction);
            if (alproto != ALPROTO_UNKNOWN)
                goto end;
        }

        if (tctx->pp) {
            alproto = AppLayerProtoDetectPP(tctx, f, buf, buflen, ipproto, direction);
            if (alproto != ALPROTO_UNKNOWN)
                goto end;
        }

        if (tctx->expectations) {
            alproto = AppLayerProtoDetectExpectation(f, ipproto, direction);
            if (alproto != ALPROTO_UNKNOWN)
                goto end;
        }
    }

end:
    return alproto;
}
