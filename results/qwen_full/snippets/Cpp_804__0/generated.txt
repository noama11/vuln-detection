void handleRoutingData(uint64_t connId, RoutingData& routingData) {
    auto pipeline = getRoutingPipeline(connId);
    if (!pipeline) {
        LOG(INFO) << "Routing pipeline not found for connection ID: " << connId;
        return;
    }

    pipeline->pauseTransport();

    uint32_t acceptorId = std::hash<std::string>()(routingData.toString()) % numAcceptors;

    Acceptor* acceptor = getAcceptor(acceptorId);
    if (!acceptor) {
        LOG(ERROR) << "Failed to find acceptor with ID: " << acceptorId;
        return;
    }

    acceptor->thread->runInEventBaseThread([&]() {
        auto socket = pipeline->getSocket();
        socket->attachEventBase(acceptor->eventBase);

        auto newPipeline = createPipeline(routingData);
        acceptor->addServerConnection(newPipeline);

        newPipeline->activateTransport();
        newPipeline->readBufferedBytes();
    });
}
