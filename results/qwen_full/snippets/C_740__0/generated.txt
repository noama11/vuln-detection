int smb2_handle_write(unsigned int xid, struct cifs_io_parms *io_parms, unsigned int *nbytes, struct kvec *iov, int n_vec) {
    int rc = 0;
    struct smb2_write_request *req;
    struct smb2_write_response *resp;
    struct TCP_Server_Info *server;

    if (!io_parms || !nbytes || !iov || n_vec <= 0) {
        return -EINVAL;
    }

    server = io_parms->cfile->tcon->ses->server;
    if (!server || !serverCanDo(server, CIFS_CAN_WRITE)) {
        return -EIO;
    }

    req = cifs_buf_get();
    if (!req) {
        return -ENOMEM;
    }

    resp = cifs_buf_get();
    if (!resp) {
        cifs_buf_release(req);
        return -ENOMEM;
    }

    req->hdr.MessageId = cpu_to_le64(xid);
    req->PersistentFileId = io_parms->persistent_fid;
    req->VolatileFileId = io_parms->volatile_fid;
    req->Offset = cpu_to_le64(io_parms->offset);
    req->Remaining = 0;
    req->DataOffset = cpu_to_le16(offsetof(struct smb2_write_request, Buffer));
    req->DataLength = cpu_to_le32(io_parms->length);
    req->Channel = 0;

    rc = cifs_send_recv(io_parms->tcon, server, req, resp, CIFS_WRITE_OP);
    if (rc) {
        cifs_buf_release(req);
        cifs_buf_release(resp);
        return rc;
    }

    *nbytes = le32_to_cpu(resp->DataLength);
    cifs_buf_release(req);
    cifs_buf_release(resp);
    return 0;
}
