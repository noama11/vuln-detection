struct block_device *get_block_device(dev_t dev, struct super_block *sb) {
    struct block_device *bdev = open_block_device(dev, O_RDWR | O_EXCL);
    if (!bdev) {
        printk(KERN_ERR "Failed to open block device for dev %d:%d\n",
               MAJOR(dev), MINOR(dev));
        return NULL;
    }

    return bdev;
}
