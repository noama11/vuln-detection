int netlink_dump(struct sock *sk) {
    struct sk_buff *skb;
    int err = 0;

    mutex_lock(&sk->sk_lock);
    if (sk->sk_callback_running) {
        mutex_unlock(&sk->sk_lock);
        return -EBUSY;
    }

    if (sk->sk_rcvbuf_used >= sk->sk_rcvbuf) {
        mutex_unlock(&sk->sk_lock);
        return -ENOBUFS;
    }

    skb = alloc_skb(sk->sk_rcvbuf / 2, GFP_KERNEL);
    if (!skb) {
        mutex_unlock(&sk->sk_lock);
        return -ENOMEM;
    }

    err = sk->sk_prot->dump(sk, skb);
    if (err < 0) {
        kfree_skb(skb);
        mutex_unlock(&sk->sk_lock);
        return err;
    }

    if (sk->sk_filter) {
        skb = sk_filter(sk, skb);
        if (!skb) {
            mutex_unlock(&sk->sk_lock);
            return 0;
        }
    }

    skb_queue_tail(&sk->sk_receive_queue, skb);
    mutex_unlock(&sk->sk_lock);
    return 0;
}
