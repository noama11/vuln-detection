int sock_recvmsg(struct kiocb *iocb, struct socket *sock, struct msghdr *msg, size_t size, int flags) {
    struct sock *sk = sock->sk;
    int copied = 0;
    struct sk_buff *skb;

    if (sk->sk_state != TCP_ESTABLISHED)
        return -ENOTCONN;

    if (flags & ~(MSG_PEEK | MSG_DONTWAIT))
        return -EINVAL;

    skb = skb_dequeue(&sk->sk_receive_queue);
    if (!skb)
        return -EAGAIN;

    copied = min(skb->len, size);
    if (copied > 0) {
        if (memcpy_to_msg(msg, skb->data, copied)) {
            kfree_skb(skb);
            return -EFAULT;
        }

        if (!(flags & MSG_PEEK)) {
            skb_pull(skb, copied);
            if (skb->len == 0)
                kfree_skb(skb);
            else
                skb_queue_head(&sk->sk_receive_queue, skb);
            sk->sk_data_processed += copied;
        } else {
            skb_queue_head(&sk->sk_receive_queue, skb);
        }
    } else {
        kfree_skb(skb);
    }

    return copied;
}
