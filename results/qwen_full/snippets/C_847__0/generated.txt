void readSingleImageFrame(MagickCore::Image *image, MagickCore::ExceptionInfo *exceptionInfo) {
  if (image != (MagickCore::Image *) NULL) {
    if (image->next != (MagickCore::Image *) NULL) {
      MagickCore::Image *nextImage = image->next;
      while (nextImage != (MagickCore::Image *) NULL) {
        MagickCore::Image *temp = nextImage->next;
        nextImage->next = (MagickCore::Image *) NULL;
        DestroyImage(nextImage);
        nextImage = temp;
      }
    }
    ReplaceImage(&image, image);
  }

  if (exceptionInfo->severity == UndefinedException && image == (MagickCore::Image *) NULL) {
    exceptionInfo->severity = NoException;
    if (!Is QuietMode) {
      (void) ThrowMagickException(exceptionInfo, GetMagickModule(), WarningError, "ImageReadWarning", "No image read");
    }
    exceptionInfo = (MagickCore::ExceptionInfo *) DestroyExceptionInfo(exceptionInfo);
  }

  image_exception(image, exceptionInfo);
}
