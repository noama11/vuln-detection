Image *ReadGroup4Image(const ImageInfo *image_info, ExceptionInfo *exception) {
  Image *image = NULL;
  FILE *file = NULL;
  FILE *temp_file = NULL;
  size_t length;
  unsigned char *buffer = NULL;

  if (!image_info || !exception)
    return NULL;

  file = fopen(image_info->filename, "rb");
  if (!file) {
    ThrowFileException(exception, FileOpenError, image_info->filename, "unknown");
    return NULL;
  }

  temp_file = tmpfile();
  if (!temp_file) {
    fclose(file);
    ThrowFileException(exception, FileOpenError, "temporary file", "unknown");
    return NULL;
  }

  // Write TIFF headers and image data to the temporary file
  // (Implementation of writing TIFF headers and data is omitted for brevity)

  // Read the TIFF image back into memory
  fseek(temp_file, 0, SEEK_END);
  length = ftell(temp_file);
  fseek(temp_file, 0, SEEK_SET);

  buffer = (unsigned char *) AcquireMemory(length);
  if (!buffer) {
    fclose(file);
    fclose(temp_file);
    ThrowMemoryException(exception);
    return NULL;
  }

  if (fread(buffer, 1, length, temp_file) != length) {
    fclose(file);
    fclose(temp_file);
    RelinquishMemory(buffer);
    ThrowFileException(exception, FileReadError, image_info->filename, "unknown");
    return NULL;
  }

  // Create and populate the Image structure
  image = CloneImageInfo(image_info);
  if (!image) {
    fclose(file);
    fclose(temp_file);
    RelinquishMemory(buffer);
    ThrowMemoryException(exception);
    return NULL;
  }

  image->filename = (char *) AcquireString(image_info->filename);
  image->format = (char *) AcquireString("TIFF");

  // Clean up
  fclose(file);
  fclose(temp_file);
  RelinquishMemory(buffer);

  return image;
}
