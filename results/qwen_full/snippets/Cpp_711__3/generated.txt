bool checkUserPassword(std::string& user_password, std::string const& owner_password, QPDF::EncryptionData const& data) {
    // Compute RC4 key based on user and owner passwords
    std::string rc4_key = computeRC4Key(user_password, owner_password);

    // Retrieve O data from encryption data
    std::string o_data = data.o;

    // Iterate RC4 algorithm to generate new user password
    std::string new_user_password = iterateRC4(rc4_key, o_data);

    // Check if the new password matches the provided data
    if (new_user_password == data.u) {
        user_password = new_user_password;
        return true;
    }

    return false;
}
