int read_quantization_matrix(MpegEncContext *s, GetBitContext *gb) {
    int i;

    if (s->workaround_bugs & FF_BUG_QP_RD) {
        s->intra_quantizer = get_bits(gb, 8);
    }

    if (gb->index + 64 * 8 > gb->size_in_bits) {
        return AVERROR_INVALIDDATA;
    }

    for (i = 0; i < 64; i++) {
        s->intra_matrix[zigzag_direct[i]] = get_bits(gb, 8);
    }

    return 0;
}
