int cipso_v4_doi_add_common(struct genl_info *info, struct cipso_v4_doi *doi_def)
{
    struct nlattr *attr;
    int i, ret = 0;
    int tag_count = 0;

    attr = info->attrs[...]; // Assuming appropriate attribute index for tag list

    if (!attr)
        return -EINVAL;

    // Process the nested tag list
    for (i = 0; i < CIPSO_V4_TAG_MAX; i++) {
        struct nlattr *tag_attr = nla_find(attr, info->netns, i + 1);

        if (!tag_attr)
            continue;

        if (tag_count >= CIPSO_V4_MAX_TAGS)
            return -EINVAL;

        // Copy tag data into DOI structure
        memcpy(&doi_def->tags[tag_count], nla_data(tag_attr), nla_len(tag_attr));
        tag_count++;
    }

    // Mark unused tag storage with invalid tag value
    for (; tag_count < CIPSO_V4_TAG_MAX; tag_count++) {
        doi_def->tags[tag_count].type = CIPSO_V4_TAG_INVALID;
    }

    return ret;
}
