void LayoutShellWindowFrameView() {
    // Retrieve the preferred size of the close button
    int close_button_width, close_button_height;
    GetPreferredSize(&close_button_width, &close_button_height);

    // Calculate vertical and horizontal offsets based on the caption height
    int caption_height = GetCaptionHeight();
    int vertical_offset = (caption_height - close_button_height) / 2;
    int horizontal_offset = (caption_height - close_button_width) / 2;

    // Position the close button within the frame
    PositionCloseButton(horizontal_offset, vertical_offset);
}
