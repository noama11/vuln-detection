int flush_key_notify_policy(const struct km_event *c)
{
    struct sk_buff *skb;
    int err;

    skb = alloc_skb(KEY_NOTIFY_MSG_SIZE, GFP_KERNEL);
    if (!skb) {
        return -ENOBUFS;
    }

    /* Initialize the header with information from the event structure */
    init_key_mgmt_msg(skb, c->seq, c->port_id);

    err = broadcast_key_mgmt_msg(skb);
    if (err < 0) {
        kfree_skb(skb);
        return err;
    }

    return 0;
}
