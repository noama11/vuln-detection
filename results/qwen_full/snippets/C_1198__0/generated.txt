bool finalize_async_pixel_transfer(base::subtle::Atomic32 submit_count) {
    SharedMemoryBuffer* buffer = get_shared_memory_buffer();
    if (!buffer || !buffer->is_valid()) {
        return false;
    }

    MemoryParameters params;
    setup_memory_parameters(&params, buffer);

    CompletionObserver* observer = create_completion_observer();
    if (!observer) {
        return false;
    }

    AsyncPixelTransferManager* manager = get_async_pixel_transfer_manager();
    if (!manager->request_notification_on_completion(observer)) {
        return false;
    }

    return manager->add_pending_transfer(submit_count, params);
}
