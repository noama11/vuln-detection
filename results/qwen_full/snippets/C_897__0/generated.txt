OPJ_BOOL read_and_decode_rle4_bitmap(FILE* IN, OPJ_UINT8* pData, OPJ_UINT32 stride, OPJ_UINT32 width, OPJ_UINT32 height) {
    OPJ_UINT32 y;
    OPJ_UINT8* row_ptr;
    OPJ_UINT8 byte;
    OPJ_UINT32 count;
    OPJ_UINT32 x = 0;

    for (y = 0; y < height; ++y) {
        row_ptr = pData + y * stride;
        x = 0;

        while (x < width) {
            if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                return OPJ_FALSE;
            }

            if (byte == 0x00) {
                if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                    return OPJ_FALSE;
                }

                if (byte == 0x00) {
                    // End of line
                    break;
                } else if (byte == 0x01) {
                    // End of packet
                    return OPJ_TRUE;
                } else if (byte == 0x02) {
                    // Move to absolute position
                    if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                        return OPJ_FALSE;
                    }
                    x = (OPJ_UINT32)byte * 2;
                    continue;
                } else {
                    // Absolute mode
                    count = (OPJ_UINT32)byte;
                    if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                        return OPJ_FALSE;
                    }
                    count = (count << 8) | byte;

                    for (OPJ_UINT32 i = 0; i < count; ++i) {
                        if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                            return OPJ_FALSE;
                        }
                        row_ptr[x++] = byte;
                        if (x >= width) {
                            break;
                        }
                    }
                }
            } else {
                // Encoded mode
                OPJ_UINT32 num_pixels = (OPJ_UINT32)byte;
                if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                    return OPJ_FALSE;
                }
                num_pixels = (num_pixels << 8) | byte;

                for (OPJ_UINT32 i = 0; i < num_pixels; ++i) {
                    if (fread(&byte, sizeof(OPJ_UINT8), 1, IN) != 1) {
                        return OPJ_FALSE;
                    }
                    row_ptr[x++] = byte;
                    if (x >= width) {
                        break;
                    }
                }
            }
        }

        if (x < width) {
            return OPJ_FALSE;
        }
    }

    return OPJ_TRUE;
}
