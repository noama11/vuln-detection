#include <stdio.h>
#include <ctype.h>

#define MAX_WORD_SIZE 1024

void process_file_characters(FILE *file, char *word, size_t word_size) {
    int c;
    int i = 0;
    int escape = 0;
    int in_quotes = 0;

    while ((c = fgetc(file)) != EOF) {
        if (escape) {
            escape = 0;
            if (c == 'n') {
                word[i++] = '\n';
            } else if (c == 't') {
                word[i++] = '\t';
            } else if (c == 'b') {
                word[i++] = '\b';
            } else if (c == 'r') {
                word[i++] = '\r';
            } else if (c == 'f') {
                word[i++] = '\f';
            } else if (c == 'a') {
                word[i++] = '\a';
            } else if (c == 'v') {
                word[i++] = '\v';
            } else if (c == '\\') {
                word[i++] = '\\';
            } else if (c == '\'') {
                word[i++] = '\'';
            } else if (c == '"') {
                word[i++] = '"';
            } else if (c == '?') {
                word[i++] = '?';
            } else if (isdigit(c)) {
                // Handle octal or hexadecimal escape sequences
                // This is a simplified version and may not handle all cases
                int value = 0;
                if (c == '0' || c == '1' || c == '2') {
                    // Octal escape sequence
                    value = c - '0';
                    int d = fgetc(file);
                    if (isdigit(d) && d - '0' < 8) {
                        value = value * 8 + (d - '0');
                        d = fgetc(file);
                        if (isdigit(d) && d - '0' < 8) {
                            value = value * 8 + (d - '0');
                        } else {
                            ungetc(d, file);
                        }
                    } else {
                        ungetc(d, file);
                    }
                } else if (c == 'x' || c == 'X') {
                    // Hexadecimal escape sequence
                    int d = fgetc(file);
                    if (isxdigit(d)) {
                        value = hex_digit_value(d);
                        d = fgetc(file);
                        if (isxdigit(d)) {
                            value = value * 16 + hex_digit_value(d);
                        } else {
                            ungetc(d, file);
                        }
                    } else {
                        ungetc(d, file);
                    }
                } else {
                    ungetc(c, file);
                    c = '\\';
                    word[i++] = c;
                }
                word[i++] = value;
            } else {
                // Invalid escape sequence
                word[i++] = '\\';
                word[i++] = c;
            }
        } else if (c == '\\') {
            escape = 1;
        } else if (c == '"') {
            in_quotes = !in_quotes;
            word[i++] = c;
        } else if (isspace(c)) {
            if (!in_quotes) {
                if (c == '#') {
                    // Skip the rest of the line as a comment
                    while ((c = fgetc(file)) != EOF && c != '\n') {
                        // Skip characters until newline
                    }
                    if (c == '\n') {
                        word[i++] = '\n';
                    }
                } else {
                    break; // End of word
                }
            } else {
                word[i++] = c;
            }
        } else {
            word[i++] = c;
        }

        if (i >= word_size - 1) {
            break; // Prevent buffer overflow
        }
    }

    word[i] = '\0'; // Null-terminate the word
}

int hex_digit_value(int c) {
    if (c >= '0' && c <= '9') {
        return c - '0';
    } else if (c >= 'a' && c <= 'f') {
        return c - 'a' + 10;
    } else if (c >= 'A' && c <= 'F') {
        return c - 'A' + 10;
    }
    return 0; // Invalid hex digit
}
