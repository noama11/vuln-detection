long sys_semtimedop(int semid, struct oabi_sembuf __user *tsops, unsigned nsops, const struct timespec __user *timeout) {
    struct oabi_sembuf *sops;
    long ret = -EINVAL;
    int i;

    if (nsops == 0)
        return -EINVAL;

    sops = kmalloc_array(nsops, sizeof(*sops), GFP_KERNEL);
    if (!sops)
        return -ENOMEM;

    if (copy_from_user(sops, tsops, nsops * sizeof(*sops)))
        goto out_free;

    if (timeout) {
        struct timespec ktimeout;
        if (copy_from_user(&ktimeout, timeout, sizeof(ktimeout)))
            goto out_free;
        ret = do_semtimedop(semid, sops, nsops, &ktimeout);
    } else {
        ret = do_semtimedop(semid, sops, nsops, NULL);
    }

out_free:
    kfree(sops);
    return ret;
}
