plist_t *parse_string_node(const char **bnode, uint64_t size) {
    plist_t *node = (plist_t *)malloc(sizeof(plist_t));
    if (!node) {
        return NULL;
    }

    char *str = (char *)malloc(size + 1);
    if (!str) {
        free(node);
        return NULL;
    }

    memcpy(str, *bnode, size);
    str[size] = '\0';

    node->type = PLIST_STRING;
    node->string_value = str;
    node->length = size;

    return node;
}
