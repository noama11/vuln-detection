int cdrom_ioctl_check_media_change(struct cdrom_device_info *cdi, unsigned long arg)
{
    int ret = -ENOSYS;

    if (!cdi || !cdi->ops || !cdi->ops->check_media_change)
        return -ENOSYS;

    if (arg > CDROM_MAX_DISC_COUNT)
        return -EINVAL;

    if (arg == 0) {
        ret = cdi->ops->check_media_change(cdi, arg);
        return ret;
    }

    struct cdrom_changer_info *info = kmalloc(sizeof(*info), GFP_KERNEL);
    if (!info)
        return -ENOMEM;

    ret = cdi->ops->get_changer_status(cdi, info);
    if (ret < 0)
        goto out;

    if (arg <= info->num_discs)
        ret = 0;
    else
        ret = -EINVAL;

out:
    kfree(info);
    return ret;
}
