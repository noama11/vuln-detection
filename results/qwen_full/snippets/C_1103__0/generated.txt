// SSLErrorInfo constructor
void SSLErrorInfo_Init(SSLErrorInfo* self, const char* title, const char* details, const char* short_description, const char** extra_info, size_t extra_info_size) {
    strncpy(self->title, title, sizeof(self->title) - 1);
    strncpy(self->details, details, sizeof(self->details) - 1);
    strncpy(self->short_description, short_description, sizeof(self->short_description) - 1);

    self->extra_info_size = extra_info_size;
    for (size_t i = 0; i < extra_info_size; ++i) {
        strncpy(self->extra_info[i], extra_info[i], sizeof(self->extra_info[i]) - 1);
    }
}
