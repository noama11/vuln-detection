struct sk_buff *fragment_udp_packet(struct sk_buff *skb, u32 features) {
    int mss = skb->dev->mtu - sizeof(struct ipv6hdr) - sizeof(struct udphdr);
    struct sk_buff *frag_skb;

    if (skb->len <= mss)
        return skb;

    if (!(features & NETIF_F_SG)) {
        return ERR_PTR(-EINVAL);
    }

    skb_checksum_help(skb);

    skb_pull(skb, sizeof(struct udphdr));
    skb_push(skb, sizeof(struct udphdr));

    frag_skb = skb_segment(skb, features);
    if (IS_ERR(frag_skb))
        return frag_skb;

    return frag_skb;
}
