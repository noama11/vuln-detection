int map_user_io(struct request_queue *q, struct request *rq, struct rq_map_data *map_data,
                const struct iov_iter *iter, gfp_t gfp_mask) {
    struct bio *bio = NULL;
    struct bio *prev_bio = NULL;
    size_t offset = 0;
    ssize_t bytes;
    int ret = 0;

    while ((bytes = iov_iter_count(iter)) > 0) {
        struct bio *new_bio = bio_alloc(GFP_NOIO, 1);
        if (!new_bio) {
            ret = -EINVAL;
            goto unmap_and_exit;
        }

        new_bio->bi_iter = iter->iov->iov_len;
        new_bio->bi_bdev = NULL;
        new_bio->bi_private = NULL;
        new_bio->bi_end_io = NULL;
        new_bio->bi_next = NULL;

        if (bio) {
            bio->bi_next = new_bio;
        } else {
            bio = new_bio;
        }

        if (map_data && !map_data->offset) {
            if (offset != 0) {
                ret = -EINVAL;
                goto unmap_and_exit;
            }
        }

        if (bio_map_user_iovec(q, iter, new_bio, gfp_mask) < 0) {
            ret = -EINVAL;
            goto unmap_and_exit;
        }

        offset += bytes;
        iov_iter_advance(iter, bytes);
        prev_bio = new_bio;
    }

    if (!bio) {
        ret = -EINVAL;
        goto exit;
    }

    rq->bio = bio;
    rq->biotail = prev_bio;
    rq->cmd_flags |= REQ_USER_MAPPED;

exit:
    return ret;

unmap_and_exit:
    if (bio) {
        bio_unmap_user(bio);
    }
    goto exit;
}
