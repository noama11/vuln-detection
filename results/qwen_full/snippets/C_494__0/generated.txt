int vc4_prepare_cmdlist_and_shader_records(struct drm_device *dev, struct vc4_exec_info *exec)
{
    int ret;
    struct vc4_dev *vc4 = to_vc4_dev(dev);
    struct vc4_bo *bo;
    void *cmdlist, *shader_records;
    unsigned int cmdlist_size, shader_records_size;

    if (!exec->cmdlist_size || !exec->shader_records_size)
        return -EINVAL;

    cmdlist_size = exec->cmdlist_size;
    shader_records_size = exec->shader_records_size;

    cmdlist = kmalloc(cmdlist_size, GFP_KERNEL);
    if (!cmdlist)
        return -ENOMEM;

    shader_records = kmalloc(shader_records_size, GFP_KERNEL);
    if (!shader_records) {
        ret = -ENOMEM;
        goto free_cmdlist;
    }

    if (copy_from_user(cmdlist, u64_to_user_ptr(exec->cmdlist_ptr), cmdlist_size)) {
        ret = -EFAULT;
        goto free_shader_records;
    }

    if (copy_from_user(shader_records, u64_to_user_ptr(exec->shader_records_ptr), shader_records_size)) {
        ret = -EFAULT;
        goto free_shader_records;
    }

    ret = vc4_validate_cmdlist(cmdlist, cmdlist_size);
    if (ret)
        goto free_shader_records;

    ret = vc4_validate_shader_records(shader_records, shader_records_size);
    if (ret)
        goto free_shader_records;

    bo = vc4_bo_create(dev, cmdlist_size + shader_records_size, 0, ttm_bo_type_kernel);
    if (IS_ERR(bo)) {
        ret = PTR_ERR(bo);
        goto free_shader_records;
    }

    ret = vc4_bo_map(bo);
    if (ret)
        goto free_bo;

    memcpy(bo->vaddr, cmdlist, cmdlist_size);
    memcpy(bo->vaddr + cmdlist_size, shader_records, shader_records_size);

    exec->cmdlist_bo = bo;
    exec->cmdlist = bo->vaddr;
    exec->shader_records = bo->vaddr + cmdlist_size;

    ret = vc4_wait_for_render_complete(vc4);
    if (ret)
        return ret;

    kfree(cmdlist);
    kfree(shader_records);

    return 0;

free_bo:
    vc4_bo_unref(bo);
free_shader_records:
    kfree(shader_records);
free_cmdlist:
    kfree(cmdlist);
    return ret;
}
