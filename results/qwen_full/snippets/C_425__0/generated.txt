size_t RegisterMPCImage(void) {
  MagickInfo
    *magick_info;

  magick_info = AcquireMagickInfo("MPC", "MPC", "Magick Persistent Cache");
  magick_info->decoder = (DecodeImageHandler *) ReadMPCImage;
  magick_info->encoder = (EncodeImageHandler *) WriteMPCImage;
  magick_info->adjoin = MagickFalse;
  magick_info->raw = MagickTrue;
  (void) SetMagickInfoFile(magick_info, "MPC", (void *) NULL);
  return (RegisterMagickInfo(magick_info));
}
