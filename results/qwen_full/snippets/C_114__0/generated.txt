ssize_t archive_write_data(struct archive* _a, const void* buff, size_t s) {
    if (_a == NULL || buff == NULL || s == 0) {
        return -1;
    }

    if (_a->magic != ARCHIVE_MAGIC) {
        archive_set_error(_a, ARCHIVE_ERRNO_MISC, "Archive not properly initialized");
        return -1;
    }

    if (_a->write_callback == NULL) {
        archive_set_error(_a, ARCHIVE_ERRNO_MISC, "No write callback available");
        return -1;
    }

    _a->bytes_written = 0;
    archive_clear_error(_a);

    return _a->write_callback(_a, buff, s);
}
