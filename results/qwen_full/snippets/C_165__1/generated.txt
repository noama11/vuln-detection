int validate_stream_offset(const cdf_stream_t *sst, const cdf_header_t *h, const void *p, size_t tail, int line) {
    const char *stream_end = (const char *)p + tail;
    const char *stream_start = (const char *)sst;
    size_t min_size = h->min_size;
    size_t dir_length = sst->dir_length;

    if ((const char *)p < stream_start || (const char *)p + tail > stream_start + min_size - dir_length) {
        errno = EFTYPE;
        return -1;
    }

    return 0;
}
