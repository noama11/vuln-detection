size_t WritePSDChannel(const PSDInfo *psd_info, const ImageInfo *image_info,
                       Image *image, Image *next_image, QuantumType quantum_type,
                       unsigned char *compact_pixels, MagickOffsetType size_offset,
                       MagickBooleanType separate) {
  size_t bytes_written = 0;
  MagickBooleanType is_monochrome = (image->depth == 1 ? MagickTrue : MagickFalse);
  CompressionType compression = psd_info->compression;

  for (size_t y = 0; y < image->rows; y++) {
    Quantum *pixels = (Quantum *) AcquireQuantumMemory(image->columns, sizeof(Quantum));
    ExportQuantumPixels(image, (CacheView *) NULL, quantum_type, image->columns, 1, pixels);

    if (compression == RLECompression) {
      unsigned char *rle_data = CompressRLE(pixels, image->columns, is_monochrome);
      bytes_written += WriteBlob(image, size_offset, rle_data);
      rle_data = (unsigned char *) RelinquishMagickMemory(rle_data);
    }
    else if (compression == ZipCompression) {
      unsigned char *zip_data = CompressZIP(pixels, image->columns, is_monochrome);
      bytes_written += WriteBlob(image, size_offset, zip_data);
      zip_data = (unsigned char *) RelinquishMagickMemory(zip_data);
    }
    else {
      bytes_written += WriteBlob(image, size_offset, (const unsigned char *) pixels);
    }

    pixels = (Quantum *) RelinquishMagickMemory(pixels);
  }

  return bytes_written;
}
