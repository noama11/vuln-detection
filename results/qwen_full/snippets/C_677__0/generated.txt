int unimac_mdio_probe(struct platform_device *pdev)
{
    struct resource *res;
    void __iomem *base;
    struct mii_bus *bus;
    struct unimac_mdio_priv *priv;
    int ret;

    priv = devm_kzalloc(&pdev->dev, sizeof(*priv), GFP_KERNEL);
    if (!priv)
        return -ENOMEM;

    res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
    if (!res)
        return -ENODEV;

    base = devm_ioremap_resource(&pdev->dev, res);
    if (IS_ERR(base))
        return PTR_ERR(base);

    bus = devm_mdiobus_alloc(&pdev->dev);
    if (!bus)
        return -ENOMEM;

    bus->name = "UniMAC MDIO Bus";
    bus->read = unimac_mdio_read;
    bus->write = unimac_mdio_write;
    bus->priv = (void *)base;
    snprintf(bus->id, MII_BUS_ID_SIZE, "unimac-mdio-%d", pdev->id);

    ret = of_mdiobus_register(bus, &pdev->dev);
    if (ret) {
        dev_err(&pdev->dev, "Failed to register MDIO bus\n");
        return ret;
    }

    platform_set_drvdata(pdev, priv);
    dev_info(&pdev->dev, "MDIO bus initialized at 0x%p\n", base);

    return 0;
}
