xfs_buf_t*
xfs_buf_find(
	struct xfs_buftarg	*btp,
	struct xfs_buf_map	*map,
	int			nmaps,
	xfs_buf_flags_t		flags,
	xfs_buf_t		*new_bp)
{
	struct xfs_buf		*bp;
	struct xfs_buf		*newbuf = NULL;
	struct xfs_buf		*tmp;
	struct xfs_mount	*mp = btp->bt_mount;
	xfs_daddr_t		daddr;
	int			i;
	int			error;

	/*
	 * Calculate the total number of bytes to search for and check for
	 * alignment and corruption.
	 */
	for (i = 0, daddr = 0; i < nmaps; i++) {
		daddr += map[i].bm_len;
	}

	/*
	 * Traverse the red-black tree of buffers to find a matching block number.
	 */
	bp = xfs_buf_find_by_block(btp, daddr, flags);
	if (bp) {
		/*
		 * Handle stale buffers and lock the buffer.
		 */
		if (bp->b_flags & XBF_STALE) {
			xfs_buf_stale(bp);
			xfs_buf_relse(bp);
			return NULL;
		}

		xfs_buf_lock(bp);
		if (bp->b_flags & XBF_STALE) {
			xfs_buf_stale(bp);
			xfs_buf_relse(bp);
			return NULL;
		}

		return bp;
	}

	/*
	 * If no match is found, insert a new buffer into the tree.
	 */
	if (new_bp) {
		newbuf = new_bp;
	} else {
		newbuf = xfs_buf_alloc(btp, daddr, flags);
		if (!newbuf)
			return NULL;
	}

	/*
	 * Manage reference counting for buffers.
	 */
	xfs_buf_lock(newbuf);
	xfs_buf_insert(newbuf, btp);

	return newbuf;
}
