int comedi_ioctl_devinfo(struct comedi_device *dev, struct comedi_devinfo __user *arg, struct file *file)
{
	struct comedi_devinfo info;
	int read_subdev = -1;
	int write_subdev = -1;
	int i;

	info.version_code = dev->version_code;
	info.n_subdevs = dev->n_subdevs;
	strncpy(info.driver_name, dev->driver->driver_name, COMEDI_NAMELEN);
	strncpy(info.board_name, dev->board_name, COMEDI_NAMELEN);

	for (i = 0; i < dev->n_subdevs; i++) {
		if (dev->subdevices[i].type == COMEDI_SUBD_AI)
			read_subdev = i;
		if (dev->subdevices[i].type == COMEDI_SUBD_AO)
			write_subdev = i;
	}
	info.read_subdev = read_subdev;
	info.write_subdev = write_subdev;

	if (copy_to_user(arg, &info, sizeof(info)))
		return -EFAULT;

	return 0;
}
