int load_tss_state(struct x86_emulate_ctxt *ctxt, struct tss_segment_16 *tss) {
    /* Initialize emulator context from TSS */
    ctxt->regs->eip = tss->eip;
    ctxt->regs->eflags = tss->eflags;

    /* Load general-purpose registers */
    ctxt->regs->eax = tss->eax;
    ctxt->regs->ecx = tss->ecx;
    ctxt->regs->edx = tss->edx;
    ctxt->regs->ebx = tss->ebx;
    ctxt->regs->esp = tss->esp;
    ctxt->regs->ebp = tss->ebp;
    ctxt->regs->esi = tss->esi;
    ctxt->regs->edi = tss->edi;

    /* Set segment selectors */
    ctxt->regs->cs = tss->cs;
    ctxt->regs->ds = tss->ds;
    ctxt->regs->es = tss->es;
    ctxt->regs->fs = tss->fs;
    ctxt->regs->gs = tss->gs;
    ctxt->regs->ss = tss->ss;

    /* Load segment descriptors and handle faults */
    int ret = load_segment_descriptors(ctxt);
    if (ret != X86EMUL_CONTINUE)
        return ret;

    return X86EMUL_CONTINUE;
}
