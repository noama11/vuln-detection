int restore_fpu_checking(struct task_struct *tsk) {
    if (cpu_has_fpu_feature) {
        /* Clear x87 state */
        __asm__ __volatile__(
            "fninit\n"
            "fldz\n"
            "fld1\n"
            "fldpi\n"
            "fldl2e\n"
            "fldl2t\n"
            "fstsw %%ax\n"
            "fwait"
            :
            :
            : "ax", "cc");
    }

    return __fpu_restore_checking(tsk);
}
