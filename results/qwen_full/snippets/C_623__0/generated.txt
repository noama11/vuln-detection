ExprDef *append_keysym_list(ExprDef *expr, ExprDef *append) {
    size_t original_size = expr->keysym_count;
    size_t append_size = append->keysym_count;

    // Resize the keysym map of the first expression to accommodate the new entries
    expr->keysym_map = realloc(expr->keysym_map, (original_size + append_size) * sizeof(*expr->keysym_map));
    if (!expr->keysym_map) {
        return NULL; // Handle memory allocation failure
    }

    // Append the size information to the respective maps
    expr->keysym_count += append_size;

    // Concatenate the keysym lists
    memcpy(expr->keysym_map + original_size, append->keysym_map, append_size * sizeof(*expr->keysym_map));

    // Free the memory associated with the second expression
    free(append->keysym_map);
    free(append);

    return expr;
}
