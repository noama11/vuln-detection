int ceph_auth_client_process_ticket(struct ceph_auth_client *ac,
                                    struct ceph_crypto_key *secret,
                                    void **p, void *end) {
    int ret = -EINVAL;
    void *ticket_start = *p;
    uint32_t type, version;

    if (*p + sizeof(type) > end) {
        goto out;
    }
    type = le32_to_cpu(*(__le32 *)*p);
    *p += sizeof(type);

    if (*p + sizeof(version) > end) {
        goto out;
    }
    version = le32_to_cpu(*(__le32 *)*p);
    *p += sizeof(version);

    struct ceph_auth_client_ticket_handler *handler =
        ceph_get_ticket_handler(type, version);
    if (!handler) {
        ret = -ENOTSUPP;
        goto out;
    }

    ret = handler->decode_ticket(ac, secret, p, end);
    if (ret < 0) {
        goto out;
    }

    if (ceph_ticket_is_valid(ac)) {
        ceph_update_session_key(ac, handler->session_key);
        ceph_set_ticket_blob(ac, ticket_start, *p - ticket_start);
    }

    ret = 0;

out:
    return ret;
}
