krb5_error_code check_kdc_public_key_identifier(krb5_context context,
                                               pkinit_plg_crypto_context plg_cryptoctx,
                                               pkinit_req_crypto_context req_cryptoctx,
                                               pkinit_identity_crypto_context id_cryptoctx,
                                               unsigned char *pdid_buf,
                                               unsigned int pkid_len,
                                               int *valid_kdcPkId) {
    krb5_error_code retval = 0;
    krb5_data pdid_data;
    pkinit_pkeyid pkeyid = NULL;
    krb5_principal kdc_principal = NULL;
    krb5_X509_CERT cert = NULL;
    krb5_principal issuer = NULL;
    krb5_data serial_number = {0, NULL};

    pdid_data.data = (char *)pdid_buf;
    pdid_data.length = pkid_len;

    *valid_kdcPkId = 0;

    if ((retval = pkinit_decode_PKINIT_PKEYID(context, &pdid_data, &pkeyid)))
        goto cleanup;

    if ((retval = krb5_kdc_realm_to_principal(context, plg_cryptoctx->realm, 0, &kdc_principal)))
        goto cleanup;

    if ((retval = plg_cryptoctx->get_kdc_cert(context, plg_cryptoctx, kdc_principal, &cert)))
        goto cleanup;

    if ((retval = plg_cryptoctx->get_issuer(context, plg_cryptoctx, cert, &issuer)))
        goto cleanup;

    if ((retval = plg_cryptoctx->get_serial_number(context, plg_cryptoctx, cert, &serial_number)))
        goto cleanup;

    if (pkeyid->issuer != NULL && issuer != NULL &&
        krb5_principal_compare(context, issuer, pkeyid->issuer) &&
        serial_number.length == pkeyid->serialNumber.length &&
        memcmp(serial_number.data, pkeyid->serialNumber.data, serial_number.length) == 0) {
        *valid_kdcPkId = 1;
    }

cleanup:
    if (pkeyid)
        free(pkeyid);
    if (kdc_principal)
        krb5_free_principal(context, kdc_principal);
    if (cert)
        plg_cryptoctx->free_cert(context, plg_cryptoctx, cert);
    if (issuer)
        krb5_free_principal(context, issuer);
    if (serial_number.data)
        free(serial_number.data);

    return retval;
}
