void add_suggestion_element(UiScene* scene, UiBrowserInterface* browser, Model* model, SuggestionBinding* element_binding) {
    // Create UI components
    UiElement* icon = create_icon_element();
    UiElement* content_text = create_text_element();
    UiElement* description_text = create_text_element();
    UiElement* background_button = create_button_element();

    // Bind UI components to the model
    bind_icon(icon, model);
    bind_text(content_text, model, "content");
    bind_text(description_text, model, "description");
    bind_button_colors(background_button, model);

    // Set up layout
    UiElement* layout = create_horizontal_layout();
    add_child(layout, icon);
    add_child(layout, content_text);
    add_child(layout, description_text);

    // Set up navigation on click
    set_button_on_click(background_button, (void*)browser, navigate_to_suggestion);

    // Add the constructed element to the scene
    add_element_to_scene(scene, layout);
}
