struct xt_table_info *alloc_xt_table_info(unsigned int size) {
    struct xt_table_info *table;
    size_t total_size = sizeof(struct xt_table_info) + size;

    table = kmalloc(total_size, GFP_KERNEL);
    if (!table) {
        table = vmalloc(total_size);
        if (!table)
            return NULL;
    }

    memset(table, 0, total_size);
    table->size = size;
    return table;
}
