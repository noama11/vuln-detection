Image *
ReadPNGImage(const ImageInfo *image_info, ExceptionInfo *exception)
{
  Image *image = NULL;
  FILE *file = NULL;
  unsigned char signature[PNG_SIG_SIZE];
  MngInfo *mng_info = NULL;

  (void) LogMagickEvent(CoderEvent, GetMagickModule(), "Enter ReadPNGImage");

  if (!image_info || !exception)
    return NULL;

  file = fopen(image_info->filename, "rb");
  if (!file)
  {
    (void) ThrowMagickException(exception, GetMagickModule(), FileOpenError,
      "UnableToOpenFile", "%s", image_info->filename);
    return NULL;
  }

  if (fread(signature, 1, PNG_SIG_SIZE, file) != PNG_SIG_SIZE)
  {
    (void) fclose(file);
    (void) ThrowMagickException(exception, GetMagickModule(), CorruptImageError,
      "CorruptedImage", "%s", image_info->filename);
    return NULL;
  }

  if (png_sig_cmp(signature, 0, PNG_SIG_SIZE))
  {
    (void) fclose(file);
    (void) ThrowMagickException(exception, GetMagickModule(), CorruptImageError,
      "NotPngFile", "%s", image_info->filename);
    return NULL;
  }

  mng_info = AcquireMagickMemory(sizeof(*mng_info));
  if (!mng_info)
  {
    (void) fclose(file);
    (void) ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
      "MemoryAllocationFailed", "Unable to allocate MngInfo structure");
    return NULL;
  }

  (void) InitializeMngInfo(mng_info);

  image = ReadPNGImageInternal(file, mng_info, image_info, exception);
  if (!image)
  {
    (void) RelinquishMagickMemory(mng_info);
    (void) fclose(file);
    return NULL;
  }

  if (image->columns <= 0 || image->rows <= 0)
  {
    (void) ThrowMagickException(exception, GetMagickModule(), CorruptImageError,
      "InvalidImageDimensions", "%s", image_info->filename);
    DestroyImageList(image);
    return NULL;
  }

  if (image->colorspace == UndefinedColorspace)
    image->colorspace = sRGBColorspace;

  (void) fclose(file);
  (void) LogMagickEvent(CoderEvent, GetMagickModule(), "Exit ReadPNGImage");

  return image;
}
