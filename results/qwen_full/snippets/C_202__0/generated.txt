int bind_pptp_socket(struct socket *sock, struct sockaddr *uservaddr, int sockaddr_len) {
    int err = 0;

    lock_sock(sock);
    sock->sk->sk_rcv_saddr = ((struct sockaddr_in *)uservaddr)->sin_addr.s_addr;
    err = add_pptp_channel(sock);
    if (err < 0) {
        sock->sk->sk_err = -err;
    }
    release_sock(sock);

    return err;
}
