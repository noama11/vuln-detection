void check_dummy_password_v1_6(kadm5_principal_ent_t entry, long mask, int n_ks_tuple, krb5_key_salt_tuple *ks_tuple, char **passptr) {
    if ((mask & KADM5_ATTRIBUTES) && (entry->attributes & KRB5_KDB_AUX_OLD_STYLE)) {
        return;
    }

    if (*passptr == NULL || (strlen(*passptr) >= 1 && strlen(*passptr) <= 255)) {
        *passptr = NULL;
    }
}
