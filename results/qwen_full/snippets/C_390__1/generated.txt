ACPI_STATUS AcpiPsCompleteFinalOp(ACPI_WALK_STATE *WalkState, ACPI_PARSE_OBJECT *Op, ACPI_STATUS Status)
{
    ACPI_PARSE_OBJECT *ScopeOp;
    ACPI_PARSE_OBJECT *Next;

    /* Check if we have a valid Op to process */
    if (!Op) {
        return (Status);
    }

    /* Get the scope operator */
    ScopeOp = Op->Common.Value.ScopeOp;

    if (ScopeOp) {
        /* Clear the scope stack */
        while (ScopeOp) {
            Next = ScopeOp->Common.Next;
            ScopeOp->Common.Value.ScopeOp = NULL;
            ScopeOp = Next;
        }
    }

    /* Check for ascending callbacks */
    if (WalkState->DescendingCallback && WalkState->AscendingCallback) {
        WalkState->AscendingCallback(WalkState, Op, Status);
    }

    return (Status);
}
