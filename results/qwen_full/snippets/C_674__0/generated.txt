int read_index_entry(git_index_entry **out, size_t *out_size, git_index *index, const void *buffer, size_t buffer_size, const char *last) {
    if (!buffer || buffer_size < sizeof(git_index_entry)) {
        return -1;
    }

    git_index_entry *entry = (git_index_entry *)malloc(buffer_size);
    if (!entry) {
        return -1;
    }

    memcpy(entry, buffer, buffer_size);

    if (entry->flags & GIT_INDEX_ENTRY_UNCOPIED) {
        if (last) {
            size_t path_len = strlen(last) + 1;
            entry->path = (char *)malloc(path_len);
            if (!entry->path) {
                free(entry);
                return -1;
            }
            strcpy((char *)entry->path, last);
        }
    } else {
        entry->path = (char *)malloc(entry->path_len + 1);
        if (!entry->path) {
            free(entry);
            return -1;
        }
        memcpy(entry->path, buffer + sizeof(git_index_entry), entry->path_len);
        entry->path[entry->path_len] = '\0';
    }

    *out = entry;
    *out_size = buffer_size;

    return 0;
}
