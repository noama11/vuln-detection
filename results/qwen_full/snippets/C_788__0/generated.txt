int archive_read_support_zip_cleanup(struct archive_read *a) {
  struct zip *zip = (struct zip *)(a->format->data);

  if (zip == NULL)
    return (ARCHIVE_OK);

  if (zip->zlib_stream != NULL)
    free(zip->zlib_stream);
  if (zip->lzma_stream != NULL)
    free(zip->lzma_stream);
  if (zip->bz2_stream != NULL)
    free(zip->bz2_stream);

  struct zip_entry *entry, *next;
  for (entry = zip->entries; entry != NULL; entry = next) {
    next = entry->next;
    if (entry->buffer != NULL)
      free(entry->buffer);
    free(entry);
  }

  if (zip->central_directory != NULL)
    free(zip->central_directory);
  if (zip->local_file_header != NULL)
    free(zip->local_file_header);
  if (zip->filename != NULL)
    free(zip->filename);

  free(zip);
  a->format->data = NULL;

  return (ARCHIVE_OK);
}
