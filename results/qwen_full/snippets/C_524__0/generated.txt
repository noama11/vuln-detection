int kvm_dev_create(struct kvm *kvm, struct kvm_create_device *cd)
{
	int r;
	struct kvm_device_ops *ops;
	struct kvm_device *dev;
	struct kvm_device *dev_list;

	ops = kvm_find_device_ops(cd->type);
	if (!ops)
		return -EINVAL;

	if (cd->flags & KVM_CREATE_DEVICE_TEST)
		return 0;

	dev = kzalloc(sizeof(*dev), GFP_KERNEL);
	if (!dev)
		return -ENOMEM;

	kvm_lock(kvm);
	list_for_each_entry(dev_list, &kvm->devices, list) {
		if (dev_list->ops == ops) {
			r = -EEXIST;
			goto unlock;
		}
	}

	dev->ops = ops;
	dev->kvm = kvm;
	dev->id = cd->id;

	r = ops->init(dev);
	if (r)
		goto unlock;

	dev->fd = anon_inode_getfd(ops->name, &kvm_device_fops, dev,
				    O_CLOEXEC);
	if (dev->fd < 0) {
		r = dev->fd;
		goto unlock;
	}

	list_add(&dev->list, &kvm->devices);
unlock:
	kvm_unlock(kvm);

	if (r)
		ops->destroy(dev);

	cd->fd = dev->fd;
	return r;
}
