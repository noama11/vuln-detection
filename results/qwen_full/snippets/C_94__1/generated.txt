ext3_fsblk_t get_superblock_block(void **data, struct super_block *sb) {
    char *options = (char *)*data;
    ext3_fsblk_t block = 1;

    if (strncmp(options, "sb=", 3) == 0) {
        char *endptr;
        block = simple_strtoul(options + 3, &endptr, 0);
        if (endptr == options + 3) {
            return 1; // Invalid number
        }

        // Check for invalid characters after the number
        while (*endptr && isspace(*endptr)) {
            endptr++;
        }

        if (*endptr != '\0') {
            printk(KERN_ERR "Invalid characters after block number in 'sb=' option\n");
        }

        *data = (void *)endptr;
    }

    return block;
}
