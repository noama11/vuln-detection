int cifs_tree_connect(unsigned int xid, struct cifs_ses *ses, const char *tree,
                      struct cifs_tcon *tcon, const struct nls_table *cp)
{
    int rc = 0;
    char *utf16_path = NULL;
    char *full_path = NULL;
    struct smb2_tree_connect_req *req;
    struct smb2_tree_connect_rsp *rsp;
    struct kvec iov[1];
    struct kvec rsp_iov;
    int rsp_buflen;

    if (!ses || !tree || !tcon || !cp)
        return -EINVAL;

    full_path = kmalloc(PATH_MAX, GFP_KERNEL);
    if (!full_path)
        return -ENOMEM;

    snprintf(full_path, PATH_MAX, "\\\\%s\\%s", ses->serverName, tree);

    utf16_path = kmalloc(PATH_MAX * sizeof(__u16), GFP_KERNEL);
    if (!utf16_path) {
        rc = -ENOMEM;
        goto out;
    }

    rc = cifs_strtoUTF16(utf16_path, full_path, PATH_MAX, cp);
    if (rc < 0)
        goto out;

    req = cifs_buf_get();
    if (!req) {
        rc = -ENOMEM;
        goto out;
    }

    iov[0].iov_base = req;
    iov[0].iov_len = sizeof(*req);

    rc = SMB2_tree_connect_init(xid, ses, iov, 1, utf16_path, tcon->tid);
    if (rc < 0)
        goto out_free_req;

    rc = cifs_call_async(ses, iov, 1, &rsp_iov, &SMB2_tree_connect_callback,
                         NULL, 0, CIFS_ASYNC_OP);
    if (rc < 0)
        goto out_free_req;

    rsp = (struct smb2_tree_connect_rsp *)rsp_iov.iov_base;
    if (!rsp) {
        rc = -EIO;
        goto out_free_rsp;
    }

    tcon->share_type = le32_to_cpu(rsp->share_type);
    tcon->capabilities = le32_to_cpu(rsp->capabilities);
    tcon->maximal_access = le32_to_cpu(rsp->maximal_access);

    if (rc == 0)
        rc = 0;
    else
        cifs_reconnect(ses);

out_free_rsp:
    kfree(rsp_iov.iov_base);
out_free_req:
    cifs_buf_release(req);
out:
    kfree(utf16_path);
    kfree(full_path);
    return rc;
}
