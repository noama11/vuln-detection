#include <stdio.h>
#include <stdlib.h>
#include <string.h>

void normalize_path(char* file) {
    char* result = malloc(strlen(file) + 1);
    if (!result) return;
    strcpy(result, "");

    char* token = strtok(file, "/");
    while (token) {
        if (strcmp(token, ".") != 0 && strcmp(token, "") != 0) {
            if (strcmp(token, "..") == 0) {
                if (strlen(result) > 0) {
                    char* lastSlash = strrchr(result, '/');
                    if (lastSlash) {
                        *lastSlash = '\0';
                    }
                }
            } else {
                if (strlen(result) == 0) {
                    strcat(result, token);
                } else {
                    strcat(result, "/");
                    strcat(result, token);
                }
            }
        }
        token = strtok(NULL, "/");
    }

    strcpy(file, result);
    free(result);
}
