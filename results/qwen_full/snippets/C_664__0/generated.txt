mrb_value
mrb_io_init_copy(mrb_state *mrb, mrb_value copy)
{
  mrb_value orig;
  struct mrb_io *io, *orig_io;
  int oflags;

  orig = mrb_get_args(mrb, "o", &orig);
  if (!mrb_io_p(orig)) {
    mrb_raise(mrb, E_TYPE_ERROR, "wrong argument type");
  }

  mrb_data_check_type(mrb, copy, "IO");
  io = (struct mrb_io *)mrb_data_get_ptr(mrb, copy, &mrb_io_type);
  if (io == NULL) {
    io = (struct mrb_io *)mrb_malloc(mrb, sizeof(*io));
    memset(io, 0, sizeof(*io));
  }

  orig_io = (struct mrb_io *)mrb_data_get_ptr(mrb, orig, &mrb_io_type);
  if (orig_io->fd != -1) {
    int new_fd = dup(orig_io->fd);
    if (new_fd == -1) {
      mrb_sys_fail(mrb, "dup");
    }
    io->fd = new_fd;
  }

  oflags = fcntl(orig_io->fd, F_GETFL, 0);
  if (oflags == -1) {
    mrb_sys_fail(mrb, "fcntl");
  }

  if (oflags & O_APPEND) {
    io->append = 1;
  }
  if (oflags & O_NONBLOCK) {
    io->nonblock = 1;
  }

  io->sync = orig_io->sync;
  io->readable = orig_io->readable;
  io->writable = orig_io->writable;

  mrb_data_set_ptr(mrb, copy, io);
  return copy;
}
