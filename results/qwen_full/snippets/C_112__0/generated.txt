#include <linux/fs.h>
#include <linux/cpu.h>
#include <linux/errno.h>

int open_msr(struct inode *inode, struct file *file) {
    int cpu;
    int ret = 0;

    cpu = smp_find_cpu_from_node(inode);
    if (cpu < 0) {
        return -ENXIO;
    }

    if (!cpu_online(cpu)) {
        return -ENXIO;
    }

    if (!cpu_has_msr(cpu)) {
        return -EIO;
    }

    if (security_msr_access(file, cpu) < 0) {
        return -EPERM;
    }

    return ret;
}
