int set_posix_acl(struct inode *inode, struct posix_acl *acl, int type) {
    int ret;

    if (type == ACL_TYPE_ACCESS) {
        ret = posix_acl_update_mode(inode, &inode->i_mode, NULL);
        if (ret)
            return ret;
    } else {
        return -EINVAL;
    }

    inode->i_ctime = current_time(inode);
    set_cached_acl(inode, type, acl);
    return 0;
}
