void GetNamedProperty(v8::Local<v8::Name> name, const v8::PropertyCallbackInfo<v8::Value>& info) {
  v8::Isolate* isolate = info.GetIsolate();
  v8::Local<v8::Object> holder = info.Holder();

  // Retrieve the LocalDOMWindow from the holder
  LocalDOMWindow* window = ToLocalDOMWindow(holder);
  if (!window)
    return;

  // Check if the window is attached to a frame
  Frame* frame = window->frame();
  if (!frame)
    return;

  // Convert the v8::Name to a string
  v8::String::Utf8Value utf8_name(isolate, name);
  if (!utf8_name)
    return;

  const String& propertyName = *utf8_name;

  // Search for sub-frames and named items in the document
  Element* element = frame->document()->getElementById(propertyName);
  if (element) {
    v8::Local<v8::Value> v8_element = ToV8(element, holder, isolate);
    if (!v8_element.IsEmpty())
      info.GetReturnValue().Set(v8_element);
    return;
  }

  // If no element is found, check for named items
  Node* node = frame->document()->GetNamedItem(propertyName);
  if (node) {
    v8::Local<v8::Value> v8_node = ToV8(node, holder, isolate);
    if (!v8_node.IsEmpty())
      info.GetReturnValue().Set(v8_node);
    return;
  }

  // If no match is found, return undefined
  info.GetReturnValue().Set(v8::Undefined(isolate));
}
