size_t RegisterJPEGImage(void) {
  MagickInfo *magick_info;
  size_t length = 0;

  const char *version = "JPEG";
  const char *description = "Joint Photographic Experts Group JFIF format";

  magick_info = SetMagickInfo("JPE");
  magick_info->decoder = (DecodeImageHandler *) ReadJPEGImage;
  magick_info->encoder = (EncodeImageHandler *) WriteJPEGImage;
  magick_info->thread_support = MagickCoreSupport;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) description;
  magick_info->version = (char *) version;
  length += RegisterMagickInfo("JPE", magick_info);

  magick_info = SetMagickInfo("JPEG");
  magick_info->decoder = (DecodeImageHandler *) ReadJPEGImage;
  magick_info->encoder = (EncodeImageHandler *) WriteJPEGImage;
  magick_info->thread_support = MagickCoreSupport;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) description;
  magick_info->version = (char *) version;
  length += RegisterMagickInfo("JPEG", magick_info);

  magick_info = SetMagickInfo("JPG");
  magick_info->decoder = (DecodeImageHandler *) ReadJPEGImage;
  magick_info->encoder = (EncodeImageHandler *) WriteJPEGImage;
  magick_info->thread_support = MagickCoreSupport;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) description;
  magick_info->version = (char *) version;
  length += RegisterMagickInfo("JPG", magick_info);

  magick_info = SetMagickInfo("JPS");
  magick_info->decoder = (DecodeImageHandler *) ReadJPEGImage;
  magick_info->encoder = (EncodeImageHandler *) WriteJPEGImage;
  magick_info->thread_support = MagickCoreSupport;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) description;
  magick_info->version = (char *) version;
  length += RegisterMagickInfo("JPS", magick_info);

  magick_info = SetMagickInfo("PJPEG");
  magick_info->decoder = (DecodeImageHandler *) ReadJPEGImage;
  magick_info->encoder = (EncodeImageHandler *) WriteJPEGImage;
  magick_info->thread_support = MagickCoreSupport;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) description;
  magick_info->version = (char *) version;
  length += RegisterMagickInfo("PJPEG", magick_info);

  return length;
}
