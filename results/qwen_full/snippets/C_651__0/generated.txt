struct RClass* get_real_class(struct RClass* cl) {
  if (cl == NULL) {
    return NULL;
  }

  while (1) {
    if (cl->flags & MRB_CLASS_FLAG_SINGLETON) {
      cl = cl->super;
      if (cl == NULL) {
        return NULL;
      }
    } else if (cl->flags & MRB_CLASS_FLAG_INCLUDED) {
      cl = cl->super;
      if (cl == NULL) {
        return NULL;
      }
    } else {
      return cl;
    }
  }
}
