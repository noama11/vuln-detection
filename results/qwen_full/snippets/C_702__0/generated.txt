int get_data_block(struct inode *inode, sector_t iblock, struct buffer_head *bh, int create, int flag, pgoff_t *next_pgofs) {
    struct address_space *mapping = inode->i_mapping;
    int ret;

    ret = mapping->a_ops->bmap(mapping, iblock);
    if (ret < 0)
        return ret;

    bh->b_blocknr = ret;
    bh->b_bdev = inode->i_sb->s_bdev;
    bh->b_size = inode->i_sb->s_blocksize;

    if (create && !ret) {
        ret = 1;
        *next_pgofs = iblock + 1;
    }

    return 0;
}
