int decodeADPCMBlock(const uint8_t* encoded, int16_t* decoded) {
    int channels = getTrackChannelCount();
    int samplesPerChannel = 0;
    int totalSamples = 0;

    for (int ch = 0; ch < channels; ++ch) {
        ADPCMState state;
        initADPCMState(&state);

        uint8_t* data = (uint8_t*)encoded;
        uint8_t predictor = *data++;
        uint8_t index = *data++;

        while (data < encoded + getPacketSize()) {
            int16_t sample1 = decodeSample(predictor, index, data);
            int16_t sample2 = decodeSample(predictor, index, data + 1);
            decoded[totalSamples++] = sample1;
            decoded[totalSamples++] = sample2;
            data += 2;
        }

        samplesPerChannel = totalSamples / channels;
    }

    return samplesPerChannel * channels * sizeof(int16_t);
}
