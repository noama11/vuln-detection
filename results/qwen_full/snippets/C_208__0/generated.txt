int get_bitmap_file(struct mddev *mddev, void __user *arg)
{
    struct bitmap_file *bf;
    int err = 0;

    bf = kmalloc(sizeof(*bf), GFP_KERNEL);
    if (!bf)
        return -ENOMEM;

    if (!mddev->bitmap) {
        bf->pathname[0] = '\0';
    } else {
        const char *path = mddev->bitmap->filename;
        if (path && strlen(path) >= sizeof(bf->pathname)) {
            err = -ENAMETOOLONG;
            goto out;
        }
        strncpy(bf->pathname, path, sizeof(bf->pathname) - 1);
        bf->pathname[sizeof(bf->pathname) - 1] = '\0';
    }

    if (copy_to_user(arg, bf, sizeof(*bf))) {
        err = -EFAULT;
        goto out;
    }

out:
    kfree(bf);
    return err;
}
