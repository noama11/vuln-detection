int rfcomm_recvmsg(struct kiocb *iocb, struct socket *sock, struct msghdr *msg,
                   size_t size, int flags)
{
    int bytes_received;

    /* Check if RFCOMM connection setup is deferred and accept it if so */
    if (rfcomm_check_deferred_setup(sock)) {
        rfcomm_accept_connection(sock);
    }

    /* Call lower-level function to receive the message */
    bytes_received = sock->sk->sk_prot->recvmsg(iocb, sock, msg, size, flags);

    if (bytes_received > 0) {
        /* Update the socket's receive memory allocation */
        sock_update_rcv_mem(sock, bytes_received);

        /* Check if we need to unthrottle the connection */
        if (sock_mem_usage_below_threshold(sock)) {
            rfcomm_unthrottle_connection(sock);
        }
    }

    return bytes_received;
}
