class ImageFrameGenerator : public ThreadSafeRefCounted<ImageFrameGenerator> {
public:
    static RefPtr<ImageFrameGenerator> create() {
        return adoptRef(new ImageFrameGenerator());
    }

    void decodeAndScaleImage() {
        MutexLocker locker(m_mutex);
        // Implementation for decoding and scaling image
        m_decodedSuccessfully = true;
        m_hasAlpha = false;
    }

    void manageYUVComponents() {
        MutexLocker locker(m_mutex);
        // Implementation for managing YUV components
    }

    class ClientMutexLocker {
    public:
        ClientMutexLocker(ImageFrameGenerator* generator, int clientId)
            : m_generator(generator), m_clientId(clientId) {
            m_generator->m_clientMutexes[clientId].lock();
        }

        ~ClientMutexLocker() {
            m_generator->m_clientMutexes[m_clientId].unlock();
        }

    private:
        ImageFrameGenerator* m_generator;
        int m_clientId;
    };

private:
    ImageFrameGenerator()
        : m_decodedSuccessfully(false)
        , m_hasAlpha(false) {
        // Constructor initializes full size, color behavior, and supported sizes
    }

    Mutex m_mutex;
    bool m_decodedSuccessfully;
    bool m_hasAlpha;
    std::unordered_map<int, std::mutex> m_clientMutexes;
};
