void rds6_info_fill(struct rds_incoming *inc, struct rds_info_iterator *iter,
                    struct in6_addr *saddr, struct in6_addr *daddr, int flip)
{
    struct rds6_info_message msg;

    msg.ri_family = AF_INET6;
    msg.ri_sport = inc->i_sport;
    msg.ri_dport = inc->i_dport;

    if (flip) {
        msg.ri_laddr = *daddr;
        msg.ri_faddr = *saddr;
        msg.ri_lport = inc->i_dport;
        msg.ri_fport = inc->i_sport;
    } else {
        msg.ri_laddr = *saddr;
        msg.ri_faddr = *daddr;
        msg.ri_lport = inc->i_sport;
        msg.ri_fport = inc->i_dport;
    }

    rds_info_copy(iter, &msg, sizeof(msg));
}
