int copy_huge_page_entry(struct mm_struct *dst_mm, pte_t *dst_pte,
                         struct vm_area_struct *dst_vma, unsigned long dst_addr,
                         unsigned long src_addr, struct page **pagep) {
    struct page *page = *pagep;
    int ret = -EFAULT;

    if (!page) {
        page = alloc_huge_page();
        if (!page)
            return -ENOMEM;
        *pagep = page;
    }

    if (copy_from_user(page_address(page), (void __user *)src_addr, HPAGE_PMD_SIZE)) {
        return -EFAULT;
    }

    if (dst_vma->vm_flags & VM_SHARED) {
        set_page_dirty(page);
    } else {
        get_page(page);
    }

    spin_lock(&dst_mm->page_table_lock);
    set_pte_at(dst_mm, dst_addr, dst_pte, pte_mkhuge(pte_mksrwx(pte_clear(pte_val(*dst_pte)), page)));
    spin_unlock(&dst_mm->page_table_lock);

    return 0;
}
