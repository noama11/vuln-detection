void motion_compensation_mmx(uint8_t* dst, uint8_t* src, int stride, int h, int ox, int oy, int dxx, int dxy, int dyx, int dyy, int shift, int r, int width, int height) {
    if (ox > 1 || oy > 1 || dxx > 1 || dxy > 1 || dyx > 1 || dyy > 1) {
        // Fall back to software implementation
        return;
    }

    __asm__ volatile (
        "pxor %%mm7, %%mm7\n"
        "movq %%"ASM_REG_MMX(7)", %%"ASM_REG_MMX(6)"\n"
        "1:\n"
        "movq (%0, %%"ASM_REG_MMX(0)", 1), %%"ASM_REG_MMX(0)"\n"
        "punpcklbw %%mm7, %%"ASM_REG_MMX(0)"\n"
        "pmullw %%"ASM_REG_MMX(1)", %%"ASM_REG_MMX(0)"\n"
        "paddw %%"ASM_REG_MMX(2)", %%"ASM_REG_MMX(0)"\n"
        "psrlw %3, %%"ASM_REG_MMX(0)"\n"
        "packuswb %%"ASM_REG_MMX(7)", %%"ASM_REG_MMX(0)"\n"
        "movq %%"ASM_REG_MMX(0)", (%1)\n"
        "addl $8, %1\n"
        "decl %4\n"
        "jnz 1b\n"
        :
        : "r"(src), "r"(dst), "m"(dxx), "r"(shift), "r"(h)
        : "memory"
    );
}
