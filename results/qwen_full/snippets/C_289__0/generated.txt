int tipc_nl_compat_dump_link(struct tipc_nl_compat_msg *msg, struct nlattr **attrs)
{
    struct nlattr *link_attr;
    struct nlattr *tb[TIPC_NL_COMPAT_LINK_MAX + 1];
    int err;

    if (!attrs[TIPC_NL_COMPAT_LINK])
        return -EINVAL;

    err = nla_parse_nested(tb, TIPC_NL_COMPAT_LINK_MAX, attrs[TIPC_NL_COMPAT_LINK], NULL);
    if (err < 0)
        return err;

    if (!tb[TIPC_NL_COMPAT_LINK_DEST] || !tb[TIPC_NL_COMPAT_LINK_STATUS] || !tb[TIPC_NL_COMPAT_LINK_NAME])
        return -EINVAL;

    link_attr = nla_nest_start(msg->skb, TIPC_NL_COMPAT_MSG_LINK);
    if (!link_attr)
        return -EMSGSIZE;

    nla_put(msg->skb, TIPC_NL_COMPAT_LINK_DEST, nla_len(tb[TIPC_NL_COMPAT_LINK_DEST]),
            nla_data(tb[TIPC_NL_COMPAT_LINK_DEST]));
    nla_put(msg->skb, TIPC_NL_COMPAT_LINK_STATUS, nla_len(tb[TIPC_NL_COMPAT_LINK_STATUS]),
            nla_data(tb[TIPC_NL_COMPAT_LINK_STATUS]));
    nla_put(msg->skb, TIPC_NL_COMPAT_LINK_NAME, nla_len(tb[TIPC_NL_COMPAT_LINK_NAME]),
            nla_data(tb[TIPC_NL_COMPAT_LINK_NAME]));

    nla_nest_end(msg->skb, link_attr);

    return 0;
}
