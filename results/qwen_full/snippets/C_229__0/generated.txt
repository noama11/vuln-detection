int path_to_string(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int i = 0;
    int j = 0;

    while (i < fromlen && j < tolen - 1) {
        unsigned char c = from[i];

        if (c == 0) {
            to[j++] = '.';
        } else if (c == 1) {
            if (j + 2 <= tolen - 1) {
                to[j++] = '.';
                to[j++] = '.';
            } else {
                return -1; // Buffer overflow
            }
        } else if (c == 2) {
            if (j + 1 <= tolen - 1) {
                to[j++] = '~';
            } else {
                return -1; // Buffer overflow
            }
        } else {
            if (j + 1 <= tolen - 1) {
                to[j++] = c;
            } else {
                return -1; // Buffer overflow
            }
        }

        i++;
    }

    if (j >= tolen) {
        return -1; // Buffer overflow
    }

    to[j] = '\0';
    return 0;
}
