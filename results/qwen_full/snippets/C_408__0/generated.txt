MagickBooleanType AutoGammaImage(Image *image, ExceptionInfo *exception) {
  double mean;
  double gamma;
  MagickBooleanType status;
  size_t channel;

  if (image == (Image *) NULL)
    return MagickFalse;

  mean = GetImageMean(image, exception);
  if (mean == 0.0)
    return MagickFalse;

  gamma = log(mean) / log(0.5);

  if (image->channel_mask == DefaultChannels)
    status = AdjustImageGamma(image, gamma, exception);
  else {
    for (channel = 0; channel < MaxChannel; channel++) {
      if ((image->channel_mask & (1UL << channel)) != 0) {
        mean = GetImageMeanByChannel(image, (ChannelType) (1UL << channel), exception);
        if (mean == 0.0)
          return MagickFalse;
        gamma = log(mean) / log(0.5);
        status = AdjustImageGammaByChannel(image, (ChannelType) (1UL << channel), gamma, exception);
        if (status == MagickFalse)
          return MagickFalse;
      }
    }
  }

  return status;
}
