enum integrity_status verify_hmac_xattr(struct dentry *dentry, const char *xattr_name,
                                        char *xattr_value, size_t xattr_value_len,
                                        struct integrity_iint_cache *iint) {
    enum integrity_status status = INTEGRITY_PASS;
    int rc;
    void *digest = NULL;
    size_t digest_len = 0;
    struct integrity_dentry_data *dentry_data;

    if (iint->flags & INTEGRITY_SIGNATURE)
        return INTEGRITY_PASS;

    dentry_data = get_dentry_data(dentry);
    if (!dentry_data)
        return INTEGRITY_NO_ATTRIBUTE;

    if (dentry_data->xattr_type == INTEGRITY_XATTR_DIGEST_HMAC) {
        digest = calculate_hmac(dentry, xattr_value, xattr_value_len);
        digest_len = HMAC_DIGEST_SIZE;
    } else if (dentry_data->xattr_type == INTEGRITY_XATTR_DIGEST_SHA1) {
        digest = calculate_sha1(xattr_value, xattr_value_len);
        digest_len = SHA1_DIGEST_SIZE;
    } else {
        return INTEGRITY_NO_ATTRIBUTE;
    }

    if (!digest)
        return INTEGRITY_FAIL;

    rc = memcmp(digest, xattr_value, digest_len);
    if (rc != 0)
        status = INTEGRITY_FAIL;

    kfree(digest);

    if (status == INTEGRITY_FAIL)
        iint->flags |= INTEGRITY_SIGNATURE_INVALID;

    return status;
}
