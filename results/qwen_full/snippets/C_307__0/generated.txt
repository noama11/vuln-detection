int attach_usb_serial_for_treo_kyocera(struct usb_serial *serial) {
    int i;
    struct usb_host_endpoint *ep;
    struct usb_endpoint_descriptor *epd;
    struct usb_serial_port *port;
    int num_interrupt_eps = 0;
    int has_required_interrupt = 0;

    for (i = 0; i < serial->num_ports; i++) {
        ep = serial->port[i].port->host_endpoint;
        epd = &ep->desc;

        if ((epd->bmAttributes & USB_ENDPOINT_XFERTYPE_MASK) == USB_ENDPOINT_XFER_INT) {
            num_interrupt_eps++;
            if (epd->bEndpointAddress & USB_DIR_IN) {
                has_required_interrupt = 1;
            }
        }
    }

    if (!has_required_interrupt || num_interrupt_eps < 2) {
        return -ENODEV;
    }

    port = kzalloc(sizeof(*port), GFP_KERNEL);
    if (!port) {
        return -ENOMEM;
    }

    for (i = 0; i < serial->num_ports; i++) {
        ep = serial->port[i].port->host_endpoint;
        epd = &ep->desc;

        if ((epd->bmAttributes & USB_ENDPOINT_XFERTYPE_MASK) == USB_ENDPOINT_XFER_INT) {
            if (epd->bEndpointAddress & USB_DIR_IN) {
                serial->port[i].interrupt_in_endpoint = ep;
            } else {
                serial->port[i].interrupt_out_endpoint = ep;
            }
        }
    }

    kfree(port);
    return 0;
}
