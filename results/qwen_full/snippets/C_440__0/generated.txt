void mrb_gc_mark_context(mrb_state *mrb, struct mrb_context *c) {
  if (!c->stack) return;

  mrb_value *sp = c->stack + c->stack_size;
  mrb_value *base = c->stack + c->base_size;

  while (sp > base) {
    mrb_value *p = sp - 1;
    mrb_gc_mark(mrb, *p);
    if (mrb_immediate_p(*p)) {
      sp = p;
    }
    else {
      sp = p;
    }
  }
}
