void StartDownloadItem(download::DownloadCreateInfo* info,
                       const download::DownloadUrlParameters::OnStartedCallback& on_started,
                       download::InProgressDownloadManager::StartDownloadItemCallback callback,
                       uint32_t id) {
  // Ensure this is called on the UI thread
  DCHECK_CURRENTLY_ON(BrowserThread::UI);

  // Create a new active download item
  download::DownloadItem* download_item = download::DownloadItem::CreateActiveDownload(
      info, id);

  // Invoke the callback with the new download item
  callback.Run(download_item);

  // Process the new download creation and call OnDownloadStarted
  OnDownloadStarted(download_item, on_started);
}
