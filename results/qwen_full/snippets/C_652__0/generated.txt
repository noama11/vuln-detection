mrb_value
mrb_clone(mrb_state *mrb, mrb_value self)
{
  mrb_value clone;

  if (mrb_immediate_p(self)) {
    mrb_raise(mrb, E_TYPE_ERROR, "can't clone immediate value");
  }

  if (mrb_obj_singleton_class_p(mrb, self)) {
    mrb_raise(mrb, E_TYPE_ERROR, "can't clone singleton class");
  }

  clone = mrb_obj_alloc(mrb, mrb_class(self), mrb_obj_class(self));
  mrb_data_object_dup(mrb, clone, self);
  if (mrb_obj_singleton_class_p(mrb, self)) {
    mrb_raise(mrb, E_TYPE_ERROR, "can't clone singleton class");
  }

  mrb_obj_singleton_class(clone) = mrb_obj_singleton_class(self);
  mrb_obj_set_flag(clone, mrb_obj_flag(self));

  return clone;
}
