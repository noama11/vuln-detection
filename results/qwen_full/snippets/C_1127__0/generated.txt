AutocompleteMatch ConvertScoredHistoryMatchToAutocompleteMatch(const ScoredHistoryMatch* history_match, int score) {
    AutocompleteMatch match;
    match.url = history_match->url;
    match.description = history_match->description;
    match.score = score;

    // Format URL and description for display
    char formatted_url[URL_MAX_LENGTH];
    snprintf(formatted_url, sizeof(formatted_url), "URL: %s", match.url);
    match.display_url = formatted_url;

    char formatted_description[DESCRIPTION_MAX_LENGTH];
    snprintf(formatted_description, sizeof(formatted_description), "Description: %s", match.description);
    match.display_description = formatted_description;

    // Set inline autocomplete offsets if applicable
    if (history_match->has_inline_offset) {
        match.inline_offset = history_match->inline_offset;
    }

    // Handle formatting of term matches for URL and title
    if (history_match->has_term_match) {
        match.has_term_match = 1;
        match.term_match_start = history_match->term_match_start;
        match.term_match_end = history_match->term_match_end;
    }

    return match;
}
