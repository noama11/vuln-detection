int hwsim_new_radio(struct sk_buff *msg, struct genl_info *info)
{
    struct hwsim_new_radio_params params = { };
    int err;

    if (info->attrs[HWSIM_ATTR_REG_STRICTNESS]) {
        params.reg_strictness = nla_get_u8(info->attrs[HWSIM_ATTR_REG_STRICTNESS]);
    }

    if (info->attrs[HWSIM_ATTR_SUPPORT_P2P_DEVICE])
        params.support_p2p_device = true;

    if (info->attrs[HWSIM_ATTR_CHANNELS]) {
        params.channels = nla_get_u32(info->attrs[HWSIM_ATTR_CHANNELS]);
    }

    if (info->attrs[HWSIM_ATTR_RADIO_NAME]) {
        params.name = kstrdup(nla_get_string(info->attrs[HWSIM_ATTR_RADIO_NAME]), GFP_KERNEL);
        if (!params.name)
            return -ENOMEM;
    }

    if (info->attrs[HWSIM_ATTR_REGulatory_DOMAIN]) {
        params.regd_idx = nla_get_u32(info->attrs[HWSIM_ATTR_REGulatory_DOMAIN]);
        if (params.regd_idx >= ARRAY_SIZE(hwsim_regdoms)) {
            err = -EINVAL;
            goto free_name;
        }
    }

    err = mac80211_hwsim_new_radio(&params);
    if (err < 0)
        goto free_name;

    return 0;

free_name:
    kfree(params.name);
    return err;
}
