int set_subtitle_palette(unsigned int fd, unsigned int cmd, struct compat_video_spu_palette __user *up) {
    struct compat_video_spu_palette spu_palette;
    int ret;

    if (copy_from_user(&spu_palette, up, sizeof(spu_palette)))
        return -EFAULT;

    ret = ioctl(fd, cmd, &spu_palette);
    if (ret < 0)
        return ret;

    return 0;
}
