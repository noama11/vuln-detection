int ieee80211_radiotap_iterator_init(struct ieee80211_radiotap_iterator *iterator,
                                     struct ieee80211_radiotap_header *radiotap_header,
                                     int max_length,
                                     const struct ieee80211_radiotap_vendor_namespaces *vns) {
    if (!iterator || !radiotap_header || max_length < 0) {
        return -EINVAL;
    }

    iterator->radiotap = radiotap_header;
    iterator->max_length = max_length;
    iterator->vns = vns;
    iterator->this_arg = NULL;
    iterator->is_present = 0;
    iterator->arg_index = 0;
    iterator->next_index = 0;
    iterator->is_radiotap = 1;

    if (iterator->max_length < sizeof(struct ieee80211_radiotap_header)) {
        return -EINVAL;
    }

    iterator->current = (uint8_t *)radiotap_header;
    iterator->end = (uint8_t *)radiotap_header + max_length;

    return 0;
}
