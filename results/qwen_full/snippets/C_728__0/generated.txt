struct mobj *allocate_user_ta_memory(size_t size) {
    struct mobj *mobj = NULL;

    if (config_paged_user_ta_enabled()) {
        mobj = mobj_paged_alloc(size);
    } else {
        mobj = mobj_mm_alloc(size, 0, 0);
    }

    if (mobj != NULL) {
        memset(mobj, 0, sizeof(*mobj));
    }

    return mobj;
}
