void APE::Properties::analyzeOld() {
    // Read the block of data from the APE file header
    // Extract various properties
    int totalFrames = readTotalFrames();
    int compressionLevel = readCompressionLevel();
    int channels = readChannels();
    int sampleRate = readSampleRate();

    // Calculate the number of blocks per frame based on the file version and compression level
    int blocksPerFrame = calculateBlocksPerFrame(compressionLevel);

    // Compute the total number of blocks
    int totalBlocks = totalFrames * blocksPerFrame;

    // Derive the length of the audio in seconds and the bitrate in kbps
    double audioLengthInSeconds = static_cast<double>(totalBlocks) / sampleRate;
    int bitrateInKbps = calculateBitrate(totalBlocks, totalFrames, channels);

    // Update the properties of the APE::Properties object with the extracted values
    setTotalFrames(totalFrames);
    setCompressionLevel(compressionLevel);
    setChannels(channels);
    setSampleRate(sampleRate);
    setAudioLengthInSeconds(audioLengthInSeconds);
    setBitrateInKbps(bitrateInKbps);
}
