int path_to_string(struct super_block *sb, unsigned char *from, int fromlen, unsigned char *to, int tolen) {
    int i = 0, j = 0;

    if (tolen <= 0 || fromlen <= 0) {
        return -EINVAL;
    }

    while (i < fromlen) {
        if (j >= tolen - 1) {
            return -ENAMETOOLONG;
        }

        if (from[i] == '/') {
            to[j++] = '/';
            i++;
            continue;
        }

        if (from[i] == '.') {
            if (i + 1 < fromlen && from[i + 1] == '.' && (i + 2 >= fromlen || from[i + 2] == '/')) {
                if (j > 0 && to[j - 1] != '/') {
                    to[j++] = '/';
                }
                i += 2;
            } else {
                to[j++] = from[i++];
            }
        } else {
            while (i < fromlen && from[i] != '/') {
                if (j >= tolen - 1) {
                    return -ENAMETOOLONG;
                }
                to[j++] = from[i++];
            }
        }

        if (i < fromlen && from[i] == '/') {
            i++;
        }
    }

    to[j] = '\0';
    return 0;
}
