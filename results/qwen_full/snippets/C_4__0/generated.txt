int compat_ioctl_get_timestamp(struct net *net, struct socket *sock,
                               unsigned int cmd, void __user *up)
{
    int ret;
    unsigned long old_fs;
    void *timestamp;

    timestamp = kmalloc(sizeof(*timestamp), GFP_KERNEL);
    if (!timestamp)
        return -ENOMEM;

    old_fs = get_fs();
    set_fs(KERNEL_DS);

    ret = sock->ops->ioctl(sock, cmd, (unsigned long)timestamp);

    set_fs(old_fs);

    if (ret >= 0) {
        if (copy_to_user(up, timestamp, sizeof(*timestamp)))
            ret = -EFAULT;
    }

    kfree(timestamp);
    return ret;
}
