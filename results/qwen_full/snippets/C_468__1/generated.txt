#include <linux/compat.h>
#include <linux/uaccess.h>

int compat_sys_mbind(compat_ulong_t start, compat_ulong_t len,
                     compat_ulong_t mode, compat_ulong_t __user *nmask,
                     compat_ulong_t maxnode, compat_ulong_t flags)
{
    unsigned long nr_bits;
    unsigned long __user *bitmap;
    int err;

    if (!nmask)
        return -EINVAL;

    nr_bits = (maxnode + 7) / 8 * 8;
    bitmap = compat_alloc_user_space(nr_bits);
    if (copy_from_user(bitmap, nmask, nr_bits))
        return -EFAULT;

    err = sys_set_mempolicy(mode, bitmap, nr_bits / sizeof(long), flags);
    return err;
}
