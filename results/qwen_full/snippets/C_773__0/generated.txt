void handle_outbound_pinhole_timeout(struct upnphttp *h, const char *action, const char *ns) {
    if (!ipv6_forwarding_enabled()) {
        upnp_http_send_error(h, 403, "IPv6 forwarding is disabled");
        return;
    }

    char *internal_client_ip = NULL;
    int internal_port = 0;
    char *remote_host = NULL;
    char *protocol = NULL;

    if (!parse_request_parameters(h, &internal_client_ip, &internal_port, &remote_host, &protocol)) {
        upnp_http_send_error(h, 400, "Failed to parse request parameters");
        return;
    }

    if (strcasecmp(protocol, "TCP") != 0 && strcasecmp(protocol, "UDP") != 0) {
        upnp_http_send_error(h, 400, "Unsupported protocol");
        return;
    }

    int timeout = check_outbound_pinhole_timeout(internal_client_ip, internal_port, remote_host, protocol);

    if (timeout < 0) {
        upnp_http_send_error(h, 500, "Failed to retrieve pinhole timeout");
        return;
    }

    char *soap_response = build_soap_response(action, ns, timeout);
    upnp_http_send_response(h, 200, soap_response);
    free(soap_response);
}
