void dvb_usb_disconnect(struct usb_interface *intf)
{
    struct dvb_usb_device *d = usb_get_intfdata(intf);
    int i;

    if (!d)
        return;

    for (i = 0; i < d->num_adapters; i++)
        dvb_adapter_unregister(&d->adapter[i]);

    if (d->props.exit)
        d->props.exit(d);

    dvb_usbv2_exit(d);

    dev_info(&intf->dev, "DVB USB device disconnected\n");
}
