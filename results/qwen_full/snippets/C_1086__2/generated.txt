gfx::Rect GetWindowBoundsForClientBounds(const gfx::Rect& client_bounds) {
  int caption_height = 30; // Assuming a default caption height
  int close_button_width = 20; // Assuming a default close button width

  // Adjust the window bounds based on the client bounds and frame state
  int window_width = client_bounds.width + close_button_width;
  int window_height = client_bounds.height + caption_height;

  // Return the calculated window bounds
  return gfx::Rect(client_bounds.x, client_bounds.y, window_width, window_height);
}
