long compute_decompression_value() {
    inflate_state *state;

    if (strm == NULL || strm->state == NULL)
        return -1; // Specific error value

    state = (inflate_state *)strm->state;

    if (state->mode == COPY)
        return (state->back << 16) + state->length;
    else if (state->mode == MATCH)
        return (state->back << 16) + (state->was - state->length);
    else
        return 0; // Default return for unknown mode
}
