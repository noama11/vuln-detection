int get_bluetooth_socket_address(struct socket *sock, struct sockaddr *addr, int *len, int peer) {
    struct sockaddr_l2 *l2addr = (struct sockaddr_l2 *)addr;

    l2addr->l2_family = AF_BLUETOOTH;

    if (peer) {
        l2addr->l2_bdaddr = sock->sk->sk_daddr;
    } else {
        l2addr->l2_bdaddr = sock->sk->sk_saddr;
    }

    *len = sizeof(struct sockaddr_l2);
    return 0;
}
