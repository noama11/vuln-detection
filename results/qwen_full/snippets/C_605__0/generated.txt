int snd_usb_audio_probe(struct usb_interface *intf, const struct usb_device_id *usb_id)
{
    struct usb_device *dev = interface_to_usbdev(intf);
    int err;

    if (dev == NULL)
        return -ENODEV;

    if (dev->quirks & USB_QUIRK_BOOT_QUIRK) {
        err = snd_usb_apply_boot_quirk(dev, intf, usb_id);
        if (err < 0)
            return err;
    }

    if (dev->registered)
        return -EBUSY;

    for (int i = 0; i < MAX_AUDIO_DEVICES; i++) {
        if (!audio_devices[i].registered) {
            audio_devices[i].dev = dev;
            audio_devices[i].intf = intf;
            audio_devices[i].registered = true;

            if (dev->num_interfaces > 1) {
                for (int j = 1; j < dev->num_interfaces; j++) {
                    if (dev->interface[j] && dev->interface[j]->cur_altsetting->desc.bInterfaceClass == USB_CLASS_HID) {
                        // Handle multiple control interfaces
                        err = snd_usb_register_control_interface(dev, j);
                        if (err < 0)
                            return err;
                    }
                }
            }

            err = snd_card_register(audio_devices[i].card);
            if (err < 0)
                return err;

            return 0;
        }
    }

    return -ENOSPC;
}
