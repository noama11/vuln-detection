PixelChannels **allocate_pixel_channels(const Image *images) {
    int max_threads = get_max_threads();
    PixelChannels **channels = (PixelChannels **)malloc(max_threads * sizeof(PixelChannels *));
    if (!channels) {
        return NULL;
    }

    int cols = images->width;
    for (int i = 0; i < max_threads; i++) {
        channels[i] = (PixelChannels *)calloc(cols, sizeof(PixelChannels));
        if (!channels[i]) {
            for (int j = 0; j < i; j++) {
                free(channels[j]);
            }
            free(channels);
            return NULL;
        }
    }

    return channels;
}
