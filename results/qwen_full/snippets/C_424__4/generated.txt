size_t RegisterJPEGImage(void) {
  MagickInfo *magick_info;
  size_t length;

  const char *jpeg_version = "6b";
  const char *description = "JPEG image format";

  magick_info = (MagickInfo *) AcquireMagickMemory(sizeof(MagickInfo));
  *magick_info = (MagickInfo) {
    .signature = MagickSignature,
    .name = (char *) "JPE",
    .decoder = JPEGReadImage,
    .encoder = JPEGWriteImage,
    .thread_support = MagickFalse,
    .mime_type = "image/jpeg",
    .description = (char *) description,
    .module = (char *) "JPEG",
    .blob_support = MagickTrue,
    .adjoin = MagickTrue
  };

  if (strlen(jpeg_version) > 0) {
    char version_string[MaxTextLength];

    (void) FormatLocaleString(version_string, MaxTextLength, "%s %s", magick_info->name, jpeg_version);
    magick_info->version = (char *) version_string;
  }

  RegisterMagickInfo(magick_info);

  magick_info = (MagickInfo *) AcquireMagickMemory(sizeof(MagickInfo));
  *magick_info = (MagickInfo) {
    .signature = MagickSignature,
    .name = (char *) "JPEG",
    .decoder = JPEGReadImage,
    .encoder = JPEGWriteImage,
    .thread_support = MagickFalse,
    .mime_type = "image/jpeg",
    .description = (char *) description,
    .module = (char *) "JPEG",
    .blob_support = MagickTrue,
    .adjoin = MagickTrue
  };

  if (strlen(jpeg_version) > 0) {
    char version_string[MaxTextLength];

    (void) FormatLocaleString(version_string, MaxTextLength, "%s %s", magick_info->name, jpeg_version);
    magick_info->version = (char *) version_string;
  }

  RegisterMagickInfo(magick_info);

  magick_info = (MagickInfo *) AcquireMagickMemory(sizeof(MagickInfo));
  *magick_info = (MagickInfo) {
    .signature = MagickSignature,
    .name = (char *) "JPG",
    .decoder = JPEGReadImage,
    .encoder = JPEGWriteImage,
    .thread_support = MagickFalse,
    .mime_type = "image/jpeg",
    .description = (char *) description,
    .module = (char *) "JPEG",
    .blob_support = MagickTrue,
    .adjoin = MagickTrue
  };

  if (strlen(jpeg_version) > 0) {
    char version_string[MaxTextLength];

    (void) FormatLocaleString(version_string, MaxTextLength, "%s %s", magick_info->name, jpeg_version);
    magick_info->version = (char *) version_string;
  }

  RegisterMagickInfo(magick_info);

  magick_info = (MagickInfo *) AcquireMagickMemory(sizeof(MagickInfo));
  *magick_info = (MagickInfo) {
    .signature = MagickSignature,
    .name = (char *) "JPS",
    .decoder = JPEGReadImage,
    .encoder = JPEGWriteImage,
    .thread_support = MagickFalse,
    .mime_type = "image/jpeg",
    .description = (char *) description,
    .module = (char *) "JPEG",
    .blob_support = MagickTrue,
    .adjoin = MagickTrue
  };

  if (strlen(jpeg_version) > 0) {
    char version_string[MaxTextLength];

    (void) FormatLocaleString(version_string, MaxTextLength, "%s %s", magick_info->name, jpeg_version);
    magick_info->version = (char *) version_string;
  }

  RegisterMagickInfo(magick_info);

  magick_info = (MagickInfo *) AcquireMagickMemory(sizeof(MagickInfo));
  *magick_info = (MagickInfo) {
    .signature = MagickSignature,
    .name = (char *) "PJPEG",
    .decoder = JPEGReadImage,
    .encoder = JPEGWriteImage,
    .thread_support = MagickFalse,
    .mime_type = "image/jpeg",
    .description = (char *) description,
    .module = (char *) "JPEG",
    .blob_support = MagickTrue,
    .adjoin = MagickTrue
  };

  if (strlen(jpeg_version) > 0) {
    char version_string[MaxTextLength];

    (void) FormatLocaleString(version_string, MaxTextLength, "%s %s", magick_info->name, jpeg_version);
    magick_info->version = (char *) version_string;
  }

  RegisterMagickInfo(magick_info);

  length = sizeof(MagickInfo) * 5;

  return length;
}
