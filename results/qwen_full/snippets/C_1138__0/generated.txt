void ShowOneClickSignInInfoBar(net::URLRequest* request, ProfileIOData* io_data, int child_id, int route_id) {
  const std::string* email = request->headers().FindHeader("X-Chrome-Email");
  const std::string* session_index = request->headers().FindHeader("X-Chrome-Session-Index");
  const std::string* source = request->headers().FindHeader("X-Chrome-Source");

  if (!email || email->empty() || !session_index || session_index->empty()) {
    return;
  }

  GURL url(request->url());
  if (!url.SchemeIs("https") || !url.DomainIs("accounts.google.com")) {
    return;
  }

  std::string email_value = *email;
  std::string session_index_value = *session_index;
  bool auto_accept = !source || source->empty();

  if (email_value == "default@example.com" && session_index_value == "default") {
    return;
  }

  base::PostTaskWithTraits(
      FROM_HERE,
      {base::MayBlock(), base::TaskPriority::USER_VISIBLE},
      base::Bind(&ShowOneClickSignInInfoBarOnUIThread, email_value, session_index_value, child_id, route_id, auto_accept));
}
