int mov_read_dref(MOVContext *c, AVIOContext *pb, MOVAtom atom) {
    int i;
    int entries;

    if (!c->nb_streams) {
        return AVERROR_INVALIDDATA;
    }

    entries = avio_rb32(pb); // Read number of entries
    if (entries < 0) {
        return AVERROR_INVALIDDATA;
    }

    for (i = 0; i < entries; i++) {
        uint32_t type = avio_rl32(pb);
        avio_skip(pb, 4); // Skip size

        if (!strcmp("alis", (char *)&type)) {
            // Handle Macintosh alias records
            int size = avio_rb32(pb);
            if (size < 0) {
                return AVERROR_INVALIDDATA;
            }

            avio_skip(pb, size - 4); // Skip the rest of the alias data
        } else {
            // Process other dref types
            int size = avio_rb32(pb);
            if (size < 0) {
                return AVERROR_INVALIDDATA;
            }

            avio_skip(pb, size - 4); // Skip the rest of the entry
        }
    }

    return 0;
}
