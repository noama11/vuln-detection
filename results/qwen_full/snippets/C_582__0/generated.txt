int init_efi_state(struct boot_params *params, unsigned long params_load_addr,
                   unsigned int efi_map_offset, unsigned int efi_map_sz,
                   unsigned int efi_setup_data_offset)
{
    if (!efi_map_sz)
        return 0;

    if (!is_1to1_mapping_enabled())
        return 0;

    copy_efi_info(params, params_load_addr);
    setup_efi_memory_map(params, efi_map_offset, efi_map_sz);
    prepare_efi_setup_data(params, efi_setup_data_offset);

    return 0;
}
