int register_ppp_channel(struct net *net, struct ppp_channel *chan) {
    struct ppp_channel *new_chan;
    int ret = -ENOMEM;

    new_chan = kmalloc(sizeof(*new_chan), GFP_KERNEL);
    if (!new_chan)
        goto out;

    *new_chan = *chan;

    spin_lock_bh(&net->ppp_lock);
    list_add(&new_chan->list, &net->ppp_channels);
    net->ppp_num_channels++;
    spin_unlock_bh(&net->ppp_lock);

    ret = 0;

out:
    return ret;
}
