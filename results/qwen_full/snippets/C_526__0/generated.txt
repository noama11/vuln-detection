Image *ExtractPostscriptImage(Image *image, const ImageInfo *image_info, MagickOffsetType PS_Offset, ssize_t PS_Size, ExceptionInfo *exception) {
  ImageInfo *clone_info;
  Image *new_image;
  StringInfo *blob;
  char filename[MaxTextLength];
  FILE *file;

  clone_info = CloneImageInfo(image_info);
  if (clone_info == (ImageInfo *) NULL)
    return (Image *) image;

  GetPathComponent(image->filename, TailPath, filename);
  (void) FormatLocaleString(filename, MaxTextLength, "%s/PS_image_%s", GetTempDirectory(0, exception), filename);

  file = fopen(filename, "wb");
  if (file == (FILE *) NULL) {
    ThrowFileException(exception, FileOpenError, filename, image->filename);
    return (Image *) image;
  }

  blob = CloneStringInfo(image->blob);
  if (blob == (StringInfo *) NULL) {
    fclose(file);
    return (Image *) image;
  }

  if (WriteBlobString(blob, (const char *) NULL, (size_t) PS_Offset, file) != MagickTrue) {
    fclose(file);
    return (Image *) image;
  }

  if (WriteBlobString(blob, (const char *) NULL, (size_t) PS_Size, file) != MagickTrue) {
    fclose(file);
    return (Image *) image;
  }

  fclose(file);

  new_image = ReadImage(clone_info, exception);
  if (new_image == (Image *) NULL)
    return (Image *) image;

  (void) CloneImageAttributes(image, new_image);
  (void) CloneImageProfiles(image, new_image);
  (void) CloneImageArtifacts(image, new_image);

  (void) DestroyImageList(image);
  image = new_image;

  return image;
}
