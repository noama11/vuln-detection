void iriap_get_attr_by_class(struct iriap_cb *self, struct sk_buff *skb)
{
    struct iriap_object *obj;
    struct iriap_attribute *attr;
    char *class_name, *attr_name;
    int class_len, attr_len;
    int err;

    if (!self || !skb) {
        net_dbg_ratelimited("Invalid parameters in iriap_get_attr_by_class\n");
        return;
    }

    class_name = skb->data;
    class_len = strnlen(class_name, skb->len);
    if (class_len == 0 || class_len >= skb->len) {
        net_dbg_ratelimited("Invalid class name in socket buffer\n");
        return;
    }

    attr_name = class_name + class_len + 1;
    attr_len = strnlen(attr_name, skb->len - (attr_name - skb->data));
    if (attr_len == 0 || attr_len > skb->len - (attr_name - skb->data)) {
        net_dbg_ratelimited("Invalid attribute name in socket buffer\n");
        return;
    }

    obj = iriap_find_object_by_name(self, class_name);
    if (!obj) {
        net_dbg_ratelimited("Object '%s' not found\n", class_name);
        iriap_send_error_response(skb, IRIAP_ERR_OBJECT_NOT_FOUND);
        return;
    }

    attr = iriap_find_attribute_by_name(obj, attr_name);
    if (!attr) {
        net_dbg_ratelimited("Attribute '%s' not found in object '%s'\n", attr_name, class_name);
        iriap_send_error_response(skb, IRIAP_ERR_ATTRIBUTE_NOT_FOUND);
        return;
    }

    net_dbg_ratelimited("Retrieving value for attribute '%s' in object '%s'\n", attr_name, class_name);

    err = iriap_send_value_response(skb, attr->value, attr->value_len);
    if (err < 0) {
        net_dbg_ratelimited("Failed to send attribute value response\n");
    }
}
