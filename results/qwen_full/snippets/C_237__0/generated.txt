void clear_dtls_queues(SSL *s) {
    /* Clear unprocessed records queue */
    while (s->unprocessed_records) {
        SSL3_RECORD *r = s->unprocessed_records;
        s->unprocessed_records = r->next;
        if (r->data) {
            OPENSSL_free(r->data);
        }
        OPENSSL_free(r);
    }

    /* Clear processed records queue */
    while (s->processed_records) {
        SSL3_RECORD *r = s->processed_records;
        s->processed_records = r->next;
        if (r->data) {
            OPENSSL_free(r->data);
        }
        OPENSSL_free(r);
    }

    /* Clear buffered messages queue */
    while (s->buffered_messages) {
        DTLS1_BUFFERED_MESSAGE *bm = s->buffered_messages;
        s->buffered_messages = bm->next;
        if (bm->data) {
            OPENSSL_free(bm->data);
        }
        OPENSSL_free(bm);
    }

    /* Clear sent messages queue */
    while (s->sent_messages) {
        WPACKET *pkt = s->sent_messages;
        s->sent_messages = pkt->next;
        WPACKET_cleanup(pkt);
        OPENSSL_free(pkt);
    }

    /* Clear buffered application data queue */
    while (s->buffered_app_data) {
        DTLS1_BUFFERED_APP_DATA *bad = s->buffered_app_data;
        s->buffered_app_data = bad->next;
        if (bad->data) {
            OPENSSL_free(bad->data);
        }
        OPENSSL_free(bad);
    }

    /* Free record buffer if allocated */
    if (s->s3->rrec.data) {
        OPENSSL_free(s->s3->rrec.data);
        s->s3->rrec.data = NULL;
        s->s3->rrec.length = 0;
    }
}
