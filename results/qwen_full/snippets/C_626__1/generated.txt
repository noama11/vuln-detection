unsigned patch_jump(void *insnbuf, const void *target, unsigned long addr, unsigned len)
{
    long offset;
    unsigned char *buf = insnbuf;

    if (len < 5)
        return len;

    offset = (long)((char *)target - (char *)addr - 5);
    *buf++ = 0xE9; /* Jump opcode */
    *(long *)buf = offset;

    return 5;
}
