bool stratum_init(struct pool *pool) {
    json_t *req, *resp;
    char *sid = NULL;
    bool retry = false;

    do {
        req = json_object();
        if (!req)
            return false;

        json_object_set_new(req, "method", json_string("mining.subscribe"));
        json_object_set_new(req, "params", json_array());
        json_object_set_new(req, "id", json_integer(1));

        if (!stratum_send(pool, req))
            goto out;

        resp = stratum_recv(pool);
        if (!resp)
            goto out;

        if (!json_is_object(resp)) {
            applog(LOG_WARNING, "Stratum response is not an object");
            goto out;
        }

        json_t *error = json_object_get(resp, "error");
        if (error && !json_is_null(error)) {
            applog(LOG_ERR, "Stratum subscription error: %s", json_dumps(error, 0));
            goto out;
        }

        json_t *result = json_object_get(resp, "result");
        if (!result || !json_is_array(result) || json_array_size(result) < 3) {
            applog(LOG_ERR, "Invalid Stratum subscription response");
            goto out;
        }

        json_t *session_id = json_array_get(result, 0);
        json_t *extradata = json_array_get(result, 2);

        if (!session_id || !json_is_string(session_id)) {
            applog(LOG_ERR, "Missing or invalid session ID in Stratum response");
            goto out;
        }

        if (!extradata || !json_is_string(extradata)) {
            applog(LOG_ERR, "Missing or invalid extradata in Stratum response");
            goto out;
        }

        sid = strdup(json_string_value(session_id));
        if (!sid)
            goto out;

        free(pool->session_id);
        pool->session_id = sid;
        pool->extradata = strdup(json_string_value(extradata));
        if (!pool->extradata)
            goto out;

        retry = false;
    } while (!retry && !sid);

    return true;

out:
    if (sid)
        free(sid);
    return false;
}
