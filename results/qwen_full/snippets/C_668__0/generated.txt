int mbedtls_ssl_get_psk_identity_hint( mbedtls_ssl_context *ssl, unsigned char **p, unsigned char *end )
{
    size_t len;
    unsigned char *q = *p;

    if( ( q + 2 ) > end )
        return( MBEDTLS_ERR_SSL_BAD_HS_CLIENT_KEY_EXCHANGE );

    len = ( ( size_t )( q[0] ) << 8 ) | ( size_t )( q[1] );
    q += 2;

    if( ( q + len ) > end )
        return( MBEDTLS_ERR_SSL_BAD_HS_CLIENT_KEY_EXCHANGE );

    ssl->psk_identity_hint = NULL;
    ssl->psk_identity_hint_len = 0;

    *p = q + len;

    return( 0 );
}
