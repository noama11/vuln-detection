#include <string>
#include <openssl/rc4.h>

std::string computeUValue(std::string const& user_password, QPDF::EncryptionData const& data) {
    // Retrieve encryption key using the password and encryption data
    std::string key = data.generateKey(user_password);

    // Prepare buffer for U value
    unsigned char u_buffer[data.getUSize()];
    RC4_KEY rc4_key;

    // Initialize RC4 key
    RC4_set_key(&rc4_key, key.size(), reinterpret_cast<const unsigned char*>(key.c_str()));

    // Process the key with RC4 encryption to generate U value
    RC4(&rc4_key, data.getUSize(), reinterpret_cast<const unsigned char*>(data.getInitializationVector()), u_buffer);

    // Convert buffer to string and return
    return std::string(reinterpret_cast<char*>(u_buffer));
}
