void CreateBufferResource(PP_Instance instance, uint32_t size, HostResource* result_resource, ppapi::proxy::SerializedHandle* result_shm_handle) {
  // Check that the dispatcher and resource creation are valid
  if (!dispatcher || !result_resource || !result_shm_handle) {
    return;
  }

  // Create the buffer resource
  HostResource buffer_resource = CreateBuffer(instance, size);
  if (buffer_resource.pp_instance() == 0) {
    return;
  }

  // Retrieve the shared memory handle
  ppapi::proxy::SerializedHandle shm_handle = GetSharedMemoryHandle(buffer_resource);
  if (shm_handle.handle_type() == ppapi::proxy::kInvalidHandle) {
    return;
  }

  // Set the result resource and shared memory handle
  *result_resource = buffer_resource;
  *result_shm_handle = shm_handle;
