int decrypt_data(struct ceph_crypto_key *secret, void **p, void *end, void **obuf, size_t olen) {
    size_t len;
    char *data;
    int ret;

    if ((char *)*p + sizeof(len) > (char *)end) {
        return -EINVAL;
    }

    memcpy(&len, *p, sizeof(len));
    *p += sizeof(len);

    if ((char *)*p + len > (char *)end) {
        return -EINVAL;
    }

    data = (char *)malloc(len);
    if (!data) {
        return -ENOMEM;
    }

    memcpy(data, *p, len);
    *p += len;

    ret = ceph_decrypt(secret, data, len, *obuf, olen);
    if (ret < 0) {
        free(data);
        return ret;
    }

    if (check_version_and_magic(*obuf)) {
        free(data);
        return -EINVAL;
    }

    *obuf += ret;
    free(data);
    return ret;
}
