void handle_syscall_exit(void *ignore, struct pt_regs *regs, long ret)
{
    int syscall_num = regs->syscallno;

    if (syscall_num < 0 || syscall_num >= __NR_syscall_max)
        return;

    struct syscall_metadata *metadata = get_syscall_metadata(syscall_num);
    if (!metadata || !metadata->enabled)
        return;

    if (!metadata->exit_event)
        return;

    struct trace_buffer *buffer = prepare_trace_buffer();
    if (!buffer)
        return;

    buffer->syscall_number = syscall_num;
    buffer->return_value = ret;

    submit_trace_buffer(buffer);
}
