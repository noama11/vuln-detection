int sctp_get_assoc_stats(struct sock *sk, int len, char __user *optval, int __user *optlen)
{
    struct sctp_association *asoc;
    struct sctp_assoc_stats stats;
    int ret = 0;

    if (len < sizeof(stats.assoc_id)) {
        return -EINVAL;
    }

    if (copy_from_user(&stats, optval, sizeof(stats))) {
        return -EFAULT;
    }

    asoc = sctp_id2assoc(sk, stats.assoc_id);
    if (!asoc) {
        return -EINVAL;
    }

    stats.cumu_acks = asoc->stat_cumu_acks;
    stats.data_pkts_sent = asoc->stat_data_pkts_sent;
    stats.data_bytes_sent = asoc->stat_data_bytes_sent;
    stats.data_pkts_recv = asoc->stat_data_pkts_recv;
    stats.data_bytes_recv = asoc->stat_data_bytes_recv;
    stats.packets_retrans = asoc->stat_packets_retrans;
    stats.bytes_retrans = asoc->stat_bytes_retrans;
    stats.packets_sacked = asoc->stat_packets_sacked;
    stats.bytes_sacked = asoc->stat_bytes_sacked;
    stats.packets_lost = asoc->stat_packets_lost;
    stats.bytes_lost = asoc->stat_bytes_lost;
    stats.packets_retrans_rtx = asoc->stat_packets_retrans_rtx;
    stats.bytes_retrans_rtx = asoc->stat_bytes_retrans_rtx;
    stats.packets_retrans_timo = asoc->stat_packets_retrans_timo;
    stats.bytes_retrans_timo = asoc->stat_bytes_retrans_timo;
    stats.packets_retrans_fast = asoc->stat_packets_retrans_fast;
    stats.bytes_retrans_fast = asoc->stat_bytes_retrans_fast;
    stats.packets_retrans_sack = asoc->stat_packets_retrans_sack;
    stats.bytes_retrans_sack = asoc->stat_bytes_retrans_sack;
    stats.packets_retrans_other = asoc->stat_packets_retrans_other;
    stats.bytes_retrans_other = asoc->stat_bytes_retrans_other;
    stats.packets_discarded = asoc->stat_packets_discarded;
    stats.bytes_discarded = asoc->stat_bytes_discarded;
    stats.packets_abandoned = asoc->stat_packets_abandoned;
    stats.bytes_abandoned = asoc->stat_bytes_abandoned;
    stats.packets_retrans_abandoned = asoc->stat_packets_retrans_abandoned;
    stats.bytes_retrans_abandoned = asoc->stat_bytes_retrans_abandoned;
    stats.packets_retrans_failed = asoc->stat_packets_retrans_failed;
    stats.bytes_retrans_failed = asoc->stat_bytes_retrans_failed;
    stats.packets_retrans_unknown = asoc->stat_packets_retrans_unknown;
    stats.bytes_retrans_unknown = asoc->stat_bytes_retrans_unknown;
    stats.packets_retrans_invalid = asoc->stat_packets_retrans_invalid;
    stats.bytes_retrans_invalid = asoc->stat_bytes_retrans_invalid;
    stats.packets_retrans_other_unknown = asoc->stat_packets_retrans_other_unknown;
    stats.bytes_retrans_other_unknown = asoc->stat_bytes_retrans_other_unknown;
    stats.packets_retrans_other_invalid = asoc->stat_packets_retrans_other_invalid;
    stats.bytes_retrans_other_invalid = asoc->stat_bytes_retrans_other_invalid;
    stats.packets_retrans_other_abandoned = asoc->stat_packets_retrans_other_abandoned;
    stats.bytes_retrans_other_abandoned = asoc->stat_bytes_retrans_other_abandoned;
    stats.packets_retrans_other_failed = asoc->stat_packets_retrans_other_failed;
    stats.bytes_retrans_other_failed = asoc->stat_bytes_retrans_other_failed;
    stats.packets_retrans_other_discarded = asoc->stat_packets_retrans_other_discarded;
    stats.bytes_retrans_other_discarded = asoc->stat_bytes_retrans_other_discarded;
    stats.packets_retrans_other_unknown_discarded = asoc->stat_packets_retrans_other_unknown_discarded;
    stats.bytes_retrans_other_unknown_discarded = asoc->stat_bytes_retrans_other_unknown_discarded;
    stats.packets_retrans_other_invalid_discarded = asoc->stat_packets_retrans_other_invalid_discarded;
    stats.bytes_retrans_other_invalid_discarded = asoc->stat_bytes_retrans_other_invalid_discarded;
    stats.packets_retrans_other_abandoned_discarded = asoc->stat_packets_retrans_other_abandoned_discarded;
    stats.bytes_retrans_other_abandoned_discarded = asoc->stat_bytes_retrans_other_abandoned_discarded;
    stats.packets_retrans_other_failed_discarded = asoc->stat_packets_retrans_other_failed_discarded;
    stats.bytes_retrans_other_failed_discarded = asoc->stat_bytes_retrans_other_failed_discarded;

    if (copy_to_user(optval, &stats, sizeof(stats))) {
        return -EFAULT;
    }

    if (put_user(sizeof(stats), optlen)) {
        return -EFAULT;
    }

    return ret;
}
