size_t RegisterJPEGImage(void) {
  static const char *jpeg_version = "6b";
  static const char *jpeg_description = "JPEG (Joint Photographic Experts Group)";
  size_t entry_size = 0;

  MagickInfo *magick_info;

  magick_info = SetMagickInfo("JPE");
  magick_info->decoder = JPEGReadImage;
  magick_info->encoder = JPEGWriteImage;
  magick_info->thread_support = MagickLibVersion >= 0x1400L;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) AcquireString(jpeg_description);
  magick_info->version = (char *) AcquireString(jpeg_version);
  entry_size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("JPEG");
  magick_info->decoder = JPEGReadImage;
  magick_info->encoder = JPEGWriteImage;
  magick_info->thread_support = MagickLibVersion >= 0x1400L;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) AcquireString(jpeg_description);
  magick_info->version = (char *) AcquireString(jpeg_version);
  entry_size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("JPG");
  magick_info->decoder = JPEGReadImage;
  magick_info->encoder = JPEGWriteImage;
  magick_info->thread_support = MagickLibVersion >= 0x1400L;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) AcquireString(jpeg_description);
  magick_info->version = (char *) AcquireString(jpeg_version);
  entry_size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("JPS");
  magick_info->decoder = JPEGReadImage;
  magick_info->encoder = JPEGWriteImage;
  magick_info->thread_support = MagickLibVersion >= 0x1400L;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) AcquireString(jpeg_description);
  magick_info->version = (char *) AcquireString(jpeg_version);
  entry_size = RegisterMagickInfo(magick_info);

  magick_info = SetMagickInfo("PJPEG");
  magick_info->decoder = JPEGReadImage;
  magick_info->encoder = JPEGWriteImage;
  magick_info->thread_support = MagickLibVersion >= 0x1400L;
  magick_info->mime_type = "image/jpeg";
  magick_info->description = (char *) AcquireString(jpeg_description);
  magick_info->version = (char *) AcquireString(jpeg_version);
  entry_size = RegisterMagickInfo(magick_info);

  return entry_size;
}
