struct ipv6_rt_hdr *ip6_rthdr_dup(struct ipv6_rt_hdr *src, gfp_t gfp,
                                  unsigned int *mtu, int *maxfraglen,
                                  unsigned int fragheaderlen,
                                  struct sk_buff *skb, struct rt6_info *rt,
                                  bool pmtuprobe) {
    if (!src) {
        return NULL;
    }

    struct ipv6_rt_hdr *dup = kmemdup(src, src->hdrlen * 8, gfp);
    if (!dup) {
        return NULL;
    }

    if (pmtuprobe && skb && rt) {
        *mtu = min(*mtu, rt->dst.metrics[RTAX_MTU - 1]);
        *maxfraglen = *mtu - fragheaderlen;
    }

    return dup;
}
