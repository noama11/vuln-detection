int sctp_get_assoc_stats(struct sock *sk, int len, char __user *optval, int __user *optlen)
{
    struct sctp_association *asoc = sctp_id2assoc(sk);
    struct sctp_assoc_stats stats;
    int ret = 0;

    if (!asoc)
        return -EINVAL;

    if (len < sizeof(stats))
        return -EINVAL;

    memset(&stats, 0, sizeof(stats));

    stats.cnt = asoc->stats.cnt;
    stats.rto = asoc->rto;
    stats.max_rto = asoc->rto_max;
    stats.min_rto = asoc->rto_min;
    stats.srtt = asoc->srtt;
    stats.rttvar = asoc->rttvar;
    stats.pmtu = asoc->pmtu;
    stats.state = asoc->state;
    stats.active = asoc->active;
    stats.inactive = asoc->inactive;
    stats.established = asoc->established;
    stats.retransmits = asoc->retransmits;
    stats.highest_tsn = asoc->highest_tsn;
    stats.cumulative_tsn = asoc->cumulative_tsn;
    stats.retrans_out = asoc->retrans_out;
    stats.retrans_in = asoc->retrans_in;
    stats.out_of_order = asoc->out_of_order;
    stats.ack_delay = asoc->ack_delay;
    stats.ack_ratio = asoc->ack_ratio;
    stats.heartbeat_interval = asoc->hbinterval;
    stats.heartbeat_missed = asoc->heartbeat_missed;
    stats.heartbeat_sent = asoc->heartbeat_sent;
    stats.heartbeat_received = asoc->heartbeat_received;

    if (copy_to_user(optval, &stats, sizeof(stats)))
        return -EFAULT;

    if (optlen && put_user(sizeof(stats), optlen))
        return -EFAULT;

    return 0;
}
