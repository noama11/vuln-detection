int xc2028_set_config(struct dvb_frontend *fe, void *priv_cfg)
{
    struct xc2028_data *cfg = (struct xc2028_data *)priv_cfg;
    int result = 0;

    mutex_lock(&fe->lock);

    if (fe->ops.tuner_ops.set_config)
        fe->ops.tuner_ops.set_config(fe, priv_cfg);

    if (cfg->firmware_name && strcmp(cfg->firmware_name, fe->ops.tuner_ops.firmware_name)) {
        kfree(fe->ops.tuner_ops.firmware_name);
        fe->ops.tuner_ops.firmware_name = kstrdup(cfg->firmware_name, GFP_KERNEL);
        if (!fe->ops.tuner_ops.firmware_name) {
            mutex_unlock(&fe->lock);
            return -ENOMEM;
        }
    }

    if (!fe->ops.tuner_ops.firmware_name) {
        result = request_firmware_nowait(NULL, 0, fe->ops.tuner_ops.firmware_name,
                                         fe->dvb->adapter->device, GFP_KERNEL,
                                         &fe->ops.tuner_ops.firmware_download);
    }

    mutex_unlock(&fe->lock);

    return result;
}
