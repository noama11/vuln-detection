bool resolve_bool_expr(struct xkb_context *ctx, const ExprDef *expr, bool *set_rtrn) {
    switch (expr->type) {
        case EXPR_BOOL: {
            if (expr->data.bool_val == true || expr->data.bool_val == false) {
                *set_rtrn = expr->data.bool_val;
                return true;
            }
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Invalid boolean value in expression.\n");
            return false;
        }

        case EXPR_IDENT: {
            if (strcmp(expr->data.ident, "true") == 0) {
                *set_rtrn = true;
                return true;
            } else if (strcmp(expr->data.ident, "false") == 0) {
                *set_rtrn = false;
                return true;
            }
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Unknown identifier '%s' in boolean expression.\n", expr->data.ident);
            return false;
        }

        case EXPR_FIELD: {
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Field references are not supported in boolean expressions.\n");
            return false;
        }

        case EXPR_UNARY: {
            if (expr->data.unary.op != OP_NOT) {
                xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Unsupported unary operation in boolean expression.\n");
                return false;
            }

            bool operand;
            if (!resolve_bool_expr(ctx, expr->data.unary.expr, &operand)) {
                return false;
            }

            *set_rtrn = !operand;
            return true;
        }

        default: {
            xkb_context_log(ctx, XKB_LOG_LEVEL_ERROR, "Unsupported expression type in boolean resolution.\n");
            return false;
        }
    }
}
