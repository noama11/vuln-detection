u64 get_nested_netlink_attr(u64 ctx, u64 A, u64 X, u64 r4, u64 r5) {
    // Assume standard netlink attribute structure and layout
    struct nlattr *nla;
    u64 offset = 0;

    // Validate the socket buffer context and offset A
    if (!ctx || A >= ctx) {
        return 0;
    }

    nla = (struct nlattr *)(ctx + A);
    if (!nla || nla->nla_len < NLA_HDRLEN) {
        return 0;
    }

    // Traverse the netlink attributes to find the nested attribute of type X
    while (offset < nla->nla_len - NLA_HDRLEN) {
        struct nlattr *child = (struct nlattr *)((u64)nla + NLA_HDRLEN + offset);
        if (!child || (u64)child >= ctx + A + nla->nla_len) {
            break;
        }

        if (child->nla_type == X) {
            return (u64)child - ctx;
        }

        offset += NLA_ALIGN(child->nla_len);
    }

    return 0;
}
