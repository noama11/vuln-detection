int dissect_unicode_string(tvbuff_t *tvb, int offset, packet_info *pinfo, proto_tree *tree, guint8 *drep, char **data, int hf_name) {
    offset = (offset + 1) & ~1; /* Ensure offset is even */
    int length = tvb_reported_length_remaining(tvb, offset);
    if (length < 0) {
        return offset;
    }

    char *utf16_str = tvb_get_string_enc(wmem_packet_scope(), tvb, offset, length, ENC_UTF_16);
    proto_tree_add_string_format(tree, hf_name, tvb, offset, length, utf16_str, "Unicode String: %s", utf16_str);

    if (data) {
        *data = utf16_str;
    }

    return offset + length;
}
