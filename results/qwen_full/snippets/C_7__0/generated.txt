int copy_creds(struct task_struct *p, unsigned long clone_flags)
{
    struct cred *new;

    if ((clone_flags & CLONE_NEWUSER) || (clone_flags & CLONE_THREAD) == 0) {
        if (!atomic_inc_not_zero(&p->cred->usage)) {
            return -ENOMEM;
        }
        p->cred = get_cred(p->cred);
        return 0;
    }

    new = prepare_creds();
    if (!new) {
        return -ENOMEM;
    }

    if (clone_flags & CLONE_NEWUSER) {
        new->user_ns = create_user_ns(new);
        if (IS_ERR(new->user_ns)) {
            abort_creds(new);
            return PTR_ERR(new->user_ns);
        }
    }

    if (clone_flags & CLONE_THREAD) {
        new->thread_keyring = p->cred->thread_keyring;
        new->request_key_auth = p->cred->request_key_auth;
        if (new->thread_keyring) {
            key_get(new->thread_keyring);
        }
        if (new->request_key_auth) {
            key_get(new->request_key_auth);
        }
    }

    if (p->cred->process_keyring) {
        new->process_keyring = p->cred->process_keyring;
        key_get(new->process_keyring);
    }

    if (p->cred->session_keyring) {
        new->session_keyring = p->cred->session_keyring;
        key_get(new->session_keyring);
    }

    put_cred(p->cred);
    p->cred = new;

    return 0;
}
