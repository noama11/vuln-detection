bool processTextNode() {
    m_lastTextNode = current_node;
    int start_offset = 0;
    int offset_in_node = handleFirstLetter();
    if (!renderer) {
        return true;
    }
    int text_length = renderer->textLength();
    if (offset_in_node < 0 || offset_in_node > text_length) {
        return false;
    }
    start_offset = offset_in_node;
    int end_offset = text_length;
    int position = 0;
    while (position < text_length) {
        if (position == start_offset) {
            break;
        }
        position++;
    }
    int computed_length = end_offset - start_offset;
    assert(computed_length >= 0);
    assert(start_offset >= 0);
    assert(end_offset <= text_length);
    m_lastCharacterProcessed = renderer->characterAt(end_offset - 1);
    return m_shouldHandleFirstLetter;
}
