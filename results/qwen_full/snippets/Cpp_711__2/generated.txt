#include <string>
#include <openssl/md5.h>
#include <openssl/rc4.h>

std::string computeUValue(std::string const& user_password, QPDF::EncryptionData const& data) {
    // Step 1: Generate encryption key from password and encryption data
    // (Assuming key generation is handled by the encryption data object)
    unsigned char key[16]; // Assuming a 128-bit key
    data.generateKey(user_password, key, sizeof(key));

    // Step 2: Compute MD5 digest of padded password and encryption data's ID
    std::string padded_password = user_password;
    while (padded_password.size() < 32) {
        padded_password += '\0';
    }

    std::string input = padded_password + data.getID();
    unsigned char md5_digest[MD5_DIGEST_LENGTH];
    MD5_CTX md5;
    MD5_Init(&md5);
    MD5_Update(&md5, input.c_str(), input.size());
    MD5_Final(md5_digest, &md5);

    // Step 3: Process digest through RC4
    RC4_KEY rc4_key;
    RC4_set_key(&rc4_key, sizeof(key), key);
    unsigned char rc4_output[MD5_DIGEST_LENGTH];
    RC4(&rc4_key, MD5_DIGEST_LENGTH, md5_digest, rc4_output);

    // Step 4: Pad result with arbitrary data to ensure consistency
    std::string u_value(reinterpret_cast<char*>(rc4_output), MD5_DIGEST_LENGTH);
    while (u_value.size() < 32) {
        u_value += '\0';
    }

    // Step 5: Return the U value as a string of a specified length
    return u_value;
}
