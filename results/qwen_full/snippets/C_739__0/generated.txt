int genl_register_family(struct genl_family *family) {
    int err;

    if (!family || !family->ops || !family->name)
        return -EINVAL;

    if (family->ops->validate && family->ops->validate(family) < 0)
        return -EINVAL;

    mutex_lock(&genl_mutex);

    if (genl_family_find(family->name)) {
        mutex_unlock(&genl_mutex);
        return -EEXIST;
    }

    if (family->id == 0) {
        for (int i = 1; i < GENL_ID_RANGE; i++) {
            if (!genl_families[i]) {
                family->id = i;
                break;
            }
        }
        if (family->id == 0) {
            mutex_unlock(&genl_mutex);
            return -ENOSPC;
        }
    } else {
        if (family->id >= GENL_ID_RANGE || genl_families[family->id]) {
            mutex_unlock(&genl_mutex);
            return -EINVAL;
        }
    }

    if (family->n_attrs > 0) {
        family->attrs = kzalloc(family->n_attrs * sizeof(struct nlattr *), GFP_KERNEL);
        if (!family->attrs) {
            mutex_unlock(&genl_mutex);
            return -ENOMEM;
        }
    }

    genl_families[family->id] = family;

    if (family->ops->validate_mc_groups && family->ops->validate_mc_groups(family) < 0) {
        kfree(family->attrs);
        genl_families[family->id] = NULL;
        mutex_unlock(&genl_mutex);
        return -EINVAL;
    }

    genl_notify_family_registered(family);

    mutex_unlock(&genl_mutex);

    return 0;
}
