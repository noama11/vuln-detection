int kvm_pit_read(struct kvm_io_device *this, gpa_t addr, int len, void *data)
{
    int ret = -EOPNOTSUPP;
    uint8_t val = 0;

    if (addr >= 0x40 && addr <= 0x43) {
        switch (addr) {
        case 0x40: /* Channel 0 */
        case 0x41: /* Channel 1 */
        case 0x42: /* Channel 2 */
            val = pit_get_channel_data(addr - 0x40);
            break;
        case 0x43: /* Command port */
            val = pit_get_command_register();
            break;
        }

        if (len == 1) {
            *(uint8_t *)data = val;
        } else if (len == 2) {
            *(uint16_t *)data = val;
        } else {
            *(uint32_t *)data = val;
        }

        ret = 0;
    }

    return ret;
}
