bool oom_reap_task_mm(struct task_struct *tsk, struct mm_struct *mm)
{
    bool success = false;
    struct vm_area_struct *vma;

    down_read(&mm->mmap_lock);

    if (!atomic_read(&mm->mm_users)) {
        up_read(&mm->mmap_lock);
        return false;
    }

    if (test_bit(MMF_OOM_SKIP, &mm->flags)) {
        up_read(&mm->mmap_lock);
        return false;
    }

    set_bit(MMF_OOM_REAPING, &mm->flags);

    for (vma = mm->mmap; vma; vma = vma->vm_next) {
        if (vma->vm_flags & (VM_ANON | VM_SHARED))
            do_munmap(mm, vma->vm_start, vma->vm_end - vma->vm_start, 0);
    }

    clear_bit(MMF_OOM_REAPING, &mm->flags);
    up_read(&mm->mmap_lock);

    success = true;
    return success;
}
