void handle_reception_of_savable_resource_links(std::vector<GURL>& resources_list, std::vector<Referrer>& referrers_list, std::vector<GURL>& frames_list) {
    // Check if in correct wait state (assumed to be a global or context variable)
    if (!is_in_correct_wait_state) {
        return;
    }

    int total_items_to_save = resources_list.size() + frames_list.size();
    if (download_in_progress) {
        download.total_bytes += total_items_to_save * AVERAGE_RESOURCE_SIZE;
    }

    for (size_t i = 0; i < resources_list.size(); ++i) {
        const GURL& url = resources_list[i];
        const Referrer& referrer = referrers_list[i];
        if (url.is_valid()) {
            SaveItem item;
            item.url = url;
            item.referrer = referrer;
            waiting_queue.push(item);
        }
    }

    for (const GURL& frame_url : frames_list) {
        if (frame_url.is_valid()) {
            SaveItem item;
            item.url = frame_url;
            waiting_queue.push(item);
        }
    }

    if (waiting_queue.empty()) {
        handle_user_cancellation();
    }

    update_wait_state();
    initiate_saving_process();
}
