int handle_invalid_entry(struct fuse_conn *fc, unsigned int size, struct fuse_copy_state *cs) {
    char *name = NULL;
    struct output_data *out = NULL;
    int ret = -EINVAL;

    if (size == 0 || size > MAX_NAME_LENGTH) {
        goto out;
    }

    name = malloc(size);
    if (!name) {
        ret = -ENOMEM;
        goto out;
    }

    if (fuse_copy_state_read(cs, name, size, 0) != size) {
        ret = -EFAULT;
        goto free_name;
    }

    out = calloc(1, sizeof(*out));
    if (!out) {
        ret = -ENOMEM;
        goto free_name;
    }

    if (validate_name_length(name, size) < 0) {
        ret = -EINVAL;
        goto free_out;
    }

    process_name_data(out, name, size);

    if (compute_hash(out) < 0) {
        ret = -EINVAL;
        goto free_out;
    }

    if (reverse_invalid_entry(fc, out) < 0) {
        ret = -EFAULT;
        goto free_out;
    }

    ret = 0;

free_out:
    free(out);
free_name:
    free(name);
out:
    return ret;
}
