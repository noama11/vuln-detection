scoped_refptr<VideoFrame> CloneVideoFrameWithLayout(const VideoFrame* src_frame, const VideoFrameLayout& dst_layout) {
    if (!src_frame->is_mappable() || src_frame->format() != dst_layout.format()) {
        LOG(ERROR) << "Source frame is not mappable or format does not match destination layout.";
        return nullptr;
    }

    scoped_refptr<VideoFrame> dst_frame = VideoFrame::Create(dst_layout, src_frame->timestamp());
    if (!dst_frame) {
        LOG(ERROR) << "Failed to create destination VideoFrame.";
        return nullptr;
    }

    for (size_t i = 0; i < src_frame->num_planes(); ++i) {
        const uint8_t* src_data = src_frame->data(i);
        uint8_t* dst_data = dst_frame->data(i);
        size_t size = src_frame->rows(i) * src_frame->stride(i);
        memcpy(dst_data, src_data, size);
    }

    return dst_frame;
}
