int add_route_to_fib6(struct fib6_node *root, struct rt6_info *rt, struct nl_info *info) {
    int err;

    if ((info->nl_net->event & RTM_NEWROUTE) == 0)
        return -EINVAL;

    err = fib6_add_1(root, rt, info);
    if (err < 0)
        return err;

    if (rt->rt6i_src) {
        struct fib6_node *subtree;

        subtree = fib6_node_create_subtree(root, rt->rt6i_src);
        if (!subtree) {
            fib6_remove_route(rt);
            return -ENOMEM;
        }

        err = fib6_add_source_route(subtree, rt, info);
        if (err < 0) {
            fib6_remove_route(rt);
            return err;
        }
    }

    rt6_info_hold(rt);
    return 0;
}
