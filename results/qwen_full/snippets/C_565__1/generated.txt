MagickBooleanType SetQuantumDepth(const Image *image, QuantumInfo *quantum_info, const size_t depth) {
  assert(image != (const Image *) NULL);
  assert(image->signature == MagickSignature);
  assert(quantum_info != (QuantumInfo *) NULL);
  assert(quantum_info->signature == QuantumInfoSignature);

#if defined(MAGICKCORE_DEBUG)
  (void) LogMagickEvent(CoderEvent, GetMagickModule(), "  Setting depth for image: %s", image->filename);
#endif

  quantum_info->depth = depth;

  if (quantum_info->quantum_format == UndefinedQuantumFormat)
    return(MagickTrue);

  if (quantum_info->quantum_format == IntegerQuantumFormat)
    {
      if (depth > sizeof(size_t)*8)
        return(MagickFalse);
    }

  return(MagickTrue);
}
