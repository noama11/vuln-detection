void fillRect(const Rect& r, const void* pix) {
    if (r.isEmpty())
        return;

    int width = r.width();
    int height = r.height();
    int bytesPerPixel = r.bytesPerPixel();

    if (bytesPerPixel == 1) {
        // Single byte per pixel, fill with a single byte value
        unsigned char pixelValue = *(unsigned char*)pix;
        for (int y = 0; y < height; ++y) {
            unsigned char* row = r.buffer + y * r.rowBytes;
            for (int x = 0; x < width; ++x) {
                row[x] = pixelValue;
            }
        }
    } else {
        // Multiple bytes per pixel, copy pixel data
        for (int y = 0; y < height; ++y) {
            unsigned char* row = r.buffer + y * r.rowBytes;
            for (int x = 0; x < width; ++x) {
                memcpy(row + x * bytesPerPixel, pix, bytesPerPixel);
            }
        }
    }

    r.commit();
}
