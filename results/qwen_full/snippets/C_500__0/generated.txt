size_t WritePSDChannelBlob(const PSDInfo *psd_info, const ImageInfo *image_info,
                           Image *image, Image *next_image,
                           QuantumType quantum_type, unsigned char *compact_pixels,
                           MagickOffsetType size_offset, MagickBooleanType separate,
                           ExceptionInfo *exception) {
  size_t number_pixels = 0;
  ssize_t y;
  unsigned char *pixels = NULL;
  MagickBooleanType status = MagickTrue;

  if (image->depth == 1)
    number_pixels = (image->columns * image->rows + 7) / 8;

  for (y = 0; y < (ssize_t) image->rows; y++) {
    pixels = (unsigned char *) AcquireQuantumMemory(image->columns, sizeof(*pixels));
    if (!pixels) {
      ThrowBinaryException(ResourceLimitError, "MemoryAllocationFailed", image->filename);
      return 0;
    }

    ExportQuantumPixels(image, (Cache) NULL, quantum_type, image->columns, 1, pixels, exception);
    if (psd_info->compression == 1) {
      /* RLE compression logic here */
    }
    else if (psd_info->compression == 2) {
      /* ZIP compression logic here */
    }

    number_pixels += image->columns;
    pixels = (unsigned char *) RelinquishMagickMemory(pixels);
  }

  return number_pixels;
}
