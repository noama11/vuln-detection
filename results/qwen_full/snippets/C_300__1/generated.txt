int receive_message_with_info(int sockfd, void *buf, size_t *buflen, int flags,
                              struct in6_addr *addr, uint32_t *ifindex, int *hoplimit) {
    struct msghdr msg;
    union {
        struct cmsghdr hdr;
        char data[CMSG_SPACE(sizeof(struct in6_pktinfo)) + CMSG_SPACE(sizeof(int))];
    } cmsgbuf;
    struct iovec iov;

    iov.iov_base = buf;
    iov.iov_len = *buflen;

    msg.msg_name = NULL;
    msg.msg_namelen = 0;
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;
    msg.msg_control = cmsgbuf.data;
    msg.msg_controllen = sizeof(cmsgbuf);
    msg.msg_flags = 0;

    ssize_t len = recvmsg(sockfd, &msg, flags);
    if (len < 0)
        return -1;

    *buflen = len;

    struct cmsghdr *cmsg;
    for (cmsg = CMSG_FIRSTHDR(&msg); cmsg != NULL; cmsg = CMSG_NXTHDR(&msg, cmsg)) {
        if (cmsg->cmsg_level == IPPROTO_IPV6) {
            if (cmsg->cmsg_type == IPV6_PKTINFO) {
                struct in6_pktinfo *pktinfo = (struct in6_pktinfo *)CMSG_DATA(cmsg);
                if (addr)
                    *addr = pktinfo->ipi6_addr;
                if (ifindex)
                    *ifindex = pktinfo->ipi6_ifindex;
            } else if (cmsg->cmsg_type == IPV6_HOPLIMIT) {
                if (hoplimit)
                    *hoplimit = *(int *)CMSG_DATA(cmsg);
            }
        }
    }

    return 0;
}
