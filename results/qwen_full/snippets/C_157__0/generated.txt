OM_uint32 spnego_gss_context_res_select(OM_uint32 *minor_status,
                                        spnego_gss_ctx_id_t sc,
                                        OM_uint32 acc_negState,
                                        gss_OID supportedMech,
                                        gss_buffer_t *responseToken,
                                        gss_buffer_t *mechListMIC,
                                        OM_uint32 *negState,
                                        send_token_flag *tokflag) {
    OM_uint32 major_status;
    gss_OID_set mech_set = NULL;
    int found = 0;

    *minor_status = 0;

    /* Delete the current security context */
    if (sc != NULL) {
        gss_delete_sec_context(&major_status, &sc, NULL);
    }

    /* Check if the supported mechanism is in the mechanism set */
    if (sc != NULL && sc->mech_set != NULL) {
        gss_test_oid_set_member(&major_status, supportedMech, sc->mech_set, &found);
        if (major_status != GSS_S_COMPLETE || !found) {
            return GSS_S_BAD_MECH;
        }
    }

    /* Validate the negotiation state */
    if (acc_negState != GSS_C_NO_CONTEXT &&
        acc_negState != GSS_S_CONTINUE_NEEDED) {
        return GSS_S_BAD_SIG;
    }

    /* Set the internal mechanism */
    if (sc != NULL) {
        sc->internal_mech = supportedMech;
    }

    /* Update the negotiation state */
    if (negState != NULL) {
        *negState = acc_negState;
    }

    /* Update the token send flag */
    if (tokflag != NULL) {
        *tokflag = SEND_TOKEN_RESPONSE;
    }

    return GSS_S_COMPLETE;
}
