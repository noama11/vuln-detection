int handle_sa_flush_notify(const struct km_event *c)
{
    struct sk_buff *skb;
    struct sadb_msg *msg;

    skb = alloc_skb(sizeof(*msg), GFP_KERNEL);
    if (!skb)
        return -ENOBUFS;

    msg = skb_put(skb, sizeof(*msg));
    msg->sadb_msg_type = SADB_MSG_FLUSH;
    msg->sadb_msg_satype = c->proto;
    msg->sadb_msg_len = sizeof(*msg) / sizeof(uint64_t);
    msg->sadb_msg_reserved = 0;
    msg->sadb_msg_seq = c->seq;
    msg->sadb_msg_pid = 0;

    netlink_broadcast(skb, 0, 0, NLG_DEFAULT, GFP_KERNEL);
    return 0;
}
