#include <stdint.h>

// Constants for shelf UI dimensions
#define ASH_SHELF_BUTTON_SIZE 48
#define ASH_SHELF_ICON_SIZE 24
#define ASH_SHELF_CONTROL_SIZE 32
#define ASH_SHELF_PADDING 8
#define ASH_SHELF_MARGIN 16

// Enumeration for focus cycling direction
typedef enum {
  FOCUS_CYCLE_FORWARD,
  FOCUS_CYCLE_BACKWARD
} FocusCycleDirection;

// Colors (as RGB values)
#define ASH_SHELF_BACKGROUND_COLOR 0x2E2E2EFF
#define ASH_SHELF_BUTTON_COLOR 0x3C3C3CFF
#define ASH_SHELF_ICON_COLOR 0xFFFFFFFF
#define ASH_SHELF_CONTROL_COLOR 0x5A5A5AFF

// Static access methods for constants
typedef struct {
  int button_size;
  int icon_size;
  int control_size;
  int padding;
  int margin;
  uint32_t background_color;
  uint32_t button_color;
  uint32_t icon_color;
  uint32_t control_color;
} ShelfConstants;

// Prevent implicit construction
static ShelfConstants* get_shelf_constants() {
  static ShelfConstants constants = {
    .button_size = ASH_SHELF_BUTTON_SIZE,
    .icon_size = ASH_SHELF_ICON_SIZE,
    .control_size = ASH_SHELF_CONTROL_SIZE,
    .padding = ASH_SHELF_PADDING,
    .margin = ASH_SHELF_MARGIN,
    .background_color = ASH_SHELF_BACKGROUND_COLOR,
    .button_color = ASH_SHELF_BUTTON_COLOR,
    .icon_color = ASH_SHELF_ICON_COLOR,
    .control_color = ASH_SHELF_CONTROL_COLOR
  };
  return &constants;
}
