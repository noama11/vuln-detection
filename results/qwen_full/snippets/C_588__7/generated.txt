MagickBooleanType SkipMipmapImages(Image* image, DDSInfo* dds_info, int pixel_size, ExceptionInfo* exception) {
  if ((dds_info->mipmap_count > 1) && ((dds_info->flags & DDSTexture) || (dds_info->flags & DDS立方图))) {
    MagickSizeType width = image->columns;
    MagickSizeType height = image->rows;
    MagickSizeType row_size = width * pixel_size;
    MagickSizeType image_size = row_size * height;

    for (MagickSizeType level = 1; level < dds_info->mipmap_count; level++) {
      MagickSizeType new_width = width / 2;
      MagickSizeType new_height = height / 2;
      MagickSizeType new_row_size = new_width * pixel_size;
      MagickSizeType new_image_size = new_row_size * new_height;

      if (new_width == 0 || new_height == 0)
        break;

      if (SeekBlob(image, image_size, SEEK_CUR) != (image_size + MagickPathOffset)) {
        (void) ThrowMagickException(exception, GetMagickModule(), CoderError, "UnableToSkipMipmap", " %s", image->filename);
        return MagickFalse;
      }

      width = new_width;
      height = new_height;
      row_size = new_row_size;
      image_size = new_image_size;
    }
  }

  return MagickTrue;
}
