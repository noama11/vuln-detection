MagickBooleanType WritePSDImageChannels(const PSDInfo *psd_info,
                                        const ImageInfo *image_info,
                                        Image *image, Image *next_image,
                                        MagickBooleanType separate,
                                        ExceptionInfo *exception) {
  MagickBooleanType status = MagickTrue;
  size_t number_pixels;
  Quantum *pixels = NULL;

  number_pixels = image->columns * image->rows;

  pixels = (Quantum *) AcquireQuantumMemory(number_pixels, sizeof(Quantum));
  if (!pixels) {
    ThrowMagickException(exception, GetMagickModule(), ResourceLimitError,
                         "UnableToAcquireMemory", "\n");
    return MagickFalse;
  }

  if (next_image != (Image *) NULL) {
    if (next_image->compression == RLECompression) {
      if (image->colorspace == GRAYColorspace || image->colorspace == RGBColorspace) {
        status = WriteRLECompressedData(image, pixels, exception);
      }
    }
  }

  if (image->alpha_trait != UndefinedPixelTrait) {
    status = WriteAlphaChannel(psd_info, image, pixels, exception);
  }

  if (separate != MagickFalse) {
    status = WriteSeparateChannels(psd_info, image, pixels, exception);
  }

  if (status != MagickFalse) {
    (void) SyncImageProgress(image, SaveImageTag, 0);
  }

  pixels = (Quantum *) RelinquishMagickMemory(pixels);
  return status;
}
