MagickBooleanType SetQuantumDepth(const Image *image, QuantumInfo *quantum_info, const size_t depth) {
  assert(image != (const Image *) NULL);
  assert(image->signature == MagickSignature);
  assert(quantum_info != (QuantumInfo *) NULL);
  assert(quantum_info->signature == QuantumInfoSignature);

  if (GetMagickBool("debug","quantum"))
    (void) LogMagickEvent(TraceEvent,GetMagickModule(),"  image: %s",image->filename);

  quantum_info->depth = depth;
  return(MagickTrue);
}
