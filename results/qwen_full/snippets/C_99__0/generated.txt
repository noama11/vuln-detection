void edgeport_drain(struct edgeport_port *port, unsigned long timeout, int flush)
{
    struct tty_struct *tty = port->tty;
    unsigned long flags;
    unsigned long start_time;
    int drained;

    if (!tty)
        return;

    start_time = jiffies;
    drained = 0;

    add_wait_queue(&port->wait, &current);

    set_current_state(TASK_INTERRUPTIBLE);

    while (!signal_pending(current) && !port->disconnected) {
        if (time_after(jiffies, start_time + timeout))
            break;

        if (port->xmit_cnt == 0) {
            drained = 1;
            break;
        }

        schedule_timeout(1);
    }

    remove_wait_queue(&port->wait, &current);

    if (flush && !port->disconnected)
        tty_flush(tty, 0);

    if (!drained && !port->disconnected) {
        spin_lock_irqsave(&port->lock, flags);
        port->drain_timeout = 1;
        spin_unlock_irqrestore(&port->lock, flags);

        schedule_timeout(timeout);
    }

    if (!port->disconnected)
        tty_wait_until_sent(tty, port->timeout);
}
