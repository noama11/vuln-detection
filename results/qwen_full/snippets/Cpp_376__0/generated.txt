AP4_HdlrAtom::AP4_HdlrAtom(AP4_UI32 size, AP4_UI08 version, AP4_UI32 flags, AP4_ByteStream& stream) :
    AP4_Atom(AP4_ATOM_TYPE_HDLR, size, version, flags)
{
    AP4_UI32  pre_defined[3];
    AP4_UI32  handler_type;
    AP4_UI32  reserved[2];
    AP4_Size  handler_name_size;
    char*     handler_name = NULL;

    // read the atom data
    stream.ReadBE32(&pre_defined[0]);
    stream.ReadBE32(&pre_defined[1]);
    stream.ReadBE32(&pre_defined[2]);
    stream.ReadBE32(&handler_type);
    stream.ReadBE32(&reserved[0]);
    stream.ReadBE32(&reserved[1]);

    // determine the handler name size
    handler_name_size = size - (AP4_Atom::GetHeaderSize() + 28);

    if (handler_name_size > 0) {
        // read the handler name
        handler_name = new char[handler_name_size];
        stream.Read(handler_name, handler_name_size);

        // handle Pascal string format
        if (handler_name_size > 1 && handler_name[0] < handler_name_size) {
            AP4_Size pascal_length = (AP4_Size)handler_name[0];
            if (pascal_length < handler_name_size - 1) {
                handler_name[pascal_length] = '\0';
            }
        } else {
            // ensure null-terminated C string
            handler_name[handler_name_size - 1] = '\0';
        }
    }

    // store the handler name
    m_HandlerType = handler_type;
    m_HandlerName = handler_name;
}
