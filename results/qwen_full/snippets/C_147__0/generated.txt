void sctp_update_association(struct sctp_association *asoc, struct sctp_association *new) {
    /* Copy new parameters from 'new' to 'asoc' */
    asoc->next_tsn = new->next_tsn;
    asoc->ssn_map = new->ssn_map;
    asoc->auth_key = new->auth_key;
    asoc->auth_key_len = new->auth_key_len;
    asoc->assoc_id = new->assoc_id;

    /* Remove peer addresses from 'asoc' not present in 'new' */
    struct sctp_peer_addr *tmp, *next;
    list_for_each_entry_safe(tmp, next, &asoc->peer_addrs, list) {
        int found = 0;
        list_for_each_entry(next, &new->peer_addrs, list) {
            if (memcmp(&tmp->addr, &next->addr, sizeof(tmp->addr)) == 0) {
                found = 1;
                break;
            }
        }
        if (!found) {
            list_del(&tmp->list);
            free(tmp);
        }
    }

    /* Copy over peer addresses from 'new' to 'asoc' if not already present */
    list_for_each_entry(tmp, &new->peer_addrs, list) {
        int found = 0;
        list_for_each_entry(next, &asoc->peer_addrs, list) {
            if (memcmp(&tmp->addr, &next->addr, sizeof(tmp->addr)) == 0) {
                found = 1;
                break;
            }
        }
        if (!found) {
            struct sctp_peer_addr *new_addr = malloc(sizeof(*new_addr));
            if (!new_addr)
                return; // Out of memory
            memcpy(new_addr, tmp, sizeof(*new_addr));
            list_add_tail(&new_addr->list, &asoc->peer_addrs);
        }
    }

    if (asoc->state == SCTP_ESTABLISHED) {
        asoc->next_tsn = new->next_tsn;
        asoc->ssn_map = 0;
        sctp_flush_ulp_reasm_queue(asoc);
        asoc->error_count = 0;
    } else {
        if (asoc->ssn_map != new->ssn_map) {
            asoc->ssn_map = new->ssn_map;
        }
    }

    /* Copy over authentication data */
    if (new->auth_key) {
        if (asoc->auth_key) {
            free(asoc->auth_key);
        }
        asoc->auth_key = malloc(new->auth_key_len);
        if (asoc->auth_key) {
            memcpy(asoc->auth_key, new->auth_key, new->auth_key_len);
            asoc->auth_key_len = new->auth_key_len;
        }
    }
}
