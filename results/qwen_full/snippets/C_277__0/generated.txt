#include <linux/sched.h>
#include <linux/errno.h>
#include <linux/audit.h>
#include <linux/string.h>

int apparmor_setprocattr(struct task_struct *task, char *name, void *value, size_t size)
{
    if (size == 0)
        return -EINVAL;

    if (current != task)
        return -EPERM;

    if (!value || strlen(name) == 0)
        return -EINVAL;

    if (value && ((char *)value)[size - 1] != '\0') {
        audit_log(current->audit_context, AUDIT_APPARMOR, "non-terminated string");
        return -EINVAL;
    }

    if (strcmp(name, "current") == 0) {
        int ret = apparmor_set_current_profile(value);
        if (ret < 0)
            return ret;
    } else if (strcmp(name, "exec") == 0) {
        int ret = apparmor_set_exec_profile(value);
        if (ret < 0)
            return ret;
    } else if (strcmp(name, "prev") == 0) {
        int ret = apparmor_set_prev_profile(value);
        if (ret < 0)
            return ret;
    } else {
        return -EINVAL;
    }

    return (int)size;
}
