int kvm_vm_create_vcpu(struct kvm *kvm, u32 id) {
    int ret;
    struct kvm_vcpu *vcpu;

    if (id >= KVM_MAX_VCPUS)
        return -EINVAL;

    vcpu = kvm_vcpu_create(kvm, id);
    if (!vcpu)
        return -ENOMEM;

    ret = kvm_vcpu_setup_preempt_notifier(vcpu);
    if (ret) {
        kvm_vcpu_free(vcpu);
        return ret;
    }

    if (atomic_read(&kvm->online_vcpus) >= KVM_MAX_VCPUS) {
        kvm_vcpu_free(vcpu);
        return -EINVAL;
    }

    ret = kvm_register_vcpu(vcpu);
    if (ret) {
        kvm_vcpu_free(vcpu);
        return ret;
    }

    atomic_inc(&kvm->online_vcpus);
    kvm_vcpu_allow_userspace(vcpu);

    ret = kvm_arch_post_vcpu_creation(vcpu);
    if (ret) {
        atomic_dec(&kvm->online_vcpus);
        kvm_unregister_vcpu(vcpu);
        kvm_vcpu_free(vcpu);
        return ret;
    }

    return 0;
}
