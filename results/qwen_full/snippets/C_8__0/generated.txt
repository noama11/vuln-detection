struct sk_buff *reassemble_ipv6_fragments(struct nf_ct_frag6_queue *fq, struct net_device *dev) {
    struct sk_buff *skb;
    int payload_len;

    if (!fq || !dev) {
        return NULL;
    }

    payload_len = fq->payload_len;
    if (payload_len <= 0) {
        return NULL;
    }

    skb = alloc_skb(payload_len, GFP_ATOMIC);
    if (!skb) {
        return NULL;
    }

    if (fq->first && fq->first->more) {
        if (!nf_ct_frag6_queue_shift(fq)) {
            kfree_skb(skb);
            return NULL;
        }
    }

    if (nf_ct_frag6_queue_reassemble(fq, skb, dev)) {
        kfree_skb(skb);
        return NULL;
    }

    skb->dev = dev;
    skb->protocol = htons(ETH_P_IPV6);

    return skb;
}
