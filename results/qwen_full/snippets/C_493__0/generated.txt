int vc4_prepare_binning(struct drm_device *dev, struct vc4_exec_info *exec)
{
    int ret;
    struct vc4_dev *vc4 = to_vc4_dev(dev);
    struct vc4_bo *bo;
    void *binning_cmd_list, *shader_records, *uniforms;
    uint32_t binning_cmd_offset, shader_records_offset, uniforms_offset;
    size_t binning_cmd_size, shader_records_size, uniforms_size;

    binning_cmd_size = exec->binning_cmd_size;
    shader_records_size = exec->shader_records_size;
    uniforms_size = exec->uniforms_size;

    binning_cmd_offset = 0;
    shader_records_offset = binning_cmd_offset + binning_cmd_size;
    uniforms_offset = shader_records_offset + shader_records_size;

    if (shader_records_offset + shader_records_size > vc4->binning_max_offset ||
        uniforms_offset + uniforms_size > vc4->binning_max_offset) {
        return -EINVAL;
    }

    binning_cmd_list = kmalloc(binning_cmd_size, GFP_KERNEL);
    if (!binning_cmd_list)
        return -ENOMEM;

    shader_records = kmalloc(shader_records_size, GFP_KERNEL);
    if (!shader_records) {
        ret = -ENOMEM;
        goto free_binning_cmd_list;
    }

    uniforms = kmalloc(uniforms_size, GFP_KERNEL);
    if (!uniforms) {
        ret = -ENOMEM;
        goto free_shader_records;
    }

    if (copy_from_user(binning_cmd_list, exec->binning_cmd, binning_cmd_size)) {
        ret = -EFAULT;
        goto free_uniforms;
    }

    if (copy_from_user(shader_records, exec->shader_records, shader_records_size)) {
        ret = -EFAULT;
        goto free_uniforms;
    }

    if (copy_from_user(uniforms, exec->uniforms, uniforms_size)) {
        ret = -EFAULT;
        goto free_uniforms;
    }

    ret = vc4_bo_create(dev, binning_cmd_offset + uniforms_offset, &bo);
    if (ret)
        goto free_uniforms;

    exec->bo = bo;

    ret = vc4_validate_binning_cmd_list(binning_cmd_list, binning_cmd_size);
    if (ret)
        goto free_bo;

    ret = vc4_validate_shader_records(shader_records, shader_records_size);
    if (ret)
        goto free_bo;

    ret = vc4_wait_for_previous_rendering(dev);
    if (ret)
        goto free_bo;

    exec->binning_cmd_list = binning_cmd_list;
    exec->shader_records = shader_records;
    exec->uniforms = uniforms;

    return 0;

free_bo:
    vc4_bo_unreference(bo);
free_uniforms:
    kfree(uniforms);
free_shader_records:
    kfree(shader_records);
free_binning_cmd_list:
    kfree(binning_cmd_list);
    return ret;
}
