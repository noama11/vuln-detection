void process_savable_resources(std::vector<GURL>& resources_list, std::vector<Referrer>& referrers_list, std::vector<GURL>& frames_list) {
    if (!is_wait_state_appropriate()) {
        return;
    }

    size_t total_items = resources_list.size() + frames_list.size();
    if (total_items == 0) {
        handle_user_cancellation();
        return;
    }

    update_download_progress(total_items);

    for (size_t i = 0; i < resources_list.size(); ++i) {
        SaveItem* item = create_save_item(resources_list[i], referrers_list[i]);
        add_to_waiting_queue(item);
    }

    for (size_t i = 0; i < frames_list.size(); ++i) {
        SaveItem* item = create_save_item(frames_list[i], Referrer());
        add_to_waiting_queue(item);
    }

    update_wait_state();
    initiate_saving_process();
}
