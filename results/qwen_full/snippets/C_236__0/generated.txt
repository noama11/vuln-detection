int verify_asn1_signature(const ASN1_ITEM *it, X509_ALGOR *a, ASN1_BIT_STRING *signature, void *asn, EVP_PKEY *pkey) {
    EVP_MD_CTX *ctx = NULL;
    const EVP_MD *md = NULL;
    int ret = -1;

    if (!a || !signature || !pkey) {
        return -1;
    }

    md = EVP_get_digestbyname(OBJ_nid2ln(a->algorithm));
    if (!md) {
        return -1;
    }

    ctx = EVP_MD_CTX_new();
    if (!ctx) {
        goto err;
    }

    if (!EVP_VerifyInit_ex(ctx, md, NULL, pkey, NULL)) {
        goto err;
    }

    if (!asn1_item_verify(ctx, it, asn, a, signature)) {
        goto err;
    }

    ret = EVP_VerifyFinal(ctx, signature->data, signature->length, pkey);
    if (ret <= 0) {
        ret = 0;
    }

err:
    if (ctx) {
        EVP_MD_CTX_free(ctx);
    }

    return ret;
}
