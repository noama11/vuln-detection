void verify_image_integrity(Gimp *gimp, gboolean with_unusual_stuff, gboolean compat_paths, gboolean use_gimp_2_8_features) {
  GimpImage *image;
  gchar *temp_file_path;
  GimpImage *loaded_image;
  GError *error = NULL;
  GFile *file;

  image = create_main_image(gimp, with_unusual_stuff, compat_paths, use_gimp_2_8_features);
  g_assert(image != NULL);

  temp_file_path = g_strdup_temp("image_XXXXXX", NULL);
  gimp_file_save(GIMP_FILE_SAVE_OVERWRITE, image, NULL, temp_file_path, &error);
  g_assert_no_error(error);

  file = g_file_new_for_path(temp_file_path);
  loaded_image = gimp_file_load(GIMP_FILE_LOAD_NORMAL, file, &error);
  g_assert_no_error(error);
  g_assert(loaded_image != NULL);

  g_assert(compare_images(image, loaded_image));

  g_file_delete(file, NULL, NULL);
  g_object_unref(file);
  g_free(temp_file_path);
  gimp_image_delete(loaded_image);
  gimp_image_delete(image);
}
