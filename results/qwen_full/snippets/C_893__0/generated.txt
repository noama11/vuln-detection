PixelChannels **allocate_pixel_channels(const Image *images) {
    // Retrieve the resource limit for threads
    long thread_limit = sysconf(_SC_THREAD_THREADS_MAX);
    if (thread_limit == -1) {
        return NULL; // Unable to determine thread limit
    }

    // Initialize variables for the number of columns and threads
    int num_columns = images[0].width * images[0].height;
    int num_threads = (int)thread_limit;

    // Allocate memory for the two-dimensional array of pixel channels
    PixelChannels **pixel_channels = (PixelChannels **)malloc(num_threads * sizeof(PixelChannels *));
    if (!pixel_channels) {
        return NULL; // Memory allocation failed
    }

    for (int i = 0; i < num_threads; ++i) {
        pixel_channels[i] = (PixelChannels *)malloc(num_columns * sizeof(PixelChannels));
        if (!pixel_channels[i]) {
            // Free previously allocated memory and return NULL
            for (int j = 0; j < i; ++j) {
                free(pixel_channels[j]);
            }
            free(pixel_channels);
            return NULL;
        }
    }

    return pixel_channels;
}
