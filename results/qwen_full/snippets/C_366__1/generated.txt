Sdb* process_verneed_section(ELFOBJ *bin, Elf_Shdr *shdr, int sz) {
    if (!bin || !shdr || sz <= 0) {
        return NULL;
    }

    Sdb *sdb = sdb_new0();
    if (!sdb) {
        return NULL;
    }

    void *data = elf_getdata(bin, shdr, &sz);
    if (!data) {
        sdb_free(sdb);
        return NULL;
    }

    Elf_Verneed *vn = data;
    while ((char *)vn < (char *)data + sz) {
        for (int i = 0; i < vn->vn_cnt; i++) {
            Elf_Verneed *vnaux = (Elf_Verneed *)((char *)vn + sizeof(Elf_Verneed));
            const char *version_name = elf_strptr(bin, shdr->sh_link, vnaux->vna_name);
            if (!version_name) {
                sdb_free(sdb);
                return NULL;
            }

            char key[256];
            snprintf(key, sizeof(key), "version.%s", version_name);
            sdb_num_set(sdb, key, vnaux->vna_version, 0);

            vn = (Elf_Verneed *)((char *)vnaux + vnaux->vna_next);
        }
    }

    return sdb;
}
