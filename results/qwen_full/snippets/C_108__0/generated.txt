int hidp_setup_hid_session(struct hidp_session *session, struct hidp_connadd_req *req)
{
    int ret = -ENOMEM;

    session->hid_dev = hid_allocate_device();
    if (!session->hid_dev)
        goto err_out;

    session->hid_dev->vendor = req->vendor;
    session->hid_dev->product = req->product;
    session->hid_dev->version = req->version;
    session->hid_dev->country = req->country;

    if (hid_is_blacklisted(session->hid_dev)) {
        ret = -ENODEV;
        goto err_free_dev;
    }

    session->hid_dev->phys = kstrdup(req->phys, GFP_KERNEL);
    if (!session->hid_dev->phys)
        goto err_free_dev;

    ret = hid_parse_report(session->hid_dev, req->report_desc, req->report_desc_size);
    if (ret)
        goto err_free_phys;

    ret = hid_register_device(session->hid_dev);
    if (ret)
        goto err_free_phys;

    return 0;

err_free_phys:
    kfree(session->hid_dev->phys);
err_free_dev:
    hid_free_device(session->hid_dev);
err_out:
    return ret;
}
