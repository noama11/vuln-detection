u32 kvm_lapic_get_current_count(struct kvm_lapic *apic)
{
    u64 period;
    u64 remaining;
    u32 count;

    if (!apic || !apic->base.shregs.divide_count)
        return 0;

    if (apic->base.shregs.initial_count == 0)
        return 0;

    period = apic->base.shregs.initial_count * 1000000000ULL;
    period /= apic->base.shregs.divide_count;

    remaining = apic->base.timer_expires - ktime_get_ns();
    if (remaining < 0)
        remaining = 0;

    count = (u32)((remaining % period) / (period / apic->base.shregs.initial_count));
    return count;
}
