ACPI_STATUS AcpiPsCompleteThisOp(ACPI_WALK_STATE *WalkState, ACPI_PARSE_OBJECT *Op, ACPI_STATUS Status)
{
    ACPI_PARSE_OBJECT *Parent;
    ACPI_PARSE_OBJECT *Prev;

    if (!WalkState || !Op) {
        return (AE_BAD_PARAMETER);
    }

    if (WalkState->ControlState) {
        WalkState->ControlState = NULL;
    }

    if (WalkState->ResultCount > 0) {
        WalkState->ResultCount--;
    }

    if (WalkState->ArgCount > 0) {
        WalkState->ArgCount--;
    }

    if (WalkState->ParseTreeRoot == Op) {
        WalkState->ParseTreeRoot = NULL;
    }

    Parent = Op->Common.Parent;
    Prev = Op->Common.Next;

    if (Parent) {
        if (Parent->Common.Value.Arg) {
            if (Parent->Common.Value.Arg == Op) {
                if (Prev) {
                    Parent->Common.Value.Arg = Prev;
                }
                else {
                    Parent->Common.Value.Arg = NULL;
                }
            }
        }
    }

    if (WalkState->ScopeInfo && WalkState->ScopeInfo->Prev) {
        WalkState->ScopeInfo = WalkState->ScopeInfo->Prev;
    }

    if (WalkState->ControlMethod) {
        WalkState->ControlMethod->ParamCount = WalkState->ArgCount;
    }

    if (ACPI_SUCCESS(Status) && Op->Common.DescType == ACPI_DESC_TYPE_PARSE_METHOD) {
        if (WalkState->ControlMethod) {
            WalkState->ControlMethod->Node = Op->Common.Node;
        }
    }

    if (WalkState->WalkCount > 0) {
        WalkState->WalkCount--;
    }

    return (Status);
}
