int sctp_peeloff(struct sock *sk, sctp_assoc_t id, struct socket **sockp) {
    struct sctp_association *asoc;
    int ret = -EINVAL;

    asoc = sctp_id2assoc(sk, id);
    if (!asoc)
        return -ENOENT;

    if (!sctp_assoc_can_be_peeled(asoc))
        return -EBUSY;

    ret = sock_create_kern(&init_net, PF_INET, SOCK_STREAM, IPPROTO_SCTP, sockp);
    if (ret < 0)
        return ret;

    sctp_copy_sock(*sockp, sk, asoc);
    sctp_assoc_migrate(asoc, *sockp);

    return 0;
}
