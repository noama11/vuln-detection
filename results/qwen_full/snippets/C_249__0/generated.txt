int handle_far_jump(struct x86_emulate_ctxt *ctxt) {
    uint16_t selector;
    struct desc_struct descriptor;

    if (ctxt->opcode == 0x1f || ctxt->opcode == 0x0f) {
        selector = ctxt->regs->ax;
    } else {
        selector = ctxt->regs->dx;
    }

    if (get_desc(&descriptor, selector, ctxt)) {
        return X86EMUL_UNHANDLEABLE;
    }

    ctxt->regs->ip = (ctxt->regs->ip & 0xffff0000) | (uint32_t)ctxt->opcode_len;
    ctxt->regs->cs = selector;
    return X86EMUL_CONTINUE;
}
