ssize_t receive_and_process_message(struct kiocb *iocb, struct socket *sock, struct msghdr *msg, size_t len, int flags) {
    struct sock *sk = sock->sk;
    struct sk_buff *skb;
    ssize_t copied = 0;
    int err = 0;

    if (sk->sk_state == TCP_LISTEN)
        return -EINVAL;

    if (sk->sk_receive_queue.all == NULL) {
        if (flags & MSG_DONTWAIT)
            return -EAGAIN;
        else
            return -EWOULDBLOCK;
    }

    skb = skb_dequeue(&sk->sk_receive_queue);
    if (!skb)
        return -EAGAIN;

    if (flags & MSG_OOB) {
        if (skb->len < 1) {
            kfree_skb(skb);
            return -EINVAL;
        }
        copied = 1;
        err = skb->data[0];
        kfree_skb(skb);
        return copied;
    }

    if (skb->len > len) {
        if (flags & MSG_TRUNC) {
            copied = skb->len;
        } else {
            copied = len;
            skb_trim(skb, len);
        }
    } else {
        copied = skb->len;
    }

    err = skb_copy_datagram_msg(skb, 0, msg, copied);
    if (err) {
        kfree_skb(skb);
        return err;
    }

    if (sock->type == SOCK_DGRAM) {
        struct sockaddr_in *addr = (struct sockaddr_in *)msg->msg_name;
        if (addr) {
            addr->sin_family = AF_INET;
            addr->sin_port = skb->skb_iif;
            addr->sin_addr.s_addr = skb->pkt_type;
        }
    }

    if (sock->type == SOCK_STREAM) {
        skb_queue_head(&sk->sk_receive_queue, skb);
    } else {
        kfree_skb(skb);
    }

    return copied;
}
