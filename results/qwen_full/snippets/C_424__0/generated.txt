size_t RegisterJPEGImage(void) {
  MagickInfo *magick_info;

  magick_info = AcquireMagickInfo("JPE", "JPEG", "Joint Photographic Experts Group JFIF format");
  magick_info->decoder = (AcquireImageHandler) JPEGReadImage;
  magick_info->encoder = (AcquireImageHandler) JPEGWriteImage;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = ConstantString("image/jpeg");
  magick_info->description = ConstantString("JPEG (Joint Photographic Experts Group)");
  magick_info->module = ConstantString("JPEG");
  (void) RegisterMagickInfo(magick_info);

  magick_info = AcquireMagickInfo("JPG", "JPEG", "Joint Photographic Experts Group JFIF format");
  magick_info->decoder = (AcquireImageHandler) JPEGReadImage;
  magick_info->encoder = (AcquireImageHandler) JPEGWriteImage;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = ConstantString("image/jpeg");
  magick_info->description = ConstantString("JPEG (Joint Photographic Experts Group)");
  magick_info->module = ConstantString("JPEG");
  (void) RegisterMagickInfo(magick_info);

  magick_info = AcquireMagickInfo("JPS", "JPEG", "Joint Photographic Experts Group JFIF format");
  magick_info->decoder = (AcquireImageHandler) JPEGReadImage;
  magick_info->encoder = (AcquireImageHandler) JPEGWriteImage;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = ConstantString("image/jpeg");
  magick_info->description = ConstantString("JPEG (Joint Photographic Experts Group)");
  magick_info->module = ConstantString("JPEG");
  (void) RegisterMagickInfo(magick_info);

  magick_info = AcquireMagickInfo("PJPEG", "JPEG", "Progressive Joint Photographic Experts Group format");
  magick_info->decoder = (AcquireImageHandler) JPEGReadImage;
  magick_info->encoder = (AcquireImageHandler) JPEGWriteImage;
  magick_info->thread_support = MagickTrue;
  magick_info->mime_type = ConstantString("image/jpeg");
  magick_info->description = ConstantString("Progressive JPEG");
  magick_info->module = ConstantString("JPEG");
  (void) RegisterMagickInfo(magick_info);

  return sizeof(MagickInfo);
}
