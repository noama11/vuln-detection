int get_sas_phy_event_counts(struct sas_phy *phy) {
    void *request;
    void *response;
    int result;

    request = kmalloc(COMMAND_BUFFER_SIZE, GFP_KERNEL);
    response = kmalloc(RESPONSE_BUFFER_SIZE, GFP_KERNEL);

    if (!request || !response) {
        kfree(request);
        kfree(response);
        return -ENOMEM;
    }

    initialize_request_command(request, phy->number);

    result = execute_task(request, response);
    if (result < 0) {
        goto cleanup;
    }

    phy->frame_status = get_frame_status(response);
    phy->invalid_dword_count = get_invalid_dword_count(response);
    phy->running_disparity_error_count = get_running_disparity_error_count(response);
    phy->loss_of_dword_sync_count = get_loss_of_dword_sync_count(response);
    phy->phy_reset_count = get_phy_reset_count(response);
    phy->task_reset_count = get_task_reset_count(response);
    phy->primitive_count = get_primitive_count(response);
    phy->ignore_primitive_count = get_ignore_primitive_count(response);

cleanup:
    kfree(request);
    kfree(response);
    return result;
}
