int open_url_and_init_io(AVFormatContext *s, AVIOContext **pb, const char *url, AVDictionary *opts, AVDictionary *opts2, int *is_http) {
    AVDictionary *tmp_opts = NULL;
    int ret;

    // Merge the options dictionaries
    if (opts) {
        av_dict_copy(&tmp_opts, opts, 0);
    }
    if (opts2) {
        av_dict_copy(&tmp_opts, opts2, AV_DICT_APPEND);
    }

    // Attempt to open the URL
    ret = avio_open2(pb, url, AVIO_FLAG_READ, &s->interrupt_callback, tmp_opts);
    if (ret < 0) {
        av_dict_free(&tmp_opts);
        return ret;
    }

    // Determine if the URL is HTTP
    if (av_strstart(url, "http://", NULL) || av_strstart(url, "https://", NULL)) {
        *is_http = 1;
    } else {
        *is_http = 0;
    }

    // Clean up the temporary options dictionary
    av_dict_free(&tmp_opts);

    return 0;
}
