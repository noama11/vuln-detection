TEE_Result populate_crypto_object(unsigned long obj, struct utee_attribute *usr_attrs, unsigned long attr_count) {
    TEE_Result res = TEE_ERROR_GENERIC;
    struct crypto_session *session = get_current_session();
    struct crypto_object *crypto_obj;

    if (!session)
        return TEE_ERROR_BAD_STATE;

    crypto_obj = session->objects[obj];
    if (!crypto_obj || crypto_obj->state != TEE_OBJECT_LIFECYCLE_UNINITIALISED || !crypto_obj->is_transient)
        return TEE_ERROR_BAD_PARAMETERS;

    crypto_obj->attributes = calloc(attr_count, sizeof(struct tee_attribute));
    if (!crypto_obj->attributes)
        return TEE_ERROR_OUT_OF_MEMORY;

    for (unsigned long i = 0; i < attr_count; i++) {
        res = copy_from_user(&crypto_obj->attributes[i], &usr_attrs[i], sizeof(struct utee_attribute));
        if (res != TEE_SUCCESS)
            break;
    }

    if (res != TEE_SUCCESS)
        goto cleanup;

    res = validate_attributes(crypto_obj, attr_count);
    if (res != TEE_SUCCESS)
        goto cleanup;

    crypto_obj->state = TEE_OBJECT_LIFECYCLE_INITIALIZED;
    res = TEE_SUCCESS;

cleanup:
    if (res != TEE_SUCCESS && crypto_obj->attributes) {
        free(crypto_obj->attributes);
        crypto_obj->attributes = NULL;
    }

    return res;
}
