bool IsDownloadDangerous(const DownloadItem& download, const FilePath& suggested_path, bool visited_referrer_before) {
  // Check if the download is an extension without approval
  if (download.is_extension && !download.is_approved) {
    // If off-store extension installations are not enabled, it's dangerous
    if (!IsOffStoreExtensionInstallationEnabled()) {
      return true;
    }
  }

  // Check if the file is dangerous based on its extension
  if (IsDangerousBasedOnExtension(suggested_path)) {
    return true;
  }

  // If the user has not interacted with the page hosting the download, it's dangerous
  if (!download.user_gesture && !visited_referrer_before) {
    return true;
  }

  return false;
}
