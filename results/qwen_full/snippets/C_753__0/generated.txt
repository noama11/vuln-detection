int set_floppy_geometry(unsigned int cmd, struct floppy_struct *g, int drive, int type, struct block_device *bdev) {
    if (!g || drive < 0 || drive >= N_DRIVE) {
        return -EINVAL;
    }

    if (type != -1) {
        if (!capable(CAP_SYS_ADMIN)) {
            return -EPERM;
        }

        if (type >= N_SUPPORTED_FLOPPY_TYPES) {
            return -EINVAL;
        }

        /* Update the floppy type and related sizes */
        /* (Assuming N_SUPPORTED_FLOPPY_TYPES and related logic are defined in file_context) */
        current_floppy_type[drive] = type;
        update_drive_sizes(drive);
    } else {
        if (check_disk_change(drive)) {
            return -ENODEV;
        }

        /* Update user parameters and manage buffer cache */
        memcpy(&user_params[drive], g, sizeof(struct floppy_struct));
        invalidate_floppy_cache(drive);
    }

    return 0;
}
