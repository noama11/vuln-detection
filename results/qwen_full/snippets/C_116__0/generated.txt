#include <linux/socket.h>
#include <linux/bluetooth.h>
#include <net/sock.h>

int get_rfcomm_address(struct socket *sock, struct sockaddr *addr, int *len, int peer) {
    struct sockaddr_rc *saddr = (struct sockaddr_rc *)addr;
    struct sock *sk = sock->sk;

    saddr->rc_family = AF_BLUETOOTH;
    saddr->rc_channel = sk->sk_rcvbuf;

    if (peer) {
        bacpy(&saddr->rc_addr, &bt_sk(sk)->dst);
    } else {
        bacpy(&saddr->rc_addr, &bt_sk(sk)->src);
    }

    *len = sizeof(struct sockaddr_rc);
    return 0;
}
