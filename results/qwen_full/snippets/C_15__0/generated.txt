void unregister_memcg_threshold(struct cgroup *cgrp, struct cftype *cft, struct eventfd_ctx *eventfd)
{
    struct mem_cgroup *memcg = cgroup_to_mem_cgroup(cgrp);
    struct memcg_threshold_ary *ary;
    struct memcg_threshold_ary *spare;
    struct memcg_threshold *thresholds;
    int i, j, new_size = 0;
    int type = get_threshold_type(cft);

    mutex_lock(&memcg->thresholds_lock);

    ary = &memcg->thresholds[type];
    spare = &memcg->thresholds_spare[type];

    for (i = 0; i < ary->size; i++) {
        if (ary->entries[i].eventfd != eventfd)
            new_size++;
    }

    thresholds = kmalloc_array(new_size, sizeof(*thresholds), GFP_KERNEL);

    for (i = 0, j = 0; i < ary->size; i++) {
        if (ary->entries[i].eventfd != eventfd) {
            thresholds[j++] = ary->entries[i];
        }
    }

    spare->size = new_size;
    spare->entries = thresholds;

    swap(ary->size, spare->size);
    swap(ary->entries, spare->entries);

    mutex_unlock(&memcg->thresholds_lock);
}
