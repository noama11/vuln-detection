xfs_buf_t*
xfs_buf_find(
	struct xfs_buftarg	*btp,
	struct xfs_buf_map	*map,
	int			nmaps,
	xfs_buf_flags_t		flags,
	xfs_buf_t		*new_bp)
{
	struct xfs_buf		*bp;
	xfs_daddr_t		daddr;
	int			i;
	uint			nblocks;
	uint			align;
	uint			total_bytes = 0;

	/*
	 * Calculate the total number of bytes to be searched and check for
	 * alignment.
	 */
	for (i = 0, nblocks = 0; i < nmaps; i++) {
		total_bytes += map[i].bm_len << btp->bt_sshift;
		nblocks += map[i].bm_len;
	}

	align = 1 << btp->bt_swidth;

	/*
	 * Search the buffer cache using the requested block mapping.
	 */
	daddr = map[0].bm_bn << btp->bt_sshift;
	bp = xfs_buf_find_by_bno(btp, daddr, total_bytes, flags);

	if (bp) {
		/*
		 * Check if the buffer is stale and attempt to lock it.
		 */
		if (xfs_buf_isstale(bp)) {
			xfs_buf_stale(bp);
			xfs_buf_relse(bp);
			return NULL;
		}

		if (!xfs_buf_lock(bp)) {
			/*
			 * Buffer is already locked, so we can't use it.
			 */
			xfs_buf_relse(bp);
			return NULL;
		}

		return bp;
	}

	/*
	 * No matching buffer found. If a new buffer is provided, insert it into
	 * the tree.
	 */
	if (new_bp) {
		new_bp->b_bn = map[0].bm_bn;
		new_bp->b_bno = daddr;
		new_bp->b_count = nblocks;
		new_bp->b_bufobj = &btp->bt_bufobj;

		xfs_buf_insert(new_bp);
		return new_bp;
	}

	return NULL;
}
