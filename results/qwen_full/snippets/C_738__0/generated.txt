int fm10k_init_module(void)
{
    int err;

    pr_info("FM10K driver version 1.0\n");
    pr_info("Copyright (c) 2023 Intel Corporation\n");

    fm10k_wq = create_workqueue("fm10k_wq");
    if (!fm10k_wq)
        return -ENOMEM;

    fm10k_debug_init();

    err = pci_register_driver(&fm10k_driver);
    if (err)
        destroy_workqueue(fm10k_wq);

    return err;
}
