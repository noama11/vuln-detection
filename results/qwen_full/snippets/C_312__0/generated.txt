int powermate_probe(struct usb_interface *intf, const struct usb_device_id *id) {
    struct usb_device *udev = interface_to_usbdev(intf);
    struct powermate_dev *pdev;
    int error;

    pdev = kzalloc(sizeof(*pdev), GFP_KERNEL);
    if (!pdev)
        return -ENOMEM;

    pdev->intf = intf;
    usb_set_intfdata(intf, pdev);

    pdev->input_dev = input_allocate_device();
    if (!pdev->input_dev) {
        error = -ENOMEM;
        goto free_dev;
    }

    pdev->input_dev->name = "Logitech PowerMate";
    pdev->input_dev->id.bustype = BUS_USB;
    pdev->input_dev->id.vendor = le16_to_cpu(udev->descriptor.idVendor);
    pdev->input_dev->id.product = le16_to_cpu(udev->descriptor.idProduct);
    pdev->input_dev->id.version = 0x0100;

    input_set_drvdata(pdev->input_dev, pdev);

    __set_bit(EV_REL, pdev->input_dev->evbit);
    __set_bit(EV_KEY, pdev->input_dev->evbit);
    __set_bit(BTN_0, pdev->input_dev->keybit);
    __set_bit(REL_DIAL, pdev->input_dev->relbit);

    error = input_register_device(pdev->input_dev);
    if (error)
        goto free_input_dev;

    pdev->urb = usb_alloc_urb(0, GFP_KERNEL);
    if (!pdev->urb) {
        error = -ENOMEM;
        goto unregister_input_dev;
    }

    usb_fill_int_urb(pdev->urb, udev, USB_DIR_IN | USB_TYPE_CLASS | USB_RECIP_INTERFACE,
                     pdev->data, sizeof(pdev->data), powermate_irq, pdev, 10);

    pdev->urb->transfer_flags |= URB_INTERRUPT;

    error = usb_submit_urb(pdev->urb, GFP_KERNEL);
    if (error)
        goto free_urb;

    switch (le16_to_cpu(udev->descriptor.idProduct)) {
        case 0x0001:
            pdev->name = "PowerMate";
            break;
        case 0x0002:
            pdev->name = "PowerMate Jog Dial";
            break;
        default:
            pdev->name = "Unknown PowerMate Device";
            break;
    }

    pdev->pulse = 0;
    pdev->brightness = 0;

    return 0;

free_urb:
    usb_free_urb(pdev->urb);
unregister_input_dev:
    input_unregister_device(pdev->input_dev);
free_input_dev:
    input_free_device(pdev->input_dev);
free_dev:
    kfree(pdev);
    return error;
}
