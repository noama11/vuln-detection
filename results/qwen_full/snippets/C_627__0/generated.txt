int git_pkt_parse(git_pkt **out, const char *line, size_t len) {
    git_pkt_ng *pkt;
    char *ref, *msg;
    size_t ref_len, msg_len;
    const char *end;

    pkt = (git_pkt_ng *)malloc(sizeof(git_pkt_ng));
    if (!pkt)
        return -1;

    // Parse reference
    ref = (char *)malloc(len + 1);
    if (!ref) {
        free(pkt);
        return -1;
    }

    // Find the end of the reference
    end = memchr(line, '\0', len);
    if (!end) {
        free(ref);
        free(pkt);
        return -1;
    }

    ref_len = (size_t)(end - line);
    if (ref_len > len) {
        free(ref);
        free(pkt);
        return -1;
    }

    memcpy(ref, line, ref_len);
    ref[ref_len] = '\0';

    pkt->ref = ref;
    pkt->ref_len = ref_len;

    // Parse message
    msg = (char *)malloc(len + 1);
    if (!msg) {
        free(ref);
        free(pkt);
        return -1;
    }

    // Skip the null byte
    line = end + 1;
    len -= ref_len + 1;

    if (len > 0) {
        msg_len = len;
        memcpy(msg, line, msg_len);
        msg[msg_len] = '\0';
    } else {
        msg[0] = '\0';
        msg_len = 0;
    }

    pkt->msg = msg;
    pkt->msg_len = msg_len;

    *out = (git_pkt *)pkt;
    return 0;
}
