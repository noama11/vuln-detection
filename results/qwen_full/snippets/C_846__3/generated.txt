ssize_t read_sound_timer_data(struct file *file, char __user *buffer, size_t count, loff_t *offset) {
    ssize_t bytes_read = 0;
    int ret;

    spin_lock(&device.spinlock);
    mutex_lock(&device.mutex);

    while (device.queue_head == device.queue_tail) {
        if (file->f_flags & O_NONBLOCK) {
            ret = -EAGAIN;
            goto error_unlock;
        }

        if (signal_pending(current)) {
            ret = -ERESTARTSYS;
            goto error_unlock;
        }

        wait_event_interruptible(device.read_queue, device.queue_head != device.queue_tail);
    }

    while (bytes_read < count && device.queue_head != device.queue_tail) {
        unsigned char data = device.queue[device.queue_tail];
        if (copy_to_user(buffer + bytes_read, &data, sizeof(data))) {
            ret = -EFAULT;
            goto error_unlock;
        }

        device.queue_tail = (device.queue_tail + 1) % DEVICE_QUEUE_SIZE;
        bytes_read += sizeof(data);
    }

    mutex_unlock(&device.mutex);
    spin_unlock(&device.spinlock);

    *offset += bytes_read;
    return bytes_read;

error_unlock:
    mutex_unlock(&device.mutex);
    spin_unlock(&device.spinlock);
    return ret;
}
