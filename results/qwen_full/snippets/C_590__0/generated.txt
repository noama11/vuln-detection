int handle_packet(int s, unsigned char* data, int len, struct sockaddr_in* s_in) {
    if (len == 5 && memcmp(data, "sorbo", 5) == 0) {
        // Prepare command for internet check
        const char* response = "internet_check";
        if (sendto(s, response, strlen(response), 0, (struct sockaddr*)s_in, sizeof(*s_in)) < 0) {
            return 1;
        }
        return 0;
    } else {
        // Process the packet
        if (len < 2) {
            return 1; // Invalid packet length
        }

        int packet_id = data[0];
        int packet_len = data[1];

        // Check for duplicate packet (example logic, adjust as needed)
        static int last_id = -1;
        if (packet_id == last_id) {
            return 0; // Duplicate packet, ignore
        }
        last_id = packet_id;

        // Send processed data back
        if (sendto(s, data, len, 0, (struct sockaddr*)s_in, sizeof(*s_in)) < 0) {
            return 1;
        }

        // Handle packet ID wrapping
        if (packet_id >= 255) {
            packet_id = 0;
        }

        return 0;
    }
}
