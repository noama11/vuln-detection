AP4_Result ReadFields(AP4_ByteStream& stream)
{
    // Call base class method to read common sample entry fields
    AP4_Result result = AP4_SampleEntry::ReadFields(stream);
    if (AP4_FAILED(result)) return result;

    // Read predefined values (6 bytes)
    AP4_UI08 predefined1[6];
    result = stream.Read(predefined1, 6);
    if (AP4_FAILED(result)) return result;

    // Read predefined (2 bytes)
    AP4_UI16 reserved1;
    result = stream.ReadUI16(reserved1);
    if (AP4_FAILED(result)) return result;

    // Read reserved (2 bytes)
    AP4_UI16 reserved2;
    result = stream.ReadUI16(reserved2);
    if (AP4_FAILED(result)) return result;

    // Read predefined values (3 x 4 bytes)
    for (int i = 0; i < 3; i++) {
        AP4_UI32 reserved;
        result = stream.ReadUI32(reserved);
        if (AP4_FAILED(result)) return result;
    }

    // Read dimensions
    AP4_UI16 width;
    AP4_UI16 height;
    result = stream.ReadUI16(width);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI16(height);
    if (AP4_FAILED(result)) return result;

    // Read resolutions (horizontal and vertical)
    AP4_UI32 horiz_resolution;
    AP4_UI32 vert_resolution;
    result = stream.ReadUI32(horiz_resolution);
    if (AP4_FAILED(result)) return result;
    result = stream.ReadUI32(vert_resolution);
    if (AP4_FAILED(result)) return result;

    // Read reserved (4 bytes)
    AP4_UI32 reserved3;
    result = stream.ReadUI32(reserved3);
    if (AP4_FAILED(result)) return result;

    // Read frame count
    AP4_UI16 frame_count;
    result = stream.ReadUI16(frame_count);
    if (AP4_FAILED(result)) return result;

    // Read and process compressor name (Pascal string)
    AP4_UI08 compressor_name_length;
    result = stream.Read(&compressor_name_length, 1);
    if (AP4_FAILED(result)) return result;

    // Ensure length is valid (max 31 bytes)
    if (compressor_name_length > 31) {
        compressor_name_length = 31;
    }

    AP4_UI08 compressor_name[32];
    if (compressor_name_length > 0) {
        result = stream.Read(compressor_name, compressor_name_length);
        if (AP4_FAILED(result)) return result;
    }

    // Ensure proper null termination
    compressor_name[compressor_name_length] = '\0';

    // Read padding to complete the 32-byte field
    AP4_UI08 padding_length = 31 - compressor_name_length;
    if (padding_length > 0) {
        AP4_UI08 padding[31];
        result = stream.Read(padding, padding_length);
        if (AP4_FAILED(result)) return result;
    }

    // Read depth
    AP4_UI16 depth;
    result = stream.ReadUI16(depth);
    if (AP4_FAILED(result)) return result;

    // Read predefined value
    AP4_SI16 predefined_final;
    result = stream.ReadUI16((AP4_UI16&)predefined_final);
    if (AP4_FAILED(result)) return result;

    return AP4_SUCCESS;
}
