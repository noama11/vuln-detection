struct RClass* get_real_class(struct RClass* cl) {
    if (cl == NULL) {
        return NULL;
    }
    while (cl != NULL) {
        if (MRB_CLASS_SINGLETONP(cl)) {
            cl = cl->superclass;
            continue;
        }
        if (cl->flags & MRB_FLAG_CLASS_INCLUDED) {
            cl = cl->superclass;
            continue;
        }
        return cl;
    }
    return NULL;
}
