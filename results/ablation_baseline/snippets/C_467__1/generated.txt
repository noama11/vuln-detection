#include <FLAC/stream_decoder.h>

typedef struct {
    const FLAC__Frame *current_frame;
    const int32_t * const *current_buffer;
} DecoderPrivateData;

void copy_buffer_data(DecoderPrivateData *private_data, const FLAC__Frame *frame);

FLAC__StreamDecoderWriteStatus flac_write_callback(
    const FLAC__StreamDecoder *decoder,
    const FLAC__Frame *frame,
    const int32_t * const *buffer,
    void *client_data
)
{
    (void)decoder;
    DecoderPrivateData *private_data = (DecoderPrivateData *)client_data;
    if (private_data == NULL) {
        return FLAC__STREAM_DECODER_WRITE_STATUS_ABORT;
    }
    private_data->current_frame = frame;
    private_data->current_buffer = buffer;
    copy_buffer_data(private_data, frame);
    return FLAC__STREAM_DECODER_WRITE_STATUS_CONTINUE;
}
