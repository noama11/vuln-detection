void
rs_filter_graph(RSFilter *filter)
{
	gchar *dot_filename, *png_filename;
	GString *str;
	FILE *f;
	gint ret;
	gchar *cmd;

	g_return_if_fail(RS_IS_FILTER(filter));

	str = g_string_new("digraph {\n\tnode [shape=plaintext];\n");
	rs_filter_graph_helper(str, filter);
	g_string_append_printf(str, "}\n");

	dot_filename = g_strdup_printf("/tmp/rs_filter_graph_%d.dot", g_random_int());
	png_filename = g_strdup_printf("/tmp/rs_filter_graph_%d.png", g_random_int());

	f = fopen(dot_filename, "w");
	if (f != NULL)
	{
		fputs(str->str, f);
		fclose(f);

		cmd = g_strdup_printf("dot -Tpng -o %s %s", png_filename, dot_filename);
		ret = system(cmd);
		if (ret != 0)
		{
			g_warning("Failed to convert DOT to PNG");
		}
		g_free(cmd);

		cmd = g_strdup_printf("gnome-open %s", png_filename);
		ret = system(cmd);
		if (ret != 0)
		{
			g_warning("Failed to open PNG file");
		}
		g_free(cmd);

		g_unlink(dot_filename);
	}

	g_string_free(str, TRUE);
	g_free(dot_filename);
	g_free(png_filename);
}
