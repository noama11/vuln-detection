ExprDef *
ExprAppendMultiKeysymList(ExprDef *expr, ExprDef *append)
{
    unsigned nSyms = darray_size(expr->keysym_list.syms);
    unsigned nEntries = darray_size(append->keysym_list.symsMapIndex);
    unsigned i;

    for (i = 0; i < nEntries; i++) {
        darray_append(expr->keysym_list.symsMapIndex,
                      nSyms + darray_item(append->keysym_list.symsMapIndex, i));
        darray_append(expr->keysym_list.symsNumEntries,
                      darray_item(append->keysym_list.symsNumEntries, i));
    }

    for (i = 0; i < darray_size(append->keysym_list.syms); i++) {
        darray_append(expr->keysym_list.syms,
                      darray_item(append->keysym_list.syms, i));
    }

    free(append);
    return expr;
}
