static
std::string
compute_U_value_R3(std::string const& user_password,
                   QPDF::EncryptionData const& data)
{
    std::string k1 = QPDF::compute_encryption_key(user_password, data);
    MD5 md5;
    md5.append(padding_string, sizeof(padding_string));
    std::string id = data.getID();
    md5.append(QUtil::unsigned_char_pointer(id), id.length());
    MD5::Digest digest;
    md5.finalize(&digest);
    unsigned char udata[sizeof(MD5::Digest)];
    memcpy(udata, digest, sizeof(MD5::Digest));
    for (int i = 0; i < 20; ++i)
    {
        unsigned char key[key_bytes];
        memcpy(key, QUtil::unsigned_char_pointer(k1), data.getLengthBytes());
        if (i % 2 == 1)
        {
            for (int j = 0; j < data.getLengthBytes(); ++j)
            {
                key[j] ^= i;
            }
        }
        iterate_rc4(udata, sizeof(MD5::Digest), key, data.getLengthBytes(), 1, false);
    }
    char result[key_bytes];
    memcpy(result, udata, sizeof(MD5::Digest));
    memcpy(result + sizeof(MD5::Digest),
           padding_string + sizeof(MD5::Digest),
           key_bytes - sizeof(MD5::Digest));
    return std::string(result, key_bytes);
}
