AP4_Result
AP4_VisualSampleEntry::ReadFields(AP4_ByteStream& stream)
{
    AP4_Result result;

    result = AP4_SampleEntry::ReadFields(stream);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(m_Predefined1);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(m_Reserved2);
    if (AP4_FAILED(result)) return result;

    result = stream.Read(m_Predefined2, sizeof(m_Predefined2));
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(m_Width);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(m_Height);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI32(m_HorizResolution);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI32(m_VertResolution);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI32(m_Reserved3);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(m_FrameCount);
    if (AP4_FAILED(result)) return result;

    AP4_UI08 compressor_name[32];
    result = stream.Read(compressor_name, 32);
    if (AP4_FAILED(result)) return result;

    AP4_UI08 length = compressor_name[0];
    if (length > 31) length = 31;
    compressor_name[length + 1] = '\0';
    m_CompressorName = (const char*)&compressor_name[1];

    result = stream.ReadUI16(m_Depth);
    if (AP4_FAILED(result)) return result;

    result = stream.ReadUI16(m_Predefined3);
    if (AP4_FAILED(result)) return result;

    return AP4_SUCCESS;
}
