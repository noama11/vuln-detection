#include <sys/types.h>
#include <limits.h>

ssize_t archive_write_data(struct archive *_a, const void *buff, size_t s)
{
    // Check the magic number of the archive to ensure correct state for writing
    if (_a == NULL || _a->magic != ARCHIVE_MAGIC) {
        return -1;
    }

    // Limit the size of the data to the maximum ssize_t value to prevent overflow
    if (s > (size_t)SSIZE_MAX) {
        s = (size_t)SSIZE_MAX;
    }

    // Clear any previous errors
    _a->error = 0;

    // Call the format-specific write function to perform actual data writing
    return _a->write_func(_a, buff, s);
}
