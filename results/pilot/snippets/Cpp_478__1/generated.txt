int decodeBlock(const uint8_t* encoded, int16_t* decoded) {
    // Retrieve the number of audio channels from the track
    int numChannels = track->getChannels();

    const uint8_t* p = encoded;

    // Iterate over each channel
    for (int ch = 0; ch < numChannels; ch++) {
        // Extract predictor and index from the encoded data
        int16_t predictor = *reinterpret_cast<const int16_t*>(p);
        p += 2;
        int index = *p++;

        // For each frame in the packet, decode two samples
        for (int frame = 0; frame < 505; frame++) {
            uint8_t nibbleByte = *p++;

            // Decode first sample (lower nibble)
            int sample1 = decodeSample(nibbleByte & 0x0F, index, predictor);
            decoded[(frame * 2) * numChannels + ch] = sample1;

            // Decode second sample (upper nibble)
            int sample2 = decodeSample((nibbleByte >> 4) & 0x0F, index, predictor);
            decoded[(frame * 2 + 1) * numChannels + ch] = sample2;
        }
    }

    // Return the total size of the decoded audio data in bytes
    int totalSamples = 505 * 2 * numChannels;
    return totalSamples * sizeof(int16_t);
}
