#include <stdlib.h>
#include <string.h>

ExprDef *AppendKeysyms(ExprDef *expr, ExprDef *append)
{
    if (expr == NULL || append == NULL) {
        return expr;
    }
    int current_size = expr->keysym_count;
    int append_size = append->keysym_count;
    int new_size = current_size + append_size;
    expr->keysyms = realloc(expr->keysyms, new_size * sizeof(int));
    if (append_size > 0) {
        memcpy(expr->keysyms + current_size, append->keysyms,
               append_size * sizeof(int));
    }
    expr->keysym_count = new_size;
    free(append->keysyms);
    free(append);
    return expr;
}
