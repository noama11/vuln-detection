#include <tiff.h>
#include <tiffio.h>

int tiff_write_directory(TIFF *tiff) {
    // Print the TIFF directory to standard output
    // The second argument (0) indicates minimal verbosity (MagickFalse equivalent)
    TIFFPrintDirectory(tiff, 0);

    // Attempt to write the TIFF directory back to the file
    if (TIFFWriteDirectory(tiff) == 0) {
        // Write operation failed; return error status
        return 0;
    }

    // Write operation succeeded
    return 1;
}
