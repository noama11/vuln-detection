// Assuming CanAlloc function exists in the codebase
extern "C" int CanAlloc(size_t size);

// OGRExpatRealloc
void* OGRExpatRealloc(void* ptr, size_t size)
{
    if (CanAlloc(size))
    {
        return std::realloc(ptr, size);
    }
    return nullptr;
}

// OGRExpatUnknownEncodingHandler
int OGRExpatUnknownEncodingHandler(void* unused_encodingHandlerData,
                                   const XML_Char* name,
                                   XML_Encoding* info)
{
    (void)unused_encodingHandlerData;

    if (std::strcmp(name, "WINDOWS-1252") == 0)
    {
        FillWINDOWS1252(info);
        return XML_STATUS_OK;
    }
    else if (std::strcmp(name, "ISO-8859-15") == 0)
    {
        FillISO885915(info);
        return XML_STATUS_OK;
    }

    return XML_STATUS_ERROR;
}
