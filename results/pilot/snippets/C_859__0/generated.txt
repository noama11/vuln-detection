#include <stdlib.h>
#include <string.h>
#include <stddef.h>

typedef struct {
    char *url;
    double duration;
    size_t data_size;
} Segment;

typedef struct {
    char *url;
    Segment *segments;
    int segment_count;
    unsigned char *packet_data;
    size_t packet_size;
} Playlist;

typedef struct {
    int bandwidth;
    char **media_types;
    int media_type_count;
} variant_info;

typedef struct {
    char **http_headers;
    int header_count;
    char *proxy_settings;
    Playlist *playlists;
    int playlist_count;
} HLSContext;

void free_segment_list(Segment *segments, int count) {
    if (segments == NULL) return;
    for (int i = 0; i < count; i++) {
        if (segments[i].url != NULL) {
            free(segments[i].url);
        }
    }
    free(segments);
}

void free_playlist_list(HLSContext *ctx) {
    if (ctx == NULL || ctx->playlists == NULL) return;

    for (int i = 0; i < ctx->playlist_count; i++) {
        if (ctx->playlists[i].url != NULL) {
            free(ctx->playlists[i].url);
        }
        if (ctx->playlists[i].segments != NULL) {
            free_segment_list(ctx->playlists[i].segments, ctx->playlists[i].segment_count);
        }
        if (ctx->playlists[i].packet_data != NULL) {
            free(ctx->playlists[i].packet_data);
        }
    }
    free(ctx->playlists);
    ctx->playlists = NULL;
    ctx->playlist_count = 0;
}

Playlist* new_playlist(HLSContext *ctx, const char *url) {
    if (ctx == NULL || url == NULL) return NULL;

    Playlist *playlists = (Playlist *)realloc(ctx->playlists, sizeof(Playlist) * (ctx->playlist_count + 1));
    if (playlists == NULL) return NULL;

    ctx->playlists = playlists;
    Playlist *new_pl = &ctx->playlists[ctx->playlist_count];

    new_pl->url = (char *)malloc(strlen(url) + 1);
    if (new_pl->url == NULL) return NULL;

    strcpy(new_pl->url, url);
    new_pl->segments = NULL;
    new_pl->segment_count = 0;
    new_pl->packet_data = NULL;
    new_pl->packet_size = 0;

    ctx->playlist_count++;
    return new_pl;
}

void reset_packet(unsigned char *packet_data, size_t packet_size) {
    if (packet_data != NULL && packet_size > 0) {
        memset(packet_data, 0, packet_size);
    }
}

void free_hls_context(HLSContext *ctx) {
    if (ctx == NULL) return;
    free_playlist_list(ctx);
    if (ctx->http_headers != NULL) {
        for (int i = 0; i < ctx->header_count; i++) {
            if (ctx->http_headers[i] != NULL) {
                free(ctx->http_headers[i]);
            }
        }
        free(ctx->http_headers);
    }
    if (ctx->proxy_settings != NULL) {
        free(ctx->proxy_settings);
    }
    free(ctx);
}
