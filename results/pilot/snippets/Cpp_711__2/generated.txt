#include <string>
#include <cstring>
#include <algorithm>
#include <openssl/md5.h>
#include <openssl/rc4.h>

std::string computeUValue(std::string const& user_password, QPDF::EncryptionData const& data) {
    // PDF standard password padding (from PDF specification)
    static const unsigned char PDF_PADDING[32] = {
        0x28, 0xBF, 0x4E, 0x5E, 0x4E, 0x75, 0x8A, 0x41,
        0x64, 0x00, 0x4E, 0x56, 0xFF, 0xFA, 0x01, 0x08,
        0x2E, 0x2E, 0x00, 0xB6, 0xD0, 0x68, 0x3E, 0x80,
        0x2F, 0x0C, 0xA9, 0xFE, 0x64, 0x53, 0x69, 0x7A
    };

    // Pad the password to 32 bytes using PDF padding
    unsigned char padded_password[32];
    size_t pwd_len = std::min(user_password.length(), size_t(32));
    std::memcpy(padded_password, user_password.c_str(), pwd_len);
    std::memcpy(padded_password + pwd_len, PDF_PADDING, 32 - pwd_len);

    // Construct hash input: padded password + encryption data ID
    std::string hash_input(reinterpret_cast<const char*>(padded_password), 32);
    hash_input += data.getId();

    // Compute MD5 digest of the hash input
    unsigned char md5_digest[MD5_DIGEST_SIZE];
    MD5(reinterpret_cast<const unsigned char*>(hash_input.c_str()),
        hash_input.length(), md5_digest);

    // Process through RC4 iteration (20 iterations for PDF R >= 3)
    unsigned char result[16];
    std::memcpy(result, md5_digest, 16);

    RC4_KEY rc4_key;
    for (int i = 0; i < 20; i++) {
        RC4_set_key(&rc4_key, 16, result);
        RC4(&rc4_key, 16, result, result);
    }

    // Pad result with arbitrary data (zeros) to 32 bytes
    std::string u_value(reinterpret_cast<const char*>(result), 16);
    u_value.append(16, '\0');

    return u_value;
}
