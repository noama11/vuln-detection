#include <stdlib.h>
#include <string.h>
#include <stdio.h>

plist_t parse_string_node(const char **bnode, uint64_t size) {
    // Initialize a new plist data structure (create an empty node)
    plist_t node = plist_new_node();
    if (!node) {
        fprintf(stderr, "Error: Failed to initialize plist node\n");
        return NULL;
    }

    // Allocate memory for the string based on the provided size (+ 1 for null terminator)
    char *string_data = (char *)malloc(size + 1);
    if (!string_data) {
        // Free the allocated node data and log error message
        plist_free(node);
        fprintf(stderr, "Error: Memory allocation failed for string data\n");
        return NULL;
    }

    // Copy the string data from the byte node
    memcpy(string_data, *bnode, size);

    // Null-terminate the string
    string_data[size] = '\0';

    // Set the string value in the node (updates the length)
    plist_set_string_val(node, string_data);

    // Free the temporary string buffer (plist_set_string_val copies the data)
    free(string_data);

    // Return the new node containing the plist data
    return node;
}
