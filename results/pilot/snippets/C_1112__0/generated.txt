#include <assert.h>
#include <stdlib.h>

void MediaStreamImpl_Destroy(MediaStreamImpl* self) {
    if (self == NULL) {
        return;
    }

    // Check that peer_connection_handler_ is not active
    assert(!self->peer_connection_handler_active);

    // Release PeerConnectionFactory if present
    if (self->dependency_factory_ != NULL) {
        PeerConnectionFactory_Release(self->dependency_factory_);
        self->dependency_factory_ = NULL;
    }

    // Verify if network_manager_ exists and if chrome_worker_thread_ is running
    if (self->network_manager_ != NULL) {
        if (self->chrome_worker_thread_running) {
            // Post a task to delete the IPC network manager on the worker thread
            WorkerThread_PostTask(self->chrome_worker_thread_,
                                DeleteNetworkManager,
                                self->network_manager_);

            // Stop the thread (ensuring all tasks are processed to prevent race conditions)
            WorkerThread_Stop(self->chrome_worker_thread_);

            self->chrome_worker_thread_ = NULL;
            self->network_manager_ = NULL;
        } else {
            // If the worker thread is not running, trigger NOTREACHED assertion
            assert(0 && "NOTREACHED");
        }
    }

    // Free the struct itself
    free(self);
}
